tyrone_germany.4 = {
	type = country_event
	placement = ROOT

	title = tyrone_germany.4.t
	desc = tyrone_germany.4.d
	flavor = tyrone_germany.4.f

	event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/german_confederation.dds"
	duration = 3

	trigger = {
		c:PRU ?= this
		is_power_bloc_leader = yes
	}

	immediate = {
		every_country = {
			limit = {
				OR = {
					country_has_primary_culture = cu:north_german
					country_has_primary_culture = cu:south_german
				}
				NOT = { this = root }
				NOT = { c:AUS ?= this }
			}
			root.power_bloc = {
				add_leverage = {
					target = prev
					value = 500
				}
			}
		}
		every_country = {
			limit = {
				OR = {
					country_has_primary_culture = cu:north_german
					country_has_primary_culture = cu:south_german
				}
				is_subject = no
				NOT = { this = root }
				NOT = { c:AUS ?= this }
			}
			join_power_bloc = root
		}
	}

	option = {
		name = tyrone_germany.4.a
		default_option = yes
		every_country = {
			limit = {
				capital = {
					OR = {
						region = sr:region_central_europe
						region = sr:region_central_europe
					}
				}
				NOT = {
					OR = {
						c:AUS?= this
						c:SWI?= this
					}
				}
				is_player = no
			}
			c:PRU.power_bloc ?= {
				add_leverage = { target = prev value = 1000 }
			}
		}
	}
}
