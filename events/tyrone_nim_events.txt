tyrone_nim.70 = {
type = country_event
    placement = ROOT
    title = tyrone_nim.70.t
    desc = tyrone_nim.70.d
    flavor = tyrone_nim.70.f
    event_image = { video = "asia_politician_parliament_motion" }
    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/religion_icons/sunni.dds"
    duration = 3
    trigger = { has_variable = tyrone_nim_started }
    immediate = {
        save_scope_as = nim_india
        c:PAK ?= {
            if = {
                limit = { NOT = { has_variable = tyrone_nim_independence_barracks } }
                save_scope_as = nim_new_country
                capital = {
                    create_building = {
                        building = building_barracks
                        add_ownership = {
                            country = { country = scope:nim_new_country levels = 50 }
                        }
                        reserves = 1
                    }
                }
                set_variable = tyrone_nim_independence_barracks
            }
            if = { limit = { has_modifier = tyrone_nim_defending_independence } remove_modifier = tyrone_nim_defending_independence }
            add_modifier = { name = tyrone_nim_defending_independence days = tyrone_nim_two_years }
        }
    }

option = { name = tyrone_nim.70.a default_option = yes }
option = {
    name = tyrone_nim.70.b
    trigger = { exists = c:PAK c:PAK = { is_subject = no } NOT = { is_at_war_with = c:PAK } }
    if = { limit = { has_modifier = tyrone_nim_reannexation } remove_modifier = tyrone_nim_reannexation }
    add_modifier = { name = tyrone_nim_reannexation days = tyrone_nim_two_years }
    create_diplomatic_play = { target_country = c:PAK type = dp_annex_war escalation = 75 }
}

option = {
    name = tyrone_nim.70.c
    trigger = { exists = c:PAK c:PAK = { is_subject = no } NOT = { is_at_war_with = c:PAK } }
    if = { limit = { has_modifier = tyrone_nim_reannexation } remove_modifier = tyrone_nim_reannexation }
    add_modifier = { name = tyrone_nim_reannexation days = tyrone_nim_two_years }
    create_diplomatic_play = { target_country = c:PAK type = dp_make_protectorate escalation = 75 }
}
}
