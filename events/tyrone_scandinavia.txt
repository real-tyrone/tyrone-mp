		video = "game/dlc/dlc004_voice_of_the_people/gfx/event_pictures/votp_monarch_holding_court.bk2"
	}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	duration = 1

	option = {
		name = scandinavian_unification.1.a
		default_option = yes
		ROOT = {change_tag = SCA}
		add_modifier = {
			name = modifier_legacy_of_bjorn_ironside
			months = 120
			}
		}
	option = {
		name = scandinavian_unification.1.b
		add_modifier = {
			name = modifier_scandinavian_academia
			months = 120
		 }
		}
	after = {
		hidden_effect = {
			set_global_variable = scandinavia_unified
		}
		ROOT = {change_tag = SCA}
		if = {
			limit = {
				NOT = {
					country_has_primary_culture = cu:finnish
				}
			}
			add_journal_entry = { type = je_scandinavia_missing_crown }
		}
	}
}
