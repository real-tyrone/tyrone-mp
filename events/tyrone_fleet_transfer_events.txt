	desc = tyrone_fleet_transfer.1.d
	flavor = tyrone_fleet_transfer.1.f

		ordered_scope_fleet = { limit = { num_ships > 0 NOT = { any_scope_ship = { is_in_battle = yes } } } order_by = { add = num_ships } position = 0 save_scope_as = tyrone_fleet_transfer_fleet_1 }

	}

	option = {
		name = tyrone_fleet_transfer.1.a
		trigger = { exists = scope:tyrone_fleet_transfer_fleet_1 }
		scope:tyrone_fleet_transfer_fleet_1 = { every_scope_ship = { set_ship_owner_multiple = scope:tyrone_fleet_transfer_taker } }
	}

	option = {
		name = tyrone_fleet_transfer.1.b
		trigger = { exists = scope:tyrone_fleet_transfer_fleet_2 }
		scope:tyrone_fleet_transfer_fleet_2 = { every_scope_ship = { set_ship_owner_multiple = scope:tyrone_fleet_transfer_taker } }
	}

	option = {
		name = tyrone_fleet_transfer.1.c
		trigger = { exists = scope:tyrone_fleet_transfer_fleet_3 }
		scope:tyrone_fleet_transfer_fleet_3 = { every_scope_ship = { set_ship_owner_multiple = scope:tyrone_fleet_transfer_taker } }

	option = {

	option = {
		name = tyrone_fleet_transfer.next
		trigger = { exists = scope:tyrone_fleet_transfer_next_fleet }
		ROOT = { trigger_event = { id = tyrone_fleet_transfer.2 days = 0 popup = yes } }
	}

	option = {
		name = tyrone_fleet_transfer.cancel
		default_option = yes
	}
}
