	type = country_event
	placement = ROOT

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3

	cooldown = {
		days = 1825
	}

	trigger = {
		tyr_gir_de_1_eligible = yes
	}

	cancellation_trigger = {
		OR = {
			has_variable = tyr_gir_ended
			AND = {
				NOT = {
					has_journal_entry = je_tyr_gir_main
				}
			}
		}
	}

	immediate = {
		set_variable = {
			name = tyr_gir_de_global_cooldown
			days = 180
		}
		set_variable = {
			name = tyr_gir_de_open
			days = 1095
		}
		set_variable = {
			name = tyr_gir_de_1_pending
			days = 1095
		}
		set_variable = {
			name = tyr_gir_de_1_cooldown
			days = 1825
		}
		set_variable = tyr_gir_de_1_done
	}

	option = {
		name = tyr_gir_de.1.1
		default_option = yes
		hidden_effect = {
			remove_variable = tyr_gir_de_open
			remove_variable = tyr_gir_de_1_pending
			set_variable = {
				name = tyr_gir_de_1_active
				days = 1095
			}
		}
		custom_tooltip = tyr_gir_de_outside_effect_tt
		every_scope_state = {
			limit = {
				tyr_gir_german_state = yes
				NOT = {
					tyr_gir_zone_state = yes
				}
			}
			hidden_effect = {
				root = {
					add_to_variable_list = {
						name = tyr_gir_de_targets
						target = prev
					}
				}
			}
			add_modifier = {
				name = tyr_gir_de_new_shafts
				days = 1095
			}
		}
		je:je_tyr_gir_main ?= {
			add_modifier = {
				name = tyr_gir_de_cost_1
				multiplier = root.tyr_gir_expense_value
				days = 1095
			}
		}
		ai_chance = {
			base = 10
		}
	}

	option = {
		name = tyr_gir_de.1.2
		hidden_effect = {
			remove_variable = tyr_gir_de_open
			remove_variable = tyr_gir_de_1_pending
			set_variable = {
				name = tyr_gir_de_1_active
				days = 1095
			}
		}
		custom_tooltip = tyr_gir_de_outside_effect_tt
		every_scope_state = {
			limit = {
				tyr_gir_german_state = yes
				NOT = {
					tyr_gir_zone_state = yes
				}
			}
			hidden_effect = {
				root = {
					add_to_variable_list = {
						name = tyr_gir_de_targets
						target = prev
					}
				}
			}
			add_modifier = {
				name = tyr_gir_de_intensive_blasting
				days = 1095
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				factor = 3
				any_scope_character = {
					has_variable = tyr_gir_chancellor
					is_character_alive = yes
					has_trait = reckless
				}
			}
		}
	}

	option = {
		name = tyr_gir_de.1.3
		trigger = {
			NOT = {
				has_technology_researched = ammonium_nitrate_explosive
			}
			has_technology_researched = dynamite
			has_technology_researched = improved_fertilizer
		}
		hidden_effect = {
			remove_variable = tyr_gir_de_open
			remove_variable = tyr_gir_de_1_pending
		}
		add_technology_progress = {
			technology = ammonium_nitrate_explosive
			progress = 4000
		}
		je:je_tyr_gir_main ?= {
			add_modifier = {
				name = tyr_gir_de_cost_1
				multiplier = root.tyr_gir_expense_value
				days = 1095
			}
		}

			base = 10

					has_variable = tyr_gir_chancellor

				}

tyr_kk.1 = {
	type = country_event

	title = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = tyr_kk_state_victory
				}
				desc = tyr_kk_state_title
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = tyr_kk_church_title
			}
		}
	}

	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = tyr_kk_state_victory
				}
				desc = tyr_kk_state_desc
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = tyr_kk_church_desc
			}
		}
	}

	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_variable = tyr_kk_state_victory
				}
				desc = tyr_kk_state_flavor
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = tyr_kk_church_flavor
			}
		}
	}

	event_image = {
		video = "votp_monarch_holding_court"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	duration = 3

	trigger = {
		has_variable = tyr_kk_completed
		NOT = {
			has_variable = tyr_kk_reward_received
		}
	}

	option = {
		name = tyr_kk_settlement_option
		default_option = yes
		if = {
			limit = {
				NOT = {
					has_variable = tyr_kk_reward_received
				}
			}
			if = {
				limit = {
					has_variable = tyr_kk_state_victory
				}
				add_modifier = {
					name = tyr_kk_state_supremacy
				}
			}
			else = {
				ig:ig_devout ?= {
					set_ig_trait = ig_trait:ig_trait_be_fruitful_and_multiply
					add_modifier = {
						name = tyr_kk_clerical_influence
					}
				}
				add_modifier = {
					name = tyr_kk_religious_settlement
				}
			}
			hidden_effect = {
				set_variable = tyr_kk_reward_received
			}
		}
	}
}
