namespace = hungry_forties

hungry_forties.1 = {
	type = country_event
	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	title = hungry_forties.1.t
	desc = hungry_forties.1.d
	flavor = hungry_forties.1.f

	trigger = {
		# Triggered by on_action
	}

	immediate = {
        if = {
            limit = { c:GBR ?= this }
            every_country = {
                limit = { tyr_monroe_usa = yes }
                trigger_event = { id = tyr_irish_migration.1 days = 0 popup = yes }
            }
            every_country = {
                every_scope_state = {
                    limit = {
                        OR = {
                            state_region = s:STATE_ULSTER
                            state_region = s:STATE_MUNSTER
                            state_region = s:STATE_LEINSTER
                            state_region = s:STATE_CONNAUGHT
                        }
                    }
                    add_modifier = { name = tyr_irish_great_famine years = 5 }
                }
            }
        }
		random_state_in_europe = {
			limit = {
				state_region = {
					has_harvest_condition = disease_outbreak
				}
			}
			save_scope_as = relevant_state
		}
	}

	duration = 3

	option = {
		name = hungry_forties.1.a
        default_option = yes
		show_as_tooltip = {
			add_modifier = {
				name = modifier_european_potato_famine
				days = 1095
			}
		}
	}

    option = {
		name = hungry_forties.1.b
		show_as_tooltip = {
			add_modifier = {
				name = modifier_european_potato_famine
				days = 1095
			}
		}
		add_modifier = {
			name = modifier_european_potato_famine_relief
			days = 1095
		}
		add_modifier = {
			name = modifier_european_potato_famine_relief_costs
			multiplier = money_amount_multiplier_small
			days = 1095
		}
	}
}
