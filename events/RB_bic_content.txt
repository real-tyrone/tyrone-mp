			save_scope_as = gbr_scope
		}

			every_scope_character = {

					exists = this
					is_character_alive = yes
					is_ruler_of_own_country = yes
				}
				save_scope_as = gbr_ruler
			}

		}
		set_variable = company_dispute_started
	}

	option = {
		name = bic_content.1.a
		add_journal_entry = {
			type = je_company_equity
		}
	}

#gov of india act

bic_content.2 = {
	type = country_event
	placement = root
	title = bic_content.2.t
	desc = bic_content.2.d
	flavor = bic_content.2.f
	cooldown = {
		days = never_fire_again_modifier_time
	}
	minor_left_icon = scope:gbr_scope
	minor_right_icon = scope:company_scope
	event_image = {
		video = "ip2_telegraph_dictate"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/company_equity_dispute.dds"
	duration = 3
	immediate = {
		tyrone_unstable_bind_india = yes
		scope:raj_scope = {
			save_scope_as = company_scope
			if = {
				limit = {
					is_subject = yes
				}
				overlord = {
					save_scope_as = gbr_scope
				}
			}
		}
	}
	option = {
		name = bic_content.2.a
		default_option = yes
		custom_tooltip = tyrone_raj_crown_act_effect_tt
		if = {
			limit = {
				scope:gbr_scope ?= this
			}
			add_modifier = {
				name = gov_of_india_act_MODIFIER
				years = 25
				is_decaying = yes
			}
			set_variable = enacted_government_of_india_act
		}
	}
}

bic_content.3 = {
	type = country_event
	placement = root
	title = bic_content.3.t
	desc = bic_content.3.d
	flavor = bic_content.3.f
	cooldown = {
		days = never_fire_again_modifier_time
	}
	minor_left_icon = scope:gbr_scope
	minor_right_icon = scope:company_scope
	event_image = {
		video = "ip2_india_countryside"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/company_equity_dispute.dds"
	duration = 3
	immediate = {
		tyrone_unstable_bind_india = yes
		scope:raj_scope = {
			save_scope_as = company_scope
			if = {
				limit = {
					is_subject = yes
				}
				overlord = {
					save_scope_as = gbr_scope
				}
			}
		}
	}
	option = {
		name = bic_content.3.a
		default_option = yes
		if = {
			limit = {
				scope:company_scope ?= this
			}
			add_modifier = {
				name = economic_privileges_returned
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
		else_if = {
			limit = {
				scope:gbr_scope ?= this
			}
			add_modifier = {
				name = charter_act_repealed
				days = long_modifier_time
				is_decaying = yes
			}
		}
		scope:company_scope = {
			set_variable = repealed_charter_act
		}
	}
}
