 type = country_event

 on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
 duration = 14
 trigger = {
  global_var:tyr_ga_phase = 5
  exists = global_var:tyr_ga_austria
  NOT = { has_variable = tyr_ga_settlement_decided }
 }

  global_var:tyr_ga_austria = { save_scope_as = tyr_ga_settlement_austria }
 }
 option = {
  name = tyr_ga.3.a
  default_option = yes
  ai_chance = { base = 50 }

  s:STATE_TYROL = { add_claim = ROOT }
  s:STATE_STYRIA = { add_claim = ROOT }
  hidden_effect = {
   set_variable = tyr_ga_settlement_decided
   tyr_ga_post_settlement_news = { NOTIFICATION = tyr_ga_claims_austria_news }
  }
  trigger_event = { id = tyr_ga.4 days = 1 popup = yes }
 }
 option = {
  name = tyr_ga.3.b
  ai_chance = { base = 50 }
  change_relations = { country = global_var:tyr_ga_austria value = 200 }
  global_var:tyr_ga_austria = { add_modifier = { name = tyr_ga_austrian_reapproach years = 5 } }
  hidden_effect = {
   set_variable = tyr_ga_settlement_decided
   tyr_ga_post_settlement_news = { NOTIFICATION = tyr_ga_reapproach_news }
  }
 }
}
