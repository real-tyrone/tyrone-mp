nihilism.1 = {
	type = country_event

	title = nihilism.1.t
	desc = nihilism.1.d
	flavor = nihilism.1.f

	event_image = {
		video = "votp_realist_household"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	trigger = {
	 	#Triggered by journal entry
	}

	immediate = {
		if = {
			limit = {
				has_dlc_feature = voice_of_the_people_content
			}
			if = {
				limit = {
					c:RUS ?= THIS
					AND = {
						game_date >= 1862.1.1
						game_date <= 1882.1.1
					}
				}
				create_character = {
					template = RUS_varfolomey_zaitsev
				}
			}
			else_if = {
				limit = {
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
					AND = {
						game_date >= 1844.1.1
						game_date <= 1866.1.1
					}
					OR = { #So that the player always has first dibs on Stirner if they're playing a German country.
						is_player = yes
						NOT = {
							any_country = {
								OR = {
									country_has_primary_culture = cu:north_german
									country_has_primary_culture = cu:south_german
								}
								is_player = yes
							}
						}
					}
					NOT = { has_global_variable = stirner_spawn }
				}
				create_character = {
					template = BAV_max_stirner
					on_created = {
						set_variable = is_stirner_var
						set_global_variable = stirner_spawn
					}
				}
			}
			else_if = {
				limit = {
					c:USA ?= THIS
					AND = {
						game_date >= 1876.1.1
						game_date <= 1930.1.1
					}
				}
				create_character = {
					template = USA_benjamin_tucker
				}
			}
			else_if = {
				limit = {
					c:GBR ?= THIS
					AND = {
						game_date >= 1846.1.1
						game_date <= 1906.1.1
					}
				}
				create_character = {
					template = GBR_george_holyoake
				}
			}
			else_if = {
				limit = {
					c:JAP ?= THIS
					AND = {
						game_date >= 1878.1.1
						game_date <= 1908.1.1
					}
				}
				create_character = {
					template = JAP_kume_kunitake
				}
			}
			else = {
				create_character = {
					is_agitator = yes
					interest_group = ig_intelligentsia
					ideology = ideology_nihilist
				}
			}
		}
		else = {
			create_character = {
				is_agitator = yes
				interest_group = ig_intelligentsia
				ideology = ideology_nihilist
			}
		}

		random_scope_character = {
			limit = {
				has_role_of_type = agitator
				has_ideology = ideology:ideology_nihilist
			}
			save_scope_as = nihilist_scope
			set_career_length = {
				years = 20
				random_range = { 1.0 1.5 } # 20-30 years
			}
		}
	}

	option = {
		name = nihilism.1.a
		default_option = yes
		add_modifier = {
			name = modifier_nihilist_movement_enhancement
			days = long_modifier_time
			multiplier = 2
		}
		custom_tooltip = {
			text = t4_nihilism_option_nihilist_progress_10_tt
			je:je_nihilist_movement ?= {
			add_progress = { value = 10 name = t4_nihilism_nihilist_progress_bar }
			}
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_nihilist
					}
				}
			}
			create_political_movement = { type = movement_nihilist }
		}

		custom_tooltip = {
			text = nihilism.1.tt
		}
		scope:nihilist_scope = {
			add_modifier = {
				name = popular_nihilist
			}
		}
		set_variable = nihilism_je_started
	}

	option = {
		name = nihilism.1.b

		add_modifier = {
			name = modifier_nihilist_movement_enhancement
			days = long_modifier_time
		}
		custom_tooltip = {
			text = t4_nihilism_option_nihilist_progress_5_tt
			je:je_nihilist_movement ?= {
			add_progress = { value = 5 name = t4_nihilism_nihilist_progress_bar }
			}
		}

		custom_tooltip = {
			text = nihilism.1.tt
		}
		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_nihilist
					}
				}
			}
			create_political_movement = { type = movement_nihilist }
		}
		scope:nihilist_scope = {
			add_modifier = {
				name = less_popular_nihilist
			}
		}
		set_variable = nihilism_je_started
	}
	option = {
		name = nihilism.1.c
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_outlawed_dissent
				has_law_or_variant = law_type:law_censorship
			}
		}
		custom_tooltip = {
			text = nihilism.1.tt
		}
		custom_tooltip = {
			text = t4_nihilism_option_devout_progress_10_tt
			je:je_nihilist_movement ?= {
			add_progress = { value = 10 name = t4_nihilism_devout_progress_bar }
			}
		}
		if = {

				NOT = {

						is_political_movement_type = movement_nihilist
					}
				}
			}
