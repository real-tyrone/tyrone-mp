namespace = tyrone_indian_mining

tyrone_indian_mining.1 = {
	type = country_event
	placement = root
	title = tyrone_indian_mining.1.t
	desc = tyrone_indian_mining.1.d
	flavor = tyrone_indian_mining.1.f
	event_image = { video = "ip4_mining_scene" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/building_icons/lead_mine.dds"
	immediate = {
		if = {
			limit = {
				tyrone_promise_is_india = yes
				has_global_variable = tyrone_promise_active_var
				has_global_variable = tyrone_mining_rajputana_active
				has_global_variable = tyrone_mining_rajputana_survey_1_queued
				NOT = { has_global_variable = tyrone_mining_rajputana_survey_1_done }
			}
			# Set the persistent result lock before applying native state-region resource changes.
			set_global_variable = tyrone_mining_rajputana_survey_1_done
				s:STATE_RAJPUTANA = { change_resource_potential = { type = building_lead_mine amount = 14 } }
				s:STATE_RAJPUTANA = { change_resource_potential = { type = building_sulfur_mine amount = 15 } }
		}
	}
	option = {
		name = tyrone_indian_mining.1.a
		default_option = yes
		custom_tooltip = tyrone_indian_mining.1.result_tt
	}
}
