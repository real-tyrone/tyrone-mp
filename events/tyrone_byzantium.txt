namespace = byzantine_reconquest

byzantine_reconquest.1 = {
	type = country_event
	placement = ROOT
	title = byzantine_reconquest.1.t
	desc = byzantine_reconquest.1.d
	flavor = byzantine_reconquest.1.f
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/byzantium_je.dds"
	duration = 3

	immediate = {
		s:STATE_KONYA = { add_homeland = cu:greek }
		s:STATE_TRABZON = { add_homeland = cu:greek }
		s:STATE_KASTAMONU = { add_homeland = cu:greek }
		s:STATE_ADANA = { add_homeland = cu:greek }
		s:STATE_ANKARA = { add_homeland = cu:greek }
	}

	option = {
		name = byzantine_reconquest.1.a
		default_option = yes
		every_scope_state = {
			limit = {
				state_region = {
					OR = {
						this = s:STATE_EASTERN_THRACE
						AND = {
							region = sr:region_near_east
							is_homeland = cu:greek
							is_homeland = cu:turkish
						}
					}
				}
			}
			add_modifier = {
				name = byz_rehellenization_of_anatolia
				days = very_long_modifier_time
			}
		}
		add_journal_entry = { type = je_byz_reclaim_levant }
		ai_chance = { factor = 1 }
	}

	option = {
		name = byzantine_reconquest.1.b
		add_modifier = { name = byz_integrated_turks }
		add_loyalists = { culture = cu:turkish value = 0.25 }
		add_journal_entry = { type = je_byz_reclaim_levant }
		ai_chance = { factor = 1 }
	}
}
