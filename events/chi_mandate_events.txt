namespace = chi_mandate

chi_mandate.1 = {
	type = country_event
	placement = ROOT
	cooldown = { years = 10 }

	title = chi_mandate.1.t
	desc = chi_mandate.1.d
	flavor = chi_mandate.1.f

	event_image = {
		video = "asia_union_leader"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		has_journal_entry = je_mandate_of_heaven
		ig:ig_intelligentsia ?= {
			ig_approval < 0
		}
	}

	option = {
		name = chi_mandate.1.a
		custom_tooltip = chi_mandate.1.a.tt
		trigger = {
			authority >= 150
		}
		change_authority = -150
		add_modifier = {
			name = chi_mandate_funding
			years = 10
		}
	}

	option = {
		name = chi_mandate.1.b
		default_option = yes
		custom_tooltip = chi_mandate.1.b.tt
		add_radicals = {
			value = small_radicals
		}
		add_modifier = {
			name = chi_mandate_suppressed_innovation
			years = 3
		}
	}

	option = {
		name = chi_mandate.1.c
		custom_tooltip = chi_mandate.1.c.tt
		add_radicals = {
			value = large_radicals
		}
		add_modifier = {
			name = chi_mandate_crackdown
			years = 5
		}
	}
}

chi_mandate.2 = {
	type = country_event
	placement = ROOT

	title = chi_mandate.2.t
	desc = chi_mandate.2.d
	flavor = chi_mandate.2.f

	event_image = {
		video = "asia_union_leader"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		has_journal_entry = je_mandate_of_heaven
		je:je_mandate_of_heaven = {
			"scripted_bar_progress(chi_centralisation_progress_bar)" <= 3
		}
	}

	option = {
		name = chi_mandate.2.a
		default_option = yes
		add_modifier = {
			name = chi_mandate_low_centralisation_landowners
			years = 10
			is_decaying = yes
		}
		trigger_event = {
			id = chi_mandate.3
			days = 3650
			popup = no
		}
	}

	option = {
		name = chi_mandate.2.b
		trigger = {
			ig:ig_intelligentsia ?= {
				is_in_government = yes
				ig_approval >= 5
			}
		}
		show_as_unavailable = {
			always = yes
		}
		add_technology_researched = empiricism
		add_modifier = {
			name = chi_mandate_empiricism
			years = 5
		}
	}

	option = {
		name = chi_mandate.2.c
		trigger = {
			has_technology_researched = medical_degrees
		}
		show_as_unavailable = {
			always = yes
		}
		add_modifier = {
			name = chi_mandate_medical_degrees
			years = 5
		}
	}

	option = {
		name = chi_mandate.2.d
		add_modifier = {
			name = chi_mandate_three_groups_discontent
			years = 10
		}
	}
}

chi_mandate.3 = {
	type = country_event
	hidden = yes
	placement = ROOT

	trigger = {
		has_journal_entry = je_mandate_of_heaven
	}

	immediate = {
		je:je_mandate_of_heaven = {
			add_progress = {
				value = 1
				name = chi_centralisation_progress_bar
			}
		}
	}
}
