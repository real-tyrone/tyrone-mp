namespace = chi_mandate

chi_mandate.1 = {
	type = country_event
	placement = ROOT
	cooldown = { years = 10 }

	title = chi_mandate.1.t
	desc = chi_mandate.1.d
	flavor = chi_mandate.1.f

	event_image = {
		video = "asia_union_leader"
	}

	duration = 3

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		has_journal_entry = je_mandate_of_heaven
		ig:ig_intelligentsia ?= {
			ig_approval < 0
		}
	}

	option = {
		name = chi_mandate.1.a
		trigger = {
			authority >= 150
		}
		add_modifier = {
			name = chi_mandate_funding
			years = 10
		}
	}

	option = {
		name = chi_mandate.1.b
		default_option = yes
		add_radicals = {
			value = small_radicals
		}
		add_modifier = {
			name = chi_mandate_suppressed_innovation
			years = 3
		}
	}

	option = {
		name = chi_mandate.1.c
		add_radicals = {
			value = large_radicals
		}
		add_modifier = {
			name = chi_mandate_crackdown
			years = 5
		}
	}
}

chi_mandate.2 = {
	type = country_event
	placement = ROOT

	title = chi_mandate.2.t
	desc = chi_mandate.2.d
	flavor = chi_mandate.2.f

	event_image = {
		video = "asia_union_leader"
	}

	duration = 3

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		has_journal_entry = je_mandate_of_heaven
		je:je_mandate_of_heaven = {
			"scripted_bar_progress(chi_centralisation_progress_bar)" <= 3
		}
	}

	option = {
		name = chi_mandate.2.a
		default_option = yes
		add_modifier = {
			name = chi_mandate_low_centralisation_landowners
			years = 10
			is_decaying = yes
		}


		custom_tooltip = {
			text = chi_mandate_centralisation_gain_tt
			je:je_mandate_of_heaven = {
				add_progress = { name = chi_centralisation_progress_bar value = 1 }
			}
		}
		hidden_effect = { chi_mandate_update_centralisation_modifier = yes }
	}

	option = {
		name = chi_mandate.2.b
		add_technology_researched = empiricism
		add_modifier = {
			name = chi_mandate_empiricism
			years = 5
		}

		custom_tooltip = {
			text = chi_mandate_centralisation_gain_2_tt
			je:je_mandate_of_heaven = {
				add_progress = { name = chi_centralisation_progress_bar value = 2 }
			}
		}
		hidden_effect = { chi_mandate_update_centralisation_modifier = yes }
	}

	option = {
		name = chi_mandate.2.c
		add_modifier = {
			name = chi_mandate_medical_degrees
			years = 5
		}
	}

	option = {
		name = chi_mandate.2.d_option
		add_modifier = {
			name = chi_mandate_three_groups_discontent
			years = 10
		}
	}
}
