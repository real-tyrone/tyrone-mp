	event_image = {

	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
		c:GBR ?= this
		exists = c:GER
		exists = c:SAF
	}

	option = {
		name = berlin_conference.5.a
		default_option = yes
		ai_chance = { base = 100 }
		change_relations = { country = c:GER value = 20 }
		c:GER = {
			remove_variable = ger_south_west_africa_request_pending
			if = {
				limit = {
					any_subject_or_below = {
						is_country_type = colonial
						OR = {
							any_scope_state = { state_region = s:STATE_HEREROLAND }
							any_scope_state = { state_region = s:STATE_NAMAQUALAND }
						}
					}
				}
				random_subject_or_below = {
					limit = {
						is_country_type = colonial
						OR = {
							any_scope_state = { state_region = s:STATE_HEREROLAND }
							any_scope_state = { state_region = s:STATE_NAMAQUALAND }
						}
					}
					save_scope_as = ger_south_west_africa_target_scope
				}
			}
			else = {
				save_scope_as = ger_south_west_africa_target_scope
			}
		}
		s:STATE_HEREROLAND = {
			every_scope_state = {
				limit = {
					OR = {
						owner = c:GBR
						owner = c:SAF
					}
				}
				set_state_owner = scope:ger_south_west_africa_target_scope
			}
		}
		s:STATE_NAMAQUALAND = {
			every_scope_state = {
				limit = {
					OR = {
						owner = c:GBR
						owner = c:SAF
					}
				}
				set_state_owner = scope:ger_south_west_africa_target_scope
			}
			create_state = {
				country = c:SAF
				owned_provinces = { x8031D0 }
			}
		}
	}

	option = {
		name = berlin_conference.5.b
		ai_chance = { base = 0 }
		change_relations = { country = c:GER value = -20 }
		c:GER = {
			remove_variable = ger_south_west_africa_request_pending
		}
	}
}
