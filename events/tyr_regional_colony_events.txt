namespace = tyr_regional_colonies

tyr_regional_colonies.1 = {
    type = country_event
    placement = scope:tyr_regional_colony
    title = colonial_administration_events.1.t
    desc = tyr_gcol_founding_desc
    flavor = tyr_gcol_founding_flavor
    event_image = { video = "votp_french_algeria" }
    icon = "gfx/interface/icons/event_icons/event_map.dds"
    duration = 3
    trigger = {
        scope:tyr_regional_colony ?= {
            is_subject_of = root
            # Refuse a duplicate offer if a founding event already granted a charter.
            tyr_colonial_administration_needs_charter = yes
        }
    }
    cancellation_trigger = {
        NOT = {
            scope:tyr_regional_colony ?= {
                is_subject_of = root
                tyr_colonial_administration_needs_charter = yes
            }
        }
    }
    option = {
        name = tyr_gcol_charter_extraction
        default_option = yes
        scope:tyr_regional_colony ?= { add_modifier = { name = resource_extraction_charter_modifier } }
    }
    option = {
        name = tyr_gcol_charter_missions
        
        scope:tyr_regional_colony ?= { add_modifier = { name = religious_missions_colonial_modifier } }
    }
    option = {
        name = tyr_gcol_charter_settlement
        
        scope:tyr_regional_colony ?= { add_modifier = { name = settlement_campaigns_modifier } }
    }
    option = {
        name = tyr_gcol_charter_profits
        
        scope:tyr_regional_colony ?= { add_modifier = { name = resource_extraction_colony_modifier } }
    }
}
