scw_events.24 = {
	type = country_event
	hidden = yes
	title = scw_events.24.t
	immediate = {
		save_scope_as = scw_sponsor
		if = {
			limit = { has_variable = scw_congress_pending }
			remove_variable = scw_vote_wait
			if = {
				limit = { has_variable = scw_voting_period }
				# The expiry and event can occur in either order on the same day.
				trigger_event = { id = scw_events.24 days = 1 }
			}
			else = { scw_resolve_congress = yes }
		}
	}
}

scw_events.25 = {
	type = country_event
	duration = 30
	title = scw_events.25.t
	desc = scw_events.25.d
	event_image = { video = "unspecific_ruler_speaking_to_people" }
	icon = "gfx/interface/icons/event_icons/spanish_commonwealth.dds"
	immediate = {
		save_scope_as = scw_sponsor
		c:ALT ?= { save_scope_as = scw_los_altos }
		c:MEX ?= { save_scope_as = scw_los_altos_mexico }
		if = {
			limit = { exists = c:UCA }
			c:UCA = { save_scope_as = scw_los_altos_central }
		}
		else_if = {
			limit = { exists = c:GUA }
			c:GUA = { save_scope_as = scw_los_altos_central }
		}
	}
	option = {
		name = scw_events.25.a
		default_option = yes
		ai_chance = { base = 50 }
		custom_tooltip = scw_los_altos_independent_tt
		if = {
			limit = {
				exists = scope:scw_los_altos
				scope:scw_los_altos = { is_subject = yes }
			}
			scope:scw_los_altos = { make_independent = yes }
		}
		hidden_effect = {
			set_variable = scw_los_altos_resolved
			remove_variable = scw_los_altos_pending
		}
	}
	option = {
		name = scw_events.25.b
		trigger = { exists = scope:scw_los_altos_mexico }
		ai_chance = { base = 25 }
		if = {
			limit = {
				exists = scope:scw_los_altos
				exists = scope:scw_los_altos_mexico
			}
			scope:scw_los_altos_mexico = { annex = scope:scw_los_altos }
		}
		hidden_effect = {
			set_variable = scw_los_altos_resolved
			remove_variable = scw_los_altos_pending
		}
	}
	option = {
		name = scw_events.25.c
		trigger = { exists = scope:scw_los_altos_central }
		ai_chance = { base = 25 }
		if = {
			limit = {
				exists = scope:scw_los_altos
				exists = scope:scw_los_altos_central
			}
			scope:scw_los_altos_central = { annex = scope:scw_los_altos }
		}
		hidden_effect = {
			set_variable = scw_los_altos_resolved
			remove_variable = scw_los_altos_pending
		}
	}
}
