# ROOT is India; shared helper definitions and duration values are supplied by the main overlay.
tyrone_nim.20 = {
	type = country_event
	placement = ROOT
	title = tyrone_nim.20.t
	desc = tyrone_nim.20.d
	flavor = tyrone_nim.20.f
	event_image = { video = "asia_politician_parliament_motion" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	duration = 3
	trigger = {
		has_variable = tyrone_nim_active
		NOT = { has_variable = tyrone_nim_constitution_done }
		NOT = { has_variable = tyrone_nim_socialist_started }
		NOT = { has_variable = tyrone_nim_emergency_pending }

	}
	immediate = { save_scope_as = nim_india }
	option = {
		name = tyrone_nim.20.a
		custom_tooltip = tyrone_nim_const_monarch_ten_tt
		tyrone_nim_shift_constitution = { VALUE = -10 }
		add_modifier = { name = tyrone_nim_event_20_a days = tyrone_nim_event_time }
		add_loyalists = { pop_type = aristocrats value = 0.05 }
		add_radicals = { pop_type = academics value = 0.05 }
	}
	option = {
		name = tyrone_nim.20.b
		default_option = yes
		custom_tooltip = tyrone_nim_const_federal_ten_tt
		tyrone_nim_shift_constitution = { VALUE = 10 }
		add_modifier = { name = tyrone_nim_event_20_b days = tyrone_nim_event_time }
		add_loyalists = { pop_type = academics value = 0.05 }
		add_radicals = { pop_type = aristocrats value = 0.05 }
	}
}
