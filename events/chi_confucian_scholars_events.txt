	placement = ROOT

			}

		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = { has_journal_entry = chi_adm_provincial_scholars }
				desc = chi_adm_provincial_scholars.1.f
			}
			chi_confucian_scholars.1.f
		}
	}

	event_image = {
		video = "asia_union_leader"
	}

	duration = 3

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	cancellation_trigger = {
		NOT = {
			OR = {
			has_journal_entry = je_confront_confucian_scholars
			AND = {
				has_journal_entry = chi_adm_provincial_scholars
				chi_adm_provincial_scholars_eligible = yes
			}
		}
		}
	}

	trigger = {
		OR = {
			has_journal_entry = je_confront_confucian_scholars
			AND = {
				has_journal_entry = chi_adm_provincial_scholars
				chi_adm_provincial_scholars_eligible = yes
			}
		}
		NOT = {
			OR = {
				has_variable = chi_confucian_scholars_end_isolation
				has_variable = chi_confucian_scholars_evict_scholars
			}
		}
	}

	option = {
		name = chi_confucian_scholars.1.a
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_religious_focus_ready = yes
					ig:ig_devout ?= { is_in_government = yes }
				}
				add = 60
			}
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_religious_focus_ready = yes
				}
				add = 30
			}
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_religious_focus_ready = no
					chi_adm_provincial_scholars_merchant_schools_ready = yes
				}
				multiply = 0
			}
		}
		default_option = yes
		set_variable = chi_confucian_scholars_end_isolation
		add_modifier = {
			name = chi_confucian_scholars_end_isolation_modifier
		}
	}

	option = {
		name = chi_confucian_scholars.1.b
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_merchant_schools_ready = yes
					ig:ig_intelligentsia ?= { is_in_government = yes }
				}
				add = 60
			}
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_merchant_schools_ready = yes
				}
				add = 30
			}
			modifier = {
				trigger = {
					has_journal_entry = chi_adm_provincial_scholars
					chi_adm_provincial_scholars_merchant_schools_ready = no
					chi_adm_provincial_scholars_religious_focus_ready = yes
				}
				multiply = 0
			}
		}
		set_variable = chi_confucian_scholars_evict_scholars
		add_modifier = {
			name = chi_confucian_scholars_evict_scholars_modifier
		}
