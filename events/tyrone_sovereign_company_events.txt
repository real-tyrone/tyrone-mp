tyrone_sovereign_company.11 = {
	type = country_event
	placement = root
	title = tyrone_sovereign_company.11.t
	desc = tyrone_sovereign_company.11.d
	flavor = tyrone_sovereign_company.11.f
	event_image = { video = "ip2_india_urban_scene" }
	icon = "gfx/interface/icons/invention_icons/central_archives.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	trigger = { has_variable = tyrone_sc_settlement_active NOT = { has_variable = tyrone_sc_incident_pending } }
	immediate = {
		save_scope_as = csr_india
		set_variable = { name = tyrone_sc_incident_pending days = 30 }
	}
	cancellation_trigger = { NOT = { has_variable = tyrone_sc_settlement_active } }
	option = {
		name = tyrone_sovereign_company.11.a
		default_option = yes

		if = {
			limit = { has_variable = tyrone_sc_settlement_active has_variable = tyrone_sc_incident_pending }
			custom_tooltip = { text = tyrone_sc_experts_ten_tt je:je_tyrone_sc_settlement ?= { add_progress = { name = tyrone_sc_settlement_bar value = 10 } } }
			add_modifier = { name = tyrone_sc_incident_11_experts days = tyrone_sc_political_duration }
			add_loyalists = { pop_type = bureaucrats value = 0.05 }
			remove_variable = tyrone_sc_incident_pending
		}
	}
	option = {
		name = tyrone_sovereign_company.11.b


		if = {
			limit = { has_variable = tyrone_sc_settlement_active has_variable = tyrone_sc_incident_pending }
			custom_tooltip = { text = tyrone_sc_wealth_ten_tt je:je_tyrone_sc_settlement ?= { add_progress = { name = tyrone_sc_settlement_bar value = -10 } } }
			add_modifier = { name = tyrone_sc_incident_11_shareholders days = tyrone_sc_political_duration }
			add_loyalists = { pop_type = capitalists value = 0.05 }
			remove_variable = tyrone_sc_incident_pending
		}
	}
}

tyrone_sovereign_company.30 = {
	type = country_event
	placement = root
	title = tyrone_sovereign_company.30.t
	desc = tyrone_sovereign_company.30.d
	flavor = tyrone_sovereign_company.30.f
	event_image = { video = "ip2_india_urban_scene" }
	icon = "gfx/interface/icons/law_icons/corporate_state.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	trigger = { has_variable = tyrone_sc_capitalists_done }
	immediate = {
		save_scope_as = csr_india
		clear_saved_scope = sc_candidate_1
		clear_saved_scope = sc_candidate_2
		clear_saved_scope = sc_candidate_3
		clear_saved_scope = sc_candidate_4
		if = {
			limit = { any_company = { count >= 1 tyrone_sc_election_candidate = yes } }
			ordered_company = { limit = { tyrone_sc_election_candidate = yes } order_by = company_owned_levels position = 0 save_scope_as = sc_candidate_1 }
		}
		if = {
			limit = { any_company = { count >= 2 tyrone_sc_election_candidate = yes } }
			ordered_company = { limit = { tyrone_sc_election_candidate = yes } order_by = company_owned_levels position = 1 save_scope_as = sc_candidate_2 }
		}
		if = {
			limit = { any_company = { count >= 3 tyrone_sc_election_candidate = yes } }
			ordered_company = { limit = { tyrone_sc_election_candidate = yes } order_by = company_owned_levels position = 2 save_scope_as = sc_candidate_3 }
		}
		if = {
			limit = { any_company = { count >= 4 tyrone_sc_election_candidate = yes } }
			ordered_company = { limit = { tyrone_sc_election_candidate = yes } order_by = company_owned_levels position = 3 save_scope_as = sc_candidate_4 }
		}

	}
	cancellation_trigger = { NOT = { is_country_alive = yes } }
	option = {
		name = tyrone_sovereign_company.30.a
		default_option = yes
		trigger = { exists = scope:sc_candidate_1 }
		tyrone_sc_elect_company = { CANDIDATE = sc_candidate_1 }
	}
	option = {
		name = tyrone_sovereign_company.30.b

		trigger = { exists = scope:sc_candidate_2 }
		tyrone_sc_elect_company = { CANDIDATE = sc_candidate_2 }
	}
	option = {
		name = tyrone_sovereign_company.30.c

		trigger = { exists = scope:sc_candidate_3 }
		tyrone_sc_elect_company = { CANDIDATE = sc_candidate_3 }
	}
	option = {
		name = tyrone_sovereign_company.30.d

		trigger = { exists = scope:sc_candidate_4 }
		tyrone_sc_elect_company = { CANDIDATE = sc_candidate_4 }
	}
	option = {
		name = tyrone_sovereign_company.30.e

		trigger = { NOT = { exists = scope:sc_candidate_1 } NOT = { exists = scope:sc_candidate_2 } NOT = { exists = scope:sc_candidate_3 } NOT = { exists = scope:sc_candidate_4 } }
		remove_variable = tyrone_sc_election_pending
	}
}
