namespace = ostend_manifesto
### Ostend Manifesto Events
# Spain receives the American offer to purchase Cuba
ostend_manifesto.1 = {
	type = country_event
	placement = root

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = ostend_manifesto.1.t
	desc = ostend_manifesto.1.d
	flavor = ostend_manifesto.1.f

	duration = 3

	trigger = {
		exists = c:USA
		exists = c:CUB
	}

	option = {
		name = ostend_manifesto.1.a
		# Accept: receive 5M over 5 years
		default_option = yes
		add_modifier = {
			name = ostend_cuba_offer_income
			months = 60
		}
		c:USA = {
			add_modifier = {
				name = ostend_cuba_offer_payment
				months = 60
			}
			trigger_event = { id = ostend_manifesto.3 days = 7 }
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = { "relations(c:USA)" > relations_threshold:cordial }
				add = 100
			}
		}
	}

	option = {
		name = ostend_manifesto.1.b
		# Counteroffer: 7.5M
		c:USA = {
			trigger_event = { id = ostend_manifesto.2 days = 7 }
		}
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = ostend_manifesto.1.c
		# Refuse
		change_relations = { country = c:USA value = -25 }
		c:USA = { set_variable = ostend_cuba_offer_refused }
		ai_chance = {
			base = 0
			modifier = {
				trigger = { "relations(c:USA)" <= relations_threshold:cordial }
				add = 100
			}
		}
	}
}

# Treaty of Havana â€” USA decides Cuba's political status
ostend_manifesto.3 = {
	type = country_event
	placement = s:STATE_WESTERN_CUBA.region_state:CUB

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = ostend_manifesto.3.t
	desc = ostend_manifesto.3.d
	flavor = ostend_manifesto.3.f

	duration = 3

	option = {
		name = ostend_manifesto.3.a
		# Grant Cuba full sovereignty
		default_option = yes
		c:SPA = {
			remove_diplomatic_pact = { country = c:CUB type = colony }
		}
		change_relations = { country = c:CUB value = 50 }
	}

	option = {
		name = ostend_manifesto.3.b
		# Establish American protectorate
		c:SPA = {
			remove_diplomatic_pact = { country = c:CUB type = colony }
		}
		create_diplomatic_pact = { country = c:CUB type = protectorate }
	}

	option = {
		name = ostend_manifesto.3.c
		# Annex Cuba as a colony
		c:SPA = {
			remove_diplomatic_pact = { country = c:CUB type = colony }
		}
		create_diplomatic_pact = { country = c:CUB type = colony }
		create_incident = { country = c:CUB value = 10 }
	}
}

namespace = spanish_american_war

spanish_american_war.1 = {
	type = country_event
	placement = root

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	title = spanish_american_war.1.t
	desc = spanish_american_war.1.d
	flavor = spanish_american_war.1.f

	duration = 3

	immediate = {
		c:USA ?= { save_scope_as = usa_scope }
		c:SPA ?= { save_scope_as = spain_scope }
		random_state = {
			limit = {
				state_region = s:STATE_EASTERN_CUBA
			}
			create_building = {
				building = building_port
				level = 1
			}
		}		
	}

	trigger = {
		c:USA ?= this
		exists = c:SPA
	}

	option = {
		name = spanish_american_war.1.a
		default_option = yes
		if = {
			limit = { has_variable = saw_scripted_war_victory }
			remove_variable = saw_scripted_war_victory
		}
		if = {
			limit = {
				exists = c:CUB
				c:CUB = { is_subject_of = c:SPA }
			}
			c:CUB = {
				make_independent = yes
				create_treaty = {
					name = treaty_name_cuba_usa_agreement
					first_country = c:USA
					second_country = c:CUB

					is_draft = no
					binding_period = { years = 50 } 

					articles_to_create = {
						{
							article = treaty_port
							source_country = c:CUB
							target_country = c:USA
							inputs = {
								{ state = s:STATE_EASTERN_CUBA.region_state:CUB }
							}
						}
					}
				}
			}

		}
		if = {
			limit = {
				exists = c:PHI
				c:PHI = { is_subject_of = c:SPA }
			}
			transfer_subject = c:PHI
			if = {
				limit = {
					OR = {
						p:xA7F7E3 = { owner = c:PHI }
						p:x6E4206 = { owner = c:PHI }
					}
				}
				s:STATE_WEST_MICRONESIA = {
					set_owner_of_provinces = {
						country = c:USA
						provinces = { xA7F7E3 x6E4206 }
					}
				}
			}
		}
		if = {
			limit = {
				exists = c:PCO
				c:PCO = { is_subject_of = c:SPA }
			}
			annex = c:PCO
		}
	}
	option = {
		name = spanish_american_war.1.b
		default_option = yes
		create_incident = { country = c:CUB value = 10 }
		if = {
			limit = { has_variable = saw_scripted_war_victory }
			remove_variable = saw_scripted_war_victory
		}
		if = {
			limit = {
				exists = c:CUB
				c:CUB = { is_subject_of = c:SPA }
			}
			transfer_subject = c:CUB
			c:CUB = {
				create_treaty = {
					name = treaty_name_cuba_usa_agreement
					first_country = c:USA
					second_country = c:CUB

					is_draft = no
					binding_period = { years = 50 } 

					articles_to_create = {
						{
							article = treaty_port
							source_country = c:CUB
							target_country = c:USA
							inputs = {
								{ state = s:STATE_EASTERN_CUBA.region_state:CUB }
							}
						}
					}
				}
			}

		}
		if = {
			limit = {
				exists = c:PHI
				c:PHI = { is_subject_of = c:SPA }
			}
			transfer_subject = c:PHI
			if = {
				limit = {
					OR = {
						p:xA7F7E3 = { owner = c:PHI }
						p:x6E4206 = { owner = c:PHI }
					}
				}
				s:STATE_WEST_MICRONESIA = {
					set_owner_of_provinces = {
						country = c:USA
						provinces = { xA7F7E3 x6E4206 }
					}
				}
			}
		}
		if = {
			limit = {
				exists = c:PCO
				c:PCO = { is_subject_of = c:SPA }
			}
			annex = c:PCO
		}
	}
}
