  }

  hidden_effect = {
   chi_adm_henan_end_without_transfer = { COOLDOWN = 24 }
   c:CHI = { trigger_event = { id = chi_adm_henan.4 popup = yes } }
  }
  ai_chance = {
   base = 15
   modifier = { trigger = { OR = { var:chi_adm_autonomy >= 50 var:chi_adm_loyalty < 40 } } add = 45 }
   modifier = { trigger = { chi_adm_henan_gdp_share >= 0.30 } add = 40 }
  }
