namespace = tyr_crown_union

tyr_crown_union.1 = {
	type = country_event
	placement = root
	title = tyr_crown_union.1.t
	desc = tyr_crown_union.1.d
	flavor = tyr_crown_union.1.f

	duration = 3

	event_image = {
			video = "europenorthamerica_capitalists_meeting"
		}
	immediate = {
		if = {
			limit = { exists = c:SPA c:SPA ?= { has_variable = tyr_crown_union_winner } }
			c:SPA = { save_scope_as = tyr_crown_union_winner_scope }
		}
		if = {
			limit = { exists = c:ITA c:ITA ?= { has_variable = tyr_crown_union_winner } }
			c:ITA = { save_scope_as = tyr_crown_union_winner_scope }
		}
		ROOT = {
			save_scope_as = tyr_crown_union_loser_scope
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:tyr_crown_union_winner_scope }
			is_subject = yes
			scope:tyr_crown_union_winner_scope ?= { is_subject = yes }
		}
	}

	option = {
		name = tyr_crown_union.1.a
		default_option = yes
		ai_chance = { base = 80 }
		scope:tyr_crown_union_winner_scope = {
			create_diplomatic_pact = { country = scope:tyr_crown_union_loser_scope type = personal_union }
			
		}
	}

	option = {
		name = tyr_crown_union.1.b
		ai_chance = { base = 20 }
		set_variable = tyr_crown_union_rejected
		scope:tyr_crown_union_winner_scope = {
			set_variable = tyr_crown_union_war_unlocked
		}
	}
}

tyr_crown_union.2 = {
	type = country_event
	placement = root
	title = tyr_crown_union.2.t
	desc = tyr_crown_union.2.d
	flavor = tyr_crown_union.2.f

	duration = 3

	event_image = {
			video = "unspecific_signed_contract"
		}
	immediate = {
		if = {
			limit = { exists = c:SPA c:SPA ?= { has_variable = tyr_crown_union_loser } }
			c:SPA = { save_scope_as = tyr_crown_union_loser_scope }
		}
		if = {
			limit = { exists = c:ITA c:ITA ?= { has_variable = tyr_crown_union_loser } }
			c:ITA = { save_scope_as = tyr_crown_union_loser_scope }
		}
	}

	cancellation_trigger = {
		NOT = {
			exists = scope:tyr_crown_union_loser_scope
			scope:tyr_crown_union_loser_scope ?= {
				is_subject_of = root
				is_subject_type = subject_type_personal_union
			}
		}
	}

	option = {
		name = tyr_crown_union.2.a
		default_option = yes
		remove_variable = tyr_crown_union_winner
		remove_variable = tyr_crown_union_war_unlocked
		scope:tyr_crown_union_loser_scope = {
			remove_variable = tyr_crown_union_loser
			remove_variable = tyr_crown_union_rejected
		}
		change_tag = MED
		scope:tyr_crown_union_loser_scope = {
			every_primary_culture = {
				save_temporary_scope_as = tyr_crown_union_culture
				ROOT = { add_primary_culture = scope:tyr_crown_union_culture }
			}
		}
		annex_with_incorporation = scope:tyr_crown_union_loser_scope
	}

	option = {
		name = tyr_crown_union.2.a
		default_option = yes
		add_modifier = { name = tyr_crown_union_benefit }
		remove_variable = tyr_crown_union_winner
		remove_variable = tyr_crown_union_war_unlocked
		scope:tyr_crown_union_loser_scope = {
			remove_variable = tyr_crown_union_loser
			remove_variable = tyr_crown_union_rejected
		}
		scope:tyr_crown_union_loser_scope = {
			add_modifier = { name = tyr_crown_union_benefit }
		}
	}
}
