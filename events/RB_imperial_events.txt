#imperial fed formed
imperial_events.2 = {
	type = country_event

	title = imperial_events.2.t
	desc = imperial_events.2.d
	flavor = imperial_events.2.f


	duration = 3

	event_image = {
		video = "unspecific_world_fair"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/imperial_federation.dds"


	immediate = {
		set_global_variable = imperial_federation_formed
	}

	option = {
		name = imperial_events.2.a
		c:GBR = {
			every_country = {
				limit = {
					is_subject_of = c:GBR
					OR = {
						c:SAF ?= this
						country_has_primary_culture = cu:australian
						country_has_primary_culture = cu:anglo_canadian
						country_has_primary_culture = cu:franco_canadian
					}
				}
				save_temporary_scope_as = imperial_federation_annex_target
				c:GBR = {
					annex = scope:imperial_federation_annex_target
				}
			}
			add_primary_culture = cu:anglo_canadian
			add_primary_culture = cu:franco_canadian
			add_primary_culture = cu:scottish
			add_primary_culture = cu:scottish_gaelic
			add_primary_culture = cu:australian
			add_primary_culture = cu:irish
		}
		add_modifier = {
			name = imperial_federation_modifier
		}
		if = {
			limit = {
				exists = c:BIC
			}
			c:BIC = {
				add_modifier = imperial_jewel
			}
		}
		activate_law = law_type:law_imperial_federal_bureaucracy
	}
}
