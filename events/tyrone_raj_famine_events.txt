namespace = tyrone_raj_famine
tyrone_raj_famine.1 = {
	type = country_event
	placement = scope:famine_state
	title = tyrone_raj_famine.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					exists = scope:famine_source_region
				}
				desc = tyrone_raj_famine.spread.d
			}
			desc = tyrone_raj_famine.1.d
		}
	}
	flavor = tyrone_raj_famine.1.f
	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_fire.dds"
	duration = 3
	trigger = {
		exists = scope:famine_region
		scope:raj_scope = {
			has_variable = tyrone_unstable_raj_active
		}
		scope:famine_region = {
			has_variable = tyrone_raj_famine_pending
		}
	}
	cancellation_trigger = {
		scope:raj_scope = {
			NOT = {
				has_variable = tyrone_unstable_raj_active
			}
		}
	}
	option = {
		name = tyrone_raj_famine.1.a
		default_option = yes
		if = {
			limit = {
				OR = {
					this = scope:raj_scope
					AND = {
						exists = scope:raj_scope.overlord
						scope:raj_scope.overlord = this
					}
				}
				scope:raj_scope = {
					has_variable = tyrone_unstable_raj_active
				}
				scope:famine_region = {
					has_variable = tyrone_raj_famine_pending
					NOT = {
						has_variable = tyrone_raj_famine_active
					}
				}
			}
			scope:famine_region = {
				hidden_effect = {
					remove_variable = tyrone_raj_famine_pending
					set_variable = tyrone_raj_famine_active
				}
				tyrone_raj_famine_apply_harvest = yes
				hidden_effect = {
					remove_variable = tyrone_raj_famine_pending_duration
				}
			}
			add_contextless_journal_entry = je_tyrone_raj_famine_1
		}
	}
}

		months = 36
	}
	trigger = {
		tyrone_raj_famine_can_start_wave = yes

		if = {

				tyrone_raj_famine_can_start_wave = yes
			}
			save_scope_as = raj_scope

				name = tyrone_raj_famine_cooldown
				months = 36
			}
			remove_variable = tyrone_raj_famine_wait
			tyrone_raj_famine_start_wave = yes
		}
	}
}
