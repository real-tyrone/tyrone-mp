namespace = berlin_colonial_enforcement

berlin_colonial_enforcement.1 = {
	type = country_event
	placement = scope:berlin_disputed_colony
	title = berlin_colonial_enforcement.1.t
	desc = berlin_colonial_enforcement.1.d
	flavor = berlin_colonial_enforcement.1.f
	duration = 3
	event_image = { video = "unspecific_signed_contract" }
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
		has_global_variable = berlin_conference_ended
		has_variable = accepted_conf_var
		any_scope_state = { tyr_berlin_colonial_violation = yes }
	}

	immediate = {
		random_scope_state = {
			limit = { tyr_berlin_colonial_violation = yes }
			save_scope_as = berlin_disputed_colony
			state_region = {
				random_scope_state = {
					limit = { owner = { is_country_type = decentralized } }
					owner = { save_scope_as = berlin_colony_return_country }
				}
			}
		}
	}

	option = {
		name = berlin_colonial_enforcement.1.a
		default_option = yes
		ai_chance = { base = 100 }
		# Returning the entire colonial state removes the signatory's foothold and growth there.
		if = {
			limit = {
				exists = scope:berlin_colony_return_country
				exists = scope:berlin_disputed_colony
				scope:berlin_disputed_colony = { owner = root }
			}
			scope:berlin_disputed_colony = {
				set_state_owner = scope:berlin_colony_return_country
			}
		}
	}

	option = {
		name = berlin_colonial_enforcement.1.b
		ai_chance = { base = 0 }
		if = {
			limit = {
				exists = scope:berlin_disputed_colony
				scope:berlin_disputed_colony = { owner = root }
			}
			change_infamy = 10
			scope:berlin_disputed_colony = {
				# One penalty for retaining this colony, rather than 10 infamy every monthly pulse.
				set_variable = tyr_berlin_colony_infamy_paid
			}
		}
	}
}
