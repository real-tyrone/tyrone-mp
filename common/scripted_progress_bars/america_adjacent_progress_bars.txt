### Spanish-American War Tension Bar

je_spanish_american_war_tension_bar = {
	name = "je_spanish_american_war_tension_bar"
	desc = "je_spanish_american_war_tension_bar_desc"

	default_bad = yes
	is_inverted = yes

	monthly_progress = {
		# Natural decay — scales with current tension so high tension decays faster
		add = {
			desc = "je_saw_tension_decay_tt"
			value = owner.var:saw_tension_current_var
			multiply = -0.05
		}

		# Rivalry between USA and Spain
		if = {
			limit = {
				owner = {
					has_diplomatic_pact = {
						who = c:SPA
						type = rivalry
					}
				}
			}
			add = {
				desc = "je_saw_tension_rivalry_tt"
				value = 3
			}
		}

		# USA has embargoed Spain, or Spain has embargoed USA
		if = {
			limit = {
				OR = {
					owner = {
						has_diplomatic_pact = {
							who = c:SPA
							type = embargo
						}
					}
					c:SPA = {
						has_diplomatic_pact = {
							who = owner
							type = embargo
						}
					}
				}
			}
			add = {
				desc = "je_saw_tension_embargo_tt"
				value = 2
			}
		}

		# USA has expelled Spanish diplomats, or Spain has expelled American diplomats
		if = {
			limit = {
				OR = {
					owner = { has_diplomats_expelled = c:SPA }
					c:SPA = { has_diplomats_expelled = owner }
				}
			}
			add = {
				desc = "je_saw_tension_expel_tt"
				value = 4
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 100
}
