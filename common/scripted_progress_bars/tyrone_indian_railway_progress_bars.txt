# One canonical bar, shared by all involved countries. Negative approval is retained.
tyrone_railway_delhi_calcutta_construction_bar = {
	name = tyrone_railway_delhi_calcutta_bar_name
	desc = tyrone_railways_construction_bar_desc
	default_green = yes
	start_value = 1
	min_value = 1
	max_value = 100
	monthly_progress = {
		# The saved country survives tag and overlord changes.
		scope:tyrone_railways_india ?= {
			if = {
				limit = {
					has_global_variable = tyrone_railways_active_var
					trigger_if = {
						limit = { has_global_variable = tyrone_railway_delhi_calcutta_stage }
						global_var:tyrone_railway_delhi_calcutta_stage = 3
					}
					trigger_else = { always = no }
				}
				add = {
					desc = tyrone_railways_monthly_base
					value = 0.5
				}
				add = {
					desc = tyrone_railways_monthly_gdp
					value = gdp
					divide = 100000000
				}
				add = {
					desc = tyrone_railways_monthly_railways
					value = 0
					every_scope_state = {
						limit = { is_in_geographic_region = geographic_region_delhi_calcutta_line }
						every_scope_building = {
							limit = { is_building_type = building_railway }
							add = this.level
						}
					}
					divide = 20
				}
				add = {
					desc = tyrone_railways_monthly_landowners
					value = 0
					ig:ig_landowners ?= { add = ig_approval }
					divide = 10
				}
				add = {
					desc = tyrone_railways_monthly_industrialists
					value = 0
					ig:ig_industrialists ?= { add = ig_approval }
					divide = 10
				}
			}
		}
	}
}
