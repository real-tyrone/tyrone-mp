# ========== Italy Tyrone Decisions ==========

# Lateran Reconciliation: removes no_papal_recognition, trades it for a 5-year decaying concordat burden.
# Requires political_agitation tech. Visible only while no_papal_recognition is active.
ita_lateran_concordat_decision = {
	is_shown = {
		has_modifier = no_papal_recognition
	}

	possible = {
		is_at_war = no
		has_technology_researched = political_agitation
	}

	when_taken = {
		remove_modifier = no_papal_recognition
		add_modifier = {
			name = ita_lateran_concordat
			years = 5
			is_decaying = yes
		}
	}

	ai_chance = {
		value = 10
	}
}

ita_eritrea_purchase_decision = {
	is_shown = {
		c:ITA ?= this
		has_global_variable = berlin_conference_ended
		NOT = { has_global_variable = italy_eritrea_purchase_in_progress }
		NOT = { has_variable = italy_eritrea_purchase_decision_taken }
		NOT = { owns_entire_state_region = STATE_ERITREA }
		s:STATE_ERITREA = {
			any_scope_state = {
				NOT = { owner = ROOT }
				owner = {
					OR = {
						this ?= c:EGY
						is_country_type = recognized
					}
				}
			}
		}
	}

	possible = {
		gdp > 50000000
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		s:STATE_ERITREA = {
			any_scope_state = {
				NOT = { owner = ROOT }
				owner = {
					OR = {
						this ?= c:EGY
						is_country_type = recognized
					}
					is_at_war = no
					is_diplomatic_play_committed_participant = no
					NOT = { has_war_with = ROOT }
				}
			}
		}
	}

	when_taken = {
		set_global_variable = italy_eritrea_purchase_in_progress
		set_variable = italy_eritrea_purchase_buyer
		set_variable = italy_eritrea_purchase_decision_taken
		s:STATE_ERITREA = {
			random_scope_state = {
				limit = {
					NOT = { owner = ROOT }
					owner = {
						OR = {
							this ?= c:EGY
							is_country_type = recognized
						}
						is_at_war = no
						is_diplomatic_play_committed_participant = no
						NOT = { has_war_with = ROOT }
					}
				}
				owner = {
					trigger_event = {
						id = italy_eritrea_purchase.1
						days = 0
						popup = yes
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { gold_reserves > 15000000 }
			add = 10
		}
		if = {
			limit = {
				net_fixed_income > 100000
				gold_reserves > 0
			}
			add = 10
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			multiply = 0
		}
	}
}

# Lombardy-Venetia requests to be downgraded from crown land / personal union to protectorate.
# Visible when Austria is weakened: lost the Brothers' War, became the Dual Monarchy, or Hungary no longer exists.
# Fires an event on Austria â€” Austria chooses to accept or refuse.
lmv_request_protectorate_decision = {
	is_shown = {
		c:LMV ?= this
		is_subject_of = c:AUS
		NOT = { is_subject_type = subject_type_protectorate }
		NOT = { has_variable = lmv_protectorate_requested }
	}

	possible = {
		is_at_war = no
		c:AUS = { is_at_war = no }
		OR = {
			c:AUS = { has_modifier = brothers_war_defeat_aus }
			c:AUS = { has_government_type = gov_dual_monarchy }
			NOT = { exists = c:HUN }
		}
	}

	when_taken = {
		set_variable = lmv_protectorate_requested
		c:AUS = {
			trigger_event = {
				id = italyEventsTyroneLV.8
				popup = yes
			}
		}
	}

	ai_chance = {
		value = 5
	}
}

# ========== Sicilian Sulphur Monopoly decision ==========

# Possible when Sicily has a sulphur mine at level >= 20, or the company is already established.
# Adds the JE (company slot bonus) and establishes the company.
sic_sulphur_company_decision = {
	is_shown = {
		NOT = { has_variable = sic_sulphur_company_je_taken }
		is_player = yes
		OR = {
			c:SIC ?= this
			AND = {
				c:ITA ?= this
				has_variable = sic_ruler_of_italy
			}
		}
	}

	possible = {
		c:GBR ?= {
				relations:root >= 50
			}
		has_company = company_type:company_anglo_sicilian_sulphur_company
	}

	when_taken = {
		set_variable = sic_sulphur_company_je_taken
		add_journal_entry = { type = sic_sulphur_monopoly_je }
	}

	ai_chance = { value = 10 }
}
