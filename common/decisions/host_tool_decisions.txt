host_tool_claim_decision = {
	is_shown = {
		has_variable = host_tool_current_host
		host_tool_is_initial_game_rule_host = yes
		NOT = { has_global_variable = host_tool_initial_claimed }
		NOT = { has_journal_entry = je_host_tool }
	}

	possible = {
		has_variable = host_tool_current_host
		host_tool_is_initial_game_rule_host = yes
		NOT = { has_global_variable = host_tool_initial_claimed }
		NOT = { has_journal_entry = je_host_tool }
	}

	when_taken = {
		set_global_variable = host_tool_initial_claimed
		host_tool_grant_host_tool = yes
	}

	ai_chance = {
		value = 0
	}
}
