tyrone_revise_indian_charters = {
	is_shown = {
		any_subject_or_below = { is_india_tag_trigger = yes is_subject_of = root }
		is_ai = no
	}
	possible = {
		custom_tooltip = {
			text = tyrone_charter_cooldown_tt
			NOT = { has_variable = tyrone_charter_revision_cooldown }
		}
		hidden_trigger = { NOT = { has_variable = tyrone_charter_revision_pending } }
	}
	when_taken = {
		random_subject_or_below = {
			limit = { is_india_tag_trigger = yes is_subject_of = root }
			save_scope_as = charter_india
		}
		set_variable = { name = tyrone_charter_revision_pending days = 30 }
		trigger_event = { id = tyrone_chartered_obligations.1 popup = yes }
	}
	ai_chance = { value = 0 }
}
