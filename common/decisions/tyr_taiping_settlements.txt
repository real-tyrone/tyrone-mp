tyr_taiping_propose_independence = {
	is_shown = {
		chi_qing_administration_tag = yes
		has_variable = tyr_heavenly_defected
		exists = c:TPG
		NOT = { has_variable = tyr_taiping_settlement_independent }
	}
	possible = {
		custom_tooltip = { text = tyr_taiping_settlement_ready_tt tyr_taiping_settlement_eligible = yes }

}
