byz_claim_province_of_spania = {
	is_shown = {
		OR = {
			c:BYZ ?= ROOT
			c:ki_RMA ?= ROOT
		}
		NOT = { has_variable = byz_claimed_province_of_spania }
	}

	possible = {
		has_state_in_state_region = STATE_AL_RIF
	}

	when_taken = {
		set_variable = byz_claimed_province_of_spania
		s:STATE_MURCIA = { add_claim = ROOT }
		s:STATE_BALEARIC_ISLANDS = { add_claim = ROOT }
		s:STATE_UPPER_ANDALUSIA = { add_claim = ROOT }
		s:STATE_LOWER_ANDALUSIA = { add_claim = ROOT }
	}

	ai_chance = {
		value = 100
	}
}

byz_reestablish_exarchate_of_africa = {
	is_shown = {
		OR = {
			c:BYZ ?= ROOT
			c:ki_RMA ?= ROOT
		}
		NOT = { has_variable = byz_reestablished_exarchate_of_africa }
	}

	possible = {
		owns_entire_state_region = STATE_TUNISIA
		owns_entire_state_region = STATE_TRIPOLI
		owns_entire_state_region = STATE_LIBYA
		owns_entire_state_region = STATE_LIBYAN_DESERT
	}

	when_taken = {
		set_variable = byz_reestablished_exarchate_of_africa
		every_scope_state = {
			limit = {
				region = sr:region_north_africa
			}
			add_modifier = {
				name = byz_reclaimed_province_incorporation
				days = very_long_modifier_time
			}
		}
	}

	ai_chance = {
		value = 100
	}
}
