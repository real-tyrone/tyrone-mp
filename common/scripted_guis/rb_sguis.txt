colonial_office_conservative_emphasize_expansion = {
	scope = country
	is_shown = { has_journal_entry = je_african_conquest }
	is_valid = { NOT = { custom_tooltip = { text = has_office_cooldown_tt has_variable = colony_office_cooldown } } }
	ai_is_valid = { always = no }
	effect = {
		custom_tooltip = cant_use_office_5_years_tt

		set_variable = {
			name = colony_office_cooldown
			years = 5
		}
		add_modifier = {
			name = emphasized_colonial_expansion
			years = 5
		}
	}
}
