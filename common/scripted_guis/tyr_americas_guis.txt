tyr_am_tip_mission = {
	scope = country
	effect = {
		custom_tooltip = tyr_am_preview_title_mission
		custom_tooltip = tyr_am_preview_cost_mission
		if = {
			limit = { tyr_am_resolve_selected_target = yes }
			custom_tooltip = tyr_am_regional_mission_effect_tt
		}
		custom_tooltip = tyr_am_regional_cooldown_tt
		if = {
			limit = {
				NOT = {
					tyr_am_actor_valid = yes
				}
			}
			custom_tooltip = tyr_am_block_actor
		}
		if = {
			limit = {
				NOT = {
					tyr_am_active = yes
				}
			}
			custom_tooltip = tyr_am_block_phase
		}
		if = {
			limit = {
				has_global_variable = tyr_am_phase
				NOT = {
					global_var:tyr_am_phase = 1
				}
			}
			custom_tooltip = tyr_am_block_phase
		}
		if = {
			limit = {
				has_variable = tyr_am_cd_mission
			}
			custom_tooltip = tyr_am_block_action_cooldown
		}
		if = {
			limit = {
				NOT = {
					tyr_am_regional_quota = {
						CATEGORY = diplomatic
					}
				}
			}
			custom_tooltip = tyr_am_block_diplomatic_quota
		}
		if = {
			limit = {
				influence < tyr_am_diplomatic_amount
			}
			custom_tooltip = tyr_am_block_diplomatic_capacity
		}
		if = {
			limit = {
				tyr_am_resolve_selected_target = yes
			}
			tyr_am_ui_action_requirements = {
				NEGOTIATE = yes
				AID_ROLE = no
				PAID = no
				INVITE = no
			}
		}
		else = {
			custom_tooltip = tyr_am_block_selection
		}
		if = {
			limit = {
				tyr_am_resolve_selected_target = yes
				scope:tyr_am_target ?= {
					OR = {
						has_variable = tyr_am_mission_spa
						has_variable = tyr_am_mission_usa
					}
				}
			}
			custom_tooltip = tyr_am_block_regional_conflict
		}
	}
}

			tyr_am_ui_action_requirements = {
				NEGOTIATE = yes
				AID_ROLE = no
				PAID = no
				INVITE = yes
			}
		}
		else = {
			custom_tooltip = tyr_am_block_selection
		}
		if = {
			limit = {
				tyr_am_resolve_selected_target = yes
				always = no
			}
			custom_tooltip = tyr_am_block_regional_conflict
		}
	}
}
