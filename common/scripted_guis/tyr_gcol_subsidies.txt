tyr_gcol_maf_building_coal_mine_on = {
	scope = country

	is_valid = {
		tyr_gcol_has_maf = yes
		c:MAF ?= {
			has_variable = tyr_gcol_subsidize_building_coal_mine
		}
	}
}

tyr_gcol_maf_building_coal_mine = {
	scope = country

	ai_is_valid = {
		always = no
	}

	is_valid = {
		tyr_gcol_has_maf = yes
	}

	effect = {
		if = {
			limit = {
				tyr_gcol_has_maf = yes
			}
			c:MAF = {
				if = {
					limit = {
						has_variable = tyr_gcol_subsidize_building_coal_mine
					}
					remove_variable = tyr_gcol_subsidize_building_coal_mine
					every_scope_building = {
						limit = {
							is_building_type = building_coal_mine
						}
						set_subsidized = no
					}
				}
				else = {
					set_variable = tyr_gcol_subsidize_building_coal_mine
					every_scope_building = {
						limit = {
							is_building_type = building_coal_mine
						}
						set_subsidized = yes
					}
				}
			}
		}
	}
}
