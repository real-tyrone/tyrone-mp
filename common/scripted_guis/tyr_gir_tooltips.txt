tyr_gir_industrial_states_tooltip = {
	scope = country

	effect = {
		if = {
			limit = {
				has_variable_list = tyr_gir_industrial_states
			}
			every_in_list = {
				variable = tyr_gir_industrial_states
				limit = {
					owner = root
					tyr_gir_zone_state = yes
				}
				custom_tooltip = tyr_gir_state_tooltip_line
			}
		}
		else = {
			every_scope_state = {
				limit = {
					tyr_gir_zone_state = yes
				}
				custom_tooltip = tyr_gir_state_tooltip_line
			}
		}
	}
}
