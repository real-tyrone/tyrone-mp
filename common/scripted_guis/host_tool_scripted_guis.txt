host_tool_refresh_gui_lists = {
	scope = country

	is_shown = {
		host_tool_is_current_host = yes
	}

	effect = {
		host_tool_refresh_runtime_lists = yes
	}
}

host_tool_player_country_list = {
	scope = country

	is_shown = {
		host_tool_is_current_host = yes
	}

	effect = {
		host_tool_refresh_runtime_lists = yes
		if = {
			limit = {
				has_variable_list = host_tool_player_countries
			}
			every_in_list = {
				variable = host_tool_player_countries
				custom_tooltip_no_bullet = host_tool_player_country_list_entry
			}
		}
		else = {
			custom_tooltip_no_bullet = host_tool_empty_player_country_list
		}
	}
}
