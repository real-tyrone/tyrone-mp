ai_strategy_tyr_scandinavian_homelands = {
    icon = "gfx/interface/icons/ai_strategy_icons/unification.dds"
    type = diplomatic

    possible = { tyr_scandinavian_return_state_available = yes }
    weight = { value = 10000 }

    state_value = {
        if = {
            limit = {
                scope:target_state ?= { tyr_scandinavian_return_state_target = yes }
                can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
            }
            add = { value = 150 desc = "STATE_VALUE_OWN_HOMELAND" }
        }
    }

    wargoal_scores = {
        return_state = {
            if = {
                limit = { scope:target_state ?= { tyr_scandinavian_return_state_target = yes } }
                add = 150
            }
        }
    }

    secret_goal_scores = {
        conquer = {
            trigger = {
                scope:target_country ?= {
                    any_scope_state = { tyr_scandinavian_return_state_target = yes }
                }
                can_reach_target_country_to_conquer = { TARGET_COUNTRY = scope:target_country }
            }
            score = { value = 300 }
        }
    }
}
