tyrone_sc_back_experts = {
	name = tyrone_sc_back_experts_name
	desc = tyrone_sc_back_experts_desc
	visible = { scope:csr_india ?= this }
	possible = {
		hidden_trigger = { scope:csr_india ?= this }
		custom_tooltip = { text = tyrone_sc_intervention_ready_tt NOT = { has_variable = tyrone_sc_intervention_cooldown } }
	}
	cooldown = { days = tyrone_sc_year }
	ai_chance = { value = 10 }
	effect = {
		if = {
			limit = { scope:csr_india ?= this NOT = { has_variable = tyrone_sc_intervention_cooldown } }
			custom_tooltip = {
				text = tyrone_sc_experts_ten_tt
				scope:journal_entry = { add_progress = { name = tyrone_sc_settlement_bar value = 10 } }
			}
			add_modifier = { name = tyrone_sc_back_experts_cost days = tyrone_sc_political_duration }
			hidden_effect = { set_variable = { name = tyrone_sc_intervention_cooldown days = tyrone_sc_year } }
		}
	}
}
