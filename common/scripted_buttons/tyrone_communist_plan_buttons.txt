tyrone_cp_choose_heavy = {
	name = tyrone_cp_choose_heavy
	desc = tyrone_cp_choose_heavy_desc
	visible = {
		has_variable = tyrone_cp_started
		NOT = {
			has_variable = tyrone_cp_heavy_done
		}
	}

		custom_tooltip = {

	}
