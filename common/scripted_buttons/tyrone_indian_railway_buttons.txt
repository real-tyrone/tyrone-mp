# Country scope: either India or its current overlord may initiate the shared survey.
tyrone_railway_delhi_calcutta_survey = {
	name = tyrone_railway_delhi_calcutta_survey_button
	desc = tyrone_railway_delhi_calcutta_survey_button_desc
	visible = { tyrone_railways_is_actor = yes }
	possible = {
		hidden_trigger = {
			tyrone_railways_is_actor = yes
			has_global_variable = tyrone_railways_active_var
			trigger_if = {
				limit = { has_global_variable = tyrone_railway_delhi_calcutta_stage }
				global_var:tyrone_railway_delhi_calcutta_stage = 2
			}
			trigger_else = { always = no }
		}
		custom_tooltip = {
			text = tyrone_railways_survey_available_tt
			NOT = { has_global_variable = tyrone_railway_delhi_calcutta_survey_started }
		}
	}
	ai_chance = { value = 10 }
	effect = {
		# Recheck the lock inside the effect to prevent two payments from queued clicks.
		if = {
			limit = {
				tyrone_railways_is_actor = yes
				has_global_variable = tyrone_railways_active_var
				trigger_if = {
					limit = { has_global_variable = tyrone_railway_delhi_calcutta_stage }
					global_var:tyrone_railway_delhi_calcutta_stage = 2
				}
				trigger_else = { always = no }
				NOT = { has_global_variable = tyrone_railway_delhi_calcutta_survey_started }
			}
			hidden_effect = {
				set_global_variable = { name = tyrone_railway_delhi_calcutta_survey_payer value = this }
			}
			set_global_variable = tyrone_railway_delhi_calcutta_survey_started
			add_modifier = {
				name = tyrone_railway_delhi_calcutta_survey_cost
				days = very_short_modifier_time
			}
		}
	}
}
