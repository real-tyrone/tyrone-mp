tyrone_nim_pakistan_negotiate = {
name = tyrone_nim_pakistan_negotiate_name
    desc = tyrone_nim_pakistan_negotiate_desc
    visible = { tyrone_nim_is_actor = yes }
    possible = { 
        custom_tooltip = { text = tyrone_nim_region_once_tt has_variable = tyrone_nim_pakistan_open NOT = { has_variable = tyrone_nim_league_negotiate_used } }
        custom_tooltip = { text = tyrone_nim_league_states_radicals_tt NOT = { any_state = { OR = { tyrone_nim_is_pakistan_state = yes tyrone_nim_is_bengal_state = yes } owner = { OR = { scope:nim_india ?= this is_subject_of = scope:nim_india } } radical_fraction = { value >= 0.10 } } } } }
    ai_chance = { value = 20 }
    effect = {
        if = {
            limit = { has_variable = tyrone_nim_pakistan_open
        custom_tooltip = { text = tyrone_nim_region_once_tt NOT = { has_variable = tyrone_nim_league_negotiate_used } }
        custom_tooltip = { text = tyrone_nim_league_states_radicals_tt NOT = { any_state = { OR = { tyrone_nim_is_pakistan_state = yes tyrone_nim_is_bengal_state = yes } owner = { OR = { scope:nim_india ?= this is_subject_of = scope:nim_india } } radical_fraction = { value >= 0.10 } } } } }
            custom_tooltip = { text = tyrone_nim_region_pressure_down_ten_tt tyrone_nim_shift_pressure = { VALUE = -10 } }
                tyrone_nim_region_states = { REGION = league EFFECT = "add_modifier = { name = tyrone_nim_regional_concessions days = tyrone_nim_ten_years }" }
            hidden_effect = { set_variable = tyrone_nim_league_negotiate_used }
        }
    }
}

    ai_chance = { value = 5 }
    effect = {

            custom_tooltip = { text = tyrone_nim_region_repression_tt tyrone_nim_shift_pressure = { VALUE = 15 } }
                hidden_effect = {

                scope:nim_india = { random_political_movement = { limit = { is_political_movement_type = movement_muslim_nationalist } add_modifier = { name = tyrone_nim_movement_repression days = tyrone_nim_two_years } } }
                tyrone_nim_region_states = { REGION = league EFFECT = "add_radicals_in_state = { value = 0.05 }" }
            hidden_effect = { set_variable = tyrone_nim_league_repress_used }
        }
    }
}
