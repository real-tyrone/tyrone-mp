tyr_holy_alliance_withdraw = {
    name = tyr_holy_alliance_withdraw
    desc = tyr_holy_alliance_withdraw_desc
    visible = { is_involved_in_journal_entry = je_tyr_holy_alliance }
    possible = {
        is_ai = no
        is_involved_in_journal_entry = je_tyr_holy_alliance
        NOT = { exists = c:GER }
        custom_tooltip = {
            text = tyr_holy_alliance_has_founding_treaty_tt
            OR = {
                AND = {
                    exists = scope:tyr_holy_treaty_pru_aus
                    scope:tyr_holy_treaty_pru_aus = {
                        binds = ROOT
                        any_scope_article = { has_type = defensive_pact }
                    }
                }
                AND = {
                    exists = scope:tyr_holy_treaty_pru_rus
                    scope:tyr_holy_treaty_pru_rus = {
                        binds = ROOT
                        any_scope_article = { has_type = defensive_pact }
                    }
                }
                AND = {
                    exists = scope:tyr_holy_treaty_rus_aus
                    scope:tyr_holy_treaty_rus_aus = {
                        binds = ROOT
                        any_scope_article = { has_type = defensive_pact }
                    }
                }
            }
        }
    }
    effect = {
        change_infamy = 25
        if = {
            limit = {
                exists = scope:tyr_holy_treaty_pru_aus
                scope:tyr_holy_treaty_pru_aus = { binds = ROOT }
            }
            scope:tyr_holy_treaty_pru_aus = {
                withdraw = { country = ROOT apply_break_penalties = no }
            }
        }
        if = {
            limit = {
                exists = scope:tyr_holy_treaty_pru_rus
                scope:tyr_holy_treaty_pru_rus = { binds = ROOT }
            }
            scope:tyr_holy_treaty_pru_rus = {
                withdraw = { country = ROOT apply_break_penalties = no }
            }
        }
        if = {
            limit = {
                exists = scope:tyr_holy_treaty_rus_aus
                scope:tyr_holy_treaty_rus_aus = { binds = ROOT }
            }
            scope:tyr_holy_treaty_rus_aus = {
                withdraw = { country = ROOT apply_break_penalties = no }
            }
        }
        scope:journal_entry = { remove_involved_country = ROOT }
    }
    ai_chance = { value = 0 }
}

tyr_holy_alliance_brothers_war = {
    name = tyr_holy_alliance_brothers_war
    desc = tyr_holy_alliance_brothers_war_desc
    visible = {
        is_involved_in_journal_entry = je_tyr_holy_alliance
        OR = { scope:tyr_holy_prussia ?= this scope:tyr_holy_austria ?= this }
    }
    possible = {
        custom_tooltip = { text = tyrone_tooltip_german_ambition_idle NOT = { has_variable = tyr_ga_active } }

        is_involved_in_journal_entry = je_tyr_holy_alliance
        NOT = { exists = c:GER }
        has_technology_researched = nationalism
        is_unification_candidate = GER
        is_at_war = no
        is_diplomatic_play_committed_participant = no
        trigger_if = {
            limit = { is_ai = yes }
            scope:tyr_holy_prussia ?= this
            army_power_projection > scope:tyr_holy_austria.tyr_holy_realm_army_power
        }
        trigger_if = {
            limit = { scope:tyr_holy_prussia ?= this }
            tyr_holy_brothers_war_target_ready = { TARGET = scope:tyr_holy_austria }
        }
        trigger_else_if = {
            limit = { scope:tyr_holy_austria ?= this }
            tyr_holy_brothers_war_target_ready = { TARGET = scope:tyr_holy_prussia }
        }
        trigger_else = { always = no }
    }
    effect = {
        if = {
            limit = { scope:tyr_holy_prussia ?= this }
            scope:tyr_holy_austria = { save_scope_as = tyr_holy_brothers_war_target }
        }
        else = {
            scope:tyr_holy_prussia = { save_scope_as = tyr_holy_brothers_war_target }
        }
        # Break only bilateral defensive treaties, without the voluntary-exit fee.
        every_scope_treaty = {
            limit = {
                binds = scope:tyr_holy_brothers_war_target
                any_scope_article = { has_type = defensive_pact }
            }
            withdraw = { country = ROOT apply_break_penalties = no }
        }
        create_diplomatic_play = {
            type = dp_tyr_holy_brothers_war
            target_country = scope:tyr_holy_brothers_war_target
        }
        random_diplomatic_play = {
            limit = {
                is_diplomatic_play_type = dp_tyr_holy_brothers_war
                initiator = ROOT
                target = scope:tyr_holy_brothers_war_target
            }
            save_scope_as = tyr_holy_brothers_war_play
        }
        every_country = {
            limit = {
                is_supporting_unification_candidate = { who = ROOT country_formation = GER }
                NOT = { this = ROOT }
            }
            save_temporary_scope_as = tyr_holy_initiator_supporter
            scope:tyr_holy_brothers_war_play ?= {
                add_initiator_backers = { scope:tyr_holy_initiator_supporter }
            }
        }
        every_country = {
            limit = {
                is_supporting_unification_candidate = {
                    who = scope:tyr_holy_brothers_war_target
                    country_formation = GER
                }
                NOT = { this = scope:tyr_holy_brothers_war_target }
            }
            save_temporary_scope_as = tyr_holy_target_supporter
            scope:tyr_holy_brothers_war_play ?= {
                add_target_backers = { scope:tyr_holy_target_supporter }
            }
        }
        if = {
            limit = {
                scope:tyr_holy_prussia ?= this
                exists = scope:tyr_holy_brothers_war_play
                c:LMV ?= { is_subject_of = scope:tyr_holy_brothers_war_target }
            }
            save_scope_as = tyr_holy_italian_offer_prussia
            if = {
                limit = { c:ITA ?= { tyr_holy_italian_offer_available = yes } }
                c:ITA = { trigger_event = { id = tyr_holy_alliance.1 popup = yes } }
            }
            else_if = {
                limit = { c:SAR ?= { tyr_holy_italian_offer_available = yes } }
                c:SAR = { trigger_event = { id = tyr_holy_alliance.1 popup = yes } }
            }
        }
    }
    ai_chance = { value = 25 }
}
