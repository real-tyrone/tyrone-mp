# Country scope: India or its current overlord may fund a survey; only the initiating country pays.
tyrone_mining_rajputana_survey_1 = {
	name = tyrone_mining_rajputana_survey_1_button
	desc = tyrone_mining_rajputana_survey_1_button_desc
	visible = { tyrone_promise_is_actor = yes }
	possible = {
		hidden_trigger = {
			tyrone_promise_is_actor = yes
			has_global_variable = tyrone_promise_active_var
			has_global_variable = tyrone_mining_rajputana_active
		}
		custom_tooltip = {
			text = tyrone_mining_survey_available_tt
			NOT = { has_global_variable = tyrone_mining_rajputana_survey_1_started }
		}
	}
	ai_chance = { value = 10 }
	effect = {
		# The same lock is checked at execution, including simultaneous multiplayer clicks.
		if = {
			limit = {
				tyrone_promise_is_actor = yes
				has_global_variable = tyrone_promise_active_var
				has_global_variable = tyrone_mining_rajputana_active
				NOT = { has_global_variable = tyrone_mining_rajputana_survey_1_started }
			}
			hidden_effect = {
				set_global_variable = { name = tyrone_mining_rajputana_survey_1_payer value = this }
			}
			set_global_variable = tyrone_mining_rajputana_survey_1_started
			add_modifier = { name = tyrone_mining_rajputana_survey_cost days = very_short_modifier_time }
		}
	}
}
