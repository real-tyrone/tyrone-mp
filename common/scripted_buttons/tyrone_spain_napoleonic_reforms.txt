spain_napoleonic_schooling_button = {
	name = spain_napoleonic_schooling_button
	desc = spain_napoleonic_schooling_button_desc
	visible = { always = yes }
	possible = {
		has_journal_entry = je_spain_napoleonic_domestic_reforms
		custom_tooltip = {
			text = spain_napoleonic_schooling_not_confirmed_tt
			NOT = { has_variable = spain_napoleonic_schooling_confirmed }
		}
		institution_investment_level = { institution = institution_schools value >= 3 }
	}
	effect = {
		hidden_effect = { set_variable = spain_napoleonic_schooling_confirmed }
		trigger_event = { id = spain_napoleonic_reforms.1 popup = yes }
		show_as_tooltip = {
			add_modifier = {
				name = spain_napoleonic_schooling_modifier
				years = 20
				is_decaying = yes
			}
		}
	}
	ai_chance = { value = 100 }
}
