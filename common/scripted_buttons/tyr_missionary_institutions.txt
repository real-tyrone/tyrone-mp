tyr_missionary_schools_button = {
 name = tyr_missionary_schools_button
 desc = tyr_missionary_schools_button_desc
 visible = { chi_adm_target_valid = yes }
 possible = {
  custom_tooltip = { text = tyr_missionary_pledge_required_tt has_variable = chi_adm_religious_taiping_pledge }
  custom_tooltip = { text = tyr_missionary_permission_required_tt has_variable = chi_adm_religious_missions_permitted }
  custom_tooltip = { text = tyr_missionary_schools_state_tt any_scope_state = { has_variable = open_to_missionaries_state NOT = { has_modifier = modifier_chi_missionary_schools } } }
  custom_tooltip = { text = tyr_missionary_schools_ready_tt NOT = { has_variable = tyr_missionary_schools_cooldown } }
 }
 effect = {
  custom_tooltip = tyr_missionary_schools_effect_tt
  hidden_effect = {
   if = {
    limit = {
     has_variable = chi_adm_religious_taiping_pledge
     has_variable = chi_adm_religious_missions_permitted
     NOT = { has_variable = tyr_missionary_schools_cooldown }
     any_scope_state = { has_variable = open_to_missionaries_state NOT = { has_modifier = modifier_chi_missionary_schools } }
    }
    set_variable = { name = tyr_missionary_schools_cooldown months = 12 }
    random_scope_state = {
     limit = { has_variable = open_to_missionaries_state NOT = { has_modifier = modifier_chi_missionary_schools } }
     save_scope_as = missionary_schools_state
    }
    c:CHI ?= { change_variable = { name = missionary_works_counter add = 1 } }
    trigger_event = { id = CHI_missionaries.3 popup = yes }
   }
  }
 }
 ai_chance = { value = 0 }
}
