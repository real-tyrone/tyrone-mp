tyrone_jewel_public_works = {
	name = tyrone_jewel_public_works_button
	desc = tyrone_jewel_public_works_button_desc
	visible = {
		tyrone_jewel_is_actor = yes
	}
	possible = {
		hidden_trigger = {
			tyrone_jewel_is_actor = yes
		}
		custom_tooltip = {
			text = tyrone_jewel_personal_ready_tt
			NOT = { has_variable = tyrone_jewel_personal_cooldown }
		}
		scope:csr_india = {
			custom_tooltip = {
				text = tyrone_jewel_initiative_ready_tt
				NOT = { has_variable = tyrone_jewel_public_works_cooldown }
			}
			custom_tooltip = {
				text = tyrone_jewel_negotiation_ready_tt
				NOT = { has_variable = tyrone_jewel_pending }
			}
		}
	}
	cooldown = { days = normal_modifier_time }
	ai_chance = { value = 5 }
	effect = {
		tyrone_jewel_begin_negotiation = {
			KEY = public_works
			NUMBER = 1
			EVENT = tyrone_jewel.10
		}
	}
}
