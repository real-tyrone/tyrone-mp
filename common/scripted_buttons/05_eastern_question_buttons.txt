je_eastern_question_russia_free_balkans_button = {
	name = "je_eastern_question_russia_free_balkans_button"
	desc = "je_eastern_question_russia_free_balkans_button_desc"

	visible = {
		c:RUS ?= THIS
		has_journal_entry = je_eastern_question_russia
	}

	possible = {
		hidden_trigger = {
			exists = c:TUR
			c:TUR = {
				is_involved_in_journal_entry = je_great_eastern_crisis
			}
		}
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		c:TUR = {
			army_power_projection < ROOT.army_power_projection
		}
		NOT = { has_truce_with = c:TUR }
	}

	effect = {
		create_diplomatic_play = {
			name = free_the_balkans
			type = dp_free_the_balkans
			target_country = c:TUR
		}
		hidden_effect = {
			if = {
				limit = {
					exists = c:AUS
					c:AUS ?= {
						NOT = { owns_entire_state_region = STATE_BOSNIA }
						NOT = {
							any_subject_or_below = {
								owns_entire_state_region = STATE_BOSNIA
							}
						}
					}
					c:TUR ?= {
						OR = {
							owns_entire_state_region = STATE_BOSNIA
							any_subject_or_below = {
								owns_entire_state_region = STATE_BOSNIA
							}
						}
					}
				}
				c:AUS = {
					trigger_event = { id = eastern_question_austria.3 popup = yes }
				}
			}
			else_if = {
				limit = {
					NOT = { exists = c:AUS }
					exists = c:KUK
					c:KUK ?= {
						NOT = { owns_entire_state_region = STATE_BOSNIA }
						NOT = {
							any_subject_or_below = {
								owns_entire_state_region = STATE_BOSNIA
							}
						}
					}
					c:TUR ?= {
						OR = {
							owns_entire_state_region = STATE_BOSNIA
							any_subject_or_below = {
								owns_entire_state_region = STATE_BOSNIA
							}
						}
					}
				}
				c:KUK = {
					trigger_event = { id = eastern_question_austria.3 popup = yes }
				}
			}
		}
	}

	ai_chance = {
		value = 100
	}
}
