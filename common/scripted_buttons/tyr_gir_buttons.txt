# Country scope; national policy modifiers belong to the main Journal Entry.
tyr_gir_button_1_1 = {
	name = tyr_gir_button_1_1
	desc = tyr_gir_button_1_1_desc

	visible = {
		has_journal_entry = je_tyr_gir_main
		has_variable = tyr_gir_family
		var:tyr_gir_family ?= 1
	}

	possible = {
		custom_tooltip = {
			text = tyr_gir_req_active_tt
			has_variable = tyr_gir_active
		}
		custom_tooltip = {
			text = tyr_gir_req_year_tt
			NOT = {
				has_variable = tyr_gir_action_cooldown
			}
		}
		custom_tooltip = {
			text = tyr_gir_req_interventions_tt
			var:tyr_gir_interventions < 3
		}
		custom_tooltip = {
			text = tyr_gir_req_unused_tt
			NOT = {
				has_variable = tyr_gir_used_1
			}
		}
		custom_tooltip = {
			text = tyr_gir_req_early_tt
			has_variable = tyr_gir_early_window
		}
		custom_tooltip = {
			text = tyr_gir_req_unchanged_tt
			var:tyr_gir_difficulty = 0
		}
	}

	effect = {
		show_as_tooltip = {
			je:je_tyr_gir_main ?= {
				add_modifier = {
					name = tyr_gir_expense
					multiplier = root.tyr_gir_next_expense
				}
			}
		}
		hidden_effect = {
			set_variable = tyr_gir_used_1
			change_variable = {
				name = tyr_gir_interventions
				add = 1
			}
			set_variable = {
				name = tyr_gir_action_cooldown
				days = 365
			}
			set_variable = {
				name = tyr_gir_difficulty
				value = -1
			}
			set_variable = {
				name = tyr_gir_increment
				value = tyr_gir_easy_increment
			}
			set_variable = {
				name = tyr_gir_target
				value = tyr_gir_new_target
			}
			set_variable = {
				name = tyr_gir_reward
				value = 1
			}
			tyr_gir_add_expense = yes
		}
		custom_tooltip = tyr_gir_button_1_1_effect_tt
	}

	ai_chance = {
		base = 0
	}
}
