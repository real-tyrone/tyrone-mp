# Deliberate replication of Victorian Century's post-Punjab annexation action.
tyrone_subcontinent_punjab_annexation = {
	name = tyrone_subcontinent_punjab_annexation
	desc = tyrone_subcontinent_punjab_annexation_desc
	visible = { tyrone_subcontinent_participant = yes }
	possible = {
		has_global_variable = tyrone_subcontinent_active_var
		tyrone_subcontinent_empire_intact = yes
		NOT = { exists = c:PAN }
		NOT = { has_global_variable = tyrone_subcontinent_punjab_annexed_var }
		OR = { exists = c:BIC exists = c:BHR }
		OR = { exists = c:GAR exists = c:PTA exists = c:BHW }
	}
	ai_chance = { value = 100 }
	effect = {
		set_global_variable = tyrone_subcontinent_punjab_annexed_var
		if = {
			limit = { exists = c:BIC }
			c:BIC = {
			if = {
				limit = { exists = c:GAR }
				annex = c:GAR
			}
			if = {
				limit = { exists = c:PTA }
				annex = c:PTA
			}
			if = {
				limit = { exists = c:BHW }
				annex = c:BHW
			}
			}
		}
		else_if = {
			limit = { exists = c:BHR }
			c:BHR = {
			if = {
				limit = { exists = c:GAR }
				annex = c:GAR
			}
			if = {
				limit = { exists = c:PTA }
				annex = c:PTA
			}
			if = {
				limit = { exists = c:BHW }
				annex = c:BHW
			}
			}
		}
	}
}

# Country scope: India launches the play and bears its mobilization costs.
tyrone_subcontinent_attack_khalsa = {
	name = tyrone_subcontinent_attack_khalsa
	desc = tyrone_subcontinent_attack_khalsa_desc
	visible = { always = yes }
	possible = {
		exists = scope:tyrone_subcontinent_india
		scope:tyrone_subcontinent_india ?= {
			exists = c:PAN
			is_at_war = no
			NOT = { has_truce_with = c:PAN }
			NOT = { is_at_war_with = c:PAN }
		}
		c:PAN ?= { is_subject = no }
	}
	ai_chance = { value = 10 }
	effect = {
		scope:tyrone_subcontinent_india ?= {
			create_diplomatic_play = { target_country = c:PAN type = dp_annex_war }
			add_modifier = { name = tyrone_subcontinent_punjab_campaign days = very_short_modifier_time }
		}
	}
}

tyrone_subcontinent_attack_burma = {
	name = tyrone_subcontinent_attack_burma
	desc = tyrone_subcontinent_attack_burma_desc
	visible = { always = yes }
	possible = {
		exists = scope:tyrone_subcontinent_india
		OR = {
			scope:tyrone_subcontinent_india ?= this
			scope:tyrone_subcontinent_india.overlord ?= this
		}
		scope:tyrone_subcontinent_india ?= {
			has_technology_researched = rifling
			has_technology_researched = logistics
			is_at_war = no
			exists = c:BUR
		}
		c:BUR ?= { is_subject = no }
	}
	ai_chance = { value = 10 }
	effect = {
		change_infamy = 15
		scope:tyrone_subcontinent_india ?= {
			if = { limit = { has_truce_with = c:BUR } end_truce = c:BUR }
			create_diplomatic_play = { target_country = c:BUR type = dp_make_protectorate }
			add_modifier = { name = tyrone_subcontinent_punjab_campaign days = very_short_modifier_time }
		}
	}
}
