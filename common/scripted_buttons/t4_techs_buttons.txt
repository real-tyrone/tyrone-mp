lebensraum_base_slavery_button = {
	name = lebensraum_base_slavery_button
	desc = lebensraum_base_slavery_button_desc

	visible = {
		has_journal_entry = je_lebensraum_slavery
	}

	cooldown = { days = very_short_modifier_time }

	possible = {
		NOT = {
			je:je_lebensraum_slavery = {
					has_modifier = lebensraum_base_slavery
				}
			}
	}

	effect = {
		je:je_lebensraum_slavery = {
				add_modifier = lebensraum_base_slavery
			}
		je:je_lebensraum_slavery = {
				remove_modifier = lebensraum_expanded_slavery
			}
		add_radicals = {
			value = medium_radicals
			strata = upper
		}
	}

	ai_chance = {
		value = 50
	}
}
