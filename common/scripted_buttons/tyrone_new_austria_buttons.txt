modernize_government_button = {
	# Presentation metadata for the German Ambition hemicycle.
	tags = { tyr_ga_gui_government }
	short_desc = tyr_ga_gui_government
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	name = modernize_government_button
	desc = modernize_government_button_desc

	visible = { always = yes }

	possible = {
		# The old completion flag could be set by a failed application; allow one verified repair.
		hidden_trigger = {
			NOT = { has_variable = tyr_ga_modernization_verified }
			NOT = { active_law:lawgroup_distribution_of_power = { has_amendment = amendment_type:amendment_geheime_staatskonferenz_modernized } }
		}
		OR = {
			has_law = law_type:law_autocracy
			has_law = law_type:law_neo_absolutism
		}
		NOR = {
			has_law_or_variant = law_type:law_hereditary_bureaucrats
			has_law_or_variant = law_type:law_land_based_taxation
			has_law_or_variant = law_type:law_serfdom
		}
	}

	selected = {
		OR = {
			has_variable = tyr_ga_modernization_verified
			active_law:lawgroup_distribution_of_power = { has_amendment = amendment_type:amendment_geheime_staatskonferenz_modernized }
		}
	}

	effect = {
		active_law:lawgroup_distribution_of_power ?= { # Autocracy
			every_scope_amendment = {
				remove_amendment = yes
			}
		}
		active_law:lawgroup_distribution_of_power ?= { # Autocracy
			add_amendment = {
				type = amendment_geheime_staatskonferenz_modernized
				sponsor = ROOT.ig:ig_landowners
				cooldown = 120
			}
		}
		# Only consume the action after the amendment is actually attached.
		if = {
			limit = { active_law:lawgroup_distribution_of_power = { has_amendment = amendment_type:amendment_geheime_staatskonferenz_modernized } }
			hidden_effect = {
				set_variable = modernize_government_var
				set_variable = tyr_ga_modernization_verified
			}
		}
	}
}

invite_south_germany_button = {
	# Presentation metadata for the German Ambition hemicycle.
	tags = { tyr_ga_gui_diplomacy }
	short_desc = tyr_ga_gui_diplomacy
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	name = invite_south_germany_button
	desc = invite_south_germany_button_desc

	visible = { always = yes }

	possible = {
		hidden_trigger = { NOT = { has_variable = invited_south_germany_button_var } }
		c:BAD = {
			relations:root >= relations_threshold:amicable
		}
		c:WUR = {
			relations:root >= relations_threshold:amicable
		}
		c:BAV = {
			relations:root >= relations_threshold:amicable
		}
	}

	selected = { has_variable = invited_south_germany_button_var }

	effect = {
		every_country = {
			limit = {
				capital = {
					region = sr:region_central_europe
				}
				country_has_primary_culture = cu:south_german
				NOT = {
					OR = {
						c:SWI ?= this
						c:HOH ?= this
						c:AUS ?= this
					}
				}
			}
			c:AUS.power_bloc ?= {
				add_leverage = { target = prev value = 1500 }
			}
		}
		hidden_effect = { set_variable = invited_south_germany_button_var }
	}
}

unify_germany_button = {
	name = unify_germany_button
	desc = unify_germany_button_desc

	visible = {
		NOT = {
			has_variable = unify_germany_button_var
		}
		OR = {
			exists = c:PRU
			exists = c:NGF
		}
	}

	possible = {
		has_technology_researched = pan-nationalism
		country_rank >= rank_value:great_power
		c:SGF ?= this
	}

	selected = {
	}

	effect = {
		if = {
			limit = {
				exists = c:PRU
			}
			create_diplomatic_play = {
				target_state = s:STATE_BRANDENBURG.region_state:PRU
				type = dp_unify_germany
				add_war_goal = {
					holder = root
					target_country = c:PRU
					type = annex_country
				}
			}
		}

		if = {
			limit = {
				exists = c:NGF
			}
			create_diplomatic_play = {
				target_state = s:STATE_BRANDENBURG.region_state:NGF
				type = dp_unify_germany
				add_war_goal = {
					holder = root
					target_country = c:NGF
					type = annex_country
				}
			}
		}

		set_variable = unify_germany_button_var
	}
}
