name = tyrone_nim_back_monarchy_name
    desc = tyrone_nim_back_monarchy_desc
    visible = { tyrone_nim_is_actor = yes }
    possible = {
        custom_tooltip = { text = tyrone_nim_assembly_open_tt tyrone_nim_constitution_open = yes }
        custom_tooltip = { text = tyrone_nim_intervention_ready_tt NOT = { has_variable = tyrone_nim_intervention_cooldown } }
    }
    cooldown = { days = tyrone_nim_year }
    ai_chance = { value = 10 }
    effect = {
        if = {
            limit = { tyrone_nim_constitution_open = yes NOT = { has_variable = tyrone_nim_intervention_cooldown } }
            custom_tooltip = { text = tyrone_nim_const_monarch_ten_tt tyrone_nim_shift_constitution = { VALUE = -10 } }
            tyrone_nim_constitution_intervention_cost = { SIDE = monarchy }
            hidden_effect = { set_variable = { name = tyrone_nim_intervention_cooldown days = tyrone_nim_year } }
        }
    }
}

    desc = tyrone_nim_industry_button_desc

    possible = {
        any_scope_state = {

            any_scope_building = { is_building_type = building_urban_center level >= 80 }
        }
    }
    cooldown = { days = tyrone_nim_two_years }
    ai_chance = { value = 100 }
    effect = {
        custom_tooltip = tyrone_nim_industry_button_effect_tt

            limit = {
                state_region = { has_state_trait = state_trait_tyrone_indian_rural_region }
                any_scope_building = { is_building_type = building_urban_center level >= 80 }
            }
            state_region = { remove_state_trait = state_trait_tyrone_indian_rural_region }
        }
    }
}
