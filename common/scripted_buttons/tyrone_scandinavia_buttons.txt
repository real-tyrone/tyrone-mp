scandinavian_public_works.1 = { #Healthcare Reform
	name = scandinavian_public_works.1.t
	desc = scandinavian_public_works.1.d
	ai_chance = {value = 50}
	visible = { always = yes }
	possible = {
		has_law = law_type:law_public_health_insurance
		NOT = { has_modifier = modifier_scandinavian_administrative_burden }
	}
	effect = {
		add_modifier = {
			name = modifier_scandinavian_administrative_burden
			months = 12
			is_decaying = yes
		}
		if = {
			limit = {
				enacting_any_law = yes
			}
			cancel_enactment = yes
		}

		start_enactment = law_type:law_scandinavian_healthcare
	}
}
