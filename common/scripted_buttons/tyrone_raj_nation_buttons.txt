tyrone_raj_nation_imperial_initiative = {
	name = tyrone_raj_nation_imperial_button
	desc = tyrone_raj_nation_imperial_button_desc
	visible = {
		hidden_trigger = {
			tyrone_raj_nation_is_actor = yes
		}
	}
	possible = {
		hidden_trigger = {
			tyrone_raj_nation_can_continue = yes
			tyrone_raj_nation_is_actor = yes
		}
		scope:raj_scope = {
			custom_tooltip = {
				text = tyrone_raj_nation_initiative_available_tt
				NOT = {
					has_variable = tyrone_raj_nation_intervention
				}
			}
		}
	}
	cooldown = {
		days = very_short_modifier_time
	}
	ai_chance = {
		value = 1
		if = {
			limit = {
				scope:raj_scope = {
					tyrone_raj_nation_law_drift >= 0
				}
			}
			add = 9
		}
	}
	effect = {
		tyrone_raj_nation_intervene = {
			DIRECTION = 1
			ACTOR_TRIGGER = tyrone_raj_nation_is_actor
			TOOLTIP = tyrone_raj_nation_imperial_initiative_tt
		}
	}
}
