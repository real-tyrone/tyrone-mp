tyr_chi_reintegrate_GNG = {
	name = tyr_chi_reintegrate_GNG
	desc = tyr_chi_reintegrate_desc
	visible = { has_journal_entry = je_tyr_china_reintegration exists = c:GNG c:GNG = { is_direct_subject_of = root is_subject_type = subject_type_tyr_chinese_warlord } }
	possible = {
		has_technology_researched = pan-nationalism
		custom_tooltip = { text = tyr_chi_cooldown_tt NOT = { has_variable = tyr_chi_reintegration_cooldown } }
		c:GNG = { liberty_desire <= 25 is_diplomatic_play_committed_participant = no }
		is_at_war = no
		is_diplomatic_play_committed_participant = no
	}
	effect = {
		set_variable = { name = tyr_chi_reintegration_cooldown value = yes months = 6 }
		if = { limit = { has_modifier = tyr_chi_reintegration_backlash } remove_modifier = tyr_chi_reintegration_backlash }
		add_modifier = { name = tyr_chi_reintegration_backlash years = 5 }
		create_diplomatic_play = { type = dp_tyr_chi_reintegration target_country = c:GNG }
	}
	ai_chance = { base = 100 }
}
