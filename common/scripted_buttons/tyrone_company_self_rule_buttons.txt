tyrone_csr_welfare_intervention = {
	name = tyrone_csr_welfare_button
	desc = tyrone_csr_welfare_button_desc
	visible = {
		this = scope:csr_india
		has_variable = tyrone_csr_active
		NOT = { has_variable = tyrone_csr_welfare_intervention_used }
	}
	possible = {
		hidden_trigger = {
			this = scope:csr_india
			has_variable = tyrone_csr_active
			NOT = { has_variable = tyrone_csr_welfare_intervention_used }
		}
		institution_investment_level = {
			institution = institution_social_security
			value >= 3
		}
	}
	ai_chance = {
		value = 10
	}
	effect = {
		hidden_effect = { set_variable = tyrone_csr_welfare_intervention_used }
		add_loyalists = { value = 0.02 }
		add_modifier = {
			name = tyrone_csr_welfare_intervention
			days = normal_modifier_time
		}
	}
}
