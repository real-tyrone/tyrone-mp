scw_congress_button = {

	name = scw_congress_button
	desc = scw_congress_button_desc
	visible = { has_variable = scw_running }
	possible = {
		custom_tooltip = {

			text = scw_congress_progress_tt
			var:scw_progress >= 25

		}
		custom_tooltip = {
			text = scw_congress_cooldown_tt
			NOT = { has_variable = scw_programme_cooldown }
		}
	}
	effect = {
		set_variable = { name = scw_programme_cooldown months = 24 }
		trigger_event = { id = scw_events.6 popup = yes }
	}
	ai_chance = { value = 5 }

}

				limit = { exists = c:CLM }

			}

				limit = { exists = c:VNZ }
				c:VNZ = { save_scope_as = scw_federal_leader }

				limit = {
					OR = {
						c:CLM ?= this
						c:VNZ ?= this
						c:ECU ?= this
					}
				}

				set_variable = { name = scw_ballot_sponsor value = scope:scw_sponsor }
				if = {
					limit = { is_ai = yes }
					set_variable = { name = scw_vote value = 1 }
				}
				else = {
					set_variable = { name = scw_vote value = 0 }
				}

					change_variable = { name = scw_expected_votes add = 1 }
				}
			}
			# Every participant receives the event; AI consent is already guaranteed.
			every_country = {
				limit = {
					OR = {
						c:CLM ?= this
						c:VNZ ?= this
						c:ECU ?= this
					}
				}
				trigger_event = { id = scw_events.20 popup = yes }
			}
			trigger_event = { id = scw_events.24 days = 30 }

		}
	}
	ai_chance = { value = 1 }

}
