# Buttons operate from the clicking country; saved India is refreshed before execution.
tyrone_raj_toggle_repayment = {
	name = tyrone_raj_toggle_repayment_button
	desc = tyrone_raj_toggle_repayment_button_desc
	cooldown = {
		months = 2
	}
	visible = {
		is_india_tag_trigger = yes
		has_variable = tyrone_unstable_raj_active
	}
	possible = {
		is_ai = no
		hidden_trigger = {
			is_india_tag_trigger = yes
			has_variable = tyrone_unstable_raj_active
			scope:raj_scope = {
				has_variable = tyrone_unstable_raj_active
				is_subject = yes
				NOR = {
					has_variable = tyrone_raj_charter_act_passed
					has_variable = tyrone_raj_crown_act_passed
					has_variable = tyrone_raj_debt_done
				}
			}
		}
		scope:raj_scope = {
			custom_tooltip = {
				text = tyrone_raj_debt_outstanding_tt
				var:tyrone_raj_debt > 0
			}
		}
	}
	ai_chance = {
		value = 0
	}
	effect = {
		tyrone_unstable_bind_india = yes
		if = {
			limit = {
				hidden_trigger = {
					is_india_tag_trigger = yes
					has_variable = tyrone_unstable_raj_active
					scope:raj_scope = {
						has_variable = tyrone_unstable_raj_active
						is_subject = yes
						NOR = {
							has_variable = tyrone_raj_charter_act_passed
							has_variable = tyrone_raj_crown_act_passed
							has_variable = tyrone_raj_debt_done
						}
					}
				}
				scope:raj_scope = {
					custom_tooltip = {
						text = tyrone_raj_debt_outstanding_tt
						var:tyrone_raj_debt > 0
					}
				}
			}
			if = {
				limit = {
					has_variable = tyrone_raj_auto_repayment
				}
				custom_tooltip = tyrone_raj_stop_repayment_tt
				remove_variable = tyrone_raj_auto_repayment
			}
			else = {
				custom_tooltip = tyrone_raj_start_repayment_tt
				set_variable = tyrone_raj_auto_repayment
			}
		}
	}
}
