    name = tyr_colony_arabia_create_button
    desc = tyr_colony_arabia_create_button_desc
    visible = {
        always = yes
        NOT = { any_subject_or_below = { has_variable = tyr_arabia_colony } }
    }
    cooldown = { days = 90 }
    possible = {
        tyr_arabia_colony_can_create = yes

        custom_tooltip = tyr_arabia_workforce_tt
    }
}
