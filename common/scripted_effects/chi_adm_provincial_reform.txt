 }

chi_adm_reform_reconcile_defeat = {

  limit = {
   has_global_variable = chi_adm_reform_defeat_window
   OR = {
    NOT = { has_variable = chi_adm_reform_defeat_seen }
    var:chi_adm_reform_defeat_seen < global_var:chi_adm_reform_defeat_serial
   }
  }
  remove_modifier = chi_adm_reform_defeat
  remove_variable = chi_adm_reform_defeat_legacy
  remove_variable = chi_adm_reform_recovery_months
  remove_variable = chi_adm_reform_recovery_review_total
  set_variable = { name = chi_adm_reform_recovery_done value = 0 }
  set_variable = { name = chi_adm_reform_defeat_seen value = global_var:chi_adm_reform_defeat_serial }
  chi_adm_reform_apply_defeat = yes
  chi_adm_reform_staff = yes
  chi_adm_reform_archive = yes
  if = {
   limit = { NOT = { has_journal_entry = je_chi_adm_reform_recovery } }
   add_journal_entry = { type = je_chi_adm_reform_recovery }
  }
 }
}

chi_adm_reform_initialize = {
 # Country scope. Record the old provincial state before the legacy sync can change it.
 if = {
  limit = { NOT = { has_variable = chi_adm_reform_v1 } }
  set_variable = chi_adm_reform_v1
  set_variable = { name = chi_adm_reform_army_done value = 0 }
  set_variable = { name = chi_adm_reform_arms_done value = 0 }
  set_variable = { name = chi_adm_reform_admin_done value = 0 }
  set_variable = { name = chi_adm_reform_opium_done value = 0 }
  set_variable = { name = chi_adm_reform_recovery_done value = 0 }
  # The actual old modifier takes precedence over a stale mirrored counter.
  if = { limit = { has_modifier = chi_early_army_stage_1 } set_variable = { name = chi_adm_reform_army_done value = 3 } }
  else_if = { limit = { has_modifier = chi_early_army_stage_2 } set_variable = { name = chi_adm_reform_army_done value = 2 } }
  else_if = { limit = { has_modifier = chi_early_army_stage_3 } set_variable = { name = chi_adm_reform_army_done value = 1 } }
  else_if = { limit = { has_modifier = chi_early_army_stage_4 } set_variable = { name = chi_adm_reform_army_done value = 0 } }
  else_if = {
   limit = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } }
   if = {
    limit = { has_variable = chi_early_army_stages }
    set_variable = { name = chi_adm_reform_army_done value = { value = 4 subtract = var:chi_early_army_stages min = 0 max = 4 } }
   }
   else_if = {
    limit = { var:chi_adm_imperial_court ?= { has_variable = chi_early_army_stages } }
    set_variable = { name = chi_adm_reform_army_done value = { value = 4 subtract = var:chi_adm_imperial_court.var:chi_early_army_stages min = 0 max = 4 } }
   }
   else_if = { limit = { game_date >= 1836.1.2 } set_variable = { name = chi_adm_reform_army_done value = 4 } }
  }
  else = { set_variable = { name = chi_adm_reform_army_done value = 4 } }
  if = {
   limit = {
    NOR = {
     has_modifier = chi_early_army_line_artillery_penalty
     AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } var:chi_adm_imperial_court ?= { has_modifier = chi_early_army_line_artillery_penalty } }
     AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } game_date < 1836.1.2 }
    }
   }
   set_variable = { name = chi_adm_reform_arms_done value = 4 }
  }
  if = {
   limit = {
    NOR = {
     has_modifier = chi_mandate_centralisation_low
     AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } var:chi_adm_imperial_court ?= { has_modifier = chi_mandate_centralisation_low } }
     AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } game_date < 1836.1.2 }
    }
   }
   if = {
    limit = { OR = { has_modifier = chi_mandate_centralisation_mid AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } var:chi_adm_imperial_court ?= { has_modifier = chi_mandate_centralisation_mid } } } }
    set_variable = { name = chi_adm_reform_admin_done value = 1 }
   }
   else_if = {
    limit = { OR = { has_modifier = chi_mandate_centralisation_high AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } var:chi_adm_imperial_court ?= { has_modifier = chi_mandate_centralisation_high } } } }
    set_variable = { name = chi_adm_reform_admin_done value = 3 }
   }
   else = { set_variable = { name = chi_adm_reform_admin_done value = 4 } }
  }
  if = {
   limit = { OR = { has_modifier = opium_wars_addiction_modifier AND = { NOT = { has_variable = chi_qing_modifier_cleanup_v3 } var:chi_adm_imperial_court ?= { has_modifier = opium_wars_addiction_modifier } } } }
   set_variable = chi_adm_reform_opium_affected
  }
  if = { limit = { has_global_variable = chi_adm_reform_opium_serial } set_variable = { name = chi_adm_reform_opium_seen value = global_var:chi_adm_reform_opium_serial } }
  if = {
   limit = {
    NOT = { has_global_variable = chi_adm_reform_defeat_serial }
    chi_adm_reform_defeat_active = yes
    OR = { has_modifier = opium_wars_lost NOT = { has_variable = chi_qing_modifier_cleanup_v3 } }
   }
   add_modifier = chi_adm_reform_defeat
   set_variable = chi_adm_reform_defeat_legacy
  }
  while = { limit = { has_modifier = chi_early_army_stage_1 } remove_modifier = chi_early_army_stage_1 }
  while = { limit = { has_modifier = chi_early_army_stage_2 } remove_modifier = chi_early_army_stage_2 }
