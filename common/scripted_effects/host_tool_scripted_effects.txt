host_tool_refresh_runtime_lists = {
	clear_global_variable_list ?= host_tool_player_countries
	clear_global_variable_list ?= host_tool_countries
	clear_global_variable_list ?= host_tool_subject_overlords
	clear_global_variable_list ?= host_tool_strategic_regions
	clear_global_variable_list ?= host_tool_incorporable_states
	clear_global_variable_list ?= host_tool_unincorporable_states
	remove_variable = host_tool_selected_incorporation_country
	remove_variable = host_tool_selected_unincorporation_country
	make_strategic_region_list = yes
	save_scope_as = host_tool_host

	every_country = {
		limit = {
			host_tool_is_player_country = yes
		}
		root = {
			add_to_global_variable_list = {
				name = host_tool_player_countries
				target = prev
			}
		}
	}

	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		root = {
			add_to_global_variable_list = {
				name = host_tool_countries
				target = prev
			}
		}
	}

	every_country = {
		limit = {
			host_tool_is_gp_or_major = yes
			host_tool_has_subjects = yes
		}
		root = {
			add_to_global_variable_list = {
				name = host_tool_subject_overlords
				target = prev
			}
		}
	}

	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		every_scope_state = {
			limit = {
				state_region = {
					is_state_region_land = yes
				}
			}
			state_region = {
				region = {
					save_temporary_scope_as = host_tool_iter_region
					if = {
						limit = {
							scope:host_tool_host = {
								NOT = {
									is_target_in_global_variable_list = {
										name = host_tool_strategic_regions
										target = scope:host_tool_iter_region
									}
								}
							}
						}
						scope:host_tool_host = {
							add_to_global_variable_list = {
								name = host_tool_strategic_regions
								target = scope:host_tool_iter_region
							}
						}
					}
				}
			}
		}
	}

	host_tool_recalculate_prestige_check = yes
}

host_tool_clear_prestige_check = {
	clear_global_variable_list ?= host_tool_prestige_left_countries
	clear_global_variable_list ?= host_tool_prestige_right_countries
	host_tool_recalculate_prestige_check = yes
}

host_tool_recalculate_prestige_check = {
	set_variable = {
		name = host_tool_prestige_left_total
		value = 0
	}
	set_variable = {
		name = host_tool_prestige_right_total
		value = 0
	}
	save_scope_as = host_tool_host

	every_in_global_list = {
		variable = host_tool_prestige_left_countries
		save_temporary_scope_as = host_tool_prestige_country
		scope:host_tool_host = {
			change_variable = {
				name = host_tool_prestige_left_total
				add = scope:host_tool_prestige_country.prestige
			}
		}
	}

	every_in_global_list = {
		variable = host_tool_prestige_right_countries
		save_temporary_scope_as = host_tool_prestige_country
		scope:host_tool_host = {
			change_variable = {
				name = host_tool_prestige_right_total
				add = scope:host_tool_prestige_country.prestige
			}
		}
	}

	set_variable = {
		name = host_tool_prestige_left_double
		value = var:host_tool_prestige_left_total
	}
	change_variable = {
		name = host_tool_prestige_left_double
		multiply = 2
	}
	set_variable = {
		name = host_tool_prestige_right_double
		value = var:host_tool_prestige_right_total
	}
	change_variable = {
		name = host_tool_prestige_right_double
		multiply = 2
	}
}

host_tool_build_states_and_countries_in_region = {
	clear_global_variable_list ?= host_tool_region_states
	clear_global_variable_list ?= host_tool_region_countries
	clear_global_variable_list ?= host_tool_region_capital_countries
	remove_variable = host_tool_selected_region_state
	remove_variable = host_tool_selected_region_country
	remove_variable = host_tool_selected_war_country
	remove_variable = host_tool_selected_war
	save_scope_as = host_tool_host

	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
		}
		every_scope_state = {
			limit = {
				state_region = {
					region = scope:strategic_region
				}
			}
			save_temporary_scope_as = host_tool_iter_state
			scope:host_tool_host = {
				add_to_global_variable_list = {
					name = host_tool_region_states
					target = scope:host_tool_iter_state
				}
			}
		}
	}

	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
			any_scope_state = {
				state_region = {
					region = scope:strategic_region
				}
			}
		}
		save_temporary_scope_as = host_tool_iter_country
		if = {
			limit = {
				scope:host_tool_host = {
					NOT = {
						is_target_in_global_variable_list = {
							name = host_tool_region_countries
							target = scope:host_tool_iter_country
						}
					}
				}
			}
			scope:host_tool_host = {
				add_to_global_variable_list = {
					name = host_tool_region_countries
					target = scope:host_tool_iter_country
				}
			}
		}
	}

	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
			capital = {
				region = scope:strategic_region
			}
		}
		save_temporary_scope_as = host_tool_iter_country
		scope:host_tool_host = {
			add_to_global_variable_list = {
				name = host_tool_region_capital_countries
				target = scope:host_tool_iter_country
			}
		}
	}
}

		limit = { has_game_rule = host_tool_initial_host_fra }

	}
	if = {
		limit = { has_game_rule = host_tool_initial_host_rus }

	}
