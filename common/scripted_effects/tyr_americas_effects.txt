	remove_global_variable = tyr_am_spain

		limit = {

			set_global_variable = {

				value = this

		}
	}
}

tyr_am_count_area = {
	if = {
		limit = {
			has_global_variable = tyr_am_$AREA$_color
		}
		set_global_variable = {
			name = tyr_am_$AREA$_previous_color
			value = global_var:tyr_am_$AREA$_color
		}
	}
	else = {
		set_global_variable = {
			name = tyr_am_$AREA$_previous_color
			value = 0
		}
	}
	set_global_variable = {
		name = tyr_am_$AREA$_spa
		value = 0
	}
	set_global_variable = {
		name = tyr_am_$AREA$_usa
		value = 0
	}
	set_global_variable = {
		name = tyr_am_$AREA$_bra
		value = 0
	}
	set_global_variable = {
		name = tyr_am_$AREA$_independent
		value = 0
	}
	set_global_variable = {
		name = tyr_am_$AREA$_bra_owned
		value = 0
	}
	every_state_in_tyr_am_$AREA$ = {
		if = {
			limit = {
				exists = global_var:tyr_am_spain
				owner = {
					tyr_am_controls_for = {
						PATRON = global_var:tyr_am_spain
					}
				}
			}
			change_global_variable = {
				name = tyr_am_$AREA$_spa
				add = tyr_am_state_share
			}
		}
		else_if = {
			limit = {
				exists = c:USA
				owner = {
					tyr_am_controls_for = {
						PATRON = c:USA
					}
				}
			}
			change_global_variable = {
				name = tyr_am_$AREA$_usa
				add = tyr_am_state_share
			}
		}
		else_if = {
			limit = {
				c:BRZ ?= {
					tyr_am_brazil_sphere = yes
				}
				owner = {
					tyr_am_controls_for = {
						PATRON = c:BRZ
					}
				}
			}
			change_global_variable = {
				name = tyr_am_$AREA$_bra
				add = tyr_am_state_share
			}
		}
		else_if = {
			limit = {
				owner = {
					tyr_am_regional_country = yes
				}
				OR = {
					owner = { is_subject = no }
					AND = {
						is_in_geographic_region = geographic_region_tyr_am_7
						owner = {
							exists = c:BOL
							exists = top_overlord
							top_overlord = c:BOL
						}
					}
				}
			}
			change_global_variable = {
				name = tyr_am_$AREA$_independent
				add = tyr_am_state_share
			}
		}
		if = {
			limit = {
				owner = c:BRZ
			}
			change_global_variable = {
				name = tyr_am_$AREA$_bra_owned
				add = tyr_am_state_share
			}
		}
	}
	change_global_variable = {
		name = tyr_am_$AREA$_spa
		multiply = 100
	}
	change_global_variable = {
		name = tyr_am_$AREA$_spa
		divide = $COUNT$
	}
	change_global_variable = {
		name = tyr_am_$AREA$_usa
		multiply = 100
	}
	change_global_variable = {
		name = tyr_am_$AREA$_usa
		divide = $COUNT$
	}
	change_global_variable = {
		name = tyr_am_$AREA$_bra
		multiply = 100
	}
	change_global_variable = {
		name = tyr_am_$AREA$_bra
		divide = $COUNT$
	}
	change_global_variable = {
		name = tyr_am_$AREA$_independent
		multiply = 100
	}
	change_global_variable = {
		name = tyr_am_$AREA$_independent
		divide = $COUNT$
	}
	change_global_variable = {
		name = tyr_am_$AREA$_bra_owned
		multiply = 100
	}
	change_global_variable = {
		name = tyr_am_$AREA$_bra_owned
		divide = $COUNT$
	}
	set_global_variable = {
		name = tyr_am_$AREA$_color
		value = 0
	}
	if = {
		limit = {
			has_global_variable = tyr_am_$AREA$_bra_owned
			has_global_variable = tyr_am_$AREA$_spa
			has_global_variable = tyr_am_$AREA$_usa
			has_global_variable = tyr_am_$AREA$_bra
			has_global_variable = tyr_am_$AREA$_independent
			has_global_variable = tyr_am_$AREA$_color
			has_global_variable = tyr_am_$AREA$_previous_color
		}
		if = {
			limit = {
				global_var:tyr_am_$AREA$_bra_owned >= 75
			}
			change_global_variable = {
				name = tyr_am_bra_owned_areas
				add = 1
			}
		}
		if = {
			limit = {
				global_var:tyr_am_$AREA$_spa >= 75
			}
			set_global_variable = {
				name = tyr_am_$AREA$_color
				value = 1
			}
			change_global_variable = {
				name = tyr_am_spa_areas
				add = 1
			}
		}
		else_if = {
			limit = {
				global_var:tyr_am_$AREA$_usa >= 75
			}
			set_global_variable = {
				name = tyr_am_$AREA$_color
				value = 2
			}
			change_global_variable = {
				name = tyr_am_usa_areas
				add = 1
			}
		}
		else_if = {
			limit = {
				global_var:tyr_am_$AREA$_bra >= 75
			}
			set_global_variable = {
				name = tyr_am_$AREA$_color
				value = 3
			}
			change_global_variable = {
				name = tyr_am_bra_areas
				add = 1
			}
		}
		else = {
			change_global_variable = {
				name = tyr_am_contested
				add = 1
			}
			if = {
				limit = {
					global_var:tyr_am_$AREA$_independent >= 75
				}
				set_global_variable = {
					name = tyr_am_$AREA$_color
					value = 4
				}
			}
		}
		if = {
			limit = {
				NOT = {
					global_var:tyr_am_$AREA$_color = global_var:tyr_am_$AREA$_previous_color
				}
			}
			set_global_variable = {
				name = tyr_am_stall_months
				value = 0
			}
		}
	}
}

	}

tyr_am_check_actor_wars = {
	save_scope_as = tyr_am_belligerent
	every_scope_war = {
		limit = {
			OR = {
				attacker_warleader = scope:tyr_am_belligerent
				defender_warleader = scope:tyr_am_belligerent
			}
		}
		save_scope_as = tyr_am_war
		if = {
			limit = {
				scope:tyr_am_belligerent = {
					OR = {
						NOT = {
							has_variable_list = tyr_am_notified_wars
						}
						NOT = {
							any_in_list = {
								variable = tyr_am_notified_wars
								this = scope:tyr_am_war
							}
						}
					}
				}
				OR = {
					AND = {
						attacker_warleader = scope:tyr_am_belligerent
						defender_warleader = {
							tyr_am_present = yes
						}
					}
					AND = {
						defender_warleader = scope:tyr_am_belligerent
						attacker_warleader = {
							tyr_am_present = yes
						}
					}
					any_war_participant = {
						tyr_am_regional_country = yes
						has_war_with = scope:tyr_am_belligerent
					}
				}
			}
			scope:tyr_am_belligerent = {
				add_to_variable_list = {
					name = tyr_am_notified_wars
					target = scope:tyr_am_war
				}
			}
			if = {
				limit = {
					attacker_warleader = scope:tyr_am_belligerent
				}
				defender_warleader = {
					save_scope_as = tyr_am_target
				}
			}
			else = {
				attacker_warleader = {
					save_scope_as = tyr_am_target
				}
			}
			# Identify an actual South American opponent even if an outside power leads its side.
			if = {
				limit = {
					any_war_participant = {
						tyr_am_regional_country = yes
						has_war_with = scope:tyr_am_belligerent
						save_temporary_scope_as = tyr_am_local_opponent
					}
				}
				scope:tyr_am_local_opponent = {
					save_scope_as = tyr_am_target
				}
			}
			if = {
				limit = {
					scope:tyr_am_belligerent = {
						country_definition = cd:BRZ
					}
				}
				global_var:tyr_am_spain ?= {
					trigger_event = { id = tyr_americas.30 popup = yes }
				}
				c:USA ?= {
					trigger_event = { id = tyr_americas.30 popup = yes }
				}
			}
			else_if = {
				limit = {
					scope:tyr_am_belligerent = {
						tyr_spanish_content_country = yes
					}
				}
				c:USA ?= {
					trigger_event = {
						id = tyr_americas.30
						popup = yes
					}
				}
			}
			else = {
				global_var:tyr_am_spain ?= {
					trigger_event = {
						id = tyr_americas.30
						popup = yes
					}
				}
			}
		}
	}
}

tyr_am_finish = {
	# Runtime-only terminal transaction. Missing state and repeat calls are no-ops.
	if = {
		limit = {
			has_global_variable = tyr_am_active
			has_global_variable = tyr_am_outcome
			NOT = {
				has_global_variable = tyr_am_finished
			}
		}
		if = {
			limit = {
				global_var:tyr_am_outcome >= 1
				global_var:tyr_am_outcome <= 5
			}
			# Cache outcome and identity before cleanup; one permanent world flag prevents repeat rewards.
			set_global_variable = {
				name = tyr_am_final_result
				value = global_var:tyr_am_outcome
			}
			set_global_variable = tyr_am_finished
			global_var:tyr_am_spain ?= {
				save_scope_as = tyr_am_spanish_final
			}
			if = {
				limit = {
					global_var:tyr_am_final_result = 1
				}
				global_var:tyr_am_spain ?= {
					save_scope_as = tyr_am_winner
				}
				c:USA ?= {
					save_scope_as = tyr_am_loser
				}
			}
			else_if = {
				limit = {
					global_var:tyr_am_final_result = 2
				}
				c:USA ?= {
					save_scope_as = tyr_am_winner
				}
				global_var:tyr_am_spain ?= {
					save_scope_as = tyr_am_loser
				}
			}
			tyr_am_cleanup = yes
			if = {
				limit = {
					global_var:tyr_am_final_result = 1
				}
				scope:tyr_am_winner ?= {
					set_variable = tyr_am_victory_reward_pending
					trigger_event = {
						id = tyr_americas.70
						popup = yes
					}
				}
			}
			else_if = {
				limit = {
					global_var:tyr_am_final_result = 2
				}
				scope:tyr_am_winner ?= {
					set_variable = tyr_am_victory_reward_pending
					trigger_event = {
						id = tyr_americas.71
						popup = yes
					}
				}
			}
			else_if = {
				limit = {
					global_var:tyr_am_final_result = 3
				}
				c:BRZ ?= {
					set_variable = tyr_am_victory_reward_pending
					trigger_event = {
						id = tyr_americas.72
						popup = yes
					}
				}
				scope:tyr_am_spanish_final ?= {
					add_modifier = {
						name = tyr_am_brazil_defeat
						years = 15
					}
					trigger_event = {
						id = tyr_americas.21
						popup = yes
					}
				}
				c:USA ?= {
					add_modifier = {
						name = tyr_am_brazil_defeat
						years = 15
					}
					trigger_event = {
						id = tyr_americas.21
						popup = yes
					}
				}
			}
			else = {
				scope:tyr_am_spanish_final ?= {
					trigger_event = {
						id = tyr_americas.24
						popup = yes
					}
				}
				c:USA ?= {
					trigger_event = {
						id = tyr_americas.24
						popup = yes
					}
				}
				c:BRZ ?= {
					trigger_event = {
						id = tyr_americas.24
						popup = yes
					}
				}
			}
			if = {
				limit = {
					OR = {
						global_var:tyr_am_final_result = 1
						global_var:tyr_am_final_result = 2
					}
				}
				scope:tyr_am_loser ?= {

# Country UI context. Only tooltip output; no treasury, variables or modifiers are changed.
# Caller has resolved scope:tyr_am_actor and scope:tyr_am_target through the existing trigger.
tyr_am_ui_action_requirements = {
	if = {
		limit = {
			NOT = {
				tyr_am_has_valid_target = yes
			}
		}
		custom_tooltip = tyr_am_block_target
	}
	if = {
		limit = {
			has_variable = tyr_am_action_cooldown
		}
		custom_tooltip = tyr_am_block_cooldown
	}
	if = {
		limit = {
			always = $NEGOTIATE$
		}
		if = {
			limit = {
				NOT = {
					tyr_am_actor_valid = yes
				}
			}
			custom_tooltip = tyr_am_block_actor
		}
		if = {
			limit = {
				scope:tyr_am_target = {
					is_subject = yes
				}
			}
			custom_tooltip = tyr_am_block_subject
		}
		if = {
			limit = {
				scope:tyr_am_target = {
					is_in_same_power_bloc = scope:tyr_am_actor
				}
			}
			custom_tooltip = tyr_am_block_same_bloc
		}
	}
	if = {
		limit = {
			always = $AID_ROLE$
		}
		if = {
			limit = {
				NOT = {
					OR = {
						tyr_am_actor_valid = yes
						AND = {
							country_rank >= rank_value:major_power
							OR = {
								var:tyr_am_sponsor_choice = 1
								var:tyr_am_sponsor_choice = 2
							}
						}
					}
				}
			}
			custom_tooltip = tyr_am_block_aid_role
		}
	}
	if = {
		limit = {
			always = $PAID$
		}
		custom_tooltip = tyr_am_payment_debt_warning_tt
	}
	if = {
		limit = {
			always = $INVITE$
		}
		if = {
			limit = {
				scope:tyr_am_target = {
					is_in_power_bloc = yes
				}
			}
			custom_tooltip = tyr_am_block_other_bloc
		}
		if = {
			limit = {
				scope:tyr_am_target = {
					has_variable = tyr_am_invitation_pending
				}
			}
			custom_tooltip = tyr_am_block_pending
		}
		if = {
			limit = {
				NOT = {
					OR = {
						AND = {
							tyr_spanish_content_country = yes
							tyr_am_selected_influence <= 25
						}
						AND = {
							c:USA ?= this
							tyr_am_selected_influence >= 75
						}
					}
				}
			}
			custom_tooltip = tyr_am_block_influence
		}
		if = {
			limit = {
				NOT = {
					OR = {
						prestige > scope:tyr_am_target.prestige
						country_rank > scope:tyr_am_target.country_rank
					}
				}
			}
			custom_tooltip = tyr_am_block_rank
		}
		if = {
			limit = {
				is_in_power_bloc = yes
			}
			if = {
				limit = {
					power_bloc = {
						any_power_bloc_member = {
							is_diplomatic_play_enemy_of = scope:tyr_am_target
						}
					}
				}
				custom_tooltip = tyr_am_block_bloc_war
			}
		}
	}
}

	every_country = {
		limit = {
			is_player = no
			is_subject = no
			exists = capital
			capital = {
				is_in_geographic_region = geographic_region_tyr_am_all
			}
			NOT = {
				tyr_am_main_power = yes
			}
		}
		if = {
			limit = {
				exists = global_var:tyr_am_spain
				global_var:tyr_am_spain = {
					is_power_bloc_leader = yes
				}
				is_in_same_power_bloc = global_var:tyr_am_spain
			}
			set_variable = {
				name = tyr_am_phase2_patron
				value = global_var:tyr_am_spain
			}
		}
		else_if = {
			limit = {
				exists = c:USA
				c:USA = {
					is_power_bloc_leader = yes
				}
				is_in_same_power_bloc = c:USA
			}
			set_variable = {
				name = tyr_am_phase2_patron
				value = c:USA
			}
		}
	}
	every_country = {
		limit = {
			exists = var:tyr_am_phase2_patron
			is_player = no
		}
		save_scope_as = tyr_am_phase2_subject
		if = {
			limit = {
				is_subject = no
			}
			var:tyr_am_phase2_patron = {
				create_diplomatic_pact = {
					country = scope:tyr_am_phase2_subject
					type = protectorate
				}
			}
		}
		remove_variable = tyr_am_phase2_patron

# Winning Spain. Establish direct colony relations before annexing donors:
# a selected player can initially be a subject of another regional donor.
tyr_am_prepare_viceroyalty = {
	if = {
		limit = {
			exists = var:tyr_am_viceroyalty_$AREA$_recipient
		}
		var:tyr_am_viceroyalty_$AREA$_recipient = {
			save_scope_as = tyr_am_viceroyalty_recipient
			if = {
				limit = {
					is_subject_of = scope:tyr_am_settlement_spain
				}
				if = {
					limit = {
						NOT = { is_subject_type = subject_type_colony }
					}
					change_subject_type = subject_type_colony
				}
			}
			else = {
				if = {
					limit = {
						is_subject = yes
					}
					make_independent = yes
				}
				scope:tyr_am_settlement_spain = {
					create_diplomatic_pact = {
						country = scope:tyr_am_viceroyalty_recipient
						type = colony
					}
				}
			}
			set_country_type = colonial
			activate_law = law_type:law_colonial_administration
			add_primary_culture = cu:spanish
			set_strategy = ai_strategy_colonial_extraction
		}
	}
}

# Country WG holder. Transfer subjects BEFORE states: the losing country
# might cease to exist when its last directly-owned state changes hands.
# No incorporation, culture changes or subject-type changes are applied.
tyr_am_enforce_area_wargoal = {
	save_scope_as = tyr_am_area_wargoal_winner
	# Snapshot the full subject tree before any overlord is changed.
	$LOSER$ ?= {
		every_subject_or_below = {
			limit = {
				exists = capital
				capital ?= {
					is_in_geographic_region = geographic_region_tyr_am_$AREA$
				}
			}
			set_variable = {
				name = tyr_am_area_wargoal_recipient
				value = scope:tyr_am_area_wargoal_winner
			}
		}
	}
	every_country = {
		limit = { has_variable = tyr_am_area_wargoal_recipient }
		scope:tyr_am_area_wargoal_winner = {
			transfer_subject = prev
		}
		remove_variable = tyr_am_area_wargoal_recipient
	}
	$LOSER$ ?= {
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_tyr_am_$AREA$
			}
			set_state_owner = scope:tyr_am_area_wargoal_winner
		}
	}
	if = {
		limit = { has_global_variable = tyr_am_active }
		tyr_am_update_control = yes
	}
}

# scope:stakeholder; the original local target is now the holder's subject.

				has_war_with = root
			}

			}

			add_war_goal = {
				holder = scope:stakeholder
				target_country = scope:target_country
				type = $TYPE$
			}
		}
	}
