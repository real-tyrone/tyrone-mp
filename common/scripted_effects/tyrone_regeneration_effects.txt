# Country scope: grant the full regeneration reward for the current economic law.
tyrone_grant_economic_regeneration_amendment = {
	if = {
		limit = {
			OR = {
				has_law_or_variant = law_type:law_agrarianism
				has_law_or_variant = law_type:law_industry_banned
			}
		}
		active_law:lawgroup_economic_system ?= {
			add_amendment = {
				type = amendment_economic_regeneration_1_agrarian
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_cooperative_ownership
		}
		active_law:lawgroup_economic_system ?= {
			add_amendment = {
				type = amendment_economic_regeneration_1_cooperative
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_command_economy
		}
		active_law:lawgroup_economic_system ?= {
			add_amendment = {
				type = amendment_economic_regeneration_1_command
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else_if = {
		limit = {
			has_law_or_variant = law_type:law_state_capitalism
		}
		active_law:lawgroup_economic_system ?= {
			add_amendment = {
				type = amendment_economic_regeneration_1_state_capitalism
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else = {
		active_law:lawgroup_economic_system ?= {
			add_amendment = {
				type = amendment_economic_regeneration_1
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
}
