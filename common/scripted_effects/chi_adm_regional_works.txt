chi_adm_regional_works_endorse_charter = {
 random_scope_state = {
  limit = { chi_adm_regional_works_$PROJECT$_site = yes chi_adm_regional_works_track_available = { TRACK = $TRACK$ TYPE = $TYPE$ TIER = $TIER$ } }
  save_temporary_scope_as = chi_adm_regional_works_registered_site
 }
 if = {
  limit = { exists = scope:chi_adm_regional_works_registered_site }
  set_variable = { name = chi_adm_regional_works_charter_site value = scope:chi_adm_regional_works_registered_site }
  set_variable = { name = chi_adm_regional_works_charter_key value = $KEY$ months = 60 }
  set_variable = { name = chi_adm_regional_works_charter_route value = $ROUTE$ }
  remove_variable = chi_adm_regional_works_offer
  if = { limit = { var:chi_adm_regional_works_charter_route = 1 } add_modifier = { name = chi_adm_regional_works_province_subscription years = 2 } }
  else = { add_modifier = { name = chi_adm_regional_works_imperial_accounts months = 18 } }
  chi_adm_regional_works_review_charter = { PROJECT = $PROJECT$ TRACK = $TRACK$ TYPE = $TYPE$ TIER = $TIER$ }
 }
}
chi_adm_regional_works_clear_charter = {
 remove_variable = chi_adm_regional_works_charter_site
 remove_variable = chi_adm_regional_works_charter_key
 remove_variable = chi_adm_regional_works_charter_route
}
chi_adm_regional_works_review_charter = {
 if = {
  limit = {
   OR = {
    NOT = { chi_adm_regional_works_$PROJECT$_identity = yes }
    NOT = { var:chi_adm_regional_works_charter_site ?= { owner = prev chi_adm_regional_works_track_available = { TRACK = $TRACK$ TYPE = $TYPE$ TIER = $TIER$ } } }
   }
  }
  chi_adm_regional_works_clear_charter = yes
 }
 else_if = {
  limit = {
   chi_adm_regional_works_development = yes
   chi_adm_regional_works_$PROJECT$_industry = yes
   var:chi_adm_regional_works_charter_site ?= { chi_adm_regional_works_$PROJECT$_site = yes var:chi_adm_provincial_development_$TRACK$ >= $TIER$ }
  }
  # Record first, then issue one temporary award at the owned project site.
  var:chi_adm_imperial_court ?= { set_variable = chi_adm_regional_works_$PROJECT$_ratified }
  var:chi_adm_regional_works_charter_site = { add_modifier = { name = chi_adm_regional_works_$PROJECT$_works years = 4 } }
  if = {
   limit = { var:chi_adm_regional_works_charter_route = 1 }
   chi_adm_adjust_autonomy = { AMOUNT = 4 }
   chi_adm_adjust_loyalty = { AMOUNT = -2 }
  }
  else = {
   chi_adm_adjust_autonomy = { AMOUNT = -3 }
   chi_adm_adjust_loyalty = { AMOUNT = 6 }
   chi_adm_adjust_corruption = { AMOUNT = -5 }
  }
  set_variable = { name = chi_adm_regional_works_last_charter value = var:chi_adm_regional_works_charter_key years = 4 }
  chi_adm_regional_works_clear_charter = yes
 }
}
