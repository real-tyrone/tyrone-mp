chi_adm_provincial_development_monthly = {
if = {
    limit = {
        has_modifier = chi_adm_provincial_development_package_6
        NOT = { has_variable = chi_adm_provincial_development_tax_concessions_timed }
    }
    chi_adm_provincial_development_start_tax_concessions = yes
}
if = { limit = { chi_adm_target_valid = yes }
    if = { limit = { chi_adm_provincial_development_broad_danger = yes }
        if = { limit = { NOT = { has_variable = chi_adm_provincial_development_warning } } set_variable = { name = chi_adm_provincial_development_warning value = 0 } }
        change_variable = { name = chi_adm_provincial_development_warning add = 1 }
    } else = { remove_variable = chi_adm_provincial_development_warning }
    if = { limit = { has_variable = chi_adm_provincial_development_package NOT = { has_variable = chi_adm_provincial_development_dispute } }
 if = { limit = { chi_adm_provincial_development_crisis = yes }
  if = { limit = { has_variable = chi_adm_provincial_development_grace }
   remove_variable = chi_adm_provincial_development_grace
   change_variable = { name = chi_adm_provincial_development_grace_revision add = 1 }
  }
 }
 else_if = { limit = { NOT = { has_variable = chi_adm_provincial_development_grace } NOT = { has_variable = chi_adm_provincial_development_case } }
  set_variable = chi_adm_provincial_development_grace
  if = { limit = { NOT = { has_variable = chi_adm_provincial_development_grace_revision } } set_variable = { name = chi_adm_provincial_development_grace_revision value = 0 } }
  change_variable = { name = chi_adm_provincial_development_grace_revision add = 1 }
  save_scope_value_as = { name = chi_adm_provincial_development_grace_revision value = var:chi_adm_provincial_development_grace_revision }
  trigger_event = { id = chi_adm_provincial_development.3 months = 3 popup = no }
 }
}
    if = { limit = { has_variable = chi_adm_provincial_development_deadline }
        change_variable = { name = chi_adm_provincial_development_deadline subtract = 1 }
        chi_adm_provincial_development_auto_warrant_response = yes
        if = { limit = {
            trigger_if = {
                limit = { has_variable = chi_adm_provincial_development_deadline }
                var:chi_adm_provincial_development_deadline <= 0
            }
            trigger_else = { always = no }
        }
            if = { limit = { var:chi_adm_provincial_development_case = 1 } chi_adm_provincial_development_refuse = yes }
            else_if = { limit = { var:chi_adm_provincial_development_case = 4 } chi_adm_provincial_development_resist_restoration = yes }
            else = { chi_adm_provincial_development_abandon = yes }
        }
    }
    if = {
        limit = { is_ai = yes NOT = { has_variable = chi_adm_provincial_development_package } }
        if = {
            limit = { chi_adm_provincial_development_war_or_hostile_neighbor = yes }
            chi_adm_provincial_development_request = { PACKAGE = 1 MILITARY = yes }
        }
        else = { chi_adm_provincial_development_request = { PACKAGE = 4 MILITARY = no } }
    }
} else = { chi_adm_provincial_development_cleanup = yes }
}

# Province scope. Apply only the human court's preference for initial warrants.
chi_adm_provincial_development_auto_warrant_response = {
    if = {
        limit = {
            chi_adm_provincial_development_case_valid = yes
            var:chi_adm_provincial_development_case = 1
            var:chi_adm_imperial_court ?= {
                is_ai = no
                chi_adm_memorials_court = yes
                NOT = { tyr_china_unified_active = yes }
                has_variable = chi_adm_memorials_auto_emergency
            }
        }
        if = {
            limit = { var:chi_adm_imperial_court = { var:chi_adm_memorials_auto_emergency = 1 } }
            chi_adm_provincial_development_approve = yes
        }
        else_if = {
            limit = { var:chi_adm_imperial_court = { var:chi_adm_memorials_auto_emergency = 2 } }
            chi_adm_provincial_development_refuse = yes
        }
    }
}
