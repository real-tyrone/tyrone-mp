# Country scope: record and reward je_tyrone_raj_nation once.
tyrone_arc_tyrone_raj_nation_grant = {
	if = {
		limit = { hidden_trigger = { NOT = { has_variable = tyrone_raj_nation_done } } }
		scope:raj_scope ?= {
			ig:ig_industrialists ?= {
				remove_ideology = ideology_john_company
				add_ideology = ideology_john_company_reformed
			}
			ig:ig_landowners ?= {
				remove_ideology = ideology_zamindar_system
				add_ideology = ideology_zamindar_system_reformed
			}
		}
		if = {
			limit = {
				je:je_tyrone_raj_nation = {
					"scripted_bar_progress(tyrone_raj_nation_progress_bar)" <= -60
				}
			}
			scope:raj_scope = {
				set_variable = {
					name = tyrone_raj_nation_result
					value = -1
				}
				activate_law = law_type:law_hindu_caste_codified
				activate_law = law_type:law_chartered_hereditary_bureaucrats
				set_variable = tyrone_raj_nation_done
				trigger_event = {
					id = tyrone_raj_nation.4
					popup = yes
				}
			}
		}
		else = {
			scope:raj_scope = {
				set_variable = {
					name = tyrone_raj_nation_result
					value = 1
				}
				tyrone_raj_nation_remove_informal_classes = yes
				set_variable = tyrone_raj_nation_done
				trigger_event = {
					id = tyrone_raj_nation.5
					popup = yes
				}
			}
		}
		tyrone_raj_nation_end = yes
	}
}
