# Country scope Russia: establish a port before the treaty article transfers it.
tyr_sj_take_port_arthur = {
 s:STATE_SHENGJING = {
  random_scope_state = {
   limit = { owner = global_var:tyr_sj_china controls_treaty_port_province = yes }
   save_scope_as = tyr_sj_port_source
   if = {
    limit = { NOT = { has_building = building_port } }
    create_building = { building = building_port level = 1 reserves = 1 }
   }
  }
 }
 create_treaty = {
  first_country = global_var:tyr_sj_china
  second_country = global_var:tyr_sj_russia
  is_draft = no
  binding_period = { years = 99 }
  articles_to_create = {
   { article = treaty_port source_country = global_var:tyr_sj_china target_country = global_var:tyr_sj_russia inputs = { { state = scope:tyr_sj_port_source } } }
  }
 }
 hidden_effect = { chi_legation_refresh_rights = yes }
}
# Native humiliation enforcement: ROOT is the winner; target_country the loser.
# Only a new victory resets that winner's one-use claim purchase.
tyr_sj_record_humiliation = {
 hidden_effect = {
  if = {
   limit = {
    tyr_sj_active = yes
    OR = {
     AND = { tyr_sj_is_japan = yes scope:target_country = { tyr_sj_is_russia = yes } }
     AND = { tyr_sj_is_russia = yes scope:target_country = { tyr_sj_is_japan = yes } }
    }
   }
   set_global_variable = { name = tyr_sj_last_rus_jap_victor value = THIS }
   remove_variable = tyr_sj_claims_bought_this_victory
  }
 }
}
tyr_sj_japanese_victory_effect = {
 hidden_effect = {
  set_global_variable = tyr_sj_finished
  global_var:tyr_sj_china ?= {
   if = { limit = { has_journal_entry = je_warlord_china } set_variable = tyr_sj_force_warlord_defeat }
  }
 }
 global_var:tyr_sj_china ?= { add_modifier = { name = tyr_sj_china_defeat years = 20 } }
 global_var:tyr_sj_japan ?= { trigger_event = { id = tyr_sj.1 popup = yes } }
}

# Country scope: corresponding protagonist. Timed flag supplies the real deadline.
tyr_sj_update_japan_clock = {
 hidden_effect = {
  if = {
   limit = { tyr_sj_japan_winning = yes }
   if = {
    limit = { NOT = { has_variable = tyr_sj_clock_running } }
    set_variable = tyr_sj_clock_running
    set_variable = { name = tyr_sj_clock_deadline months = 48 }
   }
   else_if = {
    limit = { NOT = { has_variable = tyr_sj_clock_deadline } }
    set_variable = tyr_sj_clock_ready
    je:je_sino_japanese_wars ?= { set_bar_progress = { name = tyr_sj_japan_bar value = 48 } }
   }
  }
  else = {
   remove_variable = tyr_sj_clock_running
   remove_variable = tyr_sj_clock_deadline
   remove_variable = tyr_sj_clock_ready
   je:je_sino_japanese_wars ?= { set_bar_progress = { name = tyr_sj_japan_bar value = 0 } }
  }
 }
}
