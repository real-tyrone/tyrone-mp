# ROOT is Germany. Transfer eligible states from Germany and its subject hierarchy; exclude the receiving administrations.
tyr_gcol_transfer_states = {
	if = {
		limit = {
			c:GER ?= root
			has_variable = tyr_gcol_started
			is_at_war = no
			is_diplomatic_play_committed_participant = no
		}
		if = {
			limit = {
				tyr_gcol_has_maf = yes
				c:MAF ?= {
					is_at_war = no
					is_diplomatic_play_committed_participant = no
				}
			}
			every_scope_state = {
				limit = {
					tyr_gcol_african_state = yes
					is_under_colonization = no
					is_treaty_port = no
					is_incorporated = no
				}
				state_region = {
					save_scope_as = tyr_gcol_transfer_region
				}
				set_state_owner = c:MAF
				c:MAF = {
					every_scope_state = {
						limit = {
							state_region = scope:tyr_gcol_transfer_region
						}
						set_state_type = incorporated
					}
				}
			}
			every_subject_or_below = {
				limit = {
					NOT = {
						c:MAF ?= this
					}
					NOT = {
						c:DEA ?= this
					}
					is_at_war = no
					is_diplomatic_play_committed_participant = no
				}
				every_scope_state = {
					limit = {
						tyr_gcol_african_state = yes
						is_under_colonization = no
						is_treaty_port = no
					}
					state_region = {
						save_scope_as = tyr_gcol_transfer_region
					}
					set_state_owner = c:MAF
					c:MAF = {
						every_scope_state = {
							limit = {
								state_region = scope:tyr_gcol_transfer_region
							}
							set_state_type = incorporated
						}
					}
				}
			}
		}
		if = {
			limit = {
				tyr_gcol_has_dea = yes
				c:DEA ?= {
					is_at_war = no
					is_diplomatic_play_committed_participant = no
				}
			}
			every_scope_state = {
				limit = {
					tyr_gcol_asian_state = yes
					is_under_colonization = no
					is_treaty_port = no
					is_incorporated = no
				}
				state_region = {
					save_scope_as = tyr_gcol_transfer_region
				}
				set_state_owner = c:DEA
				c:DEA = {
					every_scope_state = {
						limit = {
							state_region = scope:tyr_gcol_transfer_region
						}
						set_state_type = incorporated
					}
				}
			}
			every_subject_or_below = {
				limit = {
					NOT = {
						c:MAF ?= this
					}
					NOT = {
						c:DEA ?= this
					}
					is_at_war = no
					is_diplomatic_play_committed_participant = no
				}
				every_scope_state = {
					limit = {
						tyr_gcol_asian_state = yes
						is_under_colonization = no
						is_treaty_port = no
					}
					state_region = {
						save_scope_as = tyr_gcol_transfer_region
					}
					set_state_owner = c:DEA
					c:DEA = {
						every_scope_state = {
							limit = {
								state_region = scope:tyr_gcol_transfer_region
							}
							set_state_type = incorporated
						}
					}
				}
			}
		}
	}

	tyr_gcol_apply_subject_subsidies = yes
}

tyr_gcol_create_maf = {
	random_scope_state = {
		limit = {
			tyr_gcol_african_state = yes
			is_under_colonization = no
			is_incorporated = no
			is_treaty_port = no
		}
		save_scope_as = tyr_gcol_founding_state
	}

	create_country = {
		tag = MAF
		origin = root
		state = scope:tyr_gcol_founding_state
		on_created = {
			set_variable = tyr_gcol_maf_colony
			hidden_effect = {
				activate_law = law_type:law_slave_trade
				activate_law = law_type:law_militarized_police
				activate_law = law_type:law_total_separation
				add_modifier = {
					name = tyr_gcol_colonial_acceptance
				}
				every_scope_state = {
					set_state_type = incorporated
				}
			}
			activate_law = law_type:law_frontier_colonization
			activate_law = law_type:law_extraction_economy
			activate_law = law_type:law_colonial_administration
			activate_law = law_type:law_subjecthood
		}
	}

	create_diplomatic_pact = {
		country = c:MAF
		type = colony
	}

	c:MAF = {
		add_journal_entry = {
			type = je_tyr_gcol_colonial_charter
		}
	}

	tyr_gcol_refresh_colonies = yes

	trigger_event = {
		id = tyr_gcol_founding.1
	}
}

# Country scope: rewards use the completion snapshot, including colonies founded later.
tyr_gcol_apply_colonial_rewards = {
	if = {
		limit = {
			tyr_gcol_has_maf = yes
		}
		c:MAF = {
			remove_modifier = tyr_gcol_colonial_estates
		}
		if = {
			limit = {
				has_variable = tyr_gcol_completed
				tyr_gcol_final_score_value > 0
				c:MAF ?= {
					NOT = {
						has_modifier = tyr_gcol_maf_reward
					}
				}
			}
			c:MAF = {
				add_modifier = {
					name = tyr_gcol_maf_reward
					multiplier = root.tyr_gcol_final_score_value
				}
			}
		}
	}

	if = {
		limit = {
			tyr_gcol_has_dea = yes
		}
		c:DEA = {
			remove_modifier = tyr_gcol_colonial_estates
		}
		if = {
			limit = {
				has_variable = tyr_gcol_completed
				tyr_gcol_final_score_value > 0
				c:DEA ?= {
					NOT = {
						has_modifier = tyr_gcol_dea_reward
					}
				}
			}
			c:DEA = {
				add_modifier = {
					name = tyr_gcol_dea_reward
					multiplier = root.tyr_gcol_final_score_value
				}
			}
		}
	}
}

tyr_gcol_refresh_colonies = {
	if = {
		limit = {
			tyr_gcol_has_maf = yes
		}
		set_variable = {
			name = tyr_gcol_maf_country
			value = c:MAF
		}
	}
	else = {
		remove_variable = tyr_gcol_maf_country
	}

	if = {
		limit = {
			tyr_gcol_has_dea = yes
		}
		set_variable = {
			name = tyr_gcol_dea_country
			value = c:DEA
		}
	}
	else = {
		remove_variable = tyr_gcol_dea_country
	}

	if = {
		limit = {
			tyr_gcol_has_maf = yes
		}
		c:MAF = {
			if = {
				limit = {
					NOT = {
						has_journal_entry = je_tyr_gcol_colonial_charter
					}
				}
				add_journal_entry = {
					type = je_tyr_gcol_colonial_charter
				}
			}
		}
	}

	if = {
		limit = {
			tyr_gcol_has_dea = yes
		}
		c:DEA = {
			if = {
				limit = {
					NOT = {
						has_journal_entry = je_tyr_gcol_colonial_charter
					}
				}
				add_journal_entry = {
					type = je_tyr_gcol_colonial_charter
				}
			}
		}
	}
}

tyr_gcol_dissolve_administration = {
	save_scope_as = tyr_gcol_old_administration
	remove_modifier = tyr_gcol_maf_reward
	remove_modifier = tyr_gcol_dea_reward
	remove_modifier = tyr_gcol_colonial_acceptance

	every_scope_state = {
		limit = {
			owner = scope:tyr_gcol_old_administration
		}
		save_scope_as = tyr_gcol_successor_capital
		region = {
			save_scope_as = tyr_gcol_successor_region
		}
		scope:tyr_gcol_old_administration = {
			ordered_primary_culture = {
				order_by = "root.culture_percent_country(this)"
				position = 0
				save_scope_as = tyr_gcol_successor_culture
			}
			random_scope_culture = {
				limit = {
					scope:tyr_gcol_successor_capital.state_region = {
						is_homeland = prev
					}
				}
				save_scope_as = tyr_gcol_successor_culture
			}
			create_dynamic_country = {
				origin = scope:tyr_gcol_old_administration
				country_type = unrecognized
				tier = principality
				culture = scope:tyr_gcol_successor_culture
				capital = scope:tyr_gcol_successor_capital
				cede_state_trigger = {
					region = scope:tyr_gcol_successor_region
				}
				on_created = {
					every_scope_state = {
						set_state_type = incorporated
					}
				}
			}
		}
	}
}
