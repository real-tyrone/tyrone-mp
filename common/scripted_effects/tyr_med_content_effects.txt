tyr_resolve_spanish_content_effect = {
	if = {
		limit = { exists = c:MED c:MED = { tyr_med_spanish_origin = yes } }
		c:MED = { save_scope_as = tyr_spanish_content_scope }
	}
	else = {
		c:SPA ?= { save_scope_as = tyr_spanish_content_scope }
	}
}
