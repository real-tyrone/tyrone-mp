# Preserve the selected Zaibatsu amendment when the reform reward changes laws.
tyrone_activate_reformed_economic_law = {
	if = {
		limit = {
			active_law:lawgroup_economic_system = {
				has_amendment = amendment_type:amendment_zaibatsu_oligopoly_1
			}
		}
		activate_law = law_type:$LAW$
		active_law:lawgroup_economic_system = {
			add_amendment = {
				type = amendment_zaibatsu_oligopoly_1
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else_if = {
		limit = {
			active_law:lawgroup_economic_system = {
				has_amendment = amendment_type:amendment_zaibatsu_oligopoly_2
			}
		}
		activate_law = law_type:$LAW$
		active_law:lawgroup_economic_system = {
			add_amendment = {
				type = amendment_zaibatsu_oligopoly_2
				sponsor = PREV.ig:ig_industrialists
				cooldown = 240
			}
		}
	}
	else = {
		activate_law = law_type:$LAW$
	}
}
