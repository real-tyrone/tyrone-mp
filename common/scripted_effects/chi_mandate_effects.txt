chi_mandate_update_centralisation_modifier = {
	remove_modifier = chi_mandate_centralisation_low
	remove_modifier = chi_mandate_centralisation_mid
	remove_modifier = chi_mandate_centralisation_high
	remove_modifier = chi_mandate_centralisation_max

	if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 3
			}
		}
		add_modifier = chi_mandate_centralisation_low
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 4
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 6
			}
		}
		add_modifier = chi_mandate_centralisation_mid
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 7
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 8
			}
		}
		add_modifier = chi_mandate_centralisation_high
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 9
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 10
			}
		}
		add_modifier = chi_mandate_centralisation_max
	}
}
