chi_mandate_update_centralisation_modifier = {
	if = { limit = { has_modifier = chi_mandate_centralisation_low } remove_modifier = chi_mandate_centralisation_low }
	if = { limit = { has_modifier = chi_mandate_centralisation_mid } remove_modifier = chi_mandate_centralisation_mid }
	if = { limit = { has_modifier = chi_mandate_centralisation_high } remove_modifier = chi_mandate_centralisation_high }
	if = { limit = { has_modifier = chi_mandate_centralisation_max } remove_modifier = chi_mandate_centralisation_max }

	if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 3
			}
		}
		add_modifier = chi_mandate_centralisation_low
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 4
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 6
			}
		}
		add_modifier = chi_mandate_centralisation_mid
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 7
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 8
			}
		}
		add_modifier = chi_mandate_centralisation_high
	}
	else_if = {
		limit = {
			je:je_mandate_of_heaven = {
				"scripted_bar_progress(chi_centralisation_progress_bar)" >= 9
				"scripted_bar_progress(chi_centralisation_progress_bar)" <= 10
			}
		}
		add_modifier = chi_mandate_centralisation_max
	}
}

chi_early_army_refresh_stage_modifier = {
	if = { limit = { has_modifier = chi_early_army_stage_1 } remove_modifier = chi_early_army_stage_1 }
	if = { limit = { has_modifier = chi_early_army_stage_2 } remove_modifier = chi_early_army_stage_2 }
	if = { limit = { has_modifier = chi_early_army_stage_3 } remove_modifier = chi_early_army_stage_3 }
	if = { limit = { has_modifier = chi_early_army_stage_4 } remove_modifier = chi_early_army_stage_4 }

	if = {
		limit = { var:chi_early_army_stages = 1 }
		add_modifier = chi_early_army_stage_1
	}
	else_if = {
		limit = { var:chi_early_army_stages = 2 }
		add_modifier = chi_early_army_stage_2
	}
	else_if = {
		limit = { var:chi_early_army_stages = 3 }
		add_modifier = chi_early_army_stage_3
	}
	else_if = {
		limit = { var:chi_early_army_stages = 4 }
		add_modifier = chi_early_army_stage_4
	}
}

chi_early_army_advance_stage = {
	chi_early_army_refresh_stage_modifier = yes
}
