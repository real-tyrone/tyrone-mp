}

tyr_ga_resolve_first = {
 if = { limit = { global_var:tyr_ga_phase = 1 }
  set_variable = tyr_ga_won_first
  if = { limit = { has_variable = tyr_ga_austrian_actor }
   global_var:tyr_ga_north = { set_variable = tyr_ga_lost_leadership }
   set_global_variable = { name = tyr_ga_phase value = 3 }
  }
  else = {
   save_scope_as = tyr_ga_north_winner
   global_var:tyr_ga_austria = {
    every_scope_state = { limit = { has_variable = tyr_ga_southern_state } set_state_owner = scope:tyr_ga_north_winner set_state_type = incorporated }
    set_variable = tyr_ga_lost_leadership
   }
   change_tag = GER
   set_global_variable = { name = tyr_ga_phase value = 2 }
   global_var:tyr_ga_austria = { trigger_event = { id = tyr_ga.1 popup = yes } }
  }
 }
}
tyr_ga_return_to_danube = {

 set_global_variable = { name = tyr_ga_phase value = 5 }
 remove_variable = tyr_ga_active

  limit = { exists = c:GAL c:GAL = { is_subject_of = ROOT } }
  annex_with_incorporation = c:GAL
 }
 every_subject_or_below = {
  limit = { is_subject_of = ROOT country_has_primary_culture = cu:north_italian }
  change_subject_type = subject_type_crown_land
 }
 remove_variable = finished_hungary_var
 remove_variable = established_austria_hungary_var
 remove_variable = reclaiming_empire_aus_var
 if = { limit = { NOT = { has_journal_entry = je_matter_of_hungary } } add_journal_entry = { type = je_matter_of_hungary } }
 if = { limit = { exists = c:HUN } trigger_event = { id = austria_events.4 popup = yes } }
 hidden_effect = {
  global_var:tyr_ga_north ?= {
   if = {
    limit = { NOT = { has_variable = tyr_ga_settlement_event_queued } }
    set_variable = tyr_ga_settlement_event_queued
    trigger_event = { id = tyr_ga.3 days = 1 popup = yes }
   }
  }
 }
}
tyr_ga_resolve_final = {
 global_var:tyr_ga_austria ?= { tyr_ga_release_radetzky = yes }
 if = { limit = { global_var:tyr_ga_phase = 3 }
  set_global_variable = { name = tyr_ga_phase value = 4 }
  if = { limit = { has_variable = tyr_ga_austrian_actor }
   set_variable = tyr_ga_austrian_final_victory
   remove_variable = tyr_ga_active
   global_var:tyr_ga_north ?= { ROOT = { annex_with_incorporation = prev } }
   change_tag = GER
   add_primary_culture = cu:north_german
   trigger_event = { id = austria_germany.1 days = 1 popup = yes }
  }
  else = {
   save_scope_as = tyr_ga_northern_victor
   global_var:tyr_ga_austria = {
    # Transfer every directly held German homeland, including additional acquisitions.
    every_scope_state = {
     limit = {
      state_region = {
       OR = {
        is_homeland = cu:south_german
        is_homeland = cu:north_german
       }
      }
     }
     set_state_owner = scope:tyr_ga_northern_victor
     set_state_type = incorporated
    }
    # Preserve the Austrian player, ruler, institutions, and treasury while moving the crown to Budapest.
    tyr_ga_exile = yes
   }
   if = { limit = { NOT = { country_definition = cd:GER } } change_tag = GER }
  }
 }
}

# Called by the Austrian south-German unification event, in Austrian country scope.
# Use the same primary-culture/rank exclusions as the existing German confederation events.
# The Prussian country object (including the arc's saved reference) survives the tag change.
tyr_ga_form_north_immediately = {
 c:PRU ?= {
  if = {
   limit = { NOT = { exists = c:NGF } }
   hidden_effect = { save_scope_as = tyr_ga_north_unifier }
   every_country = {
    limit = {
     country_has_primary_culture = cu:north_german
     country_rank < rank_value:great_power
     capital = { is_in_geographic_region = geographic_region_german_confederation }
     NOR = {
      THIS = scope:tyr_ga_north_unifier
      THIS = ROOT
      c:HOH ?= THIS
      is_subject_type = subject_type_colony
      is_subject_type = subject_type_dominion
      is_subject_type = subject_type_chartered_company
     }
    }
    scope:tyr_ga_north_unifier = { annex_with_incorporation = PREV }
   }
   hidden_effect = { set_variable = je_german_confederate_unifier }
   change_tag = NGF
  }
 }
}

# Country scope: protect only the historical Radetzky and retain a direct
# character reference, so cleanup still reaches him after a country/tag change.
tyr_ga_protect_radetzky = {
 hidden_effect = {
  if = {
   limit = { NOT = { has_variable = tyr_ga_radetzky_character } }
   random_scope_character = {
    limit = { has_template = aus_general_radetzky_von_radetz }
    save_scope_as = tyr_ga_radetzky
    set_character_immortal = yes
   }
   if = {
    limit = { exists = scope:tyr_ga_radetzky }
    set_variable = { name = tyr_ga_radetzky_character value = scope:tyr_ga_radetzky }
   }
  }
 }
}
tyr_ga_release_radetzky = {
 hidden_effect = {
  var:tyr_ga_radetzky_character ?= { set_character_immortal = no }
  remove_variable = tyr_ga_radetzky_character
 }
}
