# Country scope. Remove this amendment only; preserve all sibling amendments.
tyrone_promise_remove_naval_industry_restriction = {
	if = {
		limit = { any_active_law = { has_amendment = amendment_type:amendment_royal_navy_supremacy } }
		random_active_law = {
			limit = { has_amendment = amendment_type:amendment_royal_navy_supremacy }
			random_scope_amendment = {
				limit = { amendment_type:amendment_royal_navy_supremacy ?= this.type }
				remove_amendment = yes
			}
		}
	}
	# Also end the law-enactment reapplication path if no amendment is active.
	remove_variable = bic_naval_restricted_var
}
