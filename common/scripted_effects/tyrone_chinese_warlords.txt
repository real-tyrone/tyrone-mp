# Country scope; root is the Chinese government.

tyr_chi_release_GNG = {
	save_scope_as = tyr_chi_central
	if = {
		limit = { NOT = { exists = c:GNG } any_scope_state = { OR = { state_region = s:STATE_GUANGXI } } }
		random_scope_state = { limit = { OR = { state_region = s:STATE_GUANGXI } } save_scope_as = tyr_chi_seed }
		create_country = { tag = GNG origin = this state = scope:tyr_chi_seed on_created = {
				activate_law = law_type:law_presidential_republic
				activate_law = law_type:law_autocracy
				set_variable = warlord_state
			} }
		every_scope_state = { limit = { OR = { state_region = s:STATE_GUANGXI } } set_state_owner = c:GNG }
		create_diplomatic_pact = { country = c:GNG type = tyr_chinese_warlord }
		c:GNG = {
			add_liberty_desire = -100
			add_liberty_desire = 50
			change_relations = { country = scope:tyr_chi_central value = 50 }
		}
	}
}

tyr_chi_fragment = {
 every_subject_or_below = {
  limit = { chi_qing_administrative_subject = yes }
  change_subject_type = subject_type_tyr_chinese_warlord
  activate_law = law_type:law_presidential_republic
  activate_law = law_type:law_autocracy
  set_variable = warlord_state
  remove_variable = tyr_ssm_started
  add_liberty_desire = -100
  add_liberty_desire = 50
 }

	set_global_variable = china_shatters
	set_global_variable = { name = chinese_central_government value = this }
	set_variable = tyr_chi_fragmented
	tyr_chi_release_GNG = yes
	tyr_chi_release_YUN = yes
	tyr_chi_release_XIN = yes
	tyr_chi_release_XIB = yes
	tyr_chi_release_MGL = yes
	tyr_chi_release_TUV = yes
	tyr_chi_release_ALI = yes
	tyr_chi_release_SIH = yes
	tyr_chi_release_SHA = yes
	tyr_chi_release_ZHI = yes
	add_modifier = { name = tyr_chi_initial_accommodation years = 5 }
	add_journal_entry = { type = je_tyr_china_reintegration }
}
tyr_chi_republic_cost = {
	every_interest_group = {
		limit = { law_stance = { law = law_type:law_monarchy value > neutral } }
		add_modifier = { name = tyr_chi_dynastic_opposition years = 5 }
	}
	add_radicals = { value = 0.05 }
}
