# India country scope; preserve the country object when opening the contextless JE.
tyrone_open_independent_nation = {
	if = {
		limit = { NOT = { has_variable = tyrone_nim_started } }
		save_scope_as = csr_india
		save_scope_as = nim_india
		add_contextless_journal_entry = je_tyrone_csr_independent_path
	}
}

# Country scope. The existing reward option has already applied its effects.
tyrone_after_honorable_company = {
	if = {
		limit = { is_subject = no }
		trigger_event = { id = tyrone_independent_route.1 popup = yes }
	}
	else = { tyrone_company_self_rule_start = yes }
}
