# No scope. Called once on activation and then only monthly (or explicit withdrawal).
tyr_monroe_update = {
 every_country = {
  limit = { has_modifier = tyr_monroe_trade }
  remove_modifier = tyr_monroe_trade
 }
 every_country = {
  limit = { tyr_monroe_usa = yes }
  set_variable = tyr_monroe_usa_identity
  save_scope_as = tyr_monroe_america
 }
 every_country = {
  limit = { tyr_monroe_gbr = yes }
  set_variable = tyr_monroe_gbr_identity
  save_scope_as = tyr_monroe_britain
 }
 set_global_variable = { name = tyr_monroe_protected_count value = 0 }
 set_global_variable = { name = tyr_monroe_invaded_count value = 0 }
 every_country = {
  every_scope_state = {
   limit = { tyr_monroe_area = yes is_largest_state_in_region = yes }
   if = {
    limit = { tyr_monroe_protected = yes }
    change_global_variable = { name = tyr_monroe_protected_count add = 1 }
   }
   else = { change_global_variable = { name = tyr_monroe_invaded_count add = 1 } }
  }
 }
 set_global_variable = {
  name = tyr_monroe_trade_points
  value = { value = global_var:tyr_monroe_protected_count subtract = { value = global_var:tyr_monroe_invaded_count multiply = 2 } max = 50 }
 }
 if = {
  limit = {
   exists = scope:tyr_monroe_america
   OR = {
    has_global_variable = tyr_monroe_british_withdrawal
    NOT = { exists = scope:tyr_monroe_britain }
    scope:tyr_monroe_america = { has_variable = america_interventionism_points_var var:america_interventionism_points_var >= 5 }
   }
  }
  set_global_variable = { name = tyr_monroe_american_guarantor value = 1 }
  scope:tyr_monroe_america = { add_modifier = { name = tyr_monroe_trade multiplier = global_var:tyr_monroe_trade_points } }
 }
 else_if = {
  limit = { exists = scope:tyr_monroe_britain NOT = { has_global_variable = tyr_monroe_british_withdrawal } }
  set_global_variable = { name = tyr_monroe_american_guarantor value = 0 }
  scope:tyr_monroe_britain = { add_modifier = { name = tyr_monroe_trade multiplier = global_var:tyr_monroe_trade_points } }
 }
}

# No scope. Fired once by the JE failure outcome; native decay scales every effect.
tyr_monroe_unsa_failure = {
 if = {
  limit = { c:UNSA ?= { power_bloc ?= { power_bloc_leader ?= { tyr_monroe_usa = yes } } } }

   limit = { tyr_monroe_usa = yes }

  }

  every_country = {
   limit = { tyr_monroe_usa = yes }

  }
  c:UNSA ?= { add_modifier = { name = tyr_monroe_unsa_ascendant years = 10 is_decaying = yes } }

   limit = { NOT = { exists = c:UNSA } }

# Contextless JE scope; participation latches, all accounting is monthly.
tyr_ca_refresh = {
 if = { limit = { game_date >= 1850.1.1 }

 set_global_variable = { name = tyr_ca_color_counter value = 0 }
 every_country = {
  limit = { any_scope_state = { tyr_ca_state = yes } }
  change_global_variable = { name = tyr_ca_color_counter add = 1 }
  set_variable = { name = tyr_ca_map_color value = global_var:tyr_ca_color_counter }
 }

 every_country = {
  limit = { game_date < 1870.1.1 country_rank >= rank_value:great_power has_strategic_region_interest_tier = { strategic_region = sr:region_central_america value >= 2 } }
  set_variable = tyr_ca_participant
  if = { limit = { NOT = { is_involved_in_journal_entry = je_tyr_ca_contest } } scope:journal_entry = { add_involved_country = prev } }
 }
 set_global_variable = { name = tyr_ca_best value = 0 }
 set_global_variable = { name = tyr_ca_ties value = 0 }
 remove_global_variable = tyr_ca_winner
 every_country = {
  limit = { has_variable = tyr_ca_participant }
  if = { limit = { has_modifier = tyr_ca_leading } remove_modifier = tyr_ca_leading }
  set_variable = { name = tyr_ca_points value = 0 }
 }
 every_country = {
  every_scope_state = {
   limit = { tyr_ca_state = yes is_largest_state_in_region = yes }
   if = { limit = { owner = { is_subject = no has_variable = tyr_ca_participant } } owner = { change_variable = { name = tyr_ca_points add = 1 } } }
   else_if = { limit = { owner = { is_subject = yes top_overlord = { has_variable = tyr_ca_participant } } } owner.top_overlord = { change_variable = { name = tyr_ca_points add = 1 } } }
  }
 }
 every_country = {
  limit = { has_variable = tyr_ca_participant var:tyr_ca_points > 0 }
  if = {
   limit = { var:tyr_ca_points > global_var:tyr_ca_best }
   set_global_variable = { name = tyr_ca_best value = var:tyr_ca_points }
   set_global_variable = { name = tyr_ca_ties value = 1 }
   set_global_variable = { name = tyr_ca_winner value = this }
  }
  else_if = {
   limit = { var:tyr_ca_points = global_var:tyr_ca_best }
   change_global_variable = { name = tyr_ca_ties add = 1 }
   if = { limit = { gdp > global_var:tyr_ca_winner.gdp } set_global_variable = { name = tyr_ca_winner value = this } }
  }
 }
 if = { limit = { global_var:tyr_ca_ties = 1 exists = global_var:tyr_ca_winner } global_var:tyr_ca_winner = { add_modifier = { name = tyr_ca_leading } } }

 }
}

}

# Country scope: monthly recalculation, never cumulative modifiers.
tyr_dream_refresh = {
 set_variable = { name = tyr_dream_positive value = 0 }
 set_variable = { name = tyr_dream_negative value = 0 }
 remove_variable = tyr_dream_partner
 remove_global_variable = tyr_dream_status_country
 every_country = {
  remove_variable = tyr_dream_target
  if = {
   limit = { tyr_dream_european_power = yes }
   if = {
    limit = { tyr_dream_pro = yes }
    root = { change_variable = { name = tyr_dream_positive add = 1 } }
    if = {
     limit = { tyr_dream_rights = yes root = { gdp_ownership_ratio = { target = scope:tyr_dream_investment_country value >= 0.10 } } }
     root = { change_variable = { name = tyr_dream_positive add = 1 } }
    }
   }
   if = {
    limit = { tyr_dream_anti = yes }
    root = { change_variable = { name = tyr_dream_negative add = 1 } }
   }
  }
  if = {
   limit = {
    NOT = { this = ROOT }
    tyr_dream_pro = yes
    OR = {
     tyr_dream_rights = yes
     AND = { this = c:CHI any_scope_state = { any_neighbouring_state = { owner = ROOT is_treaty_port = yes } } }
    }
   }
   set_variable = tyr_dream_target
  }
 }
 ordered_country = {
  limit = { NOT = { this = ROOT } tyr_dream_pro = yes relations:ROOT >= 30 }
  order_by = gdp position = 0 check_range_bounds = no
  set_global_variable = { name = tyr_dream_status_country value = this }
  root = { set_variable = { name = tyr_dream_partner value = prev } }
 }
 remove_modifier = tyr_dream_attraction
 remove_modifier = tyr_dream_repulsion
 if = { limit = { trigger_if = { limit = { has_variable = tyr_dream_positive } var:tyr_dream_positive > 0 } trigger_else = { always = no } } add_modifier = { name = tyr_dream_attraction multiplier = var:tyr_dream_positive } }
 if = { limit = { trigger_if = { limit = { has_variable = tyr_dream_negative } var:tyr_dream_negative > 0 } trigger_else = { always = no } } add_modifier = { name = tyr_dream_repulsion multiplier = var:tyr_dream_negative } }
}

# American country scope. Two independent monthly rolls per funded GP/Major worldwide.
tyr_dream_funded_lobby_rolls = {
 every_country = {
  limit = {
   NOT = { this = ROOT }
   OR = { country_rank = rank_value:great_power country_rank = rank_value:major_power }
   save_temporary_scope_as = tyr_dream_funded_country
   root = { has_diplomatic_pact = { who = scope:tyr_dream_funded_country type = fund_lobbies } }
  }
  if = {
   limit = { NOT = { tyr_dream_pro = yes } }
   random = {
    chance = 5
    if = {
     limit = {
      any_interest_group = { is_interest_group_type = ig_industrialists tyr_dream_lobby_founder_eligible = yes }
     }
     random_interest_group = {
      limit = { is_interest_group_type = ig_industrialists tyr_dream_lobby_founder_eligible = yes }
      save_scope_as = tyr_dream_lobby_founder
     }
     create_political_lobby = { type = lobby_pro_country target = ROOT add_interest_group = scope:tyr_dream_lobby_founder lobby_formation_reason = diplomacy }
    }
    else_if = {
     limit = { any_interest_group = { tyr_dream_lobby_founder_eligible = yes } }
     random_interest_group = { limit = { tyr_dream_lobby_founder_eligible = yes } save_scope_as = tyr_dream_lobby_founder }
     create_political_lobby = { type = lobby_pro_country target = ROOT add_interest_group = scope:tyr_dream_lobby_founder lobby_formation_reason = diplomacy }
    }
   }
  }
  if = {
   limit = { relations:ROOT >= 10 tyr_dream_anti = yes }
   random = {
    chance = 3
    every_political_lobby = {
     limit = { is_political_lobby_type = lobby_anti_country target = ROOT }
     disband_political_lobby = yes
    }
   }
  }
 }
}
