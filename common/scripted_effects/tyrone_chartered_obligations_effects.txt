# Country scope. Select the incoming law explicitly, as with patent reliance.
# Existing active amendments seed the flags when loading an older campaign.
tyrone_india_reapply_charter_restrictions = {
	if = {
		limit = { is_india_tag_trigger = yes }
		if = {
			limit = {
				OR = {
					has_variable = bic_manufacturing_monopoly_var
					any_active_law = { has_amendment = amendment_type:amendment_british_manufacturing_monopoly }
				}
			}
			set_variable = bic_manufacturing_monopoly_var
			random_law = {
				limit = {
					type = law_type:$LAW$
					NOT = { has_amendment = amendment_type:amendment_british_manufacturing_monopoly }
				}
				add_amendment = { type = amendment_british_manufacturing_monopoly sponsor = prev.ig:ig_industrialists }
			}
		}
		if = {
			limit = {
				OR = {
					has_variable = bic_cotton_guarantees_var
					any_active_law = { has_amendment = amendment_type:amendment_cotton_export_guarantees }
				}
			}
			set_variable = bic_cotton_guarantees_var
			random_law = {
				limit = {
					type = law_type:$LAW$
					NOT = { has_amendment = amendment_type:amendment_cotton_export_guarantees }
				}
				add_amendment = { type = amendment_cotton_export_guarantees sponsor = prev.ig:ig_landowners }
			}
		}
	}
}
