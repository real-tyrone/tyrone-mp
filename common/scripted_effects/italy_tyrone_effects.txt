ita_industry_calculate_sluggish_effect = {
	change_variable = {
			name = agri_padana_ita_var
			add = var:investment_in_padana_ita_var
		}
	if = {
		limit = { 
			var:investment_in_north_ita_var = var:investment_in_south_ita_var 
			var:investment_in_north_ita_var > 0
			var:investment_in_south_ita_var > 0
		}
		change_variable = {
			name = industrious_north_ita_var
			add = 1
			}
		change_variable = {
			name = sluggish_south_ita_var
			add = 1
			}
	}
	else = {
		change_variable = {
			name = industrious_north_ita_var
			add = var:investment_in_north_ita_var
			}
		change_variable = {
			name = industrious_north_ita_var
			subtract = var:investment_in_south_ita_var
			}

		change_variable = {
			name = sluggish_south_ita_var
			add = var:investment_in_south_ita_var
			}
		change_variable = {
			name = sluggish_south_ita_var
			subtract = var:investment_in_north_ita_var
			}
		}



	clamp_variable = { name = sluggish_south_ita_var max = 20 min = -20 }
	clamp_variable = { name = industrious_north_ita_var max = 20 min = -20 }
	clamp_variable = { name = agri_padana_ita_var max = 50 min = 0 }

	set_variable = {
		name = investment_in_north_ita_var
		value = 0
	}
	set_variable = {
		name = investment_in_padana_ita_var
		value = 0
	}
	set_variable = {
		name = investment_in_south_ita_var
		value = 0
	}

}

# Root: country. Refresh the multiplier only when sentiment changes or the modifier is missing.
ita_refresh_sentiment_modifier = {

	if = {
		limit = {
			NOT = {
				has_variable = italian_sentiment_var
			}
		}
		set_variable = {
			name = italian_sentiment_var
			value = 10
		}
	}

	if = {
		limit = {
			trigger_if = {
				limit = {
					has_variable = ita_sentiment_modifier_value
					has_modifier = italy_tension_modifier
				}
				NOT = {
					var:ita_sentiment_modifier_value = var:italian_sentiment_var
				}
			}
			trigger_else = {
				always = yes
			}
		}
		if = {
			limit = {
				has_modifier = italy_tension_modifier
			}
			remove_modifier = italy_tension_modifier
		}
		add_modifier = {
			name = italy_tension_modifier
			multiplier = {
				value = var:italian_sentiment_var
			}
		}
		set_variable = {
			name = ita_sentiment_modifier_value
			value = var:italian_sentiment_var
		}
	}

	set_global_variable = {
		name = ita_risorgimento_sentiment
		value = var:italian_sentiment_var
	}

	set_global_variable = {
		name = ita_risorgimento_sardinia
		value = this
	}

}

# Root: none. scope:journal_entry is the shared Risorgimento JE. Reuse the existing Mazzini.
ita_risorgimento_sync_mazzini = {

	if = {
		limit = {
			NOT = {
				has_global_variable = ita_risorgimento_mazzini
			}
		}
		random_character = {
			limit = {
				has_variable = mazzini_var
			}
			set_global_variable = {
				name = ita_risorgimento_mazzini
				value = this
			}
		}
	}

	if = {
		limit = {
			has_global_variable = ita_risorgimento_mazzini
		}
		scope:journal_entry = {
			clear_variable_list = com_risorgimento_mazzini
			add_to_variable_list = {
				name = com_risorgimento_mazzini
				target = global_var:ita_risorgimento_mazzini
			}
		}
	}

}



# Root: none, scope:journal_entry = shared diplomatic stance JE.
ita_diplo_sync_characters = {

	c:SAR ?= {
		ruler ?= {
			save_scope_as = ita_diplo_monarch
		}
		random_scope_character = {
			limit = {
				has_variable = italy_cavour
			}
			save_scope_as = ita_diplo_cavour
		}
	}

	scope:journal_entry = {
		clear_variable_list = com_journal_characters
	}

	if = {
		limit = {
			exists = scope:ita_diplo_monarch
		}
		scope:journal_entry = {
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:ita_diplo_monarch
			}
		}
	}

	if = {
		limit = {
			exists = scope:ita_diplo_cavour
		}
		scope:journal_entry = {
			add_to_variable_list = {
				name = com_journal_characters
				target = scope:ita_diplo_cavour
			}
		}
	}

	clear_global_variable_list = ita_diplo_france_flags

	c:FRA ?= {
		add_to_global_variable_list = {
			name = ita_diplo_france_flags
			target = this
		}
	}

	clear_global_variable_list = ita_diplo_austria_flags

	if = {
		limit = {
			exists = c:AUS
		}
		c:AUS = {
			add_to_global_variable_list = {
				name = ita_diplo_austria_flags
				target = this
			}
		}
	}

	else = {
		c:KUK ?= {
			add_to_global_variable_list = {
				name = ita_diplo_austria_flags
				target = this
			}
		}
	}

}
