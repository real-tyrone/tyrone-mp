# Country scope. LAW is the Indian education law being enacted.
tyrone_india_reapply_patent_reliance = {
	if = {
		limit = {
			is_china_tag_trigger = yes
			has_variable = chi_confucian_focus_amendment_active
		}
		# Select the destination law explicitly, not the outgoing active law.
		random_law = {
			limit = {
				type = law_type:$LAW$
				NOT = {
					has_amendment = amendment_confussion_focus
				}
			}
			add_amendment = {
				type = amendment_confussion_focus
				sponsor = PREV.ig:ig_devout
			}
		}
	}
}
