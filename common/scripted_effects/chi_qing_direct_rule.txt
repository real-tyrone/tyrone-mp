# Country scope: Qing. One-time migration for saves made before direct Manchurian rule.
chi_qing_restore_direct_manchuria = {
 if = {
  limit = {
   c:CHI ?= this
   has_variable = chi_qing_1836_setup
   NOT = { has_variable = chi_qing_direct_manchuria_v1 }
   NOT = { has_variable = tyr_chi_fragmented }
   is_at_war = no
   is_diplomatic_play_committed_participant = no
   c:MCH ?= { is_at_war = no is_diplomatic_play_committed_participant = no }
   c:MGL ?= { is_at_war = no is_diplomatic_play_committed_participant = no }
  }
  if = {
   limit = { c:MCH ?= { is_direct_subject_of = c:CHI chi_qing_administrative_subject = yes } }
   annex = c:MCH
  }
  chi_qing_retain_eastern_hinggan = yes
  set_variable = chi_qing_direct_manchuria_v1
 }
}
