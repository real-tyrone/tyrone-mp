# Country scope: the regime-change enforcer. Requires scope:target_country (country).
# The target receives every law required by has_all_revolutionary_laws.
tyrone_enforce_revolutionary_regime_laws_effect = {
	if = {
		limit = {
			has_law_or_variant = law_type:law_council_republic
		}
		scope:target_country ?= {
			if = {
				limit = { NOT = { has_law_or_variant = law_type:law_council_republic } }
				activate_law = law_type:law_council_republic
			}
			if = {
				limit = { NOT = { has_law_or_variant = law_type:law_factory_councils } }
				activate_law = law_type:law_factory_councils
			}
			if = {
				limit = { NOT = { has_law_or_variant = law_type:law_communal_administration } }
				activate_law = law_type:law_communal_administration
			}
			if = {
				limit = { NOT = { has_law_or_variant = law_type:law_revolutionary_guard } }
				activate_law = law_type:law_revolutionary_guard
			}
		}
	}
}
