tyrone_cp_start = {
	if = {
		limit = {
			has_variable = tyrone_nim_civil_war_done
			NOT = {
				has_variable = tyrone_cp_started
			}
		}
		set_variable = tyrone_cp_started
		add_journal_entry = {
			type = je_tyrone_cp_village
		}
	}
	tyrone_cp_apply_amendments = yes
	tyrone_ch_start = yes
}
tyrone_cp_start_heavy = {
	if = {
		limit = {
			has_variable = tyrone_cp_started
			NOR = {
				has_variable = tyrone_cp_plan_active
				has_variable = tyrone_cp_heavy_done
			}
		}
		hidden_effect = {
			set_variable = {
				name = tyrone_cp_target
				value = tyrone_cp_heavy_target
			}
			set_variable = tyrone_cp_plan_active
		}
		add_modifier = {
			name = tyrone_cp_heavy_mobilization
			days = tyrone_cp_five_years
		}
		add_journal_entry = {
			type = je_tyrone_cp_plan_heavy
		}
	}
}
tyrone_cp_resolve_heavy = {
	if = {
		limit = {
			has_variable = tyrone_cp_plan_active
		}
		remove_variable = tyrone_cp_plan_active
		if = {
			limit = {
				has_modifier = tyrone_cp_heavy_mobilization
			}
			remove_modifier = tyrone_cp_heavy_mobilization
		}
		if = {
			limit = {
				tyrone_cp_heavy_levels >= var:tyrone_cp_target
			}
			set_variable = tyrone_cp_heavy_done

			trigger_event = {
				id = tyrone_cp.1
				popup = yes
			}
		}
		else = {
			trigger_event = {
				id = tyrone_cp.11
				popup = yes
			}
		}
	}
}

tyrone_cp_check_famine = {
	ordered_scope_state = {
		limit = {
			tyrone_cp_famine_candidate = yes
		}
		order_by = tyrone_cp_state_unemployed
		position = 0
		check_range_bounds = no
		save_scope_as = cp_famine_state
		state_region = {
			save_scope_as = cp_famine_region
			set_variable = {
				name = tyrone_cp_famine_pending
				days = 10
			}
		}
		owner = {
			trigger_event = {
				id = tyrone_cp.24
				popup = yes
			}
		}
	}
}
tyrone_cp_begin_famine = {
	if = {
		limit = {
			has_variable = tyrone_cp_forced_active
			exists = scope:cp_famine_region
		}
		save_scope_as = cp_india
		if = {
			limit = {
				tyrone_raj_famine_duration_days >= 548
			}
			tyrone_cp_apply_disease = {
				DAYS = 548
			}
		}
		else_if = {
			limit = {
				tyrone_raj_famine_duration_days >= 517
			}
			tyrone_cp_apply_disease = {
				DAYS = 517
			}
		}
		else_if = {
			limit = {
				tyrone_raj_famine_duration_days >= 487
			}
			tyrone_cp_apply_disease = {
				DAYS = 487
			}
		}
		else_if = {
			limit = {
				tyrone_raj_famine_duration_days >= 457
			}
			tyrone_cp_apply_disease = {
				DAYS = 457
			}
		}
		else_if = {
			limit = {
				tyrone_raj_famine_duration_days >= 426
			}
			tyrone_cp_apply_disease = {
				DAYS = 426
			}
		}
		else_if = {
			limit = {
				tyrone_raj_famine_duration_days >= 396
			}
			tyrone_cp_apply_disease = {
				DAYS = 396
			}
		}
		else = {
			tyrone_cp_apply_disease = {
				DAYS = 365
			}
		}
	}
}
tyrone_cp_apply_disease = {
	scope:cp_famine_region = {
		remove_variable = tyrone_cp_famine_pending
		set_variable = tyrone_cp_famine_active
		set_variable = tyrone_cp_famine_end
		start_harvest_condition_with_params = {
			type = disease_outbreak
			intensity = 7
			duration = $DAYS$
		}
		every_scope_state = {
			add_modifier = {
				name = tyrone_cp_famine_mortality
				days = $DAYS$
			}
		}
	}
	hidden_effect = {
		scope:cp_famine_state = {
			tyrone_cp_open_local_famine = yes
		}
	}
}

tyrone_cp_maintain_famine_cooldowns = {
	every_state_region = {
		limit = {
			has_variable = tyrone_cp_famine_end
			NOT = {
				has_harvest_condition = disease_outbreak
			}
		}
		remove_variable = tyrone_cp_famine_end
		remove_variable = tyrone_cp_famine_active
		set_variable = {
			name = tyrone_cp_famine_cooldown
			days = tyrone_cp_three_years
		}
	}
}
