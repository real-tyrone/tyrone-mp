# Country scope. LAW is the naval-model law being enacted.
tyrone_india_reapply_naval_restrictions = {
	if = {
		limit = {
			is_india_tag_trigger = yes
			has_variable = bic_naval_restricted_var
		}
		random_law = {
			limit = {
				type = law_type:$LAW$
				NOT = {
					has_amendment = amendment_type:amendment_royal_navy_supremacy
				}
			}
			add_amendment = {
				type = amendment_royal_navy_supremacy
				sponsor = PREV.ig:ig_industrialists
			}
		}
	}
}
