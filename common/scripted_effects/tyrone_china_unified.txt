 every_subject_or_below = {
  limit = { tyr_china_unification_target = yes }
  every_scope_state = {
   state_region = { save_temporary_scope_as = tyr_china_unified_region }
   scope:tyr_china_unifier = {
    add_to_variable_list = { name = tyr_china_unified_regions target = scope:tyr_china_unified_region }
   }
  }
 }
