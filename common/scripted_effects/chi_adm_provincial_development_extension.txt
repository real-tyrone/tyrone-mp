chi_adm_provincial_development_ask_sponsor = {
	if = {
		limit = {
			chi_adm_target_valid = yes
			any_scope_state = {
				has_variable = chi_adm_provincial_development_project
				NOT = {
					has_variable = chi_adm_provincial_development_project_sponsored
				}
			}
			NOT = {
				has_variable = chi_adm_provincial_development_sponsor_case
			}
			NOT = {
				has_variable = chi_adm_provincial_development_sponsor_retry
			}
		}
		set_variable = chi_adm_provincial_development_sponsor_case
		save_scope_as = chi_adm_provincial_development_province
		c:CHI = {
			save_scope_as = chi_adm_provincial_development_court
			if = {
				limit = { is_ai = yes }
				trigger_event = { id = chi_adm_provincial_development_ext.1 popup = yes }
			}
			else = { chi_adm_provincial_development_notify_court = yes }
		}
	}
}
chi_adm_provincial_development_offer_sponsor = {
	if = {
		limit = {
			chi_adm_target_valid = yes
			has_variable = chi_adm_provincial_development_sponsor_case
			NOT = { has_variable = chi_adm_provincial_development_sponsor_offer_cooldown }
			NOT = {
				var:chi_adm_provincial_development_sponsor_case = 2
			}
			c:CHI = {
				gold_reserves >= 0
				fixed_income > 0
			}
		}
		set_variable = {
			name = chi_adm_provincial_development_sponsor_case
			value = 2
		}
		set_variable = { name = chi_adm_provincial_development_sponsor_offer_cooldown months = 1 }
		c:CHI = {
			save_scope_as = chi_adm_provincial_development_court
		}
		trigger_event = {
			id = chi_adm_provincial_development_ext.2
			popup = yes
		}
	}
}
chi_adm_provincial_development_accept_sponsor = {
	if = {
		limit = {
			chi_adm_target_valid = yes
			var:chi_adm_provincial_development_sponsor_case = 2
		}
		add_modifier = {
			name = chi_adm_provincial_development_project_inspection
		}
		every_scope_state = {
			limit = {
				has_variable = chi_adm_provincial_development_project
				NOT = {
					has_variable = chi_adm_provincial_development_project_sponsored
				}
			}
			set_variable = chi_adm_provincial_development_project_sponsored
			if = {
				limit = {
					var:chi_adm_provincial_development_project_tier = 1
				}
				change_variable = {
					name = chi_adm_provincial_development_project_left
					subtract = 3
				}
				set_variable = {
					name = chi_adm_provincial_development_project_total
					value = 9
				}
			}
			else = {
				change_variable = {
					name = chi_adm_provincial_development_project_left
					subtract = 6
				}
