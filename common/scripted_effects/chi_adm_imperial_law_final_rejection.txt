   limit = { NOT = { has_variable = chi_adm_imperial_law_refusal_used } }

   chi_adm_adjust_autonomy = { AMOUNT = chi_adm_imperial_law_refusal_autonomy }

  set_variable = chi_adm_imperial_law_court_answered

  remove_variable = chi_adm_imperial_law_delay_decision

 }
}
