tyr_heavenly_enter_hierarchy = {
	if = { limit = { NOT = { has_social_hierarchy = tyr_heavenly_hierarchy } }
		if = { limit = { has_social_hierarchy = british_indian_caste_system } set_variable = tyr_heavenly_previous_castes }
		if = { limit = { has_social_hierarchy = edo_system } set_variable = tyr_heavenly_previous_edo }
		if = { limit = { has_social_hierarchy = tyrone_company_hierarchy } set_variable = tyr_heavenly_previous_company }
		set_social_hierarchy = tyr_heavenly_hierarchy
	}
	recalculate_pop_ig_support = yes
}
tyr_heavenly_leave_hierarchy = {
	if = { limit = { has_variable = tyr_heavenly_previous_castes } set_social_hierarchy = british_indian_caste_system }
	else_if = { limit = { has_variable = tyr_heavenly_previous_edo } set_social_hierarchy = edo_system }
	else_if = { limit = { has_variable = tyr_heavenly_previous_company } set_social_hierarchy = tyrone_company_hierarchy }
	else = { set_social_hierarchy = default_social_hierarchy }
	remove_variable = tyr_heavenly_previous_castes
	remove_variable = tyr_heavenly_previous_edo
	remove_variable = tyr_heavenly_previous_company
	recalculate_pop_ig_support = yes
}

tyr_heavenly_begin_war = {
 tyr_heavenly_prepare_governors = yes
	c:TPG = { create_diplomatic_play = { name = taiping_rebellion target_country = c:CHI type = dp_tyr_heavenly_civil_war } }
	random_diplomatic_play = {
		limit = { is_diplomatic_play_type = dp_tyr_heavenly_civil_war initiator_is = c:TPG target_is = c:CHI }
  save_temporary_scope_as = tyr_heavenly_war
  c:CHI = {
   every_in_list = {
    variable = tyr_heavenly_defectors
    limit = { has_variable = tyr_heavenly_defected is_subject = no }
    scope:tyr_heavenly_war = { remove_target_backers = { prev } add_initiator_backers = { prev } }
   }
   clear_variable_list = tyr_heavenly_defectors
  }
		add_diplomatic_play_war_support = { target = c:TPG value = 50 }
		set_war = yes
  initiator = { set_variable = tyr_heavenly_war_started }
  c:CHI ?= { set_variable = tyr_heavenly_war_started }
	}

 if = {
  limit = { NOT = { has_variable = tyr_heavenly_war_started } }
  every_in_list = {
   variable = tyr_heavenly_defectors
   limit = { has_variable = tyr_heavenly_former_pact is_subject = no }
   set_variable = tyr_heavenly_neutral
  }
  tyr_heavenly_settle_governors = yes
  clear_variable_list = tyr_heavenly_defectors
  set_variable = tyr_heavenly_interrupted
  c:TPG ?= { set_variable = tyr_heavenly_interrupted }
 }
}
tyr_heavenly_assign_states = {
 every_state = {
  limit = { has_variable = tyr_heavenly_transfer owner = { OR = { chi_qing_realm = yes c:TPG ?= this } } }
		remove_variable = tyr_heavenly_transfer
		if = {
			limit = { NOT = { owner = c:TPG } }
			set_state_owner = c:TPG
		}
		set_state_type = incorporated
	}
	c:TPG = {
		create_military_formation = {
			type = army
			hq_region = scope:heavenly_kingdom_spawn.region
			combat_unit = { type = unit_type:combat_unit_type_irregular_infantry state_region = scope:heavenly_kingdom_spawn.state_region count = 10 }
			combat_unit = { type = unit_type:combat_unit_type_cannon_artillery state_region = scope:heavenly_kingdom_spawn.state_region count = 5 }
		}
	}
}

# Annexation has returned occupied territory to Qing. Restore its provincial owner
# and incorporation where possible without taking land away from third parties.
tyr_heavenly_restore_states = {
	every_scope_state = {
		limit = { has_variable = tyr_heavenly_original_owner }
		if = {
			limit = { var:tyr_heavenly_original_owner ?= { chi_qing_realm = yes } }
			var:tyr_heavenly_original_owner = { save_temporary_scope_as = tyr_heavenly_restored_owner }
		}
		else = { root = { save_temporary_scope_as = tyr_heavenly_restored_owner } }
		if = {
			limit = { NOT = { owner = scope:tyr_heavenly_restored_owner } }
			set_state_owner = scope:tyr_heavenly_restored_owner
		}
		if = {
			limit = { has_variable = tyr_heavenly_originally_incorporated }
			set_state_type = incorporated
		}
		else = { set_state_type = unincorporated }
	}
	tyr_heavenly_clear_state_records = yes
}

    NOT = { has_variable = chi_adm_religious_missions_restricted }
   }

  else_if = {
   limit = {
    relative_bureaucracy >= 0.05
    NOT = { has_variable = chi_adm_religious_missions_regulated }
    NOT = { has_variable = chi_adm_religious_missions_restricted }
   }
   chi_adm_religious_set_policy = { POLICY = chi_adm_religious_missions_regulated }
  }
 }
 if = {
  limit = {
   chi_adm_religious_society_available = yes
   NOT = { has_variable = chi_adm_religious_society_cooldown }
  }
  if = {
   limit = { authority >= 25 var:chi_adm_loyalty >= 65 }
   chi_adm_religious_suppress = yes
  }
  else_if = {
   limit = {
    var:chi_adm_loyalty <= 35 var:chi_adm_autonomy >= 55
    any_scope_state = {
     has_modifier = god_worshippers
     turmoil >= 0.20
     religion_percent_state = { target = rel:protestant value >= 0.10 }
    }
   }
   set_variable = chi_adm_religious_covert
   if = {
    limit = { gold_reserves >= 10000 NOT = { has_variable = chi_adm_religious_aided } }
    chi_adm_religious_aid_society = yes
   }
   if = {
    limit = { is_at_war = no any_scope_general = { exists = this } NOT = { has_variable = chi_adm_religious_stand_aside } }
    set_variable = chi_adm_religious_taiping_pledge
   }
   set_variable = { name = chi_adm_religious_society_cooldown months = 12 }
  }
 }
}
