tyr_heavenly_enter_hierarchy = {
	if = { limit = { NOT = { has_social_hierarchy = tyr_heavenly_hierarchy } }
		if = { limit = { has_social_hierarchy = british_indian_caste_system } set_variable = tyr_heavenly_previous_castes }
		if = { limit = { has_social_hierarchy = edo_system } set_variable = tyr_heavenly_previous_edo }
		if = { limit = { has_social_hierarchy = tyrone_company_hierarchy } set_variable = tyr_heavenly_previous_company }
		set_social_hierarchy = tyr_heavenly_hierarchy
	}
	recalculate_pop_ig_support = yes
}
tyr_heavenly_leave_hierarchy = {
	if = { limit = { has_variable = tyr_heavenly_previous_castes } set_social_hierarchy = british_indian_caste_system }
	else_if = { limit = { has_variable = tyr_heavenly_previous_edo } set_social_hierarchy = edo_system }
	else_if = { limit = { has_variable = tyr_heavenly_previous_company } set_social_hierarchy = tyrone_company_hierarchy }
	else = { set_social_hierarchy = default_social_hierarchy }
	remove_variable = tyr_heavenly_previous_castes
	remove_variable = tyr_heavenly_previous_edo
	remove_variable = tyr_heavenly_previous_company
	recalculate_pop_ig_support = yes
}
