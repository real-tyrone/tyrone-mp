# Court scope. On every terminal path the result is written before phase is removed.
# A restored administration has already consented to these terms at the pledge.

   limit = { OR = { has_variable = chi_adm_qing_security_faction has_variable = chi_adm_qing_security_informed } }
   chi_adm_qing_security_clear_office = yes
  }
  clear_variable_list = chi_adm_qing_security_detached
  remove_variable = chi_adm_qing_security_claimant
  remove_variable = chi_adm_qing_security_months
  remove_variable = chi_adm_qing_security_readiness
  remove_variable = chi_adm_qing_security_evidence
  remove_variable = chi_adm_qing_security_briefed_ratio
  remove_variable = chi_adm_qing_security_briefed
  remove_variable = chi_adm_qing_security_terms
  remove_variable = chi_adm_qing_security_terms_offered
  remove_variable = chi_adm_qing_security_ultimatum
  remove_variable = chi_adm_qing_security_failed_politics
  remove_variable = chi_adm_qing_security_native_started
  remove_variable = chi_adm_qing_security_pending_result
  remove_variable = chi_adm_qing_security_foreign_envoy
  remove_variable = chi_adm_qing_security_foreign_commission

# Root = CHI for the monthly review. All iterators are limited to its subjects.
chi_adm_qing_security_court_review = {
 if = {
  limit = { var:chi_adm_qing_security_phase = 5 }
  if = {
   limit = {
    OR = { NOT = { chi_adm_qing_security_court = yes } NOT = { exists = var:chi_adm_qing_security_claimant } }
   }
   every_diplomatic_play = {
    limit = { chi_adm_qing_security_live_play = yes }
    end_play = yes
   }
  }
  if = {
   limit = {
    NOR = {
     any_diplomatic_play = { chi_adm_qing_security_live_play = yes }
     var:chi_adm_qing_security_claimant ?= { has_war_with = c:CHI }
    }
   }
   chi_adm_qing_security_resolve = yes
  }
 }
 else_if = {
  limit = {
   OR = {
    NOT = { chi_adm_qing_security_court = yes }
    NOT = { chi_adm_qing_security_claimant_valid = yes }
    var:chi_adm_qing_security_months >= 30
   }
  }
  chi_adm_qing_security_close = { RESULT = 5 }
 }
 else = {
  change_variable = { name = chi_adm_qing_security_months add = 1 }
  if = {
   limit = { has_modifier = chi_adm_qing_security_foreign_claimant NOT = { exists = var:chi_adm_qing_security_foreign_envoy } }
   remove_modifier = chi_adm_qing_security_foreign_claimant
   remove_variable = chi_adm_qing_security_foreign_envoy
  }
  if = {
   limit = { has_modifier = chi_adm_qing_security_foreign_court NOT = { exists = var:chi_adm_qing_security_foreign_commission } }
   remove_modifier = chi_adm_qing_security_foreign_court
   remove_variable = chi_adm_qing_security_foreign_commission
  }
  if = {
   limit = { var:chi_adm_qing_security_claimant ?= { chi_adm_qing_security_claimant_politics = yes } }
   remove_variable = chi_adm_qing_security_failed_politics
   if = { limit = { var:chi_adm_qing_security_phase < 4 } change_variable = { name = chi_adm_qing_security_readiness add = chi_adm_qing_security_monthly_preparation } }
  }
  else = {
   if = { limit = { NOT = { has_variable = chi_adm_qing_security_failed_politics } } set_variable = { name = chi_adm_qing_security_failed_politics value = 0 } }
   change_variable = { name = chi_adm_qing_security_failed_politics add = 1 }
   change_variable = { name = chi_adm_qing_security_readiness subtract = 8 }
  }
  change_variable = { name = chi_adm_qing_security_evidence add = chi_adm_qing_security_monthly_evidence }
  if = {
   limit = { var:chi_adm_qing_security_phase = 1 var:chi_adm_qing_security_months >= 4 }
   set_variable = { name = chi_adm_qing_security_phase value = 2 }
  }
  else_if = {
   limit = { var:chi_adm_qing_security_phase = 2 var:chi_adm_qing_security_months >= 6 chi_adm_qing_security_armed_count >= 2 }
   set_variable = { name = chi_adm_qing_security_phase value = 3 }
   if = { limit = { var:chi_adm_qing_security_evidence < 50 } set_variable = { name = chi_adm_qing_security_evidence value = 50 } }
  }
  set_variable = { name = chi_adm_qing_security_briefed_ratio value = chi_adm_qing_security_force_ratio }
  chi_adm_qing_security_clamp = yes
  chi_adm_qing_security_brief = yes
  if = {
   limit = { var:chi_adm_qing_security_failed_politics >= 3 }
   chi_adm_qing_security_close = { RESULT = 5 }
  }
  else = { chi_adm_qing_security_ai_review = yes }
 }
}

# Root = CHI. AI pledges lapse when a human takes the seat before the play.
chi_adm_qing_security_ai_review = {
 if = {
  limit = { var:chi_adm_qing_security_phase >= 2 }
  every_subject_or_below = {
   limit = {
    is_ai = yes
    chi_adm_qing_security_office = yes
    NOT = { chi_adm_qing_security_is_claimant = yes }
    NOT = { has_variable = chi_adm_qing_security_stance_cooldown }
   }
   if = {
    limit = { var:chi_adm_qing_security_faction <= 2 has_variable = chi_adm_qing_security_faction var:chi_adm_loyalty >= 65 }
    chi_adm_qing_security_revise = { FACTION = 4 }
   }
   else_if = {
    limit = {
     NOT = { has_variable = chi_adm_qing_security_faction }
     chi_adm_qing_security_can_arm = yes
     var:chi_adm_loyalty <= 30
     var:chi_adm_corruption < 65
     OR = { has_modifier = chi_adm_provincial_development_military_1 army_power_projection >= { value = c:CHI.army_power_projection multiply = 0.12 } }
    }
    chi_adm_qing_security_pledge = { FACTION = 1 }
   }
   else_if = {
    limit = { NOT = { has_variable = chi_adm_qing_security_faction } chi_adm_qing_security_can_aid = yes var:chi_adm_loyalty <= 40 }
    chi_adm_qing_security_pledge = { FACTION = 2 }
   }
   else_if = {
    limit = {
     NOT = { has_variable = chi_adm_qing_security_faction }
     chi_adm_qing_security_free_to_pledge = yes
     var:chi_adm_loyalty >= 70
     OR = { relations:c:CHI >= relations_threshold:amicable chi_adm_frontier = yes }
    }
    chi_adm_qing_security_pledge = { FACTION = 3 }
   }
  }
 }
 if = {
  limit = { is_ai = yes }
  if = { limit = { chi_adm_qing_security_can_suppress = yes } chi_adm_qing_security_suppress = yes }
  else_if = {
   limit = { chi_adm_qing_security_can_offer = yes OR = { var:chi_adm_qing_security_phase = 4 var:chi_adm_qing_security_briefed_ratio >= 80 } }
   chi_adm_qing_security_offer_terms = yes
  }
  else_if = { limit = { chi_adm_qing_security_can_muster = yes var:chi_adm_qing_security_readiness >= 40 } chi_adm_qing_security_muster = yes }
  else_if = { limit = { chi_adm_qing_security_can_investigate = yes var:chi_adm_qing_security_evidence >= 25 var:chi_adm_qing_security_evidence < 50 } chi_adm_qing_security_investigate = yes }
 }
 if = {
  limit = { has_variable = chi_adm_qing_security_phase }
  set_variable = { name = chi_adm_qing_security_briefed_ratio value = chi_adm_qing_security_force_ratio }
  var:chi_adm_qing_security_claimant ?= {
   if = {
    limit = { is_ai = yes }
    if = {
     limit = {
      chi_adm_qing_security_can_accept = yes
      OR = { c:CHI ?= { chi_adm_qing_security_force_ratio < 85 } var:chi_adm_corruption >= 60 }
     }
     chi_adm_qing_security_accept_terms = yes
    }
    else_if = { limit = { chi_adm_qing_security_can_issue = yes c:CHI ?= { chi_adm_qing_security_force_ratio >= 80 } } chi_adm_qing_security_ultimatum = yes }
    else_if = { limit = { chi_adm_qing_security_can_launch = yes c:CHI ?= { chi_adm_qing_security_force_ratio >= 80 } } chi_adm_qing_security_launch = yes }
   }
  }
 }
}
