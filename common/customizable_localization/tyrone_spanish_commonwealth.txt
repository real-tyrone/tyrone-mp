# Spanish Commonwealth progress-condition indicators. Scope: sponsoring country.
scw_ui_regional_active = {
	type = journal_entry
	text = {
		trigger = {
			any_country = {
				OR = { c:SPA ?= this was_formed_from = SPA }
				has_variable = scw_running
				var:scw_regional_points > 0
			}
		}
		localization_key = scw_condition_met
	}
	text = { localization_key = scw_condition_unmet }
}

scw_ui_institutions_advance = {
	type = journal_entry
	text = {
		trigger = {
			any_country = {
				OR = { c:SPA ?= this was_formed_from = SPA }
				has_variable = scw_running
				NOT = { has_variable = scw_completed }
				var:scw_institutions < 40
				var:scw_regional_points >= 15
				is_at_war = no
				power_bloc = { current_cohesion_number >= 60 }
			}
		}
		localization_key = scw_condition_met
	}
	text = { localization_key = scw_condition_unmet }
}
