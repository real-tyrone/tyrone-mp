t4_nihilism_level_one_checkbox = {
	type = country
	text = {
		trigger = {
			NOT = { has_variable = t4_nihilism_event_4_fired }
		}
		localization_key = t4_nihilism_checkbox_unticked
	}
	text = {
		trigger = {
			has_variable = t4_nihilism_event_4_fired
		}
		localization_key = t4_nihilism_checkbox_ticked
	}
}
