tyrone_fleet_transfer = {
	kind = directed
	cost = 0

	icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/ship_transfer_treaties.dds"

	flags = {
		giftable
	}

	usage_limit = once_per_side_with_same_inputs
	maintenance_paid_by = source_country

	visible = {
		has_dlc_feature = ep2_content
	}

	possible = {
		OR = {
			AND = {
				any_scope_fleet = {
					num_ships > 0
					NOT = {
						any_scope_ship = {
							is_in_battle = yes
						}
					}
				}
				scope:other_country = {
					has_port_country = yes
				}
			}
			AND = {
				has_port_country = yes
				scope:other_country = {
					any_scope_fleet = {
						num_ships > 0
						NOT = {
							any_scope_ship = {
								is_in_battle = yes
							}
						}
					}
				}
			}
		}
	}

	can_ratify = {
		scope:source_country = {
			any_scope_fleet = {
				num_ships > 0
				NOT = {
					any_scope_ship = {
						is_in_battle = yes
					}
				}
			}
		}
	}

	on_entry_into_force = {
		scope:article_options = {
			target_country = {
				save_scope_as = tyrone_fleet_transfer_taker
			}
			source_country = {
				save_scope_as = tyrone_fleet_transfer_giver
				trigger_event = { id = tyrone_fleet_transfer.1 days = 0 popup = yes }
			}
		}
	}

	on_withdrawal = {}

	on_break = {
		scope:withdrawing_country = {
			change_relations = { country = scope:non_withdrawing_country value = -5 }
			change_infamy = {
				value = {
					add = 5
					multiply = {
						value = scope:treaty_options.remaining_binding_period
						min = 365
						divide = 365
					}
					min = 5
					max = 50
				}
			}
		}
	}

	ai = {
		treaty_categories = {
			none
		}

		evaluation_chance = {
			value = 0
		}

		article_ai_usage = {
			request
			offer
		}

		inherent_accept_score = {
			value = -1000
		}
	}

	wargoal = {
		execution_priority = 21
		contestion_type = control_target_country_capital

		maneuvers = {
			value = 5
		}

		infamy = {
			value = 1
		}
	}
}
