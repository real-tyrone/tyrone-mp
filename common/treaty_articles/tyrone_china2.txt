# China-only concession articles. Source is the central Chinese negotiator.
# Do not put port ownership in requirement_to_maintain: that destroys articles.
# Concession caches are rebuilt monthly; treaty callbacks never dispatch events.

regional_investment_rights = {
    kind = directed
    cost = 10
    icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/foreign_investment_agreement_treaties.dds"
    flags = { is_investment_rights can_be_enforced can_be_renegotiated }
    maintenance_paid_by = target_country
    usage_limit = once_per_side_with_same_inputs
    required_inputs = { state }
    visible = {
        chi_legation_is_central = yes
        has_global_variable = chi_legation_active
        scope:other_country = { is_involved_in_journal_entry = je_chi_legation_system }
    }
    possible = {
        chi_legation_is_central = yes
        has_global_variable = chi_legation_active
        scope:other_country = { chi_legation_eligible = yes }
    }
    can_ratify = {
        scope:source_country = { chi_legation_is_central = yes }
        scope:target_country = { is_involved_in_journal_entry = je_chi_legation_system }
        has_global_variable = chi_legation_active
    }
    can_withdraw = {
        OR = {
            NOT = { has_global_variable = chi_legation_active }
            NOT = { scope:withdrawing_country = scope:source_country }
        }
    }
    state_valid_trigger = {
        scope:input = {
            state_region = { chi_legation_region = yes }
            owner = { chi_legation_chinese_government = yes }
            any_neighbouring_state = {
                state_region = { save_temporary_scope_as = chi_legation_neighbour_region }
                scope:other_country = {
                    is_target_in_variable_list = { name = chi_legation_regions target = scope:chi_legation_neighbour_region }
                }
            }
        }
    }
    ai = {
        treaty_categories = { economy }
        article_ai_usage = { will_request }
        evaluation_chance = { value = 10 }
        inherent_accept_score = { value = -20 }
        wargoal_score_multiplier = { value = 1 }
    }
    wargoal = {
        execution_priority = 21
        maneuvers = { value = 10 }
        infamy = { value = 2 }
        contestion_type = control_target_country_capital
    }
}
