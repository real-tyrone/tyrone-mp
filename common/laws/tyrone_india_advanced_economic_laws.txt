# Indian variants: same mechanics as the corresponding parent laws.
law_india_command_economy = {
	group = lawgroup_economic_system
	parent = law_command_economy
	
	icon = "gfx/interface/icons/law_icons/command_economy.dds"
	
	progressiveness = 50
	
	disallowing_laws = {
		law_serfdom
		law_anarchy
		law_women_in_the_fields
	}

	is_visible = {
		is_india_tag_trigger = yes
		NOT = { is_china_tag_trigger = yes }
	}	

	unlocking_technologies = {
		socialism
	}
	
	unlocking_laws = {
		law_autocracy
		law_neo_absolutism
		law_oligarchy
		law_organic_regulation
		law_single_party_state
		law_technocracy
	}	
	
	can_enact = {
		has_law = law_type:law_council_republic
	}

	on_activate = {
		seize_investment_pool = yes
		custom_tooltip = {
			text = enact_law_radical_leftist_economy_warning_desc
		}	
	}
	
	modifier = {
		country_disable_investment_pool_bool = yes
		
		state_bureaucrats_investment_pool_efficiency_mult = 0.25
		state_bureaucrats_investment_pool_contribution_add = 0.1
		country_authority_mult = 0.25
		state_bureaucracy_population_base_cost_factor_mult = 0.10
		country_bureaucrats_pol_str_mult = 0.25
		building_nationalization_cost_mult = -0.5
		building_nationalization_radicals_mult = -0.75
		country_government_dividends_efficiency_add = 0.6 
		country_private_construction_allocation_mult = 0.1
		building_company_government_dividends_add = 0.75
		building_group_bg_heavy_industry_construction_efficiency_add = 0.10
		country_free_charters_add = 2	# 1
		country_max_companies_add = 2

		country_leverage_resistance_mult = 0.25

		country_disable_non_company_privatization_bool = yes
	}
	
	ai_will_do = {
		OR = {
			has_law_or_variant = law_type:law_technocracy
			AND = {
				has_law_or_variant = law_type:law_council_republic
				ruler ?= {
					has_ideology = ideology:ideology_vanguardist
					has_ideology = ideology:ideology_communist
				}			
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
		
		if = {
			limit = { 
				has_law_or_variant = law_type:law_council_republic
			}
			add = 250
		}
		
		if = {
			limit = { 
				has_law_or_variant = law_type:law_technocracy
			}
			add = 100
		}
	}
	
	ai_impose_chance = {
		value = 0
		
		if = {
			limit = {
				has_law = law_type:law_india_command_economy			
			}
			
			add = base_impose_law_weight
			multiply = 10.0
		}
			
		if = {
			limit = {
				scope:target_country = {
					is_india_tag_trigger = yes
					OR = {	
						has_journal_entry = je_british_dictate_plantations
						has_journal_entry = je_british_dictate_military
						has_journal_entry = je_british_dictate_universities
					}
				}
			}
			multiply = 0
		}
	}	
}

law_india_laissez_faire = {
	group = lawgroup_economic_system
	parent = law_laissez_faire
	
	icon = "gfx/interface/icons/law_icons/laissez_faire.dds"
	
	is_visible = {
		is_india_tag_trigger = yes
		NOT = {
			has_law = law_type:law_laissez_faire_plus
		}
		NOT = {
			is_china_tag_trigger = yes
		}
	}


	progressiveness = 50
	
	disallowing_laws = {
		law_serfdom
		law_isolationism
		law_anarchy
		law_guild_system
		law_factory_councils
		law_women_in_the_fields
	}
	
	unlocking_technologies = {
		international_trade
	}
	
	on_activate = {
	}
	
	modifier = {
		country_loan_interest_rate_mult = -0.25
		state_shopkeepers_investment_pool_efficiency_mult = 0.25	
		state_capitalists_investment_pool_efficiency_mult = 0.25
		country_private_construction_allocation_mult = 0.75
		country_government_dividends_reinvestment_add = 1.0		
		country_free_charters_add = 2
		country_max_companies_add = 1
		country_disable_nationalization_bool = yes
		country_force_privatization_bool = yes
		country_forbid_monopoly_bool = yes
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
		
		if = {
			limit = {
				OR = {
					has_journal_entry = je_meiji_main
					has_journal_entry = je_great_reforms_serfdom
					has_journal_entry = je_hawaii
				}
				has_law_or_variant = law_type:law_traditionalism							
			}
			add = 10000 # Top priority for Tanzimat/Meiji AI is to get out of Traditionalism
		}	

		# AI should really try to get out of Traditionalism under the right circumstances
		if = {
			limit = {
				has_law_or_variant = law_type:law_traditionalism
				NOT = { has_strategy = ai_strategy_reactionary_agenda }
				OR = {
					NOT = { has_strategy = ai_strategy_conservative_agenda }
					any_political_movement = {
						law_enactment_stance = {
							law = law_type:law_india_laissez_faire
							value > neutral
						}	
						political_movement_support >= 0.10
					}			
				}	
			}
			add = 1000
		}		
	}	
}
