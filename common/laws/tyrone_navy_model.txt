INJECT:law_jeune_ecole = {
	
	is_visible = {
		AND = {
			is_china_tag_trigger = no
			any_scope_state = {
				is_coastal = yes
			}
		}
	}
	
	modifier = {
		country_ship_construction_progress_max_add = 20
	}

	on_enact = {
		tyrone_india_reapply_naval_restrictions = { LAW = law_jeune_ecole }
	}

}

REPLACE:law_merchant_navy = {
	group = lawgroup_navy_model
	icon = "gfx/interface/icons/law_icons/merchant_fleet.dds"

	progressiveness = -50

	is_visible = {
		AND = {
			is_china_tag_trigger = no
			any_scope_state = {
				is_coastal = yes
			}
		}
	}

	modifier = {
		country_ship_group_supply_ships_construction_efficiency_add = 0.3
		country_ship_type_early_troop_ship_construction_efficiency_add = 0.3
		country_ship_type_troop_ship_construction_efficiency_add = 0.3
		country_ship_construction_efficiency_add = -0.1
		ship_power_projection_mult = -0.1
	}

	on_enact = {
		tyrone_india_reapply_naval_restrictions = { LAW = law_merchant_navy }
	}

	# AI will never return to merchant navy
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

REPLACE:law_professional_navy = {
	group = lawgroup_navy_model
	icon = "gfx/interface/icons/law_icons/ocean_domination.dds"

	progressiveness = 0

	unlocking_technologies = { military_drill }

	is_visible = {
		AND = {
			is_china_tag_trigger = no
			any_scope_state = {
				is_coastal = yes
			}
		}
	}

	modifier = {
		building_naval_administration_hiring_rate_mult = 0.25
		country_officers_pol_str_mult = 0.2
		country_ship_group_capital_ships_construction_efficiency_add = 0.25
		country_prestige_from_navy_power_projection_mult = 0.2
		country_ship_crew_starting_veterancy_experience_add = 125
	}

	on_enact = {
		tyrone_india_reapply_naval_restrictions = { LAW = law_professional_navy }
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		# Once the AI gets to a certain naval tech level they should try to ditch merchant navy
		if = {
			limit = {
				OR = {
					has_technology_researched = monitor_tech
					has_technology_researched = pre_dreadnought_tech
					has_technology_researched = sea_lane_strategies
				}
				has_law_or_variant = law_type:law_merchant_navy
			}

			add = 1000
		}
	}
}
