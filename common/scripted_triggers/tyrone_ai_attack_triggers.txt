# Country scope: protect players and their entire subject hierarchy from AI attack decisions.
# Player-issued decisions retain their existing conditions.
tyrone_ai_attack_country_allowed = {
    OR = {
        ROOT = { is_ai = no }
        AND = {
            is_player = no
            NOT = {
                any_country = {
                    is_player = yes
                    any_subject_or_below = { this = PREV }
                }
            }
        }
    }
}

# State scope; ROOT is Afghanistan taking the AI expansion decision.
tyrone_afg_ai_expansion_target = {
	OR = {
		state_region = s:STATE_HERAT
		state_region = s:STATE_NORTHERN_BALUCHISTAN
		state_region = s:STATE_KABUL
		state_region = s:STATE_BALKH
	}
	owner = {
		tyrone_ai_attack_country_allowed = yes
		NOT = { this = ROOT }
		NOT = { is_country_type = decentralized }
		is_subject = no
		is_adjacent_to_country = ROOT
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		NOT = { has_truce_with = ROOT }
		NOT = { has_war_with = ROOT }
		army_power_projection < ROOT.army_power_projection
	}
}
