host_tool_is_current_host = {
	has_variable = host_tool_current_host
}

host_tool_is_initial_game_rule_host = {
	OR = {
		AND = {
			has_game_rule = host_tool_initial_host_gbr
			c:GBR ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_fra
			c:FRA ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_pru
			c:PRU ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_aus
			c:AUS ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_rus
			c:RUS ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_usa
			c:USA ?= this
		}
		AND = {
			has_game_rule = host_tool_initial_host_tur
			c:TUR ?= this
		}
	}
}

host_tool_is_player_country = {
	is_player = yes
	NOT = { is_country_type = decentralized }
}

host_tool_is_valid_host_transfer_target = {
	host_tool_is_player_country = yes
	NOT = { has_variable = host_tool_current_host }
}

host_tool_is_gp_or_major = {
	OR = {
		country_rank = rank_value:great_power
		country_rank = rank_value:major_power
		country_rank = rank_value:unrecognized_major_power
	}
}

host_tool_has_subjects = {
	any_subject_or_below = {
		always = yes
	}
}

host_tool_can_be_subject_target = {
	NOT = { is_country_type = decentralized }
	NOT = { has_variable = host_tool_current_host }
}

host_tool_has_wars = {
	any_scope_war = {
		always = yes
	}
}
