# Global consensus; valid from country or contextless JE scope.
tyr_berlin_completion_ready = {
		OR = {
		any_country = {
			custom_tooltip = {
				text = 7accept_berlin_conf_tt
				has_variable = accepted_conf_var
			}
			count >= 7
		}
		any_country = {
			custom_tooltip = {
				text = 3GPaccept_berlin_conf_tt
				has_variable = accepted_conf_var
				OR = {
					var:conference_gp_number = 3
					var:conference_gp_number = 2
					var:conference_gp_number = 1
				}
			}
			count >= 3
			} 
		}
	}

# Country scope: shared availability requirement for conference actions.
tyr_berlin_negotiation_open = {
