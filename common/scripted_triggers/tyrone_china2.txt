# Country scope. A port anywhere in the fixed perimeter supports all concessions.
chi_legation_has_port = {
    any_scope_state = {
        is_treaty_port = yes
        state_region = { chi_legation_region = yes }
    }
}
chi_legation_eligible = {
    NOT = { chi_legation_chinese_government = yes }
    OR = {
        country_rank >= rank_value:great_power
        has_variable = chi_legation_admitted
    }
    chi_legation_has_port = yes
}
# State scope; investor passed by the native construction system.
# Native investment access still handles subjects/blocs/pre-existing ordinary rights.
# The restrictive branch only applies to investors relying on concession articles.
chi_legation_building_allowed = {
    trigger_if = {
        limit = {
            exists = scope:investor_country
            owner = {
                OR = {
                    chi_legation_chinese_government = yes
                    AND = {
                        exists = top_overlord
                        top_overlord = { chi_legation_chinese_government = yes }
                    }
                }
            }
            scope:investor_country = { NOT = { chi_legation_chinese_government = yes } }
        }
        owner = { save_temporary_scope_as = chi_legation_build_owner }
        state_region = { save_temporary_scope_as = chi_legation_build_region }
        OR = {
            owner = scope:investor_country
            scope:investor_country = {
                any_scope_treaty = {
                    binds = scope:chi_legation_build_owner
                    any_scope_article = {
                        trigger_if = {
                            limit = { has_type = foreign_investment_rights }
                            source_country = scope:chi_legation_build_owner
                            target_country = scope:investor_country
                        }
                        trigger_else = { always = no }
                    }
                }
            }
            AND = {
                owner = { exists = top_overlord }
                owner.top_overlord = {
                save_temporary_scope_as = chi_legation_build_overlord
                any_scope_treaty = {
                    binds = scope:investor_country
                    any_scope_article = {
                        trigger_if = {
                            limit = { has_type = foreign_investment_rights }
                            source_country = scope:chi_legation_build_overlord
                            target_country = scope:investor_country
                        }
                        trigger_else = { always = no }
                    }
                }
            }
            }
            owner = { is_subject_of = scope:investor_country }
            scope:investor_country = {
                is_subject_of = scope:chi_legation_build_owner
                any_scope_diplomatic_pact = {
                    OR = {
                        is_diplomatic_action_type = da_overlord_grant_investment_rights
                        is_diplomatic_action_type = da_subject_request_investment_rights
                    }
                    "diplomatic_pact_other_country(scope:investor_country)" = scope:chi_legation_build_owner
                }
            }
            scope:investor_country.power_bloc ?= {
                has_principle = principle:principle_foreign_investment_3
                any_power_bloc_member = { this = scope:chi_legation_build_owner }
            }
            AND = {
                # Concessions never inherit into foreign subjects such as Korea.
                owner = { chi_legation_chinese_government = yes }
                state_region = { chi_legation_region = yes }
                scope:investor_country = {
                    chi_legation_has_port = yes
                    is_target_in_variable_list = { name = chi_legation_industries target = $BUILDING_TYPE$ }
                }
                scope:investor_country = { save_temporary_scope_as = chi_legation_access_investor }
                chi_legation_investor_has_local_state_access = yes
            }
        }
    }
}

# Country scope. Optional comparison is essential before the global JE initializes.
# Same global-country comparison form used by vanilla's Chinese central government.
chi_legation_is_central = {
 global_var:chi_legation_central ?= this
}
# Country scope: only the current realm may supply NEW concessions.
chi_legation_in_central_realm = {
 OR = { chi_legation_is_central = yes top_overlord ?= { chi_legation_is_central = yes } }
}
# Country scope. Live articles, not monthly caches, determine exclusivity.
# Caller saves chi_legation_check_region (state region).
chi_legation_holds_checked_region = {
 save_temporary_scope_as = chi_legation_check_power
 any_scope_treaty = {
  any_scope_article = {
   OR = { has_type = regional_investment_rights has_type = chi_legation_local_territorial_concession }
   target_country = scope:chi_legation_check_power
   save_temporary_scope_as = chi_legation_checked_article
   target_country = { NOT = { is_target_in_variable_list = { name = chi_legation_revoked_treaties target = scope:chi_legation_checked_article.treaty } } }
   exists = input_state
   input_state = { state_region = { this = scope:chi_legation_check_region } }
  }
 }
}

# Foreign country scope; current owner determines whether a port blocks completion.
chi_boxer_port_holder = {
    has_variable = chi_legation_admitted
    NOT = { chi_legation_chinese_government = yes }
    chi_legation_has_port = yes
}
# Chinese country scope. Peace is required even if the final port changes hands.
chi_boxer_can_complete = {
    is_at_war = no
    NOT = { any_country = { chi_boxer_port_holder = yes } }
}

# Chinese country scope; the foreign negotiating party is other_country.
# A port is local when this government retains territory in its region.
chi_legation_has_requester_port = {
    any_scope_state = {
        state_region = {
            chi_legation_region = yes
            any_scope_state = { is_treaty_port = yes owner = scope:other_country }
        }
    }
}
# State scope; caller supplies chi_legation_access_investor.
chi_legation_investor_has_local_state_access = {
    owner = { save_temporary_scope_as = chi_legation_access_owner }
    state_region = { save_temporary_scope_as = chi_legation_access_region }
    scope:chi_legation_access_investor = {
        any_scope_treaty = {
            any_scope_article = {
                OR = { has_type = regional_investment_rights has_type = chi_legation_local_territorial_concession has_type = treaty_port }
                target_country = scope:chi_legation_access_investor
                exists = input_state
                input_state = { state_region = { this = scope:chi_legation_access_region } }
                save_temporary_scope_as = chi_legation_access_article
                target_country = { NOT = { is_target_in_variable_list = { name = chi_legation_revoked_treaties target = scope:chi_legation_access_article.treaty } } }
                OR = {
                    # Historical central grants retain their original regional coverage.
                    has_type = regional_investment_rights
                    source_country = scope:chi_legation_access_owner
                }
            }
        }
    }
}
