tyr_ga_je_danubian_federation_czech_button_visible = {
	exists = global_var:tyr_ga_austria
	global_var:tyr_ga_austria = ROOT
	OR = {
		any_scope_state = {
			state_region = {
				is_homeland = cu:czech
			}
		}
		any_subject_or_below = {
			OR = {
				#is_subject_type = subject_type_crown_land
				is_subject_type = subject_type_personal_union
			}
			any_scope_state = {
				state_region = {
					is_homeland = cu:czech
				}
			}
		}
	}
	NAND = {
		has_variable = tyr_ga_danubian_federation_czech_button
		NOT = {
			any_in_list = {
				variable = tyr_ga_aus_integrated_cultures
				cu:czech ?= this
			}
		}
	}
	NOT = {
		any_in_list = {
			variable = tyr_ga_aus_disqualified_cultures
			cu:czech ?= this
		}
	}
}
