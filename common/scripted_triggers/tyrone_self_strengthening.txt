tyr_ssm_literacy_ready = {
	literacy_rate >= 0.30
	NOT = { has_law_or_variant = law_type:law_no_schools }
	institution_investment_level = { institution = institution_schools value >= 3 }
}

tyr_ssm_army_ready = {
	has_technology_researched = breech_loading_artillery
	army_size >= 500
	custom_tooltip = {
		text = tyr_ssm_military_industry_ready_tt
		country_has_building_type_levels = {
			target = bt:building_arms_industry
			value >= tyr_ssm_arms_industry_levels_required
		}
	}
	NOT = {
		any_military_formation = {
			is_army = yes
			any_combat_unit = {
				OR = {
					has_unit_type = unit_type:combat_unit_type_irregular_infantry
					has_unit_type = unit_type:combat_unit_type_line_infantry
				}
			}
		}
	}
}

tyr_ssm_navy_ready = {
	has_technology_researched = ironclad_tech
	navy_size >= 150
	country_has_building_type_levels = {
		target = bt:building_shipyard
		value >= 15
	}
	naval_power_projection >= 5000
	NOT = { has_law_or_variant = law_type:law_merchant_navy }
}

tyr_ssm_all_reforms_ready = {
	OR = { has_variable = tyr_ssm_literacy_done tyr_ssm_literacy_ready = yes }
	OR = { has_variable = tyr_ssm_army_done tyr_ssm_army_ready = yes }
	OR = { has_variable = tyr_ssm_navy_done tyr_ssm_navy_ready = yes }
	tyr_ssm_industry_rounds >= 5
}
