tyrone_ch_hindustani_eligible = {
	NOT = { country_has_primary_culture = cu:hindustani }
	custom_tooltip = {
		text = tyrone_ch_hindustani_atheist_tt
		tyrone_ch_hindustani_population > 0
		tyrone_ch_hindustani_atheist_share >= 0.2
	}
	custom_tooltip = {
		text = tyrone_ch_hindustani_radical_tt
		tyrone_ch_hindustani_radical_share < 0.2
	}
	custom_tooltip = {
		text = tyrone_ch_cooldown_tt
		NOT = { has_variable = tyrone_ch_recognition_cooldown }
	}
}
