# Root = Country
# $TARGET$ = Country
# Checks for a Non-Aggression Pact between either country or their top overlords.
tyrone_has_non_aggression_pact_with_country_family = {
	$TARGET$ = {
		save_temporary_scope_as = tyrone_nap_target_country
		top_overlord ?= {
			save_temporary_scope_as = tyrone_nap_target_overlord
		}
	}

	OR = {
		AND = {
			scope:tyrone_nap_target_country != this
		any_scope_treaty = {
			binds = scope:tyrone_nap_target_country
			any_scope_article = {
				has_type = non_aggression_pact
			}
		}
		}
		AND = {
			exists = scope:tyrone_nap_target_overlord
			scope:tyrone_nap_target_overlord != this
			any_scope_treaty = {
				binds = scope:tyrone_nap_target_overlord
				any_scope_article = {
					has_type = non_aggression_pact
				}
			}
		}
		top_overlord ?= {
			AND = {
				scope:tyrone_nap_target_country != this
			any_scope_treaty = {
				binds = scope:tyrone_nap_target_country
				any_scope_article = {
					has_type = non_aggression_pact
				}
			}
		}
		}
		top_overlord ?= {
			AND = {
				exists = scope:tyrone_nap_target_overlord
				scope:tyrone_nap_target_overlord != this
				any_scope_treaty = {
					binds = scope:tyrone_nap_target_overlord
					any_scope_article = {
						has_type = non_aggression_pact
					}
				}
			}
		}
	}
}

# Saved scopes are supplied by tyrone_african_colonial_split_check.
# Recheck ownership and province counts when the player finally answers.
tyrone_colonial_split_transfer_valid = {
    exists = scope:tyrone_split_smaller_state
    exists = scope:tyrone_split_larger_state
    exists = scope:tyrone_split_smaller_colony
    exists = scope:tyrone_split_larger_colony
    scope:tyrone_split_smaller_state = {
        owner = scope:tyrone_split_smaller_colony
        state_region = {
            any_scope_state = { count = 2 }
            NOT = {
                any_scope_state = {
                    OR = {
                        is_under_colonization = yes
                        owner = { is_country_type = decentralized }
                    }
                }
            }
        }
    }
    scope:tyrone_split_larger_state = {
        owner = scope:tyrone_split_larger_colony
        state_region = scope:tyrone_split_smaller_state.state_region
        num_provinces > scope:tyrone_split_smaller_state.num_provinces
    }
	scope:tyrone_split_smaller_colony = {
		OR = {
			AND = {
				country_is_colonial_or_company = yes
				is_subject = yes
				top_overlord = root
			}
			AND = {
				is_subject = no
				NOT = { is_country_type = decentralized }
				this = root
			}
		}
	}
	scope:tyrone_split_larger_colony = {
		OR = {
			AND = {
				country_is_colonial_or_company = yes
				is_subject = yes
				NOT = { top_overlord = root }
			}
			AND = {
				is_subject = no
				NOT = { is_country_type = decentralized }
				NOT = { this = root }
			}
		}
	}
}

# Checks the engine's full availability rules for a specific law, including
# technology unlocks, law prerequisites, and mutually exclusive restrictions.
# $LAW$ = Law type key without the law_type: prefix.
tyrone_law_is_available = {
	any_law = {
		type = law_type:$LAW$
		law_is_available = yes
	}
}
