# Country scope: participant, with the rival supplied as TARGET.
tyr_ga_can_fight = {
 is_subject = no
 is_at_war = no
 is_diplomatic_play_committed_participant = no
 exists = $TARGET$
 $TARGET$ = { is_subject = no is_at_war = no is_diplomatic_play_committed_participant = no }
 NOT = { has_truce_with = $TARGET$ }
 NOT = { is_diplomatic_play_participant_with = $TARGET$ }
}

# Country scope: prospective members shown on the German Question map.
tyr_ga_map_southern_candidate = {
 OR = { c:BAD ?= THIS c:BAV ?= THIS c:WUR ?= THIS c:HOH ?= THIS }
}
tyr_ga_map_northern_candidate = {
 country_has_primary_culture = cu:north_german
 country_rank < rank_value:great_power
 capital = { is_in_geographic_region = geographic_region_german_confederation }
 NOR = {
  global_var:tyr_ga_austria ?= THIS
  global_var:tyr_ga_north ?= THIS
  c:HOH ?= THIS
  is_subject_type = subject_type_colony
  is_subject_type = subject_type_dominion
  is_subject_type = subject_type_chartered_company
 }
}
