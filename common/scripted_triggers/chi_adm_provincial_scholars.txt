# Country scope: the reforming administration.
chi_adm_provincial_scholars_eligible = {
 chi_adm_target_valid = yes
 NOT = { has_variable = chi_adm_provincial_scholars_completed }
 OR = {
  is_subject_type = subject_type_qing_viceroyalty_administration
  is_subject_type = subject_type_qing_civil_administration
  AND = {
   is_subject_type = subject_type_qing_frontier_administration
   country_has_primary_culture = cu:han
   country_has_state_religion = rel:confucian
  }
 }
}

chi_adm_provincial_scholars_religious_focus_ready = {
 has_technology_researched = rationalism
 NOR = {
  has_law_or_variant = law_type:law_total_separation
  has_law_or_variant = law_type:law_state_atheism
  has_law_or_variant = law_type:law_serfdom
 }
}

chi_adm_provincial_scholars_merchant_schools_ready = {
 has_technology_researched = banking
 NOR = {
  has_law_or_variant = law_type:law_traditionalism
  has_law_or_variant = law_type:law_chi_industry_banned
 }
}

chi_adm_provincial_scholars_laws_ready = {
 OR = {
  AND = {
   has_variable = chi_confucian_scholars_end_isolation
   chi_adm_provincial_scholars_religious_focus_ready = yes
  }
  AND = {
   has_variable = chi_confucian_scholars_evict_scholars
   chi_adm_provincial_scholars_merchant_schools_ready = yes
  }
 }
}
