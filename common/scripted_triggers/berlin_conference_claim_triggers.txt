# Country scope: overseas African claimants eligible for the research preference.
# Check actual state claims rather than historical conference allocation flags.
tyr_ai_overseas_african_claimant = {
	is_ai = yes
	NOT = { is_in_geographic_region = geographic_region_africa }
	any_country = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_africa
			has_claim_by = root
		}
	}
}

# Country scope. Players retain claims; the AI rule overrides all conference allocations.
tyr_berlin_country_can_claim = {
	OR = {
		is_ai = no
		NOT = { has_game_rule = berlin_conf_ai_claims_disabled }
	}
}

# State scope. Automatic conference transfers require an AI owner with no player overlord.
tyr_berlin_state_can_transfer_automatically = {
	owner = {
		is_ai = yes
		is_player = no
		NOT = {
			any_country = {
				is_player = yes
				any_subject_or_below = { this = PREV }
			}
		}
	}
}
