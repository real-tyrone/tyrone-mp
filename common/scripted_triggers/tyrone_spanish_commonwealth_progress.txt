# Country scope. Shared by the native bar and its condition indicators.
scw_institutions_monthly_advance = {
	has_variable = scw_running
	NOT = { has_variable = scw_completed }
	is_at_war = no
	var:scw_regional_points >= 15
	power_bloc = { current_cohesion_number >= 60 }
}
scw_institutions_monthly_retreat = {
	has_variable = scw_running
	NOT = { has_variable = scw_completed }
	power_bloc = { current_cohesion_number < 30 }
}
