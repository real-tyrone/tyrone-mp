# Country scope: territory available to found or expand Arabia.
tyr_arabia_colony_territory = {
    OR = {
        any_scope_state = {
            region = sr:region_arabia
            is_under_colonization = no
            is_treaty_port = no
        }
        any_subject_or_below = {
            NOT = { has_variable = tyr_arabia_colony }
            is_at_war = no
            is_diplomatic_play_committed_participant = no
            any_scope_state = {
                region = sr:region_arabia
                is_under_colonization = no
                is_treaty_port = no
            }
        }
    }
}

tyr_arabia_colony_can_create = {
        is_subject = no
        is_at_war = no
        is_diplomatic_play_committed_participant = no
        
        custom_tooltip = { text = tyr_regional_colony_no_duplicate_tt NOT = { exists = var:tyr_arabia_colony_country } }
        custom_tooltip = { text = tyr_regional_colony_no_subject_tt NOT = { any_subject_or_below = { has_variable = tyr_arabia_colony } } }
        custom_tooltip = { text = tyr_regional_colony_land_tt tyr_arabia_colony_territory = yes }
}
