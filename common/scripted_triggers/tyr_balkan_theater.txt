tyr_bt_active_aus = {

	trigger_if = {
		limit = {
			has_global_variable = tyr_bt_aus_country
		}
		global_var:tyr_bt_aus_country = {
			exists = this
			is_subject = no
			OR = {
				is_in_power_bloc = no
				is_power_bloc_leader = yes
			}
		}
	}

	trigger_else = {
		always = no
	}

}

tyr_bt_faction_aus = {

	trigger_if = {
		limit = {
			tyr_bt_active_aus = yes
		}
		tyr_bt_active_aus = yes
		OR = {
			AND = {
				is_in_power_bloc = yes
				power_bloc = {
					power_bloc_leader = global_var:tyr_bt_aus_country
				}
			}
			AND = {
				is_in_power_bloc = no
				OR = {
					this = global_var:tyr_bt_aus_country
					top_overlord ?= {
						this = global_var:tyr_bt_aus_country
					}
				}
			}
		}
	}

	trigger_else = {
		always = no
	}

}
