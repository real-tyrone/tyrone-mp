# Country scope: readiness for je_tyrone_raj_welfare.
tyrone_arc_tyrone_raj_welfare_ready = {
	has_journal_entry = je_tyrone_raj_welfare
	scope:raj_scope = {
		NOT = {
			has_law_or_variant = law_type:law_no_social_security
		}
		institution_investment_level = {
			institution = institution_social_security
			value >= 2
		}
	}
}

# Country scope: readiness for je_tyrone_csr_police.
tyrone_arc_tyrone_csr_police_ready = {
	has_journal_entry = je_tyrone_csr_police
	hidden_trigger = {
		scope:csr_india ?= {
			always = yes
			NOT = { has_variable = tyrone_csr_police_done }
		}
	}
	scope:csr_india = {
		NOT = { has_law_or_variant = law_type:law_no_police }
		institution_investment_level = {
			institution = institution_police
			value >= 2
		}
		radical_fraction = { value <= 0.10 }
	}
}

# Country scope: readiness for je_tyrone_csr_army.
tyrone_arc_tyrone_csr_army_ready = {
	has_journal_entry = je_tyrone_csr_army
	hidden_trigger = {
		scope:csr_india ?= {
			always = yes
			NOT = { has_variable = tyrone_csr_army_done }
		}
	}
	scope:csr_india = {
		army_size >= 250
		has_technology_researched = rifling
		has_technology_researched = logistics
		OR = {
			has_law = law_type:law_india_early_standing_army
			has_law = law_type:law_india_company_army
			has_law = law_type:law_india_great_indian_army
		}
	}
}
