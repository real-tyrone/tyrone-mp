# Country scope. Shared by completion scoring and the live agenda checkboxes.

# Guarded because completion/failure tooltips can be read before activation.
tyr_gp_leadership_failed = {
	is_power_bloc_leader = no
	trigger_if = {
		limit = { has_variable = tyr_gp_months_without_leadership }
		var:tyr_gp_months_without_leadership >= 3
	}
	trigger_else = { always = no }
}

tyr_gp_objective_rank = {
	country_rank = rank_value:great_power
	global_country_ranking <= 5
}

tyr_gp_objective_army = {
	army_power_projection > 0
	NOT = {
		any_country = {
			count >= 4
			army_power_projection > root.army_power_projection
		}
	}
}

tyr_gp_objective_navy = {
	naval_power_projection > 0
	NOT = {
		any_country = {
			count >= 4
			naval_power_projection > root.naval_power_projection
		}
	}
}

tyr_gp_objective_population = {
	tyr_gp_population_ratio >= 1.5
}

tyr_gp_objective_gdp = {
	gdp_ranking <= 5
}

tyr_gp_objective_industry = {
	OR = {
		"goods_production_rank(g:glass)" <= 3
		"goods_production_rank(g:steel)" <= 3
		"goods_production_rank(g:tools)" <= 3
	}
}

tyr_gp_objective_resources = {
	OR = {
		"goods_production_rank(g:coal)" <= 3
		"goods_production_rank(g:iron)" <= 3
		"goods_production_rank(g:lead)" <= 3
		"goods_production_rank(g:sulfur)" <= 3
	}
}

tyr_gp_objective_economy = {
	OR = {
		has_law = law_type:law_state_capitalism
		has_law = law_type:law_cooperative_ownership
		has_law = law_type:law_laissez_faire_plus
	}
}

tyr_gp_objective_citizenship = {
	OR = {
		has_law = law_type:law_ethnostate
		has_law = law_type:law_multicultural
	}
}

tyr_gp_objective_bloc_population = {
	is_power_bloc_leader = yes
	tyr_gp_bloc_population > 80000000
}

# Brazil retains this national arc after forming United South America.
tyr_brz_gp_country = {
	OR = {
		country_definition = cd:BRZ
		AND = {
			country_definition = cd:UNSA
			was_formed_from = BRZ
		}
	}
}

tyr_brz_gp_leadership_failed = {
	is_power_bloc_leader = no
	trigger_if = {
		limit = {
			has_variable = tyr_brz_gp_months_without_leadership
		}
		var:tyr_brz_gp_months_without_leadership >= 3
	}
	trigger_else = {
		always = no
	}
}

tyr_brz_gp_objective_open_society = {
	has_law = law_type:law_multicultural
	has_law = law_type:law_no_migration_controls
	has_law = law_type:law_total_separation
}

tyr_brz_gp_objective_south_america = {
	country_definition = cd:UNSA
}

tyr_brz_gp_objective_capital = {
	tyr_brz_gp_capital_population > 10000000
}

tyr_brz_gp_objective_population = {
	tyr_brz_gp_population >= 80000000
}

tyr_brz_gp_objective_liberties = {
	OR = {
		has_law = law_type:law_protected_speech
		has_law = law_type:law_guaranteed_liberties
	}
}

# Country scope. Reuse in hidden activation/visibility gates for all GP variants.
tyr_gp_player_rule_eligible = {
	is_player = yes
	OR = {
		AND = {
			OR = {
				has_game_rule = tyr_gp_countries_default
				has_game_rule = tyr_gp_countries_spain_brazil
				has_game_rule = tyr_gp_countries_with_japan
			}
			OR = {
				country_definition = cd:SPA
				tyr_brz_gp_country = yes
			}
		}
		AND = {
			OR = {
				has_game_rule = tyr_gp_countries_default
				has_game_rule = tyr_gp_countries_with_japan
			}
			OR = {
				country_definition = cd:NET
				country_definition = cd:SWE
				country_definition = cd:DEN
			}
		}
		AND = {
			has_game_rule = tyr_gp_countries_with_japan
			country_definition = cd:JAP
		}
	}
}
