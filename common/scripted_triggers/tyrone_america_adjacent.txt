# Country scope. Saved country objects survive tag changes.
tyr_monroe_usa = { OR = { country_definition = cd:USA country_definition = cd:GAC has_variable = tyr_monroe_usa_identity } }
tyr_monroe_gbr = { OR = { country_definition = cd:GBR has_variable = tyr_monroe_gbr_identity } }
tyr_monroe_usa_sphere = { OR = { tyr_monroe_usa = yes top_overlord ?= { tyr_monroe_usa = yes } power_bloc ?= { power_bloc_leader ?= { tyr_monroe_usa = yes } } } }
tyr_monroe_gbr_sphere = { OR = { tyr_monroe_gbr = yes top_overlord ?= { tyr_monroe_gbr = yes } power_bloc ?= { power_bloc_leader ?= { tyr_monroe_gbr = yes } } } }
# State scope. Explicit mainland Central America avoids the Caribbean strategic region.
tyr_monroe_area = {
 OR = {
  is_in_geographic_region = geographic_region_south_america
  state_region = s:STATE_SAN_SALVADOR
  state_region = s:STATE_GUATEMALA
  state_region = s:STATE_HONDURAS
  state_region = s:STATE_NICARAGUA
  state_region = s:STATE_COSTA_RICA
  state_region = s:STATE_PANAMA
 }
}
# Country scope: regional sovereignty, including regional subjects, but not external blocs.
tyr_monroe_local_control = {
 capital ?= { OR = { is_in_geographic_region = geographic_region_south_america tyr_monroe_area = yes } }
 OR = {
  is_subject = no
  top_overlord ?= { capital ?= { tyr_monroe_area = yes } }
 }
 OR = {
  is_in_power_bloc = no
  power_bloc ?= { power_bloc_leader ?= { OR = { tyr_monroe_usa = yes tyr_monroe_gbr = yes } } }
 }
}
# State scope; Brazilian expansion is assessed against the initial state-region snapshot.
tyr_monroe_protected = {
 OR = {
  owner ?= { NOT = { country_definition = cd:BRZ } NOT = { top_overlord ?= { country_definition = cd:BRZ } } }
  state_region = { tyr_monroe_brazil_initial_region = yes }
 }
 owner ?= {
  OR = {
   tyr_monroe_usa_sphere = yes
   tyr_monroe_gbr_sphere = yes
   AND = {
    tyr_monroe_local_control = yes
    OR = {
     AND = { NOT = { country_definition = cd:BRZ } NOT = { top_overlord ?= { country_definition = cd:BRZ } } }
     prev.state_region = { tyr_monroe_brazil_initial_region = yes }
    }
   }
  }
 }
}
