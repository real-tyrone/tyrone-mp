tyr_gcol_has_maf = {
	exists = c:MAF

	c:MAF ?= {
		is_subject_of = root
	}
}

tyr_gcol_has_dea = {
	exists = c:DEA

	c:DEA ?= {
		is_subject_of = root
	}
}

tyr_gcol_create_maf_possible = {
	c:GER ?= root

	OR = {
		has_journal_entry = je_tyr_gcol_empire
		has_journal_entry = je_tyr_gcol_administration
	}

	NOT = {
		exists = c:MAF
	}

	is_at_war = no
	is_diplomatic_play_committed_participant = no

	institution_investment_level = {
		institution = institution_colonial_affairs
		value >= 2
	}

	any_scope_state = {
		tyr_gcol_african_state = yes
		is_under_colonization = no
		is_incorporated = no
		is_treaty_port = no
	}
}

tyr_gcol_create_dea_possible = {
	c:GER ?= root

	OR = {
		has_journal_entry = je_tyr_gcol_empire
		has_journal_entry = je_tyr_gcol_administration
	}

	NOT = {
		exists = c:DEA
	}

	is_at_war = no
	is_diplomatic_play_committed_participant = no

	institution_investment_level = {
		institution = institution_colonial_affairs
		value >= 2
	}

	owns_entire_state_region = STATE_MALAYA

	any_scope_state = {
		state_region = s:STATE_MALAYA
		is_under_colonization = no
		is_incorporated = no
		is_treaty_port = no
	}
}
