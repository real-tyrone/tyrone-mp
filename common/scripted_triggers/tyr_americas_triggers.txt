# Country predicates. Parameters name their expected country scope.
tyr_am_active = {
	has_global_variable = tyr_am_active
}

tyr_am_main_power = {
	OR = {
		tyr_spanish_content_country = yes
		AND = {
			exists = c:USA
			c:USA = this
		}
	}
}

tyr_am_regional_country = {
	capital ?= {
		is_in_geographic_region = geographic_region_tyr_am_all
	}
	NOT = {
		is_country_type = decentralized
	}
}
# South American subset is reserved for Brazil's federation, not the full competition.
tyr_am_south_american = {
	capital ?= {
		is_in_geographic_region = geographic_region_tyr_am_south_america
	}
	NOT = {
		is_country_type = decentralized
	}
}

tyr_am_brazil_unified = {
	global_var:tyr_am_4_bra_owned >= 75
	global_var:tyr_am_5_bra_owned >= 75
	global_var:tyr_am_6_bra_owned >= 75
	global_var:tyr_am_7_bra_owned >= 75
	global_var:tyr_am_8_bra_owned >= 75
	global_var:tyr_am_9_bra_owned >= 75
	global_var:tyr_am_10_bra_owned >= 75
	global_var:tyr_am_11_bra_owned >= 75
	global_var:tyr_am_12_bra_owned >= 75
}

tyr_am_present = {
	any_scope_state = {
		is_in_geographic_region = geographic_region_tyr_am_all
	}
}

tyr_am_controls_for = {
	exists = $PATRON$
	OR = {
		this = $PATRON$
		AND = {
			exists = top_overlord
			top_overlord = $PATRON$
		}
		AND = {
			$PATRON$ ?= {
				is_power_bloc_leader = yes
			}
			is_in_same_power_bloc = $PATRON$
		}
	}
}

tyr_am_brazil_autonomous = {
	exists = c:BRZ
	c:BRZ = this
	is_subject = no
	is_power_bloc_leader = yes
	global_var:tyr_am_phase >= 2
}

tyr_am_actor_valid = {
	tyr_am_main_power = yes
	is_subject = no
	is_power_bloc_leader = yes
}

tyr_am_has_valid_target = {
	exists = scope:tyr_am_target
	scope:tyr_am_target = {
		exists = this
		tyr_am_regional_country = yes
		is_revolutionary = no
		NOT = {
			this = scope:tyr_am_actor
		}
		NOT = {
			has_war_with = scope:tyr_am_actor
		}
	}
}

tyr_am_can_negotiate = {
	save_temporary_scope_as = tyr_am_actor
	tyr_am_active = yes
	tyr_am_actor_valid = yes
	global_var:tyr_am_phase = 1
	tyr_am_has_valid_target = yes
	scope:tyr_am_target = {
		is_subject = no
		NOT = {
			is_in_same_power_bloc = scope:tyr_am_actor
		}
	}
}

tyr_am_can_invite = {
	tyr_am_can_negotiate = yes
	scope:tyr_am_target = {
		NOT = {
			is_in_power_bloc = yes
		}
		NOT = {
			has_variable = tyr_am_invitation_pending
		}
	}
	OR = {
		AND = {
			tyr_spanish_content_country = yes
			tyr_am_selected_influence <= 25
		}
		AND = {
			c:USA ?= this
			tyr_am_selected_influence >= 75
		}
	}
	OR = {
		prestige > scope:tyr_am_target.prestige
		country_rank > scope:tyr_am_target.country_rank
	}
	power_bloc = {
		NOT = {
			any_power_bloc_member = {
				is_diplomatic_play_enemy_of = scope:tyr_am_target
			}
		}
	}
}

tyr_am_can_pay = {
	exists = scope:tyr_am_target
}

tyr_am_brazil_play_allowed = {
	# Country scope; the pre-JE South America lock applies only to Brazil.
	trigger_if = {
		limit = {
			exists = c:BRZ
			c:BRZ ?= this
		}
		trigger_if = {
			limit = {
				OR = {
					NOT = {
						has_variable = tyr_am_brazil_south_america_unlocked
					}
					AND = {
						tyr_am_active = yes
						has_variable = tyr_am_brazil_policy_south_america_locked
					}
				}
			}
			NOT = {
				AND = {
					exists = scope:target_country
					scope:target_country ?= {
						any_scope_state = {
							is_in_geographic_region = geographic_region_tyr_am_south_america
						}
					}
				}
			}
		}
		trigger_if = {
			limit = {
				tyr_am_active = yes
			}
			NOR = {
				has_variable = tyr_am_brazil_locked
				AND = {
					exists = global_var:tyr_am_spain
					is_in_same_power_bloc = global_var:tyr_am_spain
				}
				AND = {
					exists = c:USA
					is_in_same_power_bloc = c:USA
				}
			}
		}
	}
}

tyr_am_invitation_still_valid = {
	tyr_am_active = yes
	global_var:tyr_am_phase = 1
	exists = scope:tyr_am_patron
	scope:tyr_am_patron = {
		tyr_am_actor_valid = yes
		NOT = {
			has_war_with = root
		}
		OR = {
			prestige > root.prestige
			country_rank > root.country_rank
		}
		power_bloc = {
			NOT = {
				any_power_bloc_member = {
					is_diplomatic_play_enemy_of = root
				}
			}
		}
	}
	is_subject = no
	is_revolutionary = no
	NOT = {
		is_in_power_bloc = yes
	}
}

tyr_am_war_event_valid = {
	tyr_am_active = yes
	trigger_if = {
		limit = {
			exists = scope:tyr_am_war
			exists = scope:tyr_am_belligerent
			exists = scope:tyr_am_target
		}
		scope:tyr_am_war = {
			is_war_participant = scope:tyr_am_target
			OR = {
				attacker_warleader = scope:tyr_am_belligerent
				defender_warleader = scope:tyr_am_belligerent
			}
		}
		scope:tyr_am_target = {
			has_war_with = scope:tyr_am_belligerent
		}
	}
	trigger_else = {
		always = no
	}
}

# Country candidate; requires saved scope tyr_am_settlement_spain.
tyr_am_viceroyalty_candidate = {
	trigger_if = {
		limit = {
			exists = scope:tyr_am_settlement_spain
		}
		exists = capital
		NOT = {
			this = scope:tyr_am_settlement_spain
		}
		NOT = {
			is_country_type = decentralized
		}
		OR = {
			is_subject_of = scope:tyr_am_settlement_spain
			is_in_same_power_bloc = scope:tyr_am_settlement_spain
		}
		capital = {
			tyr_am_viceroyalty_$AREA$_state = yes
		}
	}
	trigger_else = {
		always = no
	}
}

# Winning Spain; does not read selection variables or temporary scopes.
tyr_am_viceroyalty_available = {
	any_country = {
		exists = capital
		NOT = {
			this = root
		}
		NOT = {
			is_country_type = decentralized
		}
		OR = {
			is_subject_of = root
			is_in_same_power_bloc = root
		}
		capital = {
			tyr_am_viceroyalty_$AREA$_state = yes
		}
	}
	# An existing special tag must be in this settlement, so it can be reused
	# or absorbed before the surviving country changes tag.
	trigger_if = {
		limit = {
			exists = c:$TAG$
		}
		c:$TAG$ = {
			OR = {
				is_subject_of = root
				is_in_same_power_bloc = root
			}
			capital = {
				tyr_am_viceroyalty_$AREA$_state = yes
			}
		}
	}
}

# Holder = principal; target_country = a local direct subject. Its overlord
# remains the principal stakeholder, following native Transfer Subject.
tyr_am_subject_area_wargoal_possible = {
	tyr_am_area_wargoal_actor = yes
	exists = scope:target_country
	trigger_if = {
		limit = {
			exists = scope:target_country
		}
		scope:target_country = {
			is_subject = yes
			exists = capital
			capital ?= {
				is_in_geographic_region = geographic_region_tyr_am_$AREA$
			}
			exists = overlord
			overlord ?= {
				tyr_am_area_wargoal_actor = yes
				NOT = {
					this = root
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}
