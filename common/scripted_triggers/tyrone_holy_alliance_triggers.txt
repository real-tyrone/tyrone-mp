# Country scope: ROOT is the prospective initiator; TARGET is the rival court.
# Replicates script-visible leadership-play requirements. The native
# CountryFormation.CanLaunchLeadershipPlay check is not callable as a trigger.
tyr_holy_brothers_war_target_ready = {
    exists = $TARGET$
    $TARGET$ = {
        is_unification_candidate = GER
        is_at_war = no
        is_diplomatic_play_committed_participant = no
    }
    NOT = { has_truce_with = $TARGET$ }
    NOT = { is_diplomatic_play_participant_with = $TARGET$ }
}
