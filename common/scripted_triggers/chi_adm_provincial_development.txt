    OR = {
        is_at_war = yes
        custom_tooltip = {
            text = chi_adm_provincial_development_warrant_hostile_neighbor_tt
            any_country = {
                NOT = { chi_qing_realm = yes }
                is_adjacent_to_country = root
                relations:root <= relations_threshold:hostile
                army_power_projection >= { value = chi_adm_provincial_development_imperial_power multiply = 2 }
            }
        }
    }
}
