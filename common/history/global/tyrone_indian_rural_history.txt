GLOBAL = {
	# Region traits also cover Indian princely states and persist through ownership changes.
	every_country = {
		every_scope_state = {
			limit = {
				is_in_geographic_region = geographic_region_india
			}
			state_region = {
				if = {
					limit = {
						NOT = { has_state_trait = state_trait_tyrone_indian_rural_region }
					}
					add_state_trait = state_trait_tyrone_indian_rural_region
				}
			}
		}
	}
}
