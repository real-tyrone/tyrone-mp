# Country scope: the shared JE calls this once per month for India.
tyrone_jewel_monthly_events = {
	trigger = {
		tyrone_jewel_country_active = yes
		NOT = { has_variable = tyrone_jewel_monthly_pending }
	}
	random_events = {
		90 = 0
		10 = tyrone_jewel.900
	}
}

# Overlord scope; the dispatcher preserves the Indian country as csr_india.
tyrone_jewel_incident_pool = {
	trigger = {
		exists = scope:csr_india
		scope:csr_india ?= {
			tyrone_jewel_country_active = yes
			overlord ?= root
			NOT = { has_variable = tyrone_jewel_monthly_pending }
		}
	}
	random_events = {
		10 = tyrone_jewel.1 # A Disputed Route
		10 = tyrone_jewel.2 # The Relief Accounts
		10 = tyrone_jewel.3 # Places in the Service
		10 = tyrone_jewel.4 # The Supplier's Conditions
		10 = tyrone_jewel.5 # Petitions before the Council
		10 = tyrone_jewel.6 # The Governor's Dispatch
		10 = tyrone_jewel.7 # A National Delegation
		10 = tyrone_jewel.8 # The Cost of Holding India
	}
}
