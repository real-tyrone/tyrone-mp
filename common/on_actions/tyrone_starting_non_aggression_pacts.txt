on_game_started_after_lobby = {
	on_actions = { tyrone_starting_non_aggression_pacts }
}

tyrone_starting_non_aggression_pacts = {
	effect = {
		if = {
			limit = {
				game_date >= 1836.1.1
				game_date < 1837.1.1
			}

			if = {
				limit = {
					c:PER ?= { is_ai = yes }
				}
				create_treaty = {
					first_country = c:PER
					second_country = c:KAB
					is_draft = no
					binding_period = { years = 4 }
					articles_to_create = {
						{ article = non_aggression_pact }
					}
				}
				create_treaty = {
					first_country = c:PER
					second_country = c:KAN
					is_draft = no
					binding_period = { years = 4 }
					articles_to_create = {
						{ article = non_aggression_pact }
					}
				}
				create_treaty = {
					first_country = c:PER
					second_country = c:HER
					is_draft = no
					binding_period = { years = 4 }
					articles_to_create = {
						{ article = non_aggression_pact }
					}
				}
			}

			if = {
				limit = {
					c:FRA ?= { is_ai = yes }
					c:PRU ?= { is_ai = yes }
				}
				create_treaty = {
					first_country = c:FRA
					second_country = c:PRU
					is_draft = no
					binding_period = { years = 4 }
					articles_to_create = {
						{ article = non_aggression_pact }
					}
				}
			}
		}
	}
}
