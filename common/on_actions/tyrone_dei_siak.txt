# Catch Dutch puppets acquired after the one-time East Indies consolidation.
on_monthly_pulse_country = {
    on_actions = { tyrone_dei_integrate_siak }
}

tyrone_dei_integrate_siak = {
    trigger = {
        c:NET ?= this
        c:DEI ?= { is_subject_of = c:NET }
        c:SAK ?= {
            is_subject_of = c:NET
            is_subject_type = subject_type_puppet
            is_at_war = no
            is_diplomatic_play_committed_participant = no
        }
    }
    effect = {
        c:DEI = { annex = c:SAK }
    }
}
