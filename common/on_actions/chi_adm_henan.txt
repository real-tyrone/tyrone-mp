on_monthly_pulse_country = { on_actions = { chi_adm_henan_monthly } }
chi_adm_henan_monthly = {
 trigger = {
  OR = {
   has_variable = chi_adm_henan_stage
   has_modifier = chi_adm_henan_compensation
  }
 }
 effect = {
  if = {
   limit = {
    has_modifier = chi_adm_henan_compensation
    NOT = { c:AHU ?= this chi_adm_henan_viceroyalty = yes }
   }
   remove_modifier = chi_adm_henan_compensation
  }
  if = {
   limit = { has_variable = chi_adm_henan_stage }
   if = {
    limit = { NOT = { c:AHU ?= this chi_adm_henan_settlement_valid = yes } }
    chi_adm_henan_end_without_transfer = { COOLDOWN = 6 }
    c:CHI ?= { trigger_event = { id = chi_adm_henan.5 popup = yes } }
   }
   else = {
    change_variable = { name = chi_adm_henan_months add = -1 }
    if = {
     limit = { var:chi_adm_henan_months <= 0 }
     chi_adm_henan_end_without_transfer = { COOLDOWN = 6 }
     c:CHI ?= { trigger_event = { id = chi_adm_henan.5 popup = yes } }
    }
    else_if = {
     limit = { var:chi_adm_henan_stage = 2 var:chi_adm_henan_months <= 3 }
     set_variable = { name = chi_adm_henan_stage value = 1 }
     chi_adm_henan_show_proposal = yes
    }
   }
  }
 }
}
