# Country scope. Custom GUI controls are not automatically evaluated by the AI.
# Use the same proposal effect and eligibility as the player's journal controls.
on_monthly_pulse_country = {
	on_actions = { chi_adm_negotiated_integration_ai_court }
}

chi_adm_negotiated_integration_ai_court = {
	trigger = {
		is_ai = yes
		chi_adm_negotiated_integration_court = yes
		has_journal_entry = je_qing_imperial_administrations
	}
	effect = {
		# Revisit an unanswered court decision, including cases left open in old saves.
		random_country = {
			limit = {
				chi_adm_negotiated_integration_valid = yes
				var:chi_adm_negotiated_integration_stage = 2
			}
			save_scope_as = chi_adm_negotiated_integration_target
			save_scope_value_as = {
				name = chi_adm_negotiated_integration_revision
				value = var:chi_adm_negotiated_integration_revision
			}
			c:CHI = { trigger_event = { id = chi_adm_negotiated_integration.2 popup = yes } }
		}
		# Prefer a contiguous expansion of direct imperial territory.
		if = {
			limit = {
				any_country = {
					chi_adm_negotiated_integration_can_propose = yes
					is_adjacent_to_country = c:CHI
				}
			}
			random_country = {
				limit = {
					chi_adm_negotiated_integration_can_propose = yes
					is_adjacent_to_country = c:CHI
				}
				chi_adm_negotiated_integration_propose = yes
			}
		}
		else = {
			random_country = {
				limit = { chi_adm_negotiated_integration_can_propose = yes }
				chi_adm_negotiated_integration_propose = yes
			}
		}
	}
}
