    on_actions = {
    }

        200 = 0
		10 = meiji.4
		10 = meiji.5
		10 = meiji.6
    }
    events = {
        springtime.1
        eiffel_tower.1
        company_events.1
        company_events.2
        company_events.3
        company_events.4
        laissez_faire_subject.1
        flag_events.1
        flag_events.2
        austria_integrate.1
        austria_integrate.2
    }
    effect = {
		if = {
			limit = {
				c:FRA ?= this
				has_law_or_variant = law_type:law_monarchy
				is_at_war = no
				NOT = { has_variable = france_alsace_lorraine_republic_event_fired }
				NOT = { owns_entire_state_region = STATE_ALSACE_LORRAINE }
			}
			trigger_event = { id = empire_of_france.10 popup = yes }
		}
		if = { # Berlin Conference
			limit = {
				NOT = { has_global_variable = berlin_conference_happening }
				NOT = { has_global_variable = berlin_conference_ended }
                OR = {
                    has_law = law_type:law_colonial_resettlement
                    has_law = law_type:law_colonial_exploitation
                }
				is_in_geographic_region = geographic_region_europe
				has_technology_researched = civilizing_mission
				country_rank >= rank_value:major_power
			}
			set_global_variable = berlin_conference_happening
			every_country_in_europe = {
				limit = {
				    country_rank >= rank_value:major_power
                    OR = {
                        has_law = law_type:law_colonial_resettlement
                        has_law = law_type:law_colonial_exploitation
                        country_rank = rank_value:great_power
                    }
				}
				trigger_event = { id = berlin_conference.1 popup = yes }
			}
		}
    }
}

tyrone_downgrade_severe_malaria_on_start = {
	effect = {
		if = {
			limit = {
				has_game_rule = severe_malaria_downgraded
			}
			every_state = {
				limit = {
					has_state_trait = state_trait_severe_malaria
				}
				state_region = {
					remove_state_trait = state_trait_severe_malaria
					add_state_trait = state_trait_malaria
				}
			}
            set_global_variable = malaria_downgraded
		}
	}
}

on_game_started = {
	on_actions = {
		tyrone_downgrade_severe_malaria_on_start
	}
}

        }
    }
}
