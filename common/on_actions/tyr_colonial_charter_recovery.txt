on_monthly_pulse_country = {
    on_actions = { tyr_colonial_charter_recovery }
}

tyr_colonial_charter_recovery = {
    trigger = {
        tyr_colonial_administration_needs_charter = yes
        NOT = { has_variable = tyr_colonial_charter_offer_pending }
    }
    effect = {
        save_scope_as = tyr_regional_colony
        set_variable = { name = tyr_colonial_charter_offer_pending months = 2 }
        overlord ?= {
            trigger_event = { id = tyr_regional_colonies.1 popup = yes }
        }
    }
}
