REPLACE:metternich_agenda_value = {
	value = 0
	if = {
		limit = {
			NOT = {
				exists = c:ITA
			}
		}
		add = 1
	}
	if = {
		limit = {
			OR = {
				c:GER ?= this
				NOT = {
					exists = c:GER
				}
			}
		}
		add = 1
	}
	if = {
		limit = {
			leading_producer_of = g:fine_art
		}
		add = 1
	}
	if = {
		limit = {
			owns_entire_state_region = STATE_WEST_GALICIA
		}
		add = 1
	}
	if = {
		limit = {
			techs_researched >= 52
		}
		add = 1
	}
	if = {
		limit = {
			NOT = {
				any_political_movement = {
					OR = {
						movement_is_liberal = yes
						movement_is_socialist = yes
					}
					political_movement_support >= 0.05
				}
			}
		}
		add = 1
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_autocracy
			has_law_or_variant = law_type:law_secret_police
			has_law_or_variant = law_type:law_censorship
		}
		add = 1
	}
	if = {
		limit = {
			institution:institution_police.investment >= 2
			institution:institution_home_affairs.investment >= 2
		}
		add = 1
	}
	if = {
		limit = {
			power_bloc ?= {
				num_power_bloc_members >= metternich_bloc_target
			}
		}
		add = 1
	}
	if = {
		limit = {
			power_bloc ?= {
				current_cohesion_number >= metternich_bloc_cohesion_target
			}
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = ferdinand_I_abdicated_var
		}
		add = 1
	}
	if = {
		limit = {
			has_variable = support_metternich_var
		}
		add = 4
	}
}

american_exceptionalism_agenda_value = {
	value = 0
	# --- Expansion ---
	# Complete Manifest Destiny
	if = {
		limit = {
			has_variable = je_manifest_destiny_mexico_complete
			has_variable = je_manifest_destiny_frontier_wars_complete
		}
		add = 1
	}
	# Acquire Alaska
	if = {
		limit = { owns_entire_state_region = STATE_ALASKA }
		add = 1
	}
	# Acquire Hawaii
	if = {
		limit = { owns_entire_state_region = STATE_HAWAIIAN_ISLANDS }
		add = 1
	}
	# --- Diplomacy ---
	# Lead a Power Bloc with 5+ Americas members
	if = {
		limit = {
			power_bloc ?= {
				any_power_bloc_member = {
					capital = { is_in_geographic_region = geographic_region_americas }
					count >= 5
				}
			}
		}
		add = 1
	}
	# Lead a top 3-ranked Power Bloc
	if = {
		limit = {
			power_bloc ?= { power_bloc_rank <= 3 }
		}
		add = 1
	}
	# Enforce the Monroe Doctrine: 80% of North and South American states in your power bloc
	if = {
		limit = {
			ae_monroe_doctrine_ratio >= 0.8
		}
		add = 1
	}
	# --- Military ---
	# At least 100 capital ships
	if = {
		limit = {
			"num_ships_of_group(ship_group:ship_group_capital_ships)" >= 100
		}
		add = 1
	}
	# 100+ battalions with trench_works researched and 50+ trench infantry
	if = {
		limit = {
			has_technology_researched = trench_works
			army_size >= 100
			country_army_unit_type_fraction = {
				target = unit_type:combat_unit_type_trench_infantry
				value >= 0.5
			}
		}
		add = 1
	}
	# Top 3 producer of Small Arms, Artillery, Ammunition, or Ironclads
	if = {
		limit = {
			OR = {
				"goods_production_rank(g:small_arms)" <= 3
				"goods_production_rank(g:artillery)" <= 3
				"goods_production_rank(g:ammunition)" <= 3
				"goods_production_rank(g:ironclads)" <= 3
			}
		}
		add = 1
	}
	# --- Domestic ---
	# Abolish Slavery
	if = {
		limit = { has_law_or_variant = law_type:law_slavery_banned }
		add = 1
	}
	# Enact Multiculturalism
	if = {
		limit = { has_law_or_variant = law_type:law_multicultural }
		add = 1
	}
	# Enact Guaranteed Liberties
	if = {
		limit = { has_law_or_variant = law_type:law_guaranteed_liberties }
		add = 1
	}
	# --- Economy ---
	# Top 3 producer of Tools or Steel
	if = {
		limit = {
			OR = {
				"goods_production_rank(g:tools)" <= 3
				"goods_production_rank(g:steel)" <= 3
			}
		}
		add = 1
	}
	# Top 3 GDP
	if = {
		limit = { gdp_ranking <= 3 }
		add = 1
	}
	# Steel-Frame Buildings researched and active in 100% of construction sectors
	if = {
		limit = {
			has_technology_researched = steel_frame_buildings
			any_scope_state = {
				any_scope_building = {
					filter = {
						is_building_type = building_construction_sector
					}
					percent = 1
					has_active_production_method = pm_steel_frame_buildings
				}
			}
		}
		add = 1
	}
}

ae_monroe_americas_total_states = {
	value = 0
	every_state = {
		limit = {
			state_region = {
				OR = {
					is_in_geographic_region = geographic_region_north_america
					is_in_geographic_region = geographic_region_south_america
				}
			}
		}
		add = 1
	}
	min = 1
}

ae_monroe_doctrine_ratio = {
	value = 0
	every_state = {
		limit = {
			state_region = {
				OR = {
					is_in_geographic_region = geographic_region_north_america
					is_in_geographic_region = geographic_region_south_america
				}
			}
			owner = {
				is_in_same_power_bloc = root
			}
		}
		add = 1
	}
	divide = ae_monroe_americas_total_states
}
