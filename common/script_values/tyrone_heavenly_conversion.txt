# Country scope. Evaluated only at JE activation and its monthly pulse.
tyr_heavenly_christian_share = {
	value = 0
	every_scope_pop = {
		limit = {
			OR = {
				has_pop_religion = catholic
				has_pop_religion = protestant
			}
		}
		add = total_size
	}
	if = {
		limit = { total_population > 0 }
		divide = total_population
	}
	else = { multiply = 0 }
	min = 0
	max = 1
}

# Cheap UI/goal value: never iterate over Pops from a tooltip.
tyr_heavenly_cached_christian_percent = {
	value = 0
	if = {
		limit = { has_variable = tyr_heavenly_christian_share_var }
		add = var:tyr_heavenly_christian_share_var
	}
	multiply = 100
}
