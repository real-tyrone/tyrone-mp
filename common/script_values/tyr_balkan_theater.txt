tyr_bt_aus_share = {
	value = 0

	if = {
		limit = {
			has_global_variable = tyr_bt_aus_gdp
			has_global_variable = tyr_bt_total_gdp
		}
		if = {
			limit = {
				global_var:tyr_bt_total_gdp > 0
			}
			value = global_var:tyr_bt_aus_gdp
			multiply = 150
			divide = global_var:tyr_bt_total_gdp
		}
	}

}
