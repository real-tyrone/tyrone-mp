# JE scope: UI-only values; resolve the Spanish sponsor for every JE viewer.
scw_ui_institutions_advance_active = {
	value = 0
	if = {
		limit = {
			any_country = {
				OR = { c:SPA ?= this was_formed_from = SPA }
				scw_institutions_monthly_advance = yes
			}
		}
		add = 1
	}
}

scw_ui_institutions_retreat_active = {
	value = 0
	if = {
		limit = {
			any_country = {
				OR = { c:SPA ?= this was_formed_from = SPA }
				scw_institutions_monthly_retreat = yes
			}
		}
		add = 1
	}
}

scw_ui_institutions_congress_active = {
	value = 0
	if = {
		limit = {
			any_country = {
				OR = { c:SPA ?= this was_formed_from = SPA }
				has_variable = scw_running
				NOT = { has_variable = scw_programme_cooldown }
				var:scw_progress >= 25
			}
		}
		add = 1
	}
}
