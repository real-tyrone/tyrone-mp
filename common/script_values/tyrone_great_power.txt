tyr_gp_population_target = {
	value = 0
	if = {
		limit = { has_variable = tyr_gp_initial_population }
		add = var:tyr_gp_initial_population
		multiply = 1.5
	}
}

# Compare like-for-like script values, rather than a native population trigger
# against a separately evaluated target. Missing/zero baselines score nothing.
tyr_gp_population_ratio = {
	value = 0
	if = {
		limit = {
			has_variable = tyr_gp_initial_population
		}
		if = {
			limit = {
				var:tyr_gp_initial_population > 0
			}
			add = total_population
			divide = var:tyr_gp_initial_population
		}
	}
}

tyr_gp_bloc_population = {
	value = 0
	if = {
		limit = { is_in_power_bloc = yes }
		power_bloc = {
			every_power_bloc_member = {
				add = total_population
			}
		}
	}
}

	value = 0
	if = {
		limit = {
			exists = capital
		}
		capital = {
			add = state_population
		}
	}
}
