tyrone_ch_hindustani_population = {
	value = 0
	every_scope_pop = {
		limit = { has_pop_culture = hindustani }
		add = workforce
		add = dependents
	}
}
tyrone_ch_hindustani_atheist_share = {
	value = 0
	every_scope_pop = {
		limit = {
			has_pop_culture = hindustani
			has_pop_religion = atheist
		}
		add = workforce
		add = dependents
	}
	divide = {
		value = tyrone_ch_hindustani_population
		min = 1
	}
}
tyrone_ch_hindustani_radical_share = {
	value = 0
	every_scope_pop = {
		limit = { has_pop_culture = hindustani }
		add = {
			value = workforce
			add = dependents
			multiply = pop_radical_fraction
		}
	}
	divide = {
		value = tyrone_ch_hindustani_population
		min = 1
	}
}
