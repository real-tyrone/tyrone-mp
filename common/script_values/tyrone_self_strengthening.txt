tyr_ssm_duration = 7300

tyr_ssm_domestic_industry_levels = {
	value = 0
	every_scope_state = {
		every_scope_building = {
			limit = {
				OR = { is_building_group = bg_light_industry is_building_group = bg_heavy_industry }
			}
			add = level
		}
	}
}

tyr_ssm_industry_levels = {
 value = tyr_ssm_domestic_industry_levels
 if = {
  limit = { c:CHI ?= this }
  every_subject_or_below = { limit = { tyr_ssm_qing_program_administration = yes } add = tyr_ssm_domestic_industry_levels }
 }
}

# Arms Industries must supply the portion of the 30 levels not covered by the other two types.
tyr_ssm_arms_industry_levels_required = {
	value = 30
	subtract = "country_has_building_type_levels(bt:building_munition_plant)"
	subtract = "country_has_building_type_levels(bt:building_artillery_foundry)"
	min = 0
}

tyr_ssm_industry_next_target = {
	value = tyr_ssm_cached_industry_levels
	multiply = 0.20
	ceiling = yes
	min = 50
	add = tyr_ssm_cached_industry_levels
}

tyr_ssm_industry_bar_max = {
	value = 50
	if = { limit = { has_variable = tyr_ssm_industry_target } value = var:tyr_ssm_industry_target }
	min = 1
}

tyr_ssm_industry_rounds = {
	value = 0
	if = { limit = { has_variable = tyr_ssm_industrial_rounds } value = var:tyr_ssm_industrial_rounds }
	min = 0
	max = 5
}

tyr_ssm_cached_industry_levels = {
	value = 0
	if = {
		limit = { has_variable = tyr_ssm_industry_levels }
		value = var:tyr_ssm_industry_levels
	}
}
