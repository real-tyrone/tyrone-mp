tyr_kk_int_clout = {
	value = 0

	ig:ig_intelligentsia ?= {
		value = ig_clout
	}
}

tyr_kk_devout_clout = {
	value = 0

	ig:ig_devout ?= {
		value = ig_clout
	}
}

tyr_kk_monthly_change = {
	value = 0

	if = {
		limit = {
			tyr_kk_rule_public_schools = yes
		}
		add = 1
	}

	if = {
		limit = {
			tyr_kk_rule_separation = yes
		}
		add = 1
	}

	if = {
		limit = {
			tyr_kk_rule_int_clout = yes
		}
		add = 1
	}

	if = {
		limit = {
			tyr_kk_rule_int_government = yes
		}
		add = 1
	}

	if = {
		limit = {
			tyr_kk_rule_religious_schools = yes
		}
		add = -1
	}

	if = {
		limit = {
			tyr_kk_rule_state_religion = yes
		}
		add = -1
	}

	if = {
		limit = {
			tyr_kk_rule_devout_clout = yes
		}
		add = -1
	}

	if = {
		limit = {
			tyr_kk_rule_devout_government = yes
		}
		add = -1
	}

	if = {
		limit = {
			tyr_kk_rule_public_health = yes
		}
		add = 1
	}

	if = {
		limit = {
			tyr_kk_rule_charity_hospitals = yes
		}
		add = -1
	}

	min = -2
	max = 2
}

tyr_kk_balance = {
	value = 50

	if = {
		limit = {
			has_variable = tyr_kk_balance
		}
		value = var:tyr_kk_balance
	}

	min = 0
	max = 100
}

tyr_kk_next_balance = {
	value = tyr_kk_balance
	add = tyr_kk_monthly_change
	min = 0
	max = 100
}
