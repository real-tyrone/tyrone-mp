REPLACE:naval_battle_size = {
	value = 1

	if = {
		limit = {
			exists = scope:military_formation
			scope:military_formation = { is_fleet = yes }
		}
		value = {
			add = scope:military_formation.num_ships_not_in_battle
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}
	else_if = {
		limit = {
			exists = scope:military_formation
		}
		value = {
			add = scope:military_formation.num_assigned_supply_ships
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}
	else_if = {
		limit = {
			exists = scope:enemy_military_formation
		}
		value = {
			add = scope:enemy_military_formation.num_ships
			desc = "NAVAL_BATTLE_SIZE_ADMIRAL_SHARE"
		}
	}

	save_temporary_value_as = pre_max_amount_cap

	if = {
		limit = {
			exists = scope:own_commander
			scope:pre_max_amount_cap > {
				value = 0
				scope:own_commander = {
					add = this.command_limit_num_units
				}
				multiply = 0.5
			}
		}
		max = {
			add = {
				scope:own_commander = {
					add = this.command_limit_num_units
				}
			}
			multiply = 0.5
			desc = "[concept_command_limit]"
		}
		min = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
	}
	else_if = {
		limit = {
			NOT = { exists = scope:own_commander }
		}
		max = define:NMilitary|MILITARY_FORMATION_COMMAND_LIMIT_NO_COMMANDER
	}

	save_temporary_value_as = post_command_limit_base_amount_cap

	if = {
		limit = {
			exists = scope:military_formation
			exists = scope:enemy_military_formation
		}
		value = scope:post_command_limit_base_amount_cap
		add = {
			add = {
				scope:military_formation = {
					add = this.detection
				}
				divide = {
					value = scope:military_formation.num_ships_not_in_battle
					min = 1
				}
			}
			multiply = {
				scope:enemy_military_formation = {
					add = this.visibility
				}
				divide = {
					value = scope:enemy_military_formation.num_ships_not_in_battle
					min = 1
				}
			}
			divide = {
				add = scope:military_formation.num_ships_not_in_battle
				add = scope:enemy_military_formation.num_ships_not_in_battle
				min = 1
			}
		}
	}
	else_if = {
		limit = {
			exists = scope:military_formation
		}
		value = scope:post_command_limit_base_amount_cap
		add = scope:military_formation.num_ships_not_in_battle
	}
	else_if = {
		limit = {
			exists = scope:enemy_military_formation
		}
		value = scope:post_command_limit_base_amount_cap
		add = scope:enemy_military_formation.num_ships_not_in_battle
	}
	else = {
		value = scope:post_command_limit_base_amount_cap
	}
}
