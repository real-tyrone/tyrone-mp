# Country scope: India; country building iterator excludes subjects.
tyrone_raj_schooling_universities = {
	value = 0
	every_scope_building = {
		limit = { is_building_type = building_university }
		add = level
	}
	divide = 50
}

tyrone_raj_schooling_institution = {
	value = 0
	if = {
		limit = { institution_investment_level = { institution = institution_schools value >= 1 } }
		add = 1
	}
	if = {
		limit = { institution_investment_level = { institution = institution_schools value >= 2 } }
		add = 1
	}
	if = {
		limit = { institution_investment_level = { institution = institution_schools value >= 3 } }
		add = 1
	}
	if = {
		limit = { institution_investment_level = { institution = institution_schools value >= 4 } }
		add = 1
	}
	if = {
		limit = { institution_investment_level = { institution = institution_schools value >= 5 } }
		add = 1
	}
}

# Enter a valid country before evaluating triggers. Priority: BIC, BHT, BHR.
tyrone_raj_schooling_bar_universities = {
	value = 0
	c:BIC ?= {
		add = tyrone_raj_schooling_universities
	}
	c:BHT ?= {
		if = {
			limit = {
				NOT = { exists = c:BIC }
			}
			add = tyrone_raj_schooling_universities
		}
	}
	c:BHR ?= {
		if = {
			limit = {
				NOT = { exists = c:BIC }
				NOT = { exists = c:BHT }
			}
			add = tyrone_raj_schooling_universities
		}
	}
}
