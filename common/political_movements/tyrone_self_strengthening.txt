movement_tyr_self_strengthening = { # This movement loves you and wants you to succeed. It exists to help you catch up.
	category = movement_category_ideological

	ideology = ideology_tyr_self_strengthening
	character_ideologies = {
		ideology_modernizer_leader
		ideology_reformer
		ideology_protectionist
		ideology_market_liberal
		ideology_abolitionist
		ideology_state_socialist
	}

	creation_trigger = { always = no }

	creation_weight = {
		value = 100
		if = {
			limit = {
				owner ?= { # If a movement of this type exists in a neighbouring country, make it more likely to spawn here as well
					any_neighbouring_state = {
						owner = {
							prestige > ROOT.owner.prestige
							any_political_movement = {
								is_political_movement_type = movement_tyr_self_strengthening
							}
						}
					}
				}
			}
			multiply = {
				value = 2
			}
		}
	}

	on_created = { add_movement_enthusiasm_modifier = yes }

	disband_trigger = { always = no }
	on_disbanded = {
		fold_movement_into_anti_communist_movement_effect = yes
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_reformer
			has_ideology = ideology:ideology_protectionist
			has_ideology = ideology:ideology_positivist
			has_ideology = ideology:ideology_abolitionist
			has_ideology = ideology:ideology_market_liberal
			has_ideology = ideology:ideology_sovereignist
			has_ideology = ideology:ideology_utopian
			has_ideology = ideology:ideology_modernizer_leader
			has_ideology = ideology:ideology_radical
			has_ideology = ideology:ideology_state_socialist
			scope:interest_group ?= {
				OR = {
					has_ideology = ideology:ideology_cartist
					has_ideology = ideology:ideology_modernizer
					has_ideology = ideology:ideology_mercantile
					has_ideology = ideology:ideology_liberal
					has_ideology = ideology:ideology_universalist
					has_ideology = ideology:ideology_liberal_modern
					has_ideology = ideology:ideology_universalist_modern
					has_ideology = ideology:ideology_plutocratic
				}
			}
		}
	}

	character_support_weight = {
		value = 200
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_modernizer_leader
					has_ideology = ideology:ideology_protectionist
					has_ideology = ideology:ideology_reformer
					has_ideology = ideology:ideology_sovereignist
					has_ideology = ideology:ideology_market_liberal
				}
			}
			multiply = {
				value = 5
			}
		}
		else_if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_positivist
					has_ideology = ideology:ideology_abolitionist
					has_ideology = ideology:ideology_utopian
				}
			}
			multiply = {
				value = 3
			}
		}
		if = {
			limit = {
				scope:interest_group ?= {
					OR = {
						has_ideology = ideology:ideology_cartist
						has_ideology = ideology:ideology_modernizer
						has_ideology = ideology:ideology_mercantile
						has_ideology = ideology:ideology_liberal
						has_ideology = ideology:ideology_universalist
						has_ideology = ideology:ideology_liberal_modern
						has_ideology = ideology:ideology_universalist_modern
						has_ideology = ideology:ideology_plutocratic
					}
				}
			}
			multiply = {
				value = 2
			}
		}
		if = {
			limit = {
				owner ?= {
					country_should_exclusively_focus_on_modernization = no
				}
			}
			multiply = 0.5
		}
		if = {
			limit = {
				owner ?= {
					country_has_laws_in_need_of_critical_modernization = no
				}
			}
			multiply = 0.5
		}
	}

	pop_support_trigger = {
		NOR = {
			is_pop_type = clergymen
			is_pop_type = peasants
			is_pop_type = slaves
		}
	}

	pop_support_factors = {
		movement_support_je_hungarian_revolution
		movement_support_high_urbanization
		movement_support_high_literacy
		movement_support_capitalists
		movement_support_aristocrats
		movement_support_officers
		movement_support_academics
		movement_support_bureaucrats
		movement_support_engineers
		movement_support_clerks
		movement_support_soldiers
		movement_support_lost_wars
		movement_support_backwardsness
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
						OR = {
							is_being_suppressed = yes
							is_being_bolstered = yes
						}
					}
				}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 15
					desc = "POP_CAPITALISTS"
				}
			}
			if = {
				limit = {
					is_pop_type = aristocrats
				}
				if = {
					limit = {
						owner ?= {
							OR = {
								has_journal_entry = je_great_reformer
								has_journal_entry = je_tp_tsar_great_reform
							}
						}
					}
					add = {
						value = 25
						desc = "POP_ARISTOCRATS"
					}
				}
				else_if = {
					limit = {
						culture = cu:hungarian
						owner ?= {
							has_journal_entry = je_hungarian_revolution
						}
					}
					add = {
						value = 25
						desc = "je_hungarian_revolution"
					}
				}
				else = {
					add = {
						value = 10
						desc = "POP_ARISTOCRATS"
					}
				}
			}
			if = {
				limit = {
					owner = {
						has_variable = wargoal_enforced_by_recognized_power
					}
				}
				add = {
					value = 15
					desc = "wargoal_pressed_by_recognized_power_tt"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 15
					desc = "POP_OFFICERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 25
					desc = "POP_ACADEMICS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = bureaucrats
				}
				add = {
					value = 15
					desc = "POP_BUREAUCRATS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = engineers
				}
				add = {
					value = 25
					desc = "POP_BUREAUCRATS"
				}
			}
			else = {
				add = {
					value = 10
					desc = "MIDDLE_NO_ICON"
				}
			}
			if = {
				limit = {
					owner = {
						has_variable = wargoal_enforced_by_recognized_power
					}
				}
				add = {
					value = 10
					desc = "wargoal_pressed_by_recognized_power_tt"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 9
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 6
					desc = "POP_SOLDIERS"
				}
			}
			else = {
				add = {
					value = 3
					desc = "LOWER_NO_ICON"
				}
			}
		}

		multiply = { # Minority pops are substantially less likely to help modernise a horrific ethnostate
			value = 0.011
			multiply = pop_acceptance
			desc = "ACCEPTANCE"
		}

		multiply = { # Illiterate pops participate at 1/10 the rate of fully literate pops
			desc = "POP_LITERACY"
			value = literacy_rate
			add = 0.10
		}

		multiply = { # "What are you doing?" factor
			desc = "BACKWARDSNESS"
			value = 1
			add = {
				value = years_since_game_start
				divide = 100
			}
		}
	}

	revolution = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 100
		}

		state_weight = {
			value = state_urbanization_rate
			add = scope:clout
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
			scope:clout <= 0
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = state_urbanization_rate
			add = { value = this.turmoil divide = 2 }
			subtract = this.loyalists
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner = {
					has_variable = wargoal_enforced_by_recognized_power
				}
			}
			add = {
				desc = "wargoal_pressed_by_recognized_power_tt"
				value = 0.10
			}
		}
		if = {
			limit = {
				owner = {
					has_technology_researched = nationalism
					is_country_type = unrecognized
				}
			}
			add = {
				desc = "unrecognized"
				value = 0.10
			}
		}
		if = {
			limit = {
				neighboring_country_has_insurrectionary_movement_trigger = { TYPE = movement_tyr_self_strengthening }
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
		add = movement_radicalism_suppressed_after_civil_war
	}
}
