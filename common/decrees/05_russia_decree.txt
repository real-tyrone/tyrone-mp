		}
		NOT = { has_law_or_variant = law_type:law_multicultural }
	}
	unlocking_technologies = {
		centralization
		nationalism
	}

	cost = 75
	
	ai_weight = {
		value = 0

		if = {

				}
			}

			if = { # Colonies

				}

			}
			else_if = {
				limit = {
					scope:country = {
						OR = {
							has_strategy = ai_strategy_reactionary_agenda

			}
			else_if = {
				limit = {
					scope:country = {
						has_strategy = ai_strategy_nationalist_agenda
					}
				}
				add = 200
			}


			if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner
						value <= 0.1
					}
				}
				multiply = 5.0
			}
			else_if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner
						value <= 0.3

			}
			else_if = {
				limit = {
					primary_cultures_percent_state = {
						target_country = owner

			if = {
				limit = {
					is_homeland_of_country_cultures = scope:country

				multiply = 5.0

		if = {
			limit = {
				OR = {
					NOT = { scope:country = { has_law_or_variant = law_type:law_total_separation } }

					target = scope:country.religion
					value <= 0.9
				}
			}

			if = {
				limit = {
					scope:country = {
						has_strategy = ai_strategy_conservative_agenda
					}
				}
				add = 50
			}
			else_if = {
				limit = {

					}
				}

			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:country.religion
						value <= 0.3
					}
				}
				multiply = 3.0
			}
			else_if = {
				limit = {
					religion_percent_state = {
						target = scope:country.religion
						value <= 0.5
					}
				}
				multiply = 2.0
			}
		}

		if = {
			limit = {

			}
			multiply = 0
		}
		else_if = {

				state_population > 300000
			}
			multiply = 2.0
