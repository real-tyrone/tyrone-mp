# Italy Tyrone – dynamic name for Sardinia-Piedmont when North Italian Federation is formed.
# When c:SAR has variable NIF_formed (set in italyEventsTyrone.12), display as "North Italian Federation".
# Trigger follows vanilla pattern (e.g. colonial_admin_*, BUL bul_proclaimed_bulgarian_empire, SPA).

REPLACE:SAR = {
	dynamic_country_name = {
		name = dyn_c_north_italian_federation #North Italian Fed
		adjective = dyn_c_north_italian_federation_adj

		is_main_tag_only = yes
		priority = 10

		trigger = {
			exists = scope:actor
			scope:actor ?= {
				has_variable = NIF_formed
			}
		}
	}
	dynamic_country_name = {
		name = dyn_c_north_italian_theocracy_federation #North Italian Concordat
		adjective = dyn_c_north_italian_federation_adj

		is_main_tag_only = yes
		priority = 25

		trigger = {
			exists = scope:actor
			scope:actor ?= {
				has_variable = NIF_formed
				has_law = law_type:law_theocracy
			}
		}
	}
}
