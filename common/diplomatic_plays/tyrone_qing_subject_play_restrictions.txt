REPLACE:dp_reduce_autonomy = {
	war_goal = reduce_autonomy

	requires_interest_marker = no

	texture = "gfx/interface/icons/war_goals/reduce_autonomy.dds"

	possible = {
		tyr_qing_standard_play_allowed = yes

			NOT = {
				scope:target_country = { is_subject_type = subject_type_personal_union }
			}
			has_technology_researched = nationalism
		}

		OR = {
			NOT = {
				scope:target_country = { has_government_type = gov_chartered_company }
			}
			has_technology_researched = civilizing_mission
		}
		NAND = {
			is_involved_in_journal_entry = je_ryukyu_rivalry
			scope:target_country = {
				c:RYU ?= this
			}
		}
	}

	ai_acceptance_max = 90

	on_weekly_pulse = {}

	on_war_begins = {}

	on_war_end = {}

	on_demand_accepted = {
		scope:target_country = {
			add_liberty_desire = 50
		}
		custom_tooltip = {
			text = "ADD_LIBERTY_DESIRE_TO_ALL_SUBJECTS"
			every_direct_subject = {
				limit = {
					NOT = { this = scope:target_country }
				}
				add_liberty_desire = decrease_autonomy_liberty_desire_to_add
			}
		}
		hidden_effect = {
			scope:target_country = {
				create_diplomatic_catalyst = {
					type = catalyst_autonomy_change_imposed
					target = root
				}
			}
		}
	}

REPLACE:dp_annex_subject = {
	war_goal = annex_country

	texture = "gfx/interface/icons/war_goals/annex_country.dds"
	confirmation_sound = "event:/SFX/UI/SubjectInteractions/subject_button_small_annex"
	possible = {
		tyr_qing_standard_play_allowed = yes
		trigger_if = {
			limit = {
				c:BIC ?= this
				is_at_war = no
				NOT = { has_variable = doctrine_of_lapse_cooldown }
				NOT = { has_variable = doctrine_of_lapse_ended }
				scope:target_country = {
					is_subject_of = ROOT
					has_variable = princely_state
					NOT = { country_tier = empire }
					is_player = no
				}
			}
			custom_tooltip = {
				text = CAN_USE_DOCTRINE_OF_LAPSE_INSTEAD
				always = no
			}
		}
		trigger_if = {
			limit = {
				c:RUS ?= this
				c:POL ?= scope:target_country
			}
			NOT = { has_journal_entry = je_congress_poland }
		}

		NOT = { is_country_type = decentralized }
		aggressive_diplomatic_plays_permitted = yes
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_direct_subject_of = root
			OR = {
				has_diplomatic_pact = { who = root type = colony }
				has_diplomatic_pact = { who = root type = puppet }
				has_diplomatic_pact = { who = root type = vassal }
				has_diplomatic_pact = { who = root type = crown_land }
			}
		}
		NAND = {
			is_involved_in_journal_entry = je_peru_bolivia
			scope:target_country = {
				OR = {
					country_has_primary_culture = cu:quechua
					country_has_primary_culture = cu:south_andean
				}
			}
		}
		NAND = {
			is_involved_in_journal_entry = je_ryukyu_rivalry
			scope:target_country = {
				c:RYU ?= this
			}
		}
		NAND = {
			c:JAP ?= this
			has_law_or_variant = law_type:law_serfdom # Proxy for feudalism
		}
	}

	ai_acceptance_max = 75

	on_weekly_pulse = {}

	on_war_begins = {
		scope:initiator = {
			save_temporary_scope_as = annexing_overlord
			every_direct_subject = {
				limit = {
					NOT = { this = scope:target_country }
				}
				add_liberty_desire = annex_subject_base_liberty_desire_change
				add_modifier = {
					name = modifier_has_annexed_another_subject
					months = annex_subject_liberty_desire_modifier_months
					is_decaying = no
					multiplier = annex_subject_liberty_desire_weekly_change
				}
			}
		}
	}

	on_war_end = {}

	on_demand_accepted = {
		save_temporary_scope_as = annexing_overlord
		every_direct_subject = {
			limit = {
				NOT = { this = scope:target_country }
			}
			add_liberty_desire = annex_subject_base_liberty_desire_change
			add_modifier = {
				name = modifier_has_annexed_another_subject
				months = annex_subject_liberty_desire_modifier_months
				is_decaying = no
				multiplier = annex_subject_liberty_desire_weekly_change
			}
		}
	}
}

REPLACE:dp_unify_russia = {
	war_goal = annex_country

	requires_interest_marker = no
	mirror_war_goal = yes
	add_infamy_for_starting_initiator_wargoals = no

	texture = "gfx/interface/icons/war_goals/unification.dds"

	selectable_in_lens = {
		has_variable = rcw_russian_faction
	}

	possible = {
		tyr_qing_standard_play_allowed = yes
		has_variable = rcw_russian_faction
		NOT = { is_country_type = decentralized }
		is_subject = no
		is_russia_unification_pair = { TARGET = scope:target_country }

		scope:target_country = {
			NOT = { is_country_type = decentralized }
			aggressive_diplomatic_plays_permitted = yes
		}
	}

	additional_involvement_trigger = {
		has_variable = rcw_russian_faction
	}

	ai_acceptance_max = 75

	on_weekly_pulse = {
		if = {
			limit = {
				escalation < 100
			}
			add_escalation = 7
		}
	}

	on_war_begins = {}

	on_war_end = {
		every_scope_play_involved = {
			limit = {
				has_variable = rcw_russian_faction
				is_country_alive = yes
			}
			set_variable = {
				name = rcw_recent_war
				days = 90
			}
		}
	}
}

REPLACE:dp_restore_union = {
	war_goal = restore_union

	texture = "gfx/interface/icons/war_goals/enforce_treaty_article.dds"

	selectable_in_lens = {
		has_law = law_type:law_monarchy
		has_variable = por_seeking_restore_union_var
	}

	possible = {
		tyr_qing_standard_play_allowed = yes
		aggressive_diplomatic_plays_permitted = yes
	}

	ai_acceptance_max = 90

	on_weekly_pulse = {}

	on_war_begins = {
		if = {
			limit = {
				any_scope_play_involved = {
					has_variable = por_seeking_restore_union_var
				}
			}
			random_scope_play_involved = { # This is typically a temp variable. Starting this war makes it permanent for the duration of the war.
				limit = {
					has_variable = por_seeking_restore_union_var
				}
				remove_variable = por_seeking_restore_union_var
				set_variable = por_seeking_restore_union_var
			}
		}
	}

	on_war_end = {
		if = {
			limit = {
				any_scope_play_involved = {
					has_variable = por_seeking_restore_union_var
				}
			}
			random_scope_play_involved = { # Removes variable upon war conclusion, regardless of outcome
				limit = {
					has_variable = por_seeking_restore_union_var
				}
				remove_variable = por_seeking_restore_union_var
			}
		}
	}
}
