je_tyrone_sc_settlement = {
	icon = "gfx/interface/icons/law_icons/technocracy.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_sc_is_actor = yes }
	should_show_when_not_involved = { always = no }
	immediate = {
		tyrone_sc_bind_india = yes
		scope:csr_india = { set_variable = tyrone_sc_settlement_active }
		tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_sc_settlement }
	}
	scripted_progress_bar = tyrone_sc_settlement_bar
	scripted_button = tyrone_sc_back_experts
	scripted_button = tyrone_sc_back_shareholders
	on_monthly_pulse = {
		effect = {
			tyrone_sc_bind_india = yes
			tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_sc_settlement }
			scope:csr_india = { trigger_event = { on_action = tyrone_sc_settlement_monthly_events } }
		}
	}
	complete = {
		custom_tooltip = {
			text = tyrone_sc_settlement_goal_tt
			scope:journal_entry = { OR = { "scripted_bar_progress(tyrone_sc_settlement_bar)" >= 50 "scripted_bar_progress(tyrone_sc_settlement_bar)" <= -50 } }
		}
	}
	on_complete = {
		scope:csr_india = {
			set_variable = tyrone_sc_settlement_done
			remove_variable = tyrone_sc_settlement_active
			if = {
				limit = { scope:journal_entry = { "scripted_bar_progress(tyrone_sc_settlement_bar)" >= 50 } }
				activate_law = law_type:law_india_company_technocracy
			}
			else = { activate_law = law_type:law_india_shareholder_voting }
			tyrone_sc_domesticate_company = yes
			trigger_event = { id = tyrone_sovereign_company.2 popup = yes }
		}
	}
	invalid = { OR = { NOT = { exists = scope:csr_india } scope:csr_india ?= { is_country_alive = no } } }
	progressbar = no
	weight = 30
	can_revolution_inherit = yes

}
