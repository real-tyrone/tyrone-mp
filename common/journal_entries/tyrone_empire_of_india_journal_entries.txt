je_tyrone_eoi_calcutta = {
	icon = "gfx/interface/icons/invention_icons/political_agitation.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_eoi_is_actor = yes }
	should_show_when_not_involved = { always = no }
	immediate = {
		tyrone_eoi_bind_india = yes
		scope:csr_india ?= {
			set_variable = tyrone_eoi_calcutta_active
			set_variable = { name = tyrone_eoi_calcutta_score value = 0 }
			set_variable = { name = tyrone_eoi_calcutta_months value = 0 }
			set_variable = { name = tyrone_eoi_calcutta_round value = 0 }
			if = {
				limit = { is_subject = yes }
				tyrone_eoi_calcutta_send_round = yes
			}
		}
		tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_eoi_calcutta }
	}
	scripted_progress_bar = tyrone_eoi_calcutta_bar
	scripted_button = tyrone_eoi_independence_ultimatum
	on_monthly_pulse = {
		effect = {
			tyrone_eoi_bind_india = yes
			tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_eoi_calcutta }
			scope:csr_india ?= { tyrone_eoi_calcutta_monthly = yes }
		}
	}
	complete = {
		scope:csr_india ?= {
			custom_tooltip = {
				text = tyrone_eoi_calcutta_resolved_tt
				OR = {
					is_subject = no
					has_variable = tyrone_eoi_calcutta_ready
					has_variable = tyrone_eoi_ultimatum_issued
				}
			}
		}
	}
	on_complete = { scope:csr_india ?= { tyrone_eoi_calcutta_conclude = yes } }
	invalid = {
		OR = {
			NOT = { exists = scope:csr_india }
			scope:csr_india ?= { is_country_alive = no }
		}
	}
	on_invalid = {
		scope:csr_india ?= {
			remove_variable = tyrone_eoi_calcutta_active
			remove_variable = tyrone_eoi_negotiation_pending
		}
	}
	progress_desc = tyrone_eoi_calcutta_progress
	progressbar = no
	weight = 40
	can_revolution_inherit = yes
}

je_tyrone_eoi_coronation = {
	icon = "gfx/interface/icons/law_icons/monarchy.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_eoi_is_actor = yes }
	should_show_when_not_involved = { always = no }
	immediate = {
		tyrone_eoi_bind_india = yes
		scope:csr_india ?= {
			set_variable = { name = tyrone_eoi_coronation_gdp value = gdp }
			set_variable = { name = tyrone_eoi_coronation_target value = { value = gdp multiply = 1.5 } }
		}
		tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_eoi_coronation }
	}
	scripted_button = tyrone_eoi_hold_coronation
	on_monthly_pulse = {
		effect = {
			tyrone_eoi_bind_india = yes
			tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_eoi_coronation }
			
		}
	}
	complete = {
		scope:csr_india ?= {
			custom_tooltip = { text = tyrone_eoi_coronation_held_tt has_variable = tyrone_eoi_coronation_held }
		}
	}
	on_complete = { scope:csr_india ?= { set_variable = tyrone_eoi_coronation_done } }
	invalid = {
		OR = {
			NOT = { exists = scope:csr_india }
			scope:csr_india ?= { is_country_alive = no }
		}
	}
	current_value = {
		value = 0
		scope:csr_india ?= {
			if = {
				limit = { has_variable = tyrone_eoi_coronation_gdp }
				add = gdp
				subtract = var:tyrone_eoi_coronation_gdp
			}
		}
		min = 0
	}
	goal_add_value = {
		value = 0
		scope:csr_india ?= { add = gdp }
		multiply = 0.5
	}
	progress_desc = tyrone_eoi_coronation_progress
	progressbar = yes
	weight = 30
	can_revolution_inherit = yes
}
