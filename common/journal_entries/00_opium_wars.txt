je_opium_obsession = {
	activation_description = { always = yes }

	context = country
	is_shown_when_inactive = { c:CHI ?= this has_variable = tyr_opium_chinese_started }
	possible = { always = yes }

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_internal_affairs

	is_shown_in_lobby = {
		c:CHI = THIS
	}

	scripted_button = ban_opium_button
	scripted_button = unban_opium_button

	on_monthly_pulse = {
		random_events = {
			200 = 0
			10 = opium_wars.6
			5 = opium_wars.7
			5 = opium_wars.8
		}
		effect = {
			if = {
				limit = {
					is_banning_goods = opium
					is_at_war = no
				}
				change_variable = {
					name = opium_wars_completion_tracker
					add = 1
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
			custom_tooltip = {
				text = opium_ban_tt
				is_banning_goods = opium
			}
			is_at_war = no
		}
		effect = {
			remove_variable = opium_wars_target
			trigger_event = {
				id = opium_wars.3
			}
		}
	}



	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = { custom_tooltip = { text = tyr_opium_port_tt tyr_opium_britain_has_port = yes } }
		effect = {
			trigger_event = {
				id = opium_wars.4
			}
		}
	}



	current_value = {
		value = root.var:opium_wars_completion_tracker
	}

	goal_add_value = {
		add = 60
	}

	timeout = 3650
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			hidden_effect = {
				remove_modifier = opium_wars_addiction_modifier
				remove_variable = opium_wars_unresolved
			}
			add_modifier = {
				name = opium_wars_addiction_modifier
				days = long_modifier_time
			}
			add_radicals = {
				value = large_radicals
			}
			if = {
				limit = { has_journal_entry = je_warlord_china }
				chi_remove_1_fragile_unity = yes
			}
		}
	}

	progressbar = yes
	display_progressbar_as_months = yes

	weight = 10000

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_opium_wars = {
	activation_description = { always = yes }

	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_foreign_affairs
	context = country
	is_shown_when_inactive = { c:GBR ?= this has_variable = tyr_opium_british_started }
	possible = { always = yes }
	should_be_pinned_by_default_uninvolved_or_context = yes
	timeout = 1825
	weight = 10000
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = { text = tyr_opium_port_tt tyr_opium_britain_has_port = yes }
			custom_tooltip = { text = tyr_opium_reparations_tt tyr_opium_britain_has_reparations = yes }
		}
		effect = { trigger_event = { id = opium_wars.5 days = 5 } }
	}
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			if = {
				limit = { tyr_opium_britain_has_port = yes tyr_opium_britain_has_reparations = yes }
				trigger_event = { id = opium_wars.5 days = 5 }
			}
			else = {
				tyr_opium_british_failure_effect = yes
				trigger_event = { id = tyr_opium.1 }
			}
		}
	}
}
