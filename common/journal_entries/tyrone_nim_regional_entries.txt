je_tyrone_nim_pakistan = {
	icon = "gfx/interface/icons/event_icons/raj_pakistan_icon.png"
	group = je_group_british_india_global

	should_be_involved = {
		is_india_tag_trigger = yes
	}

	should_show_when_not_involved = {
		always = no
	}

	immediate = {
		tyrone_nim_bind_india = yes
		tyrone_nim_refresh_involvement = {
			ENTRY = je_tyrone_nim_pakistan
		}
		scope:nim_india ?= {
			tyrone_nim_sync_league_bars = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			tyrone_nim_refresh_involvement = {
				ENTRY = je_tyrone_nim_pakistan
			}
			scope:nim_india ?= {
				tyrone_nim_sync_league_bars = yes
			}
			scope:nim_india ?= {
				tyrone_nim_monthly_pakistan = yes
			}
		}
	}

	scripted_button = tyrone_nim_pakistan_negotiate
	scripted_button = tyrone_nim_pakistan_develop
	scripted_button = tyrone_nim_pakistan_repress
	scripted_button = tyrone_nim_pakistan_autonomy

	timeout = tyrone_nim_five_years
	on_timeout = {
		scope:nim_india ?= {
			tyrone_nim_resolve_league = yes
		}
	}

	complete = {
		exists = scope:nim_india
		scope:nim_india = {
			custom_tooltip = {
				text = tyrone_nim_pakistan_resolved_tt
				has_variable = tyrone_nim_pakistan_done
			}
		}
	}

	on_complete = {
		scope:nim_india ?= {
			tyrone_nim_finish_regional_crisis = {
				REGION = pakistan
			}
		}
	}

	on_invalid = {
		scope:nim_india ?= {
			if = {
				limit = {
					NOT = {
						tyrone_nim_pakistan_territory_exists = yes
					}
				}
				set_variable = tyrone_nim_pakistan_done
			}
			tyrone_nim_finish_regional_crisis = {
				REGION = pakistan
			}
		}
	}

	invalid = {
		OR = {
			NOT = {
				exists = scope:nim_india
			}
			scope:nim_india ?= {
				has_variable = tyrone_nim_socialist_started
			}
			scope:nim_india ?= {
				NOT = {
					tyrone_nim_pakistan_territory_exists = yes
				}
			}
		}
	}

	current_value = {
		value = 0
		scope:nim_india ?= {
			if = {
				limit = {
					has_variable = tyrone_nim_league_pressure
				}
				add = var:tyrone_nim_league_pressure
			}
		}
	}

	goal_add_value = {
		value = 50
		scope:nim_india ?= {
			subtract = var:tyrone_nim_league_pressure
		}
	}

	progress_desc = tyrone_nim_pakistan_progress
	progressbar = no
	scripted_progress_bar = tyrone_nim_league_pressure_bar
	weight = 40
	can_revolution_inherit = yes
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
