je_tyr_americas = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"
	group = je_group_global_international_situations
	is_shown_when_inactive = {
		NOT = {
			has_global_variable = tyr_am_finished
		}
		exists = c:USA
	}
	possible = {
		tyr_spanish_content_exists = yes
		tyr_resolve_spanish_content_trigger = yes
		exists = scope:tyr_spanish_content_scope
		scope:tyr_spanish_content_scope ?= {
			is_subject = no
			NOT = {
				has_journal_entry = je_the_first_carlist_war
			}
		}
		c:USA ?= {
			is_subject = no
		}
		NOT = {
			has_global_variable = tyr_am_finished
		}
	}
	should_be_involved = {
		OR = {
			tyr_am_main_power = yes
			tyr_am_regional_country = yes
		}
	}
	should_show_when_not_involved = {
		OR = {
			tyr_am_present = yes
			AND = {
				country_rank >= rank_value:major_power
				OR = {
					has_strategic_region_interest_tier = {
						strategic_region = sr:region_central_america
						value >= 1
					}
					has_strategic_region_interest_tier = {
						strategic_region = sr:region_brazil
						value >= 1
					}
					has_strategic_region_interest_tier = {
						strategic_region = sr:region_andes
						value >= 1
					}
					has_strategic_region_interest_tier = {
						strategic_region = sr:region_la_plata
						value >= 1
					}
					has_strategic_region_interest_tier = {
						strategic_region = sr:region_gran_colombia
						value >= 1
					}
				}
			}
		}
	}
	immediate = {
		tyr_am_initialize = yes
		c:ARG ?= {
			if = {
				limit = {
					is_player = no
					any_state_in_tyr_am_10 = {
						owner = {
							is_country_type = decentralized
						}
					}
				}
				trigger_event = {
					id = tyr_americas.60
					popup = yes
				}
			}
		}
		c:CHL ?= {
			if = {
				limit = {
					is_player = no
					any_state_in_tyr_am_9 = {
						owner = {
							is_country_type = decentralized
						}
					}
				}
				trigger_event = {
					id = tyr_americas.61
					popup = yes
				}
			}
		}
		c:PRG ?= {
			if = {
				limit = {
					is_player = no
					any_state_in_tyr_am_11 = {
						owner = {
							is_country_type = decentralized
						}
					}
				}
				trigger_event = {
					id = tyr_americas.62
					popup = yes
				}
			}
		}
	}
	immediate_all_involved = {
		tyr_am_prepare_actor = yes
	}
	on_become_involved_after_activation = {
		tyr_am_prepare_actor = yes
	}
	on_no_longer_involved = {
		tyr_am_clear_leverage_exemption = yes
	}
	on_complete_all_involved = {
		tyr_am_clear_leverage_exemption = yes
	}
	on_fail_all_involved = {
		tyr_am_clear_leverage_exemption = yes
	}
	on_invalid_all_involved = {
		tyr_am_clear_leverage_exemption = yes
	}
	widget = {
		gui = "gui/tyr_americas.gui"
		name = "tyr_am_journal_widget"
		container = "custom_widget_container_2"
	}
	scripted_button = tyr_am_policy_button
	scripted_button = tyr_am_federation_button
	scripted_button = tyr_am_proclaim_button
	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					NOT = {
						has_global_variable = tyr_am_active
					}
				}
				desc = tyr_am_status_potential
			}
			triggered_desc = {
				trigger = {
					global_var:tyr_am_phase = 3
				}
				desc = tyr_am_status_3_settlement
			}
			triggered_desc = {
				trigger = {
					global_var:tyr_am_phase = 2
				}
				desc = tyr_am_status_2
			}
			desc = tyr_am_status_1
		}
	}
	on_monthly_pulse = {
		effect = {
			tyr_am_monthly = yes
		}
	}
	on_weekly_pulse = {
		effect = {
			tyr_am_check_wars = yes
		}
	}
	complete = {
		custom_tooltip = {
			text = tyr_am_completion_summary_tt
			trigger_if = {
				limit = {
					has_global_variable = tyr_am_active
					has_global_variable = tyr_am_outcome
				}
				global_var:tyr_am_outcome > 0
				global_var:tyr_am_outcome < 4
			}
			trigger_else = {
				always = no
			}
		}
	}
	on_complete = {
		hidden_effect = {
			tyr_am_finish = yes
		}
	}
	invalid = {
		trigger_if = {
			limit = {
				has_global_variable = tyr_am_active
				has_global_variable = tyr_am_outcome
			}
			global_var:tyr_am_outcome = 5
		}
		trigger_else = {
			always = no
		}
	}
	on_invalid = {
		hidden_effect = {
			tyr_am_finish = yes
		}
	}
	event_outcome_completed_effect_desc = {
		header = tyr_am_preview_heading
		effect = { tyr_am_preview_final_settlement = yes }
	}
	event_outcome_invalidated_effect_desc = {
		header = tyr_am_preview_heading
		effect = { tyr_am_preview_final_settlement = yes }
	}
	weight = 250
	should_be_pinned_by_default_uninvolved_or_context = yes
}

REPLACE:je_hispanoamerica = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_historical_content

	immediate = {
		every_country_in_americas = {
			limit = {
				any_primary_culture = {
					has_discrimination_trait = language_hispanophone
				}
				has_modifier = modifier_not_recognized_by_spain
				NAND = {
					is_country_type = colonial
					is_subject_of = ROOT
				}
			}
			root = {
				add_to_variable_list = {
					name = hispanoamerica_recognition_countries
					target = prev
				}
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			clear_variable_list = hispanoamerica_recognition_countries
			every_country_in_americas = {
				limit = {
					any_primary_culture = {
						has_discrimination_trait = language_hispanophone
					}
					has_modifier = modifier_not_recognized_by_spain
					NAND = {
						is_country_type = colonial
						is_subject_of = ROOT
					}
				}
				root = {
					add_to_variable_list = {
						name = hispanoamerica_recognition_countries
						target = prev
					}
				}
			}
		}
	}

	complete = {
		any_country_in_americas = {
			filter = {
				custom_tooltip = {
					text = FORMER_SPANISH_COLONY_TT
					any_primary_culture = {
						has_discrimination_trait = language_hispanophone
					}
					NAND = {
						is_country_type = colonial
						is_subject_of = ROOT
					}
				}
			}
			percent = 1
			NOT = {
				has_modifier = modifier_not_recognized_by_spain
			}
			custom_tooltip = HISPANOAMERICA_RECOGNITION_COUNTRIES_TT
		}
	}

	on_complete = {
		clear_variable_list = hispanoamerica_recognition_countries
		trigger_event = {
			id = spanish_new_world_events.3
			popup = yes
		}
		show_as_tooltip = {
			add_modifier = {
				name = modifier_hispanoamerica_reward
				days = long_modifier_time
			}
		}
		custom_tooltip = {
			text = je_hispanoamerica_improve_relations
		}
	}


    fail = {
    }

	on_fail = {
	}

	weight = 6000

    	
	should_be_pinned_by_default_uninvolved_or_context = yes
}
