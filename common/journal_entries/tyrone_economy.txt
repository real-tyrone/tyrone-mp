je_reform_economy = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		has_technology_researched = stock_exchange
	}

	is_shown_in_lobby = {
	}


	possible = {
		OR = {
			has_law_or_variant = law_type:law_laissez_faire
            has_law_or_variant = law_type:law_interventionism
		}
		has_technology_researched = investment_banks
	}
















	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.5
	}

	progressbar = yes

	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 100

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		any_company = {
			company_is_prosperous = yes
			count >= 4
		}
		OR = {
			AND = {
				ig:ig_industrialists ?= {
					ig_counts_as_marginal = no
					ig_approval >= 10
				}
				has_law_or_variant = law_type:law_laissez_faire
			}
			AND = {
           		has_law_or_variant = law_type:law_interventionism
				OR = {
					has_law_or_variant = law_type:law_autocracy
					has_law_or_variant = law_type:law_oligarchy
					has_law_or_variant = law_type:law_single_party_state
					has_law_or_variant = law_type:law_technocracy
					any_active_law = {
						has_amendment = amendment_type:amendment_taikunate
					}
				}			
			}
		}

		scaled_debt <= 0.5
	}
		effect = {
		trigger_event = economic_reform.1
	}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		OR = {
			has_modifier = in_default
			has_modifier = declared_bankruptcy						 
		}
	}
		effect = {
	}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		NOR = {
			has_law_or_variant = law_type:law_laissez_faire
            has_law_or_variant = law_type:law_interventionism
		}
	}
	}
}
