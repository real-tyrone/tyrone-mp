# Tyrone: a shared Spain–Italy dynastic contest and a post-contest union arc.

je_tyr_spain_italy_dynastic_contest = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/invention_icons/rationalism.dds"
	scripted_progress_bar = tyr_spain_italy_dynastic_contest_bar

	is_shown_when_inactive = {
		NOT = { has_global_variable = tyr_dynastic_contest_resolved }
		exists = c:SPA
		exists = c:ITA
	}

	possible = {
		exists = c:SPA
		exists = c:ITA
		OR = {
			c:SPA = {
				has_technology_researched = pan-nationalism
			}
			c:ITA = {
				has_technology_researched = pan-nationalism
			}
		}
		c:SPA = { is_subject = no }
		c:ITA = { is_subject = no }
		OR = {
			AND = {
				custom_tooltip = {
					text = tyr_crown_union_spain_ruler_bourbon_tt
					c:SPA.ruler = { has_variable = house_bourbon }
				}
				custom_tooltip = {
					text = tyr_crown_union_italy_ruler_bourbon_tt
					c:ITA.ruler = { has_variable = house_bourbon }
				}
			}
			AND = {
				custom_tooltip = {
					text = tyr_crown_union_spain_ruler_savoy_tt
					c:SPA.ruler = { has_variable = house_savoy }
				}
				custom_tooltip = {
					text = tyr_crown_union_italy_ruler_savoy_tt
					c:ITA.ruler = { has_variable = house_savoy }
				}
			}
		}
	}

	should_be_involved = {
		OR = { c:SPA = root c:ITA = root }
	}
	should_show_when_not_involved = {
		OR = { this = c:SPA this = c:ITA }
	}

	# A contextless JE has no country root; this hook runs for each participant.
	immediate_all_involved = {
		set_variable = tyr_dynastic_contest_active
	}

	complete = {
		exists = c:SPA
		exists = c:ITA
		c:SPA = { is_subject = no }
		c:ITA = { is_subject = no }
		OR = {
			custom_tooltip = {
				text = tyr_spain_italy_spain_wins_contest_tt
				scope:journal_entry = { "scripted_bar_progress(tyr_spain_italy_dynastic_contest_bar)" <= 0 }
			}
			custom_tooltip = {
				text = tyr_spain_italy_italy_wins_contest_tt
				scope:journal_entry = { "scripted_bar_progress(tyr_spain_italy_dynastic_contest_bar)" >= 100 }
			}
		}
	}

	on_complete = {
		set_global_variable = tyr_dynastic_contest_resolved
		if = {
			limit = { scope:journal_entry = { "scripted_bar_progress(tyr_spain_italy_dynastic_contest_bar)" <= 0 } }
			c:SPA = {
				set_variable = tyr_crown_union_winner
			}
			c:ITA = {
				set_variable = tyr_crown_union_loser
				trigger_event = { id = tyr_crown_union.1 popup = yes }
			}
		}
		else = {
			c:ITA = {
				set_variable = tyr_crown_union_winner
			}
			c:SPA = {
				set_variable = tyr_crown_union_loser
				trigger_event = { id = tyr_crown_union.1 popup = yes }
			}
		}
		c:SPA ?= { remove_variable = tyr_dynastic_contest_active }
		c:ITA ?= { remove_variable = tyr_dynastic_contest_active }
	}

	invalid = {
		OR = {
			NOT = { exists = c:SPA }
			NOT = { exists = c:ITA }
			c:SPA ?= { is_subject = yes }
			c:ITA ?= { is_subject = yes }
		}
	}

	on_invalid = {
		c:SPA ?= { remove_variable = tyr_dynastic_contest_active }
		c:ITA ?= { remove_variable = tyr_dynastic_contest_active }
	}

	weight = 5000
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_tyr_crown_union_consolidation = {
	group = je_group_foreign_affairs
	icon = "gfx/interface/icons/event_icons/event_political.dds"
	scripted_progress_bar = tyr_crown_union_consolidation_bar

	is_shown_when_inactive = {
		has_variable = tyr_crown_union_winner
	}

	possible = {
		OR = {
			AND = {
				this = c:SPA
				exists = c:ITA
				c:ITA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
			}
			AND = {
				this = c:ITA
				exists = c:SPA
				c:SPA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
			}
		}
	}

	immediate = {
		remove_variable = tyr_crown_union_war_unlocked
		if = {
			limit = { this = c:SPA }
			c:ITA = { remove_variable = tyr_crown_union_rejected }
		}
		else = {
			c:SPA = { remove_variable = tyr_crown_union_rejected }
		}
	}

	complete = {
		OR = {
			AND = {
				this = c:SPA
				exists = c:ITA
				c:ITA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
			}
			AND = {
				this = c:ITA
				exists = c:SPA
				c:SPA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
			}
		}
		scope:journal_entry = { "scripted_bar_progress(tyr_crown_union_consolidation_bar)" >= 36 }
	}

	on_complete = {
		trigger_event = { id = tyr_crown_union.2 popup = yes }
	}

	invalid = {
		NOT = {
			OR = {
				AND = {
					this = c:SPA
					exists = c:ITA
					c:ITA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
				}
				AND = {
					this = c:ITA
					exists = c:SPA
					c:SPA ?= { is_subject_of = root is_subject_type = subject_type_personal_union }
				}
			}
		}
	}

	on_invalid = {
		remove_variable = tyr_crown_union_winner
		remove_variable = tyr_crown_union_war_unlocked
		c:SPA ?= { remove_variable = tyr_crown_union_loser remove_variable = tyr_crown_union_rejected }
		c:ITA ?= { remove_variable = tyr_crown_union_loser remove_variable = tyr_crown_union_rejected }
	}

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}
