 je_crimean_war = {
	activation_description = {
		always = yes
	}


	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	should_be_involved = {
		OR = {
			c:TUR ?= this
			c:WAL ?= this
			c:MOL ?= this
			c:RUS ?= this
		}
	}

	should_show_when_not_involved = {
		OR = {
			is_in_geographic_region = geographic_region_balkans_old
			is_in_geographic_region = geographic_region_megali_greece # Proxy for Anatolia
			country_has_interest_marker_in_great_eastern_crisis_region = yes
            top_overlord ?= {
                capital = {
					OR = {
						is_in_geographic_region = geographic_region_balkans_old
						is_in_geographic_region = geographic_region_megali_greece
					}
                }
            }
		}
	}

	is_shown_in_lobby = {
		has_dlc_feature = ip3_content #National Awakening
		c:TUR ?= this
	}

	is_shown_when_inactive = { #Now failure effect of sick man, and separatism tanzimat, so should be visible at game start.
		has_dlc_feature = ip3_content
		NOT = {
			has_global_variable = crimean_war_starting
		}
	}

	possible = {
		c:TUR ?= {
			country_or_subject_owns_entire_state_region = STATE_MOLDAVIA
			country_or_subject_owns_entire_state_region = STATE_WALLACHIA
		}
		c:RUS ?= {
			OR = {
			has_technology_researched = nationalism
				game_date >= 1853.1.1
			}
			"relations(c:TUR)" <= relations_threshold:cold
			NOT = { has_truce_with = c:TUR }
			is_at_war = no
			is_diplomatic_play_committed_participant = no
		}
	}

	scripted_button = je_crimean_war_begin_war_button

	modifiers_while_active = {
	}

	immediate = {
		set_global_variable = crimean_war_starting
		c:TUR ?= {
			save_scope_as = ottoman_scope
			set_variable = crimean_war_ottoman_side
		}
		c:RUS ?= {
			save_scope_as = russia_scope
			set_variable = crimean_war_russian_side
		}
		c:WAL ?= {
			save_scope_as = wallachia_scope
		}
		c:MOL ?= {
			save_scope_as = moldavia_scope
		}
		c:FRA ?= {
			trigger_event = { id = crimean_war.1 popup = yes }
		}
		c:GBR ?= {
			trigger_event = { id = crimean_war.2 popup = yes }
		}
		c:AUS ?= {
			trigger_event = { id = crimean_war.3 popup = yes }
		}
		c:KUK ?= {
			trigger_event = { id = crimean_war.3 popup = yes }
		}
	}

	immediate_all_involved = {
		trigger_event = { id = crimean_war.7 popup = yes }
	}

	on_monthly_pulse = {
	}











	timeout = 1825 # 5 years







	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
	
	weight = 1000
	transferable = no
	can_revolution_inherit = yes

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		OR = {
			custom_tooltip = {
				text = crimean_war_russian_victory_tt
				has_global_variable = crimean_war_russian_victory
			}
			custom_tooltip = {
				text = crimean_war_ottoman_victory_tt
				has_global_variable = crimean_war_ottoman_victory
			}
		}
		NOT = {
			any_diplomatic_play = {
				is_diplomatic_play_type = dp_crimean_war
			}
		}
		c:TUR ?= {
			NOT = { has_war_with = c:RUS }
		}
	}
		effect_all_involved = {
		trigger_event = { id = crimean_war.5 popup = yes }
	}
		event_outcome_desc = {
		first_valid = {
			triggered_desc = {
				desc = event_effects_crimean_war.5_tt
			}
		}
	}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		OR = {
		NOT = {
			exists = c:TUR
		}
		has_global_variable = crimean_war_russian_intent_declined
		}

	}
		effect_all_involved = {
	}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
	}
		effect_all_involved = {
		trigger_event = { id = crimean_war.6 popup = yes }
	}
		event_outcome_desc = {
	}
	}
}
