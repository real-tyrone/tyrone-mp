je_post_bonapartists = {
	icon = "gfx/interface/icons/event_icons/napoleon_empire_icon.dds"
	group = je_group_historical_content

	is_shown_when_inactive = {
		exists = c:FRA
		c:FRA = ROOT
		has_variable = bonapartist_restoration_var
		has_journal_entry = je_cement_the_rightful_dynasty
	}

	possible = {
		exists = c:FRA
		c:FRA = ROOT
		c:FRA = {
		country_or_subject_owns_entire_state_region = STATE_ALSACE_LORRAINE
		}
		government_legitimacy >= 50
		custom_tooltip = {
			text = progress_cement_half_tt
			var:french_dynastic_primary_progress_var >= 60
		}
	}

	immediate = {
		c:FRA = {
			save_scope_as = france_scope
		}
		trigger_event ={
			id = empire_of_france.1
		}
		
		set_variable = napoleon_empire_variable

		every_scope_character = {
			limit = {
				is_ruler = yes
			}
			set_variable = napoleonic_emperor_variable
		}

		scope:journal_entry = {
			add_modifier = grande_armee_return_modifier
		}
		scope:journal_entry = {
			add_modifier = power_of_the_marshals_modifier
		}
	}

	on_monthly_pulse = {
		every_scope_character = {
			limit = {
				is_ruler = yes
				NOT = {
					has_variable = napoleonic_emperor_variable
				}
			}
			set_variable = napoleonic_emperor_variable
		}
	}
	
	complete = {	
		custom_tooltip = {
			text = has_cemented_dynasty_TT
			has_variable = one_true_dynasty_var
		}
		custom_tooltip = {
			text = has_completed_rebuilt_paris_je_tt
			c:FRA = {
				has_variable = {
					rebuilt_paris_NAPOLEON
				}
			}
		}
		custom_tooltip = {
			text = has_rebuilt_railways_je_tt
			c:FRA = {
				has_variable = {
					rebuilt_railroads_NAPOLEON
				}
			}
		}
		custom_tooltip = {
			text = has_improved_living_NAPOLEON_TT
			c:FRA = { 
				has_variable = {
					public_health_NAPOLEON
				}
			}
		}
		custom_tooltip = {
			text = has_completed_natural_borders_je_tt
			has_variable = completed_natural_borders_je
		}
		custom_tooltip = {
			text = has_conquered_italy_TT
			has_variable = conquered_italy_NAPOLEON
		}
		custom_tooltip = {
			text = has_conquered_spain_TT
			has_variable = conquered_spain_NAPOLEON
		}
	}

	on_complete = {
		trigger_event ={
			id = empire_of_france.8
		}
		set_variable = napoleonic_victory	
	}
	
	fail = {
		
	}
	
	invalid = {
		NOT = {
			has_law_or_variant = law_type:law_monarchy
		}
	}

	weight = 1000
	should_be_pinned_by_default = yes
	
	scripted_button = add_je_rebuild_paris
	
}
