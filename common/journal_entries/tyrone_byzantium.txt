je_byz_reclaim_levant = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/byzantium_je.dds"
	group = je_group_historical_content
	should_be_pinned_by_default_uninvolved_or_context = yes





	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		owns_entire_state_region = STATE_ALEPPO
		owns_entire_state_region = STATE_SYRIA
		owns_entire_state_region = STATE_PALESTINE
		owns_entire_state_region = STATE_TRANSJORDAN
		owns_entire_state_region = STATE_LEBANON
	}
		effect = {
		every_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_ALEPPO
					state_region = s:STATE_SYRIA
					state_region = s:STATE_PALESTINE
					state_region = s:STATE_TRANSJORDAN
					state_region = s:STATE_LEBANON
				}
			}
			add_modifier = { name = byz_reclaimed_province_incorporation days = very_long_modifier_time }
		}
		every_state_region = {
			limit = {
				is_in_geographic_region = geographic_region_egypt
				any_scope_state = {
					NOT = { owner = ROOT }
					NOT = { has_claim_by = ROOT }
				}
			}
			add_claim = ROOT
		}
		add_journal_entry = { type = je_byz_reclaim_egypt }
		trigger_event = { id = byzantine_reconquest.2 popup = yes }
	}
	}
}
