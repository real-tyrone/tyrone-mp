# Replaces the original utilitarian arc with an open-ended reform portfolio.
# Shared between India and its current overlord; only India must satisfy the requirements.
# Requirements are inline. State/building checks follow Vic Century's
# bangladesh_industrialize entry, retaining Tyrone's own objectives and thresholds.
REPLACE:je_imperialism_of_promise = {
	icon = "gfx/interface/icons/invention_icons/central_planning.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_promise_is_actor = yes }
	should_show_when_not_involved = { tyrone_promise_is_actor = yes }
	is_shown_when_inactive = { tyrone_promise_india_exists = yes }
	possible = { tyrone_promise_india_exists = yes }
	immediate = {
		tyrone_promise_bind_india = yes
		set_global_variable = tyrone_promise_active_var
		add_contextless_journal_entry = je_tyrone_promise_naval_industry
		add_contextless_journal_entry = je_tyrone_promise_plantation_labor
		add_contextless_journal_entry = je_tyrone_promise_manufacturing_monopoly
		add_contextless_journal_entry = je_tyrone_promise_bonded_labor
		add_contextless_journal_entry = je_tyrone_promise_british_patents
		tyrone_india_portfolio_refresh_involvement = {
			INDIA = scope:tyrone_promise_india
			ENTRY = je_imperialism_of_promise
		}
	}
	on_monthly_pulse = {
		effect = {
			tyrone_promise_bind_india = yes
			# Add the new objective to an Imperialism of Promise already active in a save.
			if = {
				limit = {
					exists = scope:tyrone_promise_india
					scope:tyrone_promise_india ?= {
						NOT = { has_variable = tyrone_patents_entry_started }
						any_active_law = { has_amendment = amendment_type:amendment_reliance_on_british_patents }
					}
				}
				add_contextless_journal_entry = je_tyrone_promise_british_patents
			}
			tyrone_india_portfolio_refresh_involvement = {
				INDIA = scope:tyrone_promise_india
				ENTRY = je_imperialism_of_promise
			}
		}
	}
	widget = {
		gui = "gui/journal_entry_widgets/tyrone_imperial_promise_widgets.gui"
		name = "tyrone_imperial_promise_subentries_widget"
		container = "custom_widget_container_1"
	}
	# Deliberately unfinished: more reforms will be added to this portfolio.
	complete = { always = no }
	invalid = { NOT = { tyrone_promise_india_exists = yes } }
	on_invalid = { remove_global_variable = tyrone_promise_active_var }
	progressbar = no
	should_be_pinned_by_default_uninvolved_or_context = yes
	should_be_pinned_by_default_involved = yes
	weight = 100
}

je_tyrone_promise_naval_industry = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_promise_is_actor = yes }
	should_show_when_not_involved = { always = no }
	immediate = {
		tyrone_promise_bind_india = yes
		tyrone_india_portfolio_refresh_involvement = {
			INDIA = scope:tyrone_promise_india
			ENTRY = je_tyrone_promise_naval_industry
		}
	}
	on_monthly_pulse = {
		effect = {
			tyrone_promise_bind_india = yes
			tyrone_india_portfolio_refresh_involvement = {
				INDIA = scope:tyrone_promise_india
				ENTRY = je_tyrone_promise_naval_industry
			}
		}
	}

	complete = {
		hidden_trigger = {
			has_global_variable = tyrone_promise_active_var
			exists = scope:tyrone_promise_india
			scope:tyrone_promise_india ?= { is_country_alive = yes }
			scope:tyrone_promise_india ?= {
				OR = {
					has_variable = bic_naval_restricted_var
					any_active_law = { has_amendment = amendment_type:amendment_royal_navy_supremacy }
				}
			}
		}
		scope:tyrone_promise_india ?= {
			has_technology_researched = gantry_cranes
			any_scope_state = {
				state_region = s:STATE_EAST_BENGAL
				any_scope_building = {
					is_building_type = building_shipyard
					level >= 25
				}
				any_scope_building = {
					is_building_type = building_urban_center
					level >= 50
				}
				any_scope_building = {
					is_building_type = building_logging_camp
					level >= 20
					has_active_production_method = pm_mech_mills
				}
			}
		}
	}
	on_complete = {
		scope:tyrone_promise_india ?= { tyrone_promise_remove_naval_industry_restriction = yes }
	}
	event_outcome_completed_desc = {
		first_valid = { triggered_desc = { desc = tyrone_promise_naval_industry_completed } }
	}
	# Manual repeal and removal through law changes remain valid alternatives.
	# Once the restriction is gone, this objective silently disappears.
	invalid = {
		OR = {
			NOT = {
				AND = {
					has_global_variable = tyrone_promise_active_var
					exists = scope:tyrone_promise_india
					scope:tyrone_promise_india ?= { is_country_alive = yes }
				}
			}
			NOT = {
				scope:tyrone_promise_india ?= {
					OR = {
						has_variable = bic_naval_restricted_var
						any_active_law = { has_amendment = amendment_type:amendment_royal_navy_supremacy }
					}
				}
			}
		}
	}
	progressbar = no
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 20
}
