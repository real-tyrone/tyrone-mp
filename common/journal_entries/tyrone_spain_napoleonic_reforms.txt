je_spain_napoleonic_reforms = {
	icon = "gfx/interface/icons/event_icons/napoleon_empire_icon.dds"
	group = je_group_historical_content
	is_shown_when_inactive = { 
          tyr_spanish_content_country = yes
	}
	possible = { 
		custom_tooltip = {
			text = spain_napoleonic_bonaparte_king_tt
			ruler = { has_variable = house_bonaparte }
		}
		has_law_or_variant = law_type:law_monarchy
	 }

	# The restoration event adds the central entry, which opens both reform branches.
	immediate = {
		ruler = { add_trait = ambitious }
		add_journal_entry = { type = je_spain_napoleonic_domestic_reforms }
		add_journal_entry = { type = je_spain_napoleonic_military_reforms }
	}
	complete = {
		custom_tooltip = {
			text = je_spain_napoleonic_domestic_completed_tt
			has_variable = spain_napoleonic_domestic_reforms_completed
		}
		custom_tooltip = {
			text = je_spain_napoleonic_military_completed_tt
			has_variable = spain_napoleonic_military_reforms_completed
		}
	}
	on_complete = {
		set_variable = spain_napoleonic_reforms_completed
		trigger_event = { id = spain_napoleonic_reforms.11 popup = yes }
	}
	invalid = {
		OR = {
			NOT = { has_law_or_variant = law_type:law_monarchy }
			NOT = { ruler = { has_variable = house_bonaparte } }
		}
	}
	weight = 1000
    should_be_pinned_by_default_uninvolved_or_context = yes
}
