GGGG_je_canned_goods = {
	activation_description = {
		always = yes
	}

    icon = "gfx/interface/icons/event_icons/event_military.dds"
    
    group = je_group_technology
    
    is_shown_when_inactive = {
        NOR = {
            has_variable = GGGG_has_rations_3
            has_variable = GGGG_has_rations_2
        }
        has_technology_researched = lathe
        GGGG_sv_count_military_building_levels >= 5
    }
    
    possible = {
        has_technology_researched = canneries
    }
    
    immediate = {
        ROOT = {
            save_scope_as = target_country
        }
    }
    

    

    

    
    timeout = 3650

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
        custom_tooltip = {
            text = GGGG_tt_canned_goods_je
            GGGG_sv_mo_usage_canned_rations > GGGG_CANNED_GOODS_ADOPTION_THRESHOLD
        }
    }
		effect = {
        trigger_event = {
            id = GGGG_ev_rations.1
        }
    }
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
        OR = {
            hidden_trigger = {
                has_variable = GGGG_has_rations_3
            }
            has_journal_entry = GGGG_je_refrigerated_produce
        }
    }
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
	}
}
