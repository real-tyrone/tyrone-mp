je_veiled_protectorate = {
	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_foreign_affairs

	is_shown_when_inactive = {
		country_rank >= rank_value:major_power
		OR = {
			AND = {
				any_scope_state = {
					state_region = s:STATE_SINAI
				}
				any_neighbouring_state = {
					OR = {
						state_region = s:STATE_LOWER_EGYPT
						state_region = s:STATE_UPPER_EGYPT
						state_region = s:STATE_MIDDLE_EGYPT
						state_region = s:STATE_SINAI
					}
				}
			}
			s:STATE_SINAI = {
				any_scope_state = {
					owner = {
						is_subject_of = root
					}
					any_scope_building = {
						is_building_type = building_suez_canal
					}
				}
			}
		}
		NOT = { c:TUR ?= THIS }
		NOT = { c:EGY ?= THIS }
	}

	possible = {
		s:STATE_SINAI = {
			any_scope_state = {
				owner = {
					OR = {
						THIS = root
						is_subject_of = root
					}
				}
				any_scope_building = {
					is_building_type = building_suez_canal
				}
			}
		}
		custom_tooltip = {
			text = berlin_conference_ended_tt
			has_global_variable = berlin_conference_ended
		}
	}

	immediate = {
		custom_tooltip = {
			text = egypt_claims_tt
			s:STATE_SINAI = {
				add_claim = ROOT
			}
			s:STATE_LOWER_EGYPT = {
				add_claim = ROOT
			}
			s:STATE_MIDDLE_EGYPT = {
				add_claim = ROOT
			}
			s:STATE_UPPER_EGYPT = {
				add_claim = ROOT
			}
			s:STATE_MATRUH = {
				add_claim = ROOT
			}
			s:STATE_EGYPTIAN_DESERT = {
				add_claim = ROOT
			}
		}
	}







	timeout = 3650

	weight = 10000
		
	should_be_pinned_by_default_uninvolved_or_context = yes

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		custom_tooltip = {
			text = control_egypt_tt
			AND = {
				OR = {
					owns_entire_state_region = STATE_SINAI
					any_subject_or_below = {
						owns_entire_state_region = STATE_SINAI
					}
					AND = {
						any_scope_state = {
							state_region = s:STATE_SINAI
						}
						any_subject_or_below = {
							any_scope_state = {
								state_region = s:STATE_SINAI
							}
						}
					}
				}
				OR = {
					owns_entire_state_region = STATE_LOWER_EGYPT
					any_subject_or_below = {
						owns_entire_state_region = STATE_LOWER_EGYPT
					}
				}
				OR = {
					owns_entire_state_region = STATE_MIDDLE_EGYPT
					any_subject_or_below = {
						owns_entire_state_region = STATE_MIDDLE_EGYPT
					}
				}
				OR = {
					owns_entire_state_region = STATE_UPPER_EGYPT
					any_subject_or_below = {
						owns_entire_state_region = STATE_UPPER_EGYPT
					}
				}
				OR = {
					owns_entire_state_region = STATE_MATRUH
					any_subject_or_below = {
						owns_entire_state_region = STATE_MATRUH
					}
				}
				OR = {
					owns_entire_state_region = STATE_EGYPTIAN_DESERT
					any_subject_or_below = {
						owns_entire_state_region = STATE_EGYPTIAN_DESERT
					}
				}
			}
		}
	}
		effect = {
		trigger_event = {
			id = veiled_protectorate.1
		}
	}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		NOT = {
			s:STATE_SINAI = {
				any_scope_state = {
					owner = {
						OR = {
							THIS = root
							is_subject_of = root
						}
					}
				}
			}
		}
	}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
	}
}
