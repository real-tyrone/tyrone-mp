    should_show_when_not_involved = { always = no }
    immediate = {
        tyrone_nim_bind_india = yes
        
        tyrone_nim_refresh_involvement = { ENTRY = je_tyrone_nim_civil_war }
    }
    on_monthly_pulse = {
        effect = {
            tyrone_nim_refresh_involvement = { ENTRY = je_tyrone_nim_civil_war }

            tyrone_nim_bind_india = yes

        }
    }
    complete = { NOT = { exists = c:NIM } }

    invalid = { NOT = { exists = scope:nim_india } }
    progressbar = no
    should_be_pinned_by_default_involved = yes

    weight = 40
    can_revolution_inherit = yes
}
