REPLACE:je_consolidate_colonial_rule = {
	icon = "gfx/interface/icons/event_icons/tyrone_unified_subcontinent.png"
	group = je_group_british_india_global
	should_be_involved = { tyrone_subcontinent_participant = yes }
	should_show_when_not_involved = { tyrone_subcontinent_participant = yes }

	is_shown_when_inactive = {
		OR = { exists = c:BIC exists = c:BHR }
		NOT = { has_global_variable = tyrone_subcontinent_resolved_var }
	}
	possible = {
		OR = {
			AND = { exists = c:BIC c:BIC ?= { has_technology_researched = general_staff } }
			AND = { exists = c:BHR c:BHR ?= { has_technology_researched = general_staff } }
		}
	}
	widget = {
		gui = "gui/journal_entry_widgets/tyrone_subcontinent_widgets.gui"
		name = "tyrone_subcontinent_subentries_widget"
		container = "custom_widget_container_1"
	}
	immediate = {
		if = {
			limit = { exists = c:BIC }
			c:BIC = { save_scope_as = tyrone_subcontinent_india }
		}
		else_if = {
			limit = { exists = c:BHR }
			c:BHR = { save_scope_as = tyrone_subcontinent_india }
		}
		c:GBR ?= {
			save_scope_as = GBR_scope
			ruler ?= { save_scope_as = british_ruler }
		}
		c:BUR ?= { save_scope_as = BUR_scope }
		c:SIN ?= { save_scope_as = SIN_scope }
		c:PAN ?= { save_scope_as = PAN_scope }
		c:RUS ?= { save_scope_as = RUS_scope }
		set_global_variable = tyrone_subcontinent_active_var
		set_global_variable = { name = tyrone_subcontinent_progress_var value = 0 }
		add_contextless_journal_entry = je_tyrone_conquer_burma
		add_contextless_journal_entry = je_tyrone_conquer_sindh
		add_contextless_journal_entry = je_tyrone_conquer_punjab
		add_contextless_journal_entry = je_tyrone_conquer_himalayas
	}
	immediate_all_involved = {
		if = {
			limit = { is_india_tag_trigger = yes }
			add_modifier = { name = tyrone_subcontinent_conquest_ambition }
		}
	}
	on_become_involved_after_activation = {
		if = {
			limit = { is_india_tag_trigger = yes }
			add_modifier = { name = tyrone_subcontinent_conquest_ambition }
		}
	}
	on_no_longer_involved = {
		if = {
			limit = { has_modifier = tyrone_subcontinent_conquest_ambition }
			remove_modifier = tyrone_subcontinent_conquest_ambition
		}
	}

	complete = {
		scope:journal_entry = { is_goal_complete = yes }
	}
	on_complete = {
		set_global_variable = tyrone_subcontinent_resolved_var
		remove_global_variable = tyrone_subcontinent_active_var
	}
	on_complete_all_involved = {
		if = {
			limit = { has_modifier = tyrone_subcontinent_conquest_ambition }
			remove_modifier = tyrone_subcontinent_conquest_ambition
		}
		if = {
			limit = { is_india_tag_trigger = yes }
			set_variable = consolidated_colonial_rule
			add_modifier = {
				name = tyrone_subcontinent_consolidated_administration
				days = 3650
				is_decaying = yes
			}
		}
		trigger_event = { id = tyrone_subcontinent.5 days = 0 popup = yes }
	}
	event_outcome_completed_desc = {
		first_valid = { triggered_desc = { desc = tyrone_subcontinent_completion_desc } }
	}

	fail = {
		exists = c:GBR
		OR = { exists = c:BIC exists = c:BHR }
		NOT = { tyrone_subcontinent_empire_intact = yes }
	}
	on_fail = {
		set_global_variable = tyrone_subcontinent_resolved_var
		remove_global_variable = tyrone_subcontinent_active_var
		remove_global_variable = tyrone_subcontinent_afghanistan_active_var
	}
	on_fail_all_involved = {
		if = {
			limit = { has_modifier = tyrone_subcontinent_conquest_ambition }
			remove_modifier = tyrone_subcontinent_conquest_ambition
		}
		trigger_event = { id = tyrone_subcontinent.7 days = 0 popup = yes }
	}
	invalid = {
		OR = {
			NOT = { exists = c:GBR }
			AND = { NOT = { exists = c:BIC } NOT = { exists = c:BHR } }
		}
	}
	on_invalid = {
		set_global_variable = tyrone_subcontinent_resolved_var
		remove_global_variable = tyrone_subcontinent_active_var
		remove_global_variable = tyrone_subcontinent_afghanistan_active_var
	}
	on_invalid_all_involved = {
		if = {
			limit = { has_modifier = tyrone_subcontinent_conquest_ambition }
			remove_modifier = tyrone_subcontinent_conquest_ambition
		}
	}

	current_value = {
		value = 0
		if = {
			limit = { has_global_variable = tyrone_subcontinent_progress_var }
			value = global_var:tyrone_subcontinent_progress_var
		}
	}
	goal_add_value = { add = 5 }
	progress_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_global_variable = tyrone_subcontinent_progress_var }
				desc = tyrone_subcontinent_progress
			}
			desc = tyrone_subcontinent_progress_zero
		}
	}
	progressbar = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
	should_be_pinned_by_default_involved = yes
	weight = 100
}

	}

	immediate_all_involved = {

		if = { limit = { is_india_tag_trigger = yes } add_modifier = { name = tyrone_burma_colonial_growth } }

	on_invalid_all_involved = { if = { limit = { has_modifier = tyrone_burma_colonial_growth } remove_modifier = tyrone_burma_colonial_growth } }

	complete = {
		tyrone_subcontinent_burma_controlled = yes
		hidden_trigger = {
			has_global_variable = tyrone_subcontinent_active_var
		}
	}
	on_complete = {

		hidden_effect = {
			if = {
				limit = { NOT = { has_global_variable = tyrone_subcontinent_presidency_bonus } }
				set_global_variable = { name = tyrone_subcontinent_presidency_bonus value = 0 }
			}
			set_global_variable = {
				name = tyrone_subcontinent_presidency_bonus
				value = { value = global_var:tyrone_subcontinent_presidency_bonus add = 0.1 max = 0.5 }
			}
		}

		change_global_variable = { name = tyrone_subcontinent_progress_var add = 1 }
	}
	on_complete_all_involved = {
		if = { limit = { has_modifier = tyrone_burma_colonial_growth } remove_modifier = tyrone_burma_colonial_growth }
		trigger_event = { id = tyrone_subcontinent.1 days = 0 popup = yes }
	}
	invalid = { NOT = { has_global_variable = tyrone_subcontinent_active_var } }
	progressbar = no
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 20
}
