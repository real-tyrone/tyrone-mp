### American Exceptionalism

je_american_exceptionalism = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_usa_manifest_destiny

	is_shown_when_inactive = {
		c:USA = THIS
	}

	is_shown_in_lobby = {
		c:USA = THIS
	}

	possible = {
		c:USA = THIS
		country_rank >= rank_value:great_power
		is_power_bloc_leader = yes
		has_technology_researched = nationalism
	}

	scripted_button = je_american_exceptionalism_button

	complete = {
		custom_tooltip = {
			text = je_american_exceptionalism_complete_tt
			has_variable = je_american_exceptionalism_declared
		}
	}

	event_outcome_completed_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_american_exceptionalism_completed_tt
			}
		}
	}

	fail = {
		NOT = { is_power_bloc_leader = yes }
	}

	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}

### Spanish-American War

je_spanish_american_war = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_global_international_situations

	is_shown_when_inactive = {
		tyr_resolve_spanish_content_trigger = yes
		exists = c:USA
		exists = scope:tyr_spanish_content_scope
	}

	is_shown_in_lobby = {
		c:USA ?= THIS
	}

	should_be_involved = {
		OR = {
			c:USA ?= THIS
			tyr_spanish_content_country = yes
		}
	}

	should_show_when_not_involved = {
		OR = {
			c:USA ?= THIS
			tyr_spanish_content_country = yes
		}
	}

	possible = {
		tyr_resolve_spanish_content_trigger = yes
		exists = c:USA
		exists = scope:tyr_spanish_content_scope
		c:USA = {
			has_technology_researched = civilizing_mission
			custom_tooltip = {
				text = je_spanish_american_war_ostend_fail
				has_variable = ostend_negotiations_failed
			}
			custom_tooltip = {
				text = je_spanish_american_war_relations_tt
				"relations(scope:tyr_spanish_content_scope)" <= relations_threshold:cold
			}
			custom_tooltip = {
				text = je_spanish_american_war_no_truce_tt
				NOT = { has_truce_with = scope:tyr_spanish_content_scope }
			}
		}
	}

	scripted_button = je_spanish_american_war_start_button

	timeout = 7300

	immediate = {
		tyr_resolve_spanish_content_effect = yes
		scope:tyr_spanish_content_scope = {
			set_variable = { name = saw_spain_americas_subjects value = 0 }
			every_subject_or_below = {
				limit = {
					capital = {
						OR = {
							region = sr:region_central_america
							region = sr:region_great_plains
							region = sr:region_pacific_coast
							region = sr:region_atlantic_coast
							region = sr:region_canada
							region = sr:region_brazil
							region = sr:region_andes
							region = sr:region_la_plata
							region = sr:region_gran_colombia
						}
					}
				}
				scope:tyr_spanish_content_scope = {
					change_variable = { name = saw_spain_americas_subjects add = 1 }
				}
			}
			set_variable = { name = saw_spain_americas_subjects_threshold value = var:saw_spain_americas_subjects }
			change_variable = { name = saw_spain_americas_subjects_threshold multiply = 0.5 }
		}
	}

	custom_completion_header = je_spanish_american_war_american_victory_header

	complete = {
		tyr_resolve_spanish_content_trigger = yes
		custom_tooltip = {
			text = je_saw_american_victory_humiliation_tt
			scope:tyr_spanish_content_scope = {
				has_modifier = country_humiliated
				has_truce_with = c:USA
			}
		}
		custom_tooltip = {
			text = je_saw_american_victory_subjects_tt
			OR = {
				c:USA = {
					has_variable = saw_scripted_war_victory
				}
				NOT = {
					scope:tyr_spanish_content_scope = {
						any_subject_or_below = {
							capital = {
								OR = {
									region = sr:region_central_america
									region = sr:region_great_plains
									region = sr:region_pacific_coast
									region = sr:region_atlantic_coast
									region = sr:region_canada
									region = sr:region_brazil
									region = sr:region_andes
									region = sr:region_la_plata
									region = sr:region_gran_colombia
								}
							}
						}
					}
				}
				NOT = {
					scope:tyr_spanish_content_scope = {
						any_subject_or_below = {
							count >= scope:tyr_spanish_content_scope.var:saw_spain_americas_subjects_threshold
							capital = {
								OR = {
									region = sr:region_central_america
									region = sr:region_great_plains
									region = sr:region_pacific_coast
									region = sr:region_atlantic_coast
									region = sr:region_canada
									region = sr:region_brazil
									region = sr:region_andes
									region = sr:region_la_plata
									region = sr:region_gran_colombia
								}
							}
						}
					}
				}
			}
		}
	}

	on_complete = {
		tyr_resolve_spanish_content_effect = yes
		c:USA ?= {
			add_modifier = { name = saw_american_victory years = 10 }
			trigger_event = { id = spanish_american_war.1 popup = yes }
		}
		scope:tyr_spanish_content_scope ?= {
			add_modifier = { name = saw_spanish_defeat years = 10 }
		}
	}

	custom_failure_header = je_spanish_american_war_spanish_victory_header

	fail = {
		tyr_resolve_spanish_content_trigger = yes
		custom_tooltip = {
			text = je_saw_spanish_victory_humiliation_tt
			c:USA = {
				has_modifier = country_humiliated
				has_truce_with = scope:tyr_spanish_content_scope
			}
		}
	}

	on_fail = {
		tyr_resolve_spanish_content_effect = yes
		scope:tyr_spanish_content_scope ?= {
			add_modifier = { name = saw_spanish_victory years = 10 }
		}
		c:USA ?= {
			add_modifier = { name = saw_american_defeat years = 10 }
		}
	}

	invalid = {
		tyr_resolve_spanish_content_trigger = yes
		c:USA = {
			"relations(scope:tyr_spanish_content_scope)" > relations_threshold:cordial
		}
	}

	weight = 100
	should_be_pinned_by_default_involved = yes
}

### Ostend Manifesto

je_ostend_manifesto = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_global_international_situations

	is_shown_when_inactive = {
		tyr_resolve_spanish_content_trigger = yes
		exists = c:USA
		exists = scope:tyr_spanish_content_scope
		exists = c:CUB
	}

	is_shown_in_lobby = {
		c:USA ?= THIS
	}

	should_be_involved = {
		OR = {
			c:USA ?= THIS
			tyr_spanish_content_country = yes
			c:CUB ?= THIS
			c:PCO ?= THIS
		}
	}

	should_show_when_not_involved = {
		OR = {
			c:USA ?= THIS
			tyr_spanish_content_country = yes
			c:CUB ?= THIS
			c:PCO ?= THIS
		}
	}

	scripted_button = je_ostend_offer_cuba_button
	scripted_button = je_ostend_offer_pco_button
	scripted_button = je_ostend_conclude_negotiations_button
	scripted_button = je_ostend_spa_conclude_negotiations_button
	scripted_button = je_ostend_usa_rejects_button
	scripted_button = je_ostend_spain_rejects_button
	scripted_button = je_ostend_cuba_liberation_button
	scripted_button = je_ostend_cuba_spain_button
	scripted_button = je_ostend_pco_liberation_button
	scripted_button = je_ostend_pco_spain_button

	possible = {
		tyr_resolve_spanish_content_trigger = yes
		exists = c:USA
		exists = scope:tyr_spanish_content_scope
		exists = c:CUB
		c:USA = { has_technology_researched = nationalism }
	}

	immediate = {
		tyr_resolve_spanish_content_effect = yes
		c:USA = {
			set_variable = {
				name = ostend_rejection_cooldown
				days = 365
			}
		}
		scope:tyr_spanish_content_scope = {
			set_variable = {
				name = ostend_rejection_cooldown
				days = 365
			}
		}
	}

	on_complete = {
		c:USA = {
			add_loyalists = {
				value = 0.10
				culture = cu:yankee
			}
			add_loyalists = {
				value = 0.10
				culture = cu:dixie
			}
		}
		hidden_effect = {
			c:CUB ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
			c:PCO ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
		}
	}

	timeout = 365

	on_fail = {
		hidden_effect = {
			c:USA ?= { set_variable = ostend_negotiations_failed }
			c:CUB ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
			c:PCO ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
		}
	}

	on_timeout = {
		hidden_effect = {
			c:USA ?= { set_variable = ostend_negotiations_failed }
			c:CUB ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
			c:PCO ?= {
				if = {
					limit = { has_modifier = ostend_liberation_alignment }
					remove_modifier = ostend_liberation_alignment
				}
				if = {
					limit = { has_modifier = ostend_spain_alignment }
					remove_modifier = ostend_spain_alignment
				}
			}
		}
	}

	fail = {
		tyr_resolve_spanish_content_trigger = yes
		OR = {
			custom_tooltip = {
				text = je_ostend_manifesto_fail_negotiations_tt
				OR = {
					scope:tyr_spanish_content_scope = { has_variable = ostend_spain_rejected }
					c:USA = { has_variable = ostend_usa_rejected }
				}
			}
			custom_tooltip = {
				text = je_ostend_manifesto_fail_war_tt
				exists = c:USA
				exists = scope:tyr_spanish_content_scope
				c:USA = { has_war_with = scope:tyr_spanish_content_scope }
			}
		}
	}

	complete = {
		tyr_resolve_spanish_content_trigger = yes
		c:USA = { exists = yes }
		OR = {
			custom_tooltip = {
				text = je_ostend_manifesto_complete_concluded_tt
				OR = {
					c:USA = { has_variable = ostend_usa_negotiations_concluded }
					scope:tyr_spanish_content_scope = { has_variable = ostend_spa_negotiations_concluded }
				}
			}
			AND = {
				custom_tooltip = {
					text = je_ostend_manifesto_complete_cuba_tt
					c:CUB = {
						OR = {
							is_subject_of = c:USA
							NOT = { is_subject_of = scope:tyr_spanish_content_scope }
						}
					}
				}
				custom_tooltip = {
					text = je_ostend_manifesto_complete_pco_tt
					c:PCO = {
						OR = {
							is_subject_of = c:USA
							NOT = { is_subject_of = scope:tyr_spanish_content_scope }
						}
					}
				}
			}
		}
	}

	weight = 100
	should_be_pinned_by_default_involved = yes
}
