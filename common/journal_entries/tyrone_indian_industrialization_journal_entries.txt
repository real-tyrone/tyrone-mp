# Regional follow-up to Ending the Manufacturing Monopoly.
# Contextless entries shared by India and its current overlord; all requirements stay inline.
je_tyrone_promise_industrialize_punjab = {
	icon = "gfx/interface/icons/invention_icons/steelworking.dds"
	group = je_group_british_india_global
	should_be_involved = { tyrone_promise_is_actor = yes }
	should_show_when_not_involved = { tyrone_promise_is_actor = yes }
	# Second wave: hidden until all three initial industrializations are complete.
	is_shown_when_inactive = {
		has_global_variable = tyrone_promise_active_var
		tyrone_promise_india_exists = yes
	}
	possible = {
		custom_tooltip = {
			text = tyrone_promise_industry_second_wave_unlock_tt
			hidden_trigger = { has_global_variable = tyrone_promise_active_var }
			hidden_trigger = { has_global_variable = tyrone_promise_industrialize_bombay_done }
			hidden_trigger = { has_global_variable = tyrone_promise_industrialize_bengal_done }
			hidden_trigger = { has_global_variable = tyrone_promise_industrialize_carnatic_malabar_done }
			hidden_trigger = { tyrone_promise_india_exists = yes }
		}
	}
	immediate = {
		tyrone_promise_bind_india = yes
		tyrone_india_portfolio_refresh_involvement = {
			INDIA = scope:tyrone_promise_india
			ENTRY = je_tyrone_promise_industrialize_punjab
		}
	}
	on_monthly_pulse = {
		effect = {
			tyrone_promise_bind_india = yes
			tyrone_india_portfolio_refresh_involvement = {
				INDIA = scope:tyrone_promise_india
				ENTRY = je_tyrone_promise_industrialize_punjab
			}
		}
	}
	complete = {
		hidden_trigger = {
			has_global_variable = tyrone_promise_active_var
			exists = scope:tyrone_promise_india
			scope:tyrone_promise_india ?= { is_country_alive = yes }
		}
		scope:tyrone_promise_india ?= {
			owns_entire_state_region = STATE_PUNJAB
			any_scope_state = {
				state_region = s:STATE_PUNJAB
				state_has_building_group_levels = {
					type = bg_manufacturing
					value >= 100
				}
				any_scope_building = {
					is_building_type = building_urban_center
					level >= 50
				}
				any_scope_building = {
					is_building_type = building_railway
					level >= 10
				}
			}
		}
	}
	on_complete = {
		s:STATE_PUNJAB = {
			if = {
				limit = { has_state_trait = state_trait_tyrone_indian_rural_region }
				remove_state_trait = state_trait_tyrone_indian_rural_region
			}
			if = {
				limit = { NOT = { has_state_trait = state_trait_tyrone_early_industrialized } }
				add_state_trait = state_trait_tyrone_early_industrialized
			}
		}
	}
	invalid = {
		hidden_trigger = {
			NOT = {
				AND = {
					has_global_variable = tyrone_promise_active_var
					exists = scope:tyrone_promise_india
					scope:tyrone_promise_india ?= { is_country_alive = yes }
				}
			}
		
		}
	}
	progressbar = no
	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 20
}
