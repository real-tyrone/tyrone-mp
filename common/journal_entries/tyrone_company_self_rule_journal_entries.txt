je_tyrone_company_self_rule = {
	icon = "gfx/interface/icons/law_icons/colonial_administration.dds"
	group = je_group_british_india_global
	should_be_involved = {
		OR = {
			has_variable = tyrone_csr_started
			any_subject_or_below = {
				has_variable = tyrone_csr_started
				is_subject_of = root
			}
		}
	}
	should_show_when_not_involved = { always = no }
	immediate = {
		tyrone_csr_bind_india = yes
		tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_company_self_rule }
	}
	timeout = tyrone_csr_reform_period
	complete = {
		hidden_trigger = {
			exists = scope:csr_india
			scope:csr_india ?= { NOT = { has_variable = tyrone_csr_resolved } }
		}
		scope:csr_india ?= {
			custom_tooltip = {
				text = tyrone_csr_all_reforms_done_tt
				var:tyrone_csr_reforms >= 3
			}
		}
	}
	on_complete = {
		hidden_effect = { scope:csr_india ?= { tyrone_csr_resolve = yes } }
	}
	fail = {
		hidden_trigger = {
			exists = scope:csr_india
			scope:csr_india ?= {
				is_country_alive = yes
				NOT = { has_variable = tyrone_csr_resolved }
			}
		}
		scope:csr_india ?= {
			custom_tooltip = {
				text = tyrone_csr_failure_condition_tt
				NOT = { has_variable = tyrone_csr_active }
				var:tyrone_csr_reforms = 0
			}
		}
	}
	on_fail = {
		custom_tooltip = tyrone_csr_failure_outcome_tt
		show_as_tooltip = {
			scope:csr_india ?= {
				if = {
					limit = { is_subject = yes }
					activate_law = law_type:law_india_crown_administration
					activate_law = law_type:law_india_crown_industrial_obligations
					if = {
						limit = { NOT = { has_law = law_type:law_colonial_administration } }
						activate_law = law_type:law_colonial_administration
					}
					activate_law = law_type:law_autocracy
					if = {
						limit = {
							NOR = {
								has_law = law_type:law_hindu_caste_enforced
								has_law = law_type:law_hindu_caste_codified
							}
						}
						activate_law = law_type:law_hindu_caste_codified
					}
				}
			}
		}
		hidden_effect = {
			scope:csr_india ?= { tyrone_csr_resolve = yes }
		}
	}
	on_timeout = {
		hidden_effect = { scope:csr_india ?= { tyrone_csr_resolve = yes } }
	}
	event_outcome_completed_effect_desc = {
		header = tyrone_csr_settlement_preview_header
		effect = { custom_tooltip = tyrone_csr_evaluation_tt }
	}
	event_outcome_completed_effect_desc = {
		header = tyrone_csr_public_dividends_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_public_dividends days = long_modifier_time }
			}
		}
	}
	event_outcome_completed_effect_desc = {
		header = tyrone_csr_private_investment_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_private_investment days = long_modifier_time }
				add_loyalists = { pop_type = capitalists value = 0.20 }
			}
		}
	}
	event_outcome_completed_effect_desc = {
		header = tyrone_csr_construction_drive_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_construction_drive days = long_modifier_time }
				add_loyalists = { value = 0.02 }
			}
		}
	}
	event_outcome_timeout_effect_desc = {
		header = tyrone_csr_settlement_preview_header
		effect = { custom_tooltip = tyrone_csr_evaluation_tt }
	}
	event_outcome_timeout_effect_desc = {
		header = tyrone_csr_public_dividends_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_public_dividends days = long_modifier_time }
			}
		}
	}
	event_outcome_timeout_effect_desc = {
		header = tyrone_csr_private_investment_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_private_investment days = long_modifier_time }
				add_loyalists = { pop_type = capitalists value = 0.20 }
			}
		}
	}
	event_outcome_timeout_effect_desc = {
		header = tyrone_csr_construction_drive_preview_header
		effect = {
			scope:csr_india ?= {
				add_modifier = { name = tyrone_csr_construction_drive days = long_modifier_time }
				add_loyalists = { value = 0.02 }
			}
		}
	}
	invalid = {
		OR = {
			NOT = { exists = scope:csr_india }
			scope:csr_india ?= { is_country_alive = no }
		}
	}
	on_invalid = {
		scope:csr_india ?= { remove_variable = tyrone_csr_active }
	}
	on_weekly_pulse = {
		effect = { tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_company_self_rule } }
	}
	current_value = { value = scope:csr_india.var:tyrone_csr_reforms }
	goal_add_value = { value = 3 }
	progressbar = yes
	progress_desc = {}
	widget = {
		gui = "gui/journal_entry_widgets/tyrone_unstable_raj_widgets.gui"
		name = "tyrone_company_self_rule_subentries_widget"
		container = "custom_widget_container_1"
	}
	weight = 100
	should_be_pinned_by_default_involved = yes
	can_revolution_inherit = yes
}

			}
		}
		scope:csr_india = {
			NOT = { has_law_or_variant = law_type:law_no_police }

			set_variable = tyrone_csr_police_reward_pending
			trigger_event = { id = tyrone_company_self_rule.1 popup = yes }
		}
	}
	invalid = {
		OR = {
			NOT = { exists = scope:csr_india }
			scope:csr_india ?= {
				OR = {
					is_country_alive = no
					NOT = { has_variable = tyrone_csr_active }
				}
			}
		}
	}
	on_weekly_pulse = {
		effect = { tyrone_csr_refresh_involvement = { ENTRY = je_tyrone_csr_police } }
	}
	weight = 80
	can_revolution_inherit = yes
}
