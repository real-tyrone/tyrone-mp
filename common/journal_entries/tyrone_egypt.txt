je_oriental_crisis = {
	activation_description = {
		always = yes
	}


	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
	transferable = no
	weight = 10000

	immediate = {
		c:TUR = {
			save_scope_as = ottoman_scope
			create_bidirectional_truce = {
				country = c:EGY
				months = 6
			}			
		}
		c:EGY = {
			save_scope_as = egypt_scope
			trigger_event = {
				id = egyptian_crisis_events.3
				months = 6
			}
		}
	}

	should_be_involved = {
		OR = {
			c:TUR ?= THIS
       	 	c:EGY ?= THIS
		}

	}

	is_shown_when_inactive = {
	}









	weight = 1000

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
		custom_tooltip = {
			text = ottoman_victory_tt
			has_global_variable = oriental_crisis_ottoman_victory
		}
	}
		effect = {
		c:TUR = {
			if = {
				limit = {
					has_variable = defeated_egypt_militarily
				}
				trigger_event = { id = egyptian_crisis_events.5 }
			}
		}
	}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
		custom_tooltip = {
			text = egyptian_victory_tt
			has_global_variable = oriental_crisis_egyptian_victory
		}
	}
		effect = {
		c:EGY = {
			trigger_event = { id = egyptian_crisis_events.6 }
		}
	}
	}
}
