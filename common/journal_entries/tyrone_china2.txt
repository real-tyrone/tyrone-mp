	}

		set_variable = chi_boxer_campaign_active

		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { chi_legation_is_central = yes }
		}
		effect = {
			remove_variable = chi_boxer_campaign_active
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header

			custom_tooltip = {
				text = chi_boxer_ports_recovered_tt

			}
		}
		effect = {
			chi_boxer_success = yes
		}
	}
	outcome = {

		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC

			if = {
				limit = { chi_legation_is_central = yes }
