je_tyrone_raj_famine_1 = {
	icon = "gfx/interface/icons/invention_icons/quinine.dds"
	group = je_group_tyrone_india_unstable
	should_be_involved = {
		OR = {
			is_india_tag_trigger = yes
			any_subject_or_below = {
				is_india_tag_trigger = yes
				is_direct_subject_of = root
			}
		}
	}
	should_show_when_not_involved = {
		OR = {
			is_india_tag_trigger = yes
			any_subject_or_below = {
				is_india_tag_trigger = yes
				is_direct_subject_of = root
			}
		}
	}
	scripted_button = tyrone_raj_famine_food_1
	scripted_button = tyrone_raj_famine_works_1
	scripted_button = tyrone_raj_famine_medical_1
	immediate = {
		tyrone_unstable_bind_india = yes
		scope:journal_entry = {
			remove_variable = tyrone_famine_food_funded
			remove_variable = tyrone_famine_works_funded
			remove_variable = tyrone_famine_medical_funded
		}
		scope:raj_scope = {
			var:tyrone_raj_famine_region_1 = {
				save_scope_as = famine_region
			}
		}
		tyrone_raj_famine_refresh_state = yes
	}

	on_weekly_pulse = {
		effect = {
			tyrone_raj_famine_refresh_state = yes
			scope:journal_entry = {
				every_scope_je_involved = {
					limit = {
						NOR = {
							scope:raj_scope ?= this
							scope:raj_scope.overlord ?= this
						}
					}
					scope:journal_entry = {
						remove_involved_country = prev
					}
				}
			}
			scope:raj_scope = {
				if = {
					limit = {
						NOT = {
							is_involved_in_journal_entry = je_tyrone_raj_famine_1
						}
					}
					scope:journal_entry = {
						add_involved_country = prev
					}
				}
				if = {
					limit = {
						is_subject = yes
					}
					overlord = {
						if = {
							limit = {
								NOT = {
									is_involved_in_journal_entry = je_tyrone_raj_famine_1
								}
							}
							scope:journal_entry = {
								add_involved_country = prev
							}
						}
					}
				}
			}
		}
	}
	on_monthly_pulse = {
		effect = {
			tyrone_raj_famine_monthly_spread = {
				EVENT = 6
			}
		}
	}
	weight = 85

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			scope:raj_scope = {
				NOT = {
					has_variable = tyrone_unstable_raj_active
				}
			}
		}
		effect = {
			tyrone_raj_famine_close = {
				SLOT = 1
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:raj_scope = {
				hidden_trigger = { has_variable = tyrone_unstable_raj_active }
			}
			scope:famine_region = {
				NOT = {
					has_harvest_condition = disease_outbreak
				}
			}
		}
		effect = {
			tyrone_raj_famine_close = {
				SLOT = 1
			}
		}
	}
}
