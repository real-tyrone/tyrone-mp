je_tyrone_great_power = {
	icon = "gfx/interface/icons/invention_icons/military_drill.dds"
	group = je_group_historical_content
	
	is_shown_when_inactive = {
		has_variable = can_have_ty_gp_var
		hidden_trigger = {
			tyr_gp_player_rule_eligible = yes
		}
		NOT = { tyr_brz_gp_country = yes }
	}
	
	possible = {
		hidden_trigger = {
			tyr_gp_player_rule_eligible = yes
		}
		is_power_bloc_leader = yes
		country_rank = rank_value:great_power
	}
	
	immediate = {
		scope:journal_entry = {
			add_modifier = {
				name = tyr_gp_rising_great_power
				years = 15
				is_decaying = yes
			}
		}
		set_variable = {
			name = tyr_gp_initial_population
			value = total_population
		}
		set_variable = {
			name = tyr_gp_months_without_leadership
			value = 0
		}
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = { NOT = { has_variable = tyr_gp_initial_population } }
				desc = tyr_gp_status_potential
			}
			triggered_desc = { desc = tyr_gp_status_active }
		}
	}
	
	on_monthly_pulse = {
		effect = {
			if = {
				limit = { is_power_bloc_leader = yes }
				set_variable = {
					name = tyr_gp_months_without_leadership
					value = 0
				}
			}
			else = {
				change_variable = {
					name = tyr_gp_months_without_leadership
					add = 1
				}
			}
		}
	}
	
	complete = {
		# Failure takes precedence when both terminal conditions are true.
		NOT = { tyr_gp_leadership_failed = yes }
		OR = {
			game_date >= 1860.1.1
			custom_tooltip = {
				text = tyr_gp_all_objectives_tt
				tyr_gp_objectives_met = 10
			}
		}
	}
	
	on_complete = {
		set_variable = {
			name = tyr_gp_reward_scale
			value = tyr_gp_objectives_met
		}
		# Effect previews do not execute set_variable; guard all subsequent reads.
		if = {
			limit = {
				has_variable = tyr_gp_reward_scale
			}
			if = {
				limit = { var:tyr_gp_reward_scale > 0 }
				add_modifier = {
					name = tyr_gp_national_ascendancy
					multiplier = var:tyr_gp_reward_scale
				}
				if = {
					limit = { is_in_power_bloc = yes }
					save_scope_as = tyr_gp_reward_country
					power_bloc = {
						add_modifier = {
							name = tyr_gp_bloc_ascendancy
							multiplier = scope:tyr_gp_reward_country.var:tyr_gp_reward_scale
						}
					}
				}
			}
		}
		remove_variable = tyr_gp_initial_population
		remove_variable = tyr_gp_months_without_leadership
		remove_variable = tyr_gp_reward_scale
		remove_variable = can_have_ty_gp_var
	}
	
	event_outcome_completed_effect_desc = {
		header = tyr_gp_completion_header
		effect = {
			custom_tooltip = tyr_gp_reward_tt
		}
	}
	
	fail = {
		custom_tooltip = {
			text = tyr_gp_failure_tt
			tyr_gp_leadership_failed = yes
		}
	}
	
	on_fail = {
		tyr_gp_apply_failure = yes
		remove_variable = tyr_gp_initial_population
		remove_variable = tyr_gp_months_without_leadership
		remove_variable = tyr_gp_reward_scale
		remove_variable = can_have_ty_gp_var
	}
	
	on_invalid = {
		remove_variable = tyr_gp_initial_population
		remove_variable = tyr_gp_months_without_leadership
		remove_variable = tyr_gp_reward_scale
		remove_variable = can_have_ty_gp_var
	}
	
	transferable = no
	can_revolution_inherit = yes
	weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
}
