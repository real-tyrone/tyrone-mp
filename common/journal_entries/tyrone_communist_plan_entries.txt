je_tyrone_cp_village = {
	icon = "gfx/interface/icons/invention_icons/urbanization.dds"
	group = je_group_tyrone_india_socialist
	context = country
	status_desc = { desc = tyrone_cp_village_status }
	immediate = {
		save_scope_as = cp_india
		set_variable = {
			name = tyrone_cp_peasant_share
			value = {
				value = tyrone_cp_peasants
				divide = {
					value = total_population
					min = 1
				}
			}
		}
		set_variable = {
			name = tyrone_cp_initial_peasants
			value = tyrone_cp_peasants
		}
		set_variable = {
			name = tyrone_cp_village_months
			value = 0
		}
	}
	on_monthly_pulse = {
		effect = {
			set_variable = {
				name = tyrone_cp_peasant_share
				value = {
					value = tyrone_cp_peasants
					divide = {
						value = total_population
						min = 1
					}
				}
			}
			if = {
				limit = {
					tyrone_cp_peasants <= {
						value = var:tyrone_cp_initial_peasants
						multiply = 0.5
					}
					tyrone_cp_lower_sol >= 12
				}
				change_variable = {
					name = tyrone_cp_village_months
					add = 1
				}
			}
			else = {
				set_variable = {
					name = tyrone_cp_village_months
					value = 0
				}
			}
		}
	}

	current_value = {
		value = var:tyrone_cp_village_months
	}
	goal_add_value = {
		value = 24
	}
	progressbar = yes
	weight = 75

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = tyrone_cp_below_twenty_million_tt
					tyrone_cp_peasants < 20000000
				}
				custom_tooltip = {
					text = tyrone_cp_village_twentyfour_tt
					var:tyrone_cp_village_months >= 24
				}
			}
		}
		effect = {
			hidden_effect = {
				set_variable = tyrone_cp_village_done
				trigger_event = {
					id = tyrone_cp.20
					popup = yes
				}
			}
			show_as_tooltip = {
				add_modifier = {
					name = tyrone_cp_village_reward
					days = tyrone_cp_ten_years
				}
				add_loyalists = {
					value = 0.05
				}
			}
		}
	}
}
