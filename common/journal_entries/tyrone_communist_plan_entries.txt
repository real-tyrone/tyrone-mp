je_tyrone_cp_plan_heavy = {
	icon = "gfx/interface/icons/invention_icons/manufacturies.dds"
	group = je_group_historical_content
	immediate = {
		save_scope_as = cp_india
	}
	complete = {
		custom_tooltip = {
			text = tyrone_cp_heavy_deadline_tt
			always = no
		}
	}
	on_complete = {
		show_as_tooltip = {
			active_law:lawgroup_economic_system = {
				add_amendment = {
					type = amendment_tyrone_cp_heavy
					sponsor = root.ig:ig_trade_unions
				}
			}
		}
	}
	timeout = tyrone_cp_five_years
	on_timeout = {
		tyrone_cp_resolve_heavy = yes
	}
	invalid = {
		NOT = {
			has_variable = tyrone_cp_plan_active
		}
	}
	current_value = {
		value = tyrone_cp_heavy_levels
	}
	goal_add_value = {
		value = var:tyrone_cp_target
		subtract = tyrone_cp_heavy_levels
	}
	progressbar = yes
	weight = 80
}

je_tyrone_cp_village = {
	icon = "gfx/interface/icons/invention_icons/urbanization.dds"
	group = je_group_historical_content
	status_desc = { desc = tyrone_cp_village_status }
	immediate = {
		save_scope_as = cp_india
		set_variable = {
			name = tyrone_cp_peasant_share
			value = {
				value = tyrone_cp_peasants
				divide = {
					value = total_population
					min = 1
				}
			}
		}
		set_variable = {
			name = tyrone_cp_initial_peasants
			value = tyrone_cp_peasants
		}
		set_variable = {
			name = tyrone_cp_village_months
			value = 0
		}
	}
	on_monthly_pulse = {
		effect = {
			set_variable = {
				name = tyrone_cp_peasant_share
				value = {
					value = tyrone_cp_peasants
					divide = {
						value = total_population
						min = 1
					}
				}
			}
			if = {
				limit = {
					tyrone_cp_peasants <= {
						value = var:tyrone_cp_initial_peasants
						multiply = 0.5
					}
					tyrone_cp_lower_sol >= 12
				}
				change_variable = {
					name = tyrone_cp_village_months
					add = 1
				}
			}
			else = {
				set_variable = {
					name = tyrone_cp_village_months
					value = 0
				}
			}
		}
	}
	complete = {
		OR = {
			custom_tooltip = {
				text = tyrone_cp_below_twenty_million_tt
				tyrone_cp_peasants < 20000000
			}
			custom_tooltip = {
				text = tyrone_cp_village_twentyfour_tt
				var:tyrone_cp_village_months >= 24
			}
		}
	}
	on_complete = {
		hidden_effect = {
			set_variable = tyrone_cp_village_done
			trigger_event = {
				id = tyrone_cp.20
				popup = yes
			}
		}
		show_as_tooltip = {
			add_modifier = {
				name = tyrone_cp_village_reward
				days = tyrone_cp_ten_years
			}
			add_loyalists = {
				value = 0.05
			}
		}
	}
	current_value = {
		value = var:tyrone_cp_village_months
	}
	goal_add_value = {
		value = 24
	}
	progressbar = yes
	weight = 75
}
je_tyrone_cp_forced = {
	icon = "gfx/interface/icons/invention_icons/steelworking.dds"
	group = je_group_historical_content
	immediate = {
		save_scope_as = cp_india
	}
	timeout = tyrone_cp_ten_years
	complete = {
		custom_tooltip = {
			text = tyrone_cp_rank_steel_tt
			"goods_production_rank(g:steel)" <= 3
		}
		custom_tooltip = {
			text = tyrone_cp_rank_glass_tt
			"goods_production_rank(g:glass)" <= 3
		}
		custom_tooltip = {
			text = tyrone_cp_rank_explosives_tt
			"goods_production_rank(g:explosives)" <= 3
		}
		custom_tooltip = {
			text = tyrone_cp_rank_tools_tt
			"goods_production_rank(g:tools)" <= 3
		}
		custom_tooltip = {
			text = tyrone_cp_rank_fertilizer_tt
			"goods_production_rank(g:fertilizer)" <= 3
		}
		custom_tooltip = {
			text = tyrone_cp_rank_engines_tt
			"goods_production_rank(g:engines)" <= 3
		}
	}
	on_complete = {
		tyrone_cp_end_forced = yes
		set_variable = tyrone_cp_forced_done
		trigger_event = {
			id = tyrone_cp.22
			popup = yes
		}
	}
	on_timeout = {
		if = {
			limit = {
				tyrone_cp_forced_goal = yes
			}
			set_variable = tyrone_cp_forced_done
			trigger_event = {
				id = tyrone_cp.22
				popup = yes
			}
		}
		else = {
			trigger_event = {
				id = tyrone_cp.23
				popup = yes
			}
		}
		tyrone_cp_end_forced = yes
	}
	on_monthly_pulse = {
		effect = {
			tyrone_cp_check_famine = yes
		}
	}
	progressbar = no
	weight = 80
}
