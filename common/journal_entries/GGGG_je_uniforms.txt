GGGG_je_full_dress = {
	activation_description = {
		always = yes
	}

    icon = "gfx/interface/icons/event_icons/event_military.dds"
    
    group = je_group_technology
    
    is_shown_when_inactive = {
        NOR = {
            has_variable = GGGG_has_uniforms_3
            has_variable = GGGG_has_uniforms_2
        }
        OR = {
            has_technology_researched = mandatory_service
            has_technology_researched = military_drill
        }
        GGGG_sv_count_military_building_levels >= 5
    }
    
    possible = {
        has_technology_researched = line_infantry
    }
    
    immediate = {
        ROOT = {
            save_scope_as = target_country
        }
    }
    

    

    

    
    timeout = 3650

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
        custom_tooltip = {
            text = GGGG_tt_full_dress_je
            GGGG_sv_mo_usage_full_dress > GGGG_FULL_DRESS_ADOPTION_THRESHOLD
        }
    }
		effect = {
        trigger_event = {
            id = GGGG_ev_uniforms.1
        }
    }
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
        OR = {
            hidden_trigger = {
                has_variable = GGGG_has_uniforms_3
            }
            has_journal_entry = GGGG_je_helmets_drab
            has_technology_researched = bolt_action_rifles
        }
    }
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
	}
}
