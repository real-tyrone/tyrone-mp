	icon = "gfx/interface/icons/event_icons/raj_indian_assembly.dds"
	group = je_group_british_india_global

	should_be_involved = {
		is_india_tag_trigger = yes
	}

	should_show_when_not_involved = {
		always = no
	}

	immediate = {
		tyrone_nim_bind_india = yes
		scope:nim_india ?= {
			tyrone_nim_start = yes
		}
		tyrone_nim_refresh_involvement = {
			ENTRY = je_tyrone_csr_independent_path
		}
	}

	on_monthly_pulse = {
		effect = {
			tyrone_nim_refresh_involvement = {
				ENTRY = je_tyrone_csr_independent_path
			}
			scope:nim_india ?= {
				if = {
					limit = {
						NOT = {
							has_variable = tyrone_nim_started
						}
					}
					tyrone_nim_start = yes
				}
			}
		}
	}

	complete = {
		exists = scope:nim_india
		scope:nim_india = {
			custom_tooltip = {
				text = tyrone_nim_constitution_done_tt
				has_variable = tyrone_nim_constitution_done
			}
			custom_tooltip = {
				text = tyrone_nim_administration_done_tt
				has_variable = tyrone_nim_administration_done
			}
			custom_tooltip = {
				text = tyrone_nim_schools_done_tt
				has_variable = tyrone_nim_schools_done
			}
			custom_tooltip = {
				text = tyrone_nim_army_done_tt
				has_variable = tyrone_nim_army_done
			}
			custom_tooltip = {
				text = tyrone_nim_regions_done_tt
				has_variable = tyrone_nim_pakistan_done
				has_variable = tyrone_nim_bengal_done
			}
		}
	}

	on_complete = {
		hidden_effect = {
			scope:nim_india ?= {
				set_variable = tyrone_nim_main_done
				remove_variable = tyrone_nim_active
				every_subject_or_below = {
					limit = {
						is_princely_state = yes
						NOT = { c:PAK ?= this }
						NOT = { c:BGL ?= this }
					}
					save_scope_as = nim_integrating_prince
					scope:nim_india = { annex = scope:nim_integrating_prince }
				}
				trigger_event = { id = tyrone_nim.4 popup = yes }
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = tyrone_nim_bloc_reward_header
		effect = { custom_tooltip = tyrone_nim_bloc_reward_tt }
	}

	event_outcome_completed_effect_desc = {
		header = tyrone_nim_princes_reward_header
		effect = { custom_tooltip = tyrone_nim_annex_princely_subjects_tt }
	}

	progressbar = no
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes

	widget = {
		gui = "gui/journal_entry_widgets/tyrone_nim_widgets.gui"
		name = "tyrone_nim_subentries_widget"
		container = "custom_widget_container_1"
	}

	invalid = {
		OR = {
			NOT = {
				exists = scope:nim_india
			}
			scope:nim_india ?= {
				has_variable = tyrone_nim_socialist_started
			}
		}
	}

	weight = 40
	can_revolution_inherit = yes
}

je_tyrone_nim_assembly = {
	icon = "gfx/interface/icons/law_icons/parliamentary_republic.dds"
	group = je_group_british_india_global

	should_be_involved = {
		is_india_tag_trigger = yes
	}

	should_show_when_not_involved = {
		always = no
	}

	immediate = {
		tyrone_nim_bind_india = yes
		scope:journal_entry = {
			set_global_variable = {
				name = tyrone_nim_assembly_je
				value = this
			}
		}
		tyrone_nim_refresh_involvement = {
			ENTRY = je_tyrone_nim_assembly
		}
	}

	on_monthly_pulse = {
		effect = {
			tyrone_nim_refresh_involvement = {
				ENTRY = je_tyrone_nim_assembly
			}
			scope:journal_entry = {
				set_global_variable = {
					name = tyrone_nim_assembly_je
					value = this
				}
			}
			scope:nim_india ?= {
				if = {
					limit = {
						tyrone_nim_constitution_open = yes
					}
					trigger_event = {
						on_action = tyrone_nim_constitution_monthly
					}
					if = {
						limit = {
							has_technology_researched = socialism
						}
						trigger_event = {
							on_action = tyrone_nim_socialism_monthly
						}
					}
				}
				if = {
					limit = {
						tyrone_nim_constitution_open = yes
						has_technology_researched = socialism
						global_var:tyrone_nim_assembly_je ?= {
							"scripted_bar_progress(tyrone_nim_socialism_bar)" >= 100
						}
					}
					if = {
						limit = {
							has_variable = tyrone_nim_emergency_used
						}
						tyrone_nim_start_revolution = yes
					}
					else = {
						set_variable = tyrone_nim_emergency_pending
						trigger_event = {
							id = tyrone_nim.10
							popup = yes
						}
					}
				}
			}
		}
	}

	scripted_progress_bar = tyrone_nim_constitution_bar
	scripted_progress_bar = tyrone_nim_socialism_bar
	scripted_button = tyrone_nim_back_monarchy
	scripted_button = tyrone_nim_back_federal
	scripted_button = tyrone_nim_adopt_monarchy
	scripted_button = tyrone_nim_adopt_federal

	complete = {
		exists = scope:nim_india
		scope:nim_india = {
			custom_tooltip = {
				text = tyrone_nim_constitution_done_tt
				has_variable = tyrone_nim_constitution_done
			}
		}
	}

	progressbar = no

	invalid = {
		OR = {
			NOT = {
				exists = scope:nim_india
			}
			scope:nim_india ?= {
				has_variable = tyrone_nim_socialist_started
			}
		}
	}

	event_outcome_completed_effect_desc = {
		header = tyrone_nim_assembly_monarchy_preview_header
		effect = {
			scope:nim_india ?= {
				activate_law = law_type:law_monarchy
				activate_law = law_type:law_autocracy
				activate_law = law_type:law_india_national_secretariat
			}
			custom_tooltip = tyrone_nim_assembly_reforms_preview_tt
		}
	}

	event_outcome_completed_effect_desc = {
		header = tyrone_nim_assembly_federal_preview_header
		effect = {
			scope:nim_india ?= {
				activate_law = law_type:law_parliamentary_republic
				activate_law = law_type:law_census_voting
				activate_law = law_type:law_india_federal_secretariats
			}
			custom_tooltip = tyrone_nim_assembly_reforms_preview_tt
		}
	}

	event_outcome_completed_effect_desc = {
		header = tyrone_nim_assembly_socialist_preview_header
		effect = { custom_tooltip = tyrone_nim_assembly_socialist_preview_tt }
	}

	weight = 40
	can_revolution_inherit = yes
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes
}
