je_tyr_kk_kulturkampf = {
	icon = "gfx/interface/icons/invention_icons/bureaucracy.dds"
	group = je_group_germany_tyrone

	is_shown_when_inactive = {
		c:GER ?= root
		NOT = {
			has_variable = tyr_kk_completed
		}
	}

	possible = {
		any_scope_character = {
			is_character_alive = yes
			has_ideology = ideology:ideology_bismarck_paternalistic
		}
		has_technology_researched = philosophical_pragmatism
	}

	immediate = {
		custom_tooltip = {
			text = tyr_kk_start_tt
			hidden_effect = {
				random_scope_character = {
					limit = {
						is_character_alive = yes
						has_ideology = ideology:ideology_bismarck_paternalistic
					}
					save_scope_as = tyr_kk_bismarck_scope
				}
				ig:ig_intelligentsia = {
					save_scope_as = tyr_kk_intelligentsia_scope
				}
				ig:ig_devout = {
					save_scope_as = tyr_kk_devout_scope
				}
				set_variable = {
					name = tyr_kk_balance
					value = 50
				}
			}
		}
	}

	widget = {
		gui = "gui/journal_entry_widgets/tyr_kk_widgets.gui"
		name = "widget_tyr_kk_balance"
		container = "custom_widget_container_1"
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = {
						exists = scope:tyr_kk_bismarck_scope
					}
					any_scope_character = {
						is_character_alive = yes
						has_ideology = ideology:ideology_bismarck_paternalistic
					}
				}
				random_scope_character = {
					limit = {
						is_character_alive = yes
						has_ideology = ideology:ideology_bismarck_paternalistic
					}
					save_scope_as = tyr_kk_bismarck_scope
				}
			}
			ig:ig_intelligentsia = {
				save_scope_as = tyr_kk_intelligentsia_scope
			}
			ig:ig_devout = {
				save_scope_as = tyr_kk_devout_scope
			}
			set_variable = {
				name = tyr_kk_balance
				value = tyr_kk_next_balance
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = tyr_kk_complete_tt
			OR = {
				tyr_kk_balance <= 0
				tyr_kk_balance >= 100
			}
		}
	}

	on_complete = {
		hidden_effect = {
			set_variable = tyr_kk_completed
			if = {
				limit = {
					tyr_kk_balance >= 100
				}
				set_variable = tyr_kk_state_victory
			}
			trigger_event = {
				id = tyr_kk.1
				popup = yes
			}
		}
	}

	event_outcome_completed_desc = {
		first_valid = {
			triggered_desc = {
				desc = tyr_kk_outcome_preview
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					has_journal_entry = je_tyr_kk_kulturkampf
				}
				desc = je_tyr_kk_kulturkampf_status_active
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = je_tyr_kk_kulturkampf_status
			}
		}
	}

	transferable = no
	can_revolution_inherit = no
	weight = 145
}
