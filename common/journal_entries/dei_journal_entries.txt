REPLACE:je_reform_indies = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		AND = {
			exists = c:DEI
			c:DEI = THIS
		}
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_free_trade
			has_law_or_variant = law_type:law_mercantilism
			NOT = { has_law_or_variant = law_type:law_land_based_taxation }
			NOT = { has_law_or_variant = law_type:law_serfdom }
		}
	}

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
		OR = {
			has_law_or_variant = law_type:law_free_trade
			has_law_or_variant = law_type:law_mercantilism
		}
		NOT = { has_law_or_variant = law_type:law_land_based_taxation }
		NOT = { has_law_or_variant = law_type:law_serfdom }
		NOT = { has_law_or_variant = law_type:law_extraction_economy}
		has_law_or_variant = law_type:law_slavery_banned
	}

	on_complete = {
		trigger_event = { id = east_indies.2 }
		show_as_tooltip = {
			add_journal_entry = {
				type = je_ethical_policy
			}
		}
	}

	invalid = {
		OR = {
			has_variable = bic_collapse_var
			NOT = { c:DEI = THIS }
		}
	}


	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.5
	}

	progressbar = yes

	modifiers_while_active = {
		dei_company_men
	}

	should_be_pinned_by_default = yes

	weight = 100
}

REPLACE:je_ethical_policy = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_internal_affairs

	complete = {
		literacy_rate > 0.5
		root.average_sol > 15
	}

	on_complete = {
		trigger_event = { id = east_indies.3 }
	}

	invalid = {
		OR = {
			has_variable = bic_collapse_var
			NOT = { c:DEI = THIS }
		}
	}

	weight = 100
}
