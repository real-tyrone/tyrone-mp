je_mandate_of_heaven = {
	icon = "gfx/interface/icons/event_icons/Mandate_Heaven.dds"
	group = je_group_tyrone_china_mandate
	context = country
	highlight = yes
	scripted_progress_bar = chi_centralisation_progress_bar

	widget = {
		gui = "gui/journal_entry_widgets/chi_mandate_subentries.gui"
		name = "chi_mandate_subentries"
		container = "custom_widget_container_3"
	}
	on_monthly_pulse = {
		random_events = {
			21 = 0
			2 = chi_mandate.1
			1 = chi_mandate.2
		}
		effect = {
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_monarchy }
					NOT = { has_variable = chi_mandate_monarchy_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = -3
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = chi_mandate_monarchy_penalty_applied
			}

			if = {
				limit = {
					has_modifier = opium_wars_lost
					NOT = { has_variable = chi_mandate_opium_wars_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = -1
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = chi_mandate_opium_wars_penalty_applied
			}

			if = {
				limit = {
					exists = c:TPG
					NOT = { has_variable = chi_mandate_taiping_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = -1
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = chi_mandate_taiping_penalty_applied
			}

			if = {
				limit = {
					taking_loans = yes
					NOT = { has_variable = chi_mandate_loan_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = -1
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = chi_mandate_loan_penalty_applied
			}
			if = {
				limit = {
					has_law = law_type:law_chi_mixed_economic_policy
					NOT = { has_variable = chi_mandate_mixed_policy_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = 2
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = {
					name = chi_mandate_mixed_policy_penalty_applied
					value = 1
				}
			}

			if = {
				limit = {
					has_law = law_type:law_chi_industry_banned
					NOT = { has_variable = chi_mandate_industry_banned_penalty_applied }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = 1
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = {
					name = chi_mandate_industry_banned_penalty_applied
					value = 1
				}
			}

			if = {
				limit = {
					any_civil_war = {
						civil_war_progress > 0
					}
					NOT = { has_variable = chi_mandate_civil_war_penalty_active }
				}
				je:je_mandate_of_heaven = {
					add_progress = {
						value = -1
						name = chi_centralisation_progress_bar
					}
				}
				set_variable = chi_mandate_civil_war_penalty_active
			}

			if = {
				limit = {
					NOT = {
						any_civil_war = {
							civil_war_progress > 0
						}
					}
					has_variable = chi_mandate_civil_war_penalty_active
				}
				remove_variable = chi_mandate_civil_war_penalty_active
			}

			chi_mandate_update_centralisation_modifier = yes
		}
	}
	weight = 1000
	should_be_pinned_by_default_uninvolved_or_context = yes
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = chi_mandate_centralisation_complete_tt
				scope:journal_entry = {
					"scripted_bar_progress(chi_centralisation_progress_bar)" >= 10
				}
			}
		}
		effect = {
			hidden_effect = {
				chi_mandate_update_centralisation_modifier = yes
			}
			add_modifier = { name = chi_mandate_consolidated_authority }
		}
	}
}

je_economic_direction = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	group = je_group_tyrone_china_mandate
	context = country
	weight = 1000
	modifiers_while_active = { chi_economic_direction_consumption }
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				AND = {
					relative_bureaucracy >= 0.25
					NOT = { has_law_or_variant = law_type:law_no_schools }
					NOT = { has_law_or_variant = law_type:law_serfdom }
					has_technology_researched = romanticism
				}
				AND = {
					has_law_or_variant = law_type:law_local_police
					has_technology_researched = romanticism
					chi_qing_army_power >= 1500
				}
			}
		}
		effect = {
			if = {
				limit = {
					relative_bureaucracy >= 0.25
					NOT = { has_law_or_variant = law_type:law_no_schools }
					NOT = { has_law_or_variant = law_type:law_serfdom }
					has_technology_researched = romanticism
				}
				activate_law = law_type:law_chi_mixed_economic_policy
				add_modifier = {
					name = economic_direction_mixed_policy
				}
			}
			else_if = {
				limit = {
					has_law_or_variant = law_type:law_local_police
					has_technology_researched = romanticism
					chi_qing_army_power >= 1500
				}
				activate_law = law_type:law_chi_industry_banned
				add_modifier = {
					name = economic_direction_industry_banned
				}
			}
			add_journal_entry = {
				type = je_homeland_development
			}
		}

		event_outcome_desc = {
			first_valid = {
				desc = chi_economic_direction_preview
			}
		}
	}
}

je_confront_confucian_scholars = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	group = je_group_tyrone_china_mandate
	context = country
	scripted_button = chi_confucian_scholars_button
	immediate = {
		set_variable = {
			name = chi_confucian_scholars_progress
			value = 0
		}
	}
	on_monthly_pulse = {
		random_events = {
			10 = 0
			1 = chi_confucian_scholars.3
			1 = chi_confucian_scholars.4
		}
		effect = {
			if = {
				limit = {
					OR = {
						has_variable = chi_confucian_scholars_end_isolation
						has_variable = chi_confucian_scholars_evict_scholars
					}
					var:chi_confucian_scholars_progress >= 75
					NOT = { has_variable = chi_confucian_manchu_customs_event_fired }
				}
				set_variable = chi_confucian_manchu_customs_event_fired
				trigger_event = {
					id = chi_confucian_scholars.6
					popup = yes
				}
			}

			if = {
				limit = {
					has_variable = chi_confucian_scholars_evict_scholars
					var:chi_confucian_scholars_progress > 40
					has_journal_entry = je_chinese_missions
					NOT = { has_variable = chi_confucian_western_practices_event_fired }
				}
				set_variable = chi_confucian_western_practices_event_fired
				trigger_event = {
					id = chi_confucian_scholars.5
					popup = yes
				}
			}

			if = {
				limit = {
					OR = {
						has_variable = chi_confucian_scholars_end_isolation
						has_variable = chi_confucian_scholars_evict_scholars
					}
					var:chi_confucian_scholars_progress < 96
				}
				change_variable = {
					name = chi_confucian_scholars_progress
					add = 1
				}
			}
		}
	}
	current_value = {
		value = var:chi_confucian_scholars_progress
	}
	goal_add_value = {
		value = 96
		subtract = var:chi_confucian_scholars_progress
	}
	progressbar = yes
	weight = 1000
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		effect = {
			if = {
				limit = {
					has_variable = chi_confucian_scholars_end_isolation
				}
				activate_law = law_type:law_chi_religious_focus
				ig:ig_devout ?= {
					remove_ideology = ideology_chi_devout_traditionalist
					add_ideology = ideology_chi_devout_scholar
				}
			}
			if = {
				limit = {
					has_variable = chi_confucian_scholars_evict_scholars
				}
				activate_law = law_type:law_freedom_of_conscience
				activate_law = law_type:law_chi_merchent_led
			}
			je:je_mandate_of_heaven = {
				add_progress = {
					value = 1
					name = chi_centralisation_progress_bar
				}
			}
			if = { limit = { has_modifier = chi_confucian_scholars_button_modifier } remove_modifier = chi_confucian_scholars_button_modifier }
			remove_variable = chi_confucian_scholars_progress
		}
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
}
