	}

		NOT = { exists = c:TPG }
		custom_tooltip = {
			text = missionaries_enough_time_tt
			OR = {
				AND = {
					has_variable = missionary_works_counter

			}
		}

    limit = { tyr_heavenly_founding_state = yes }

   if = {
    limit = { NOT = { has_modifier = god_worshippers } }
    add_modifier = { name = god_worshippers years = 5 }
   }
   set_variable = cult_center
  }
  trigger_event = { id = heavenly.2 }
 }

	on_monthly_pulse = {
		random_events = {
			3 = 0
			1 = heavenly.3
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			NOT = {
				any_state = {

					has_modifier = god_worshippers
				}
			}
		}
		effect = {
			je_heavenly_kingdom_main_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
