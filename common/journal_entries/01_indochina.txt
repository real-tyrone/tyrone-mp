je_indochina = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_foreign_affairs

		is_shown_when_inactive = {
		has_dlc_feature = voice_of_the_people_content
		OR = {
			country_has_state_religion = rel:catholic
			country_has_primary_culture = cu:north_german
		}
		country_rank = rank_value:great_power
		is_player = yes
		capital = {
			NOT = { region = sr:region_indochina }
		}
		any_country = {
			capital = {
				region = sr:region_indochina
			}
			NOT = {
				country_has_state_religion = rel:catholic
			}
			has_law_or_variant = law_type:law_state_religion
		}
		NOT = {
			has_global_variable = indochina_intervention_happened
		}
	}

	possible = {
		OR = {
			country_has_state_religion = rel:catholic
			has_journal_entry = je_tyr_gcol_empire
		}
		has_technology_researched = nationalism
		has_interest_marker_in_region = sr:region_indochina
		OR = {
			has_technology_researched = civilizing_mission
			any_interest_group = {
				is_powerful = yes
				is_in_government = yes
				leader = {
					OR = {
						has_ideology = ideology:ideology_jingoist_leader
						has_ideology = ideology:ideology_ethno_nationalist
						has_ideology = ideology:ideology_fascist
						has_ideology = ideology:ideology_integralist
						has_ideology = ideology:ideology_bonapartist
					}
				}
			}
			ig:ig_devout ?= { is_powerful = yes }
			ig:ig_armed_forces ?= { is_powerful = yes }
		}
		OR = {
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_indochina
				value >= 3
			}
			has_technology_researched = civilizing_mission
		}
		OR = {
			has_law_or_variant = law_type:law_colonial_resettlement
			has_law_or_variant = law_type:law_colonial_exploitation
		}
	}

	immediate = {
		trigger_event = { id = indochina.2 }
		random_country = {
			limit = {
				has_variable = indochina_campaign_target
			}
			save_scope_as = indochina_conquest_target
			remove_variable = indochina_campaign_target
			capital = {
				state_region = {
					set_variable = indochina_capital_state
					save_scope_as = indochina_capital_state_scope
				}
			}
		}
		set_global_variable = indochina_intervention_happened
	}

	complete = {
		is_subject = no
		is_at_war = no
		OR = {
			trigger_if = {
				limit = {
					exists = scope:indochina_conquest_target
				}
				custom_tooltip = {
					text = conquered_indochina_target_tt
					NOT = { exists = scope:indochina_conquest_target }
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = conquered_indochina_target_blank_tt
					NOT = { exists = scope:indochina_conquest_target }
				}
			}
			AND = {
				exists = scope:indochina_conquest_target
				scope:indochina_conquest_target ?= { is_subject_of = ROOT }
			}
			custom_tooltip = {
				text = indochina_five_states_conquered
				any_state_in_indochina = {
					OR = {
						owner = ROOT
						owner = { is_subject_of = ROOT }
					}
					region = sr:region_indochina
					count >= 5
				}
			}
		}
		custom_tooltip = {
			text = indochina_state_controlled_by_us_or_puppet
			any_state_in_indochina = {
				OR = {
					owner = ROOT
					owner = { is_subject_of = ROOT }
				}
				state_region = {
					has_variable = indochina_capital_state
				}
			}
		}
	}

	on_complete = {
		hidden_effect = {
			s:STATE_CAMBODIA = {
				add_claim = ROOT
			}
		}
		trigger_event = { id = indochina.4 }
	}

	weight = 100

	transferable = no

	invalid = {
	}

		
	should_be_pinned_by_default_uninvolved_or_context = yes
}
